S500 · P055 a Java 21 Spring Boot

P055 se registra en COMS y XATMI y queda en espera de mensajes; cada vez que el minuto del reloj es 00 o 30 revisa los archivos S500/FILE/S500/ARCHBANDERA/{CSI}/{fecha}/{secuencia} del 20 al 00. Por cada archivo residente valida el header (tipo 01, CSI local, fecha, paso P055, sistema S500 y secuencia), lee el registro de datos tipo 03 y, si la hora del host supera la hora programada, confirma la residencia de la pieza y arma un WFL según el tipo de proceso: con parámetros, sin parámetros, lote o remoción de archivos. Marca el registro como procesado o rechazado y renombra el archivo con el prefijo APLICADOP055 para no repetirlo. Termina cuando COMS le envía el evento de fin de paso.

public class ProcesosEspecialesScheduler {

    private final RevisionBanderas revision;     // 50204000-DESPIERTO

    /**
     * ESPERACOMS devuelve un evento (mensaje, ACCEPT, HI o timeout) y después de cada uno el programa
     * revisa el reloj; solo actúa si el minuto es múltiplo de 30 y distinto del último atendido.
     * Un cron explícito reemplaza ese sondeo y elimina la dependencia del tiempo de espera de COMS.

    @Scheduled(cron = "${p055.cron:0 0/30 * * * *}", zone = "America/Mexico_City")

    }

/** WS-REG-I01-HEADER: 120 bytes, tipo 01. */
public record HeaderBandera(int tipo, int csi, LocalDate fecha,       // 9(02), 9(02), 9(08) AAAAMMDD
                            String paso, String sistema, int secuencia) {
    public boolean esValido(int csiLocal, LocalDate fechaLinea, int secuenciaEsperada) {
        return tipo == 1 && csi == csiLocal && fecha.equals(fechaLinea)
            && "P055".equals(paso) && "S500".equals(sistema) && secuencia == secuenciaEsperada;
    }
}

/** WS-REG-I01-DATOS: tipo 03 con el proceso a ejecutar. */
public record DatosBandera(TipoProceso tipo,
                           String nombreProceso,          // X(40): WFL, lote o archivo a remover
                           LocalTime horaProceso,         // 9(08) HHMMSSCC
                           TipoParametro tipoParametro,   // "V" TASKVALUE, "P" parámetro
                           String parametro,              // X(50)
                           Estado estado) {

    public enum TipoProceso {
        WFL_SIN_PARAMETRO(1), WFL_CON_PARAMETRO(2), LOTE_CON_PARAMETRO(6), REMOVER_ARCHIVOS(90);
        final int codigo;
        TipoProceso(int c) { codigo = c; }
        static Optional<TipoProceso> de(int c) {             // W88-TPO-PRO-OK
            return Arrays.stream(values()).filter(t -> t.codigo == c).findFirst();
        }
    }
    public enum TipoParametro { TASKVALUE, PARAMETRO }
    public enum Estado { PENDIENTE(0), PROCESADO(1), RECHAZADO(9); final int codigo; Estado(int c) { codigo = c; } }
}
@Component
@RequiredArgsConstructor
public class RevisionBanderas {

    private final P055Properties props;
    private final AlmacenArchivos archivos;          // ATTRIBUTE RESIDENT OF ...
    private final ProcesadorBandera procesador;      // 50204600-BUSCA-PROCESO

    /** 50204300-PROCESA-ARCHIVOS: secuencias de la máxima a 00, todas las residentes del día. */
    public void ejecuta(LocalDateTime ahora) {
        LocalDate fecha = ahora.toLocalDate();       // WKS-FEC-LINEA = fecha de la máquina
        for (int seq = props.secuenciaMaxima(); seq >= 0; seq--) {
            Path bandera = Path.of(props.rutaBanderas()
                    .replace("{csi}", "%02d".formatted(props.csiLocal()))
                    .replace("{fecha}", fecha.format(DateTimeFormatter.BASIC_ISO_DATE))
                    .replace("{seq}", "%02d".formatted(seq)));
            if (archivos.existe(bandera)) {
                procesador.procesa(bandera, seq, fecha, ahora.toLocalTime());
            }
        }
    }
}
@Component
@RequiredArgsConstructor
public class ProcesadorBandera {

    private final ArchivoBanderaReader lector;
    private final BanderaDecider decider;
    private final P055Properties props;

    public void procesa(Path bandera, int seq, LocalDate fecha, LocalTime hora) {
        Optional<HeaderBandera> header = lector.header(bandera);
        if (header.isEmpty()) {
            log.warn(" ERROR EN ARCHIVO {} VACIO SIN REGS, NO SE APLICA", bandera);
            lector.renombraAplicado(bandera);                // W77-ERR-ARCH = 2: se renombra sin marcar
            return;
        }
        if (!header.get().esValido(props.csiLocal(), fecha, seq)) {
            log.warn(" ERROR HEADER {}", header.get());
            lector.renombraAplicado(bandera);
            return;
        }
        decider.evalua(bandera, lector.datos(bandera), hora);   // 50204610-LEE-REG-ARCH
    }
}
@Component
@RequiredArgsConstructor
public class BanderaDecider {

    private static final int MAX_INTENTOS = 6;            // W77-INTENTOS > 6
    private static final Duration ESPERA = Duration.ofSeconds(20);   // WAIT (20)
    private final AlmacenArchivos archivos;
    private final CatalogoLotes lotes;                    // 98700000-TIT-LOTE (CTLVERS S500JBATCH)
    private final LanzadorProcesoEspecial lanzador;

    public Decision evalua(Path bandera, DatosBandera d, LocalTime ahora) {
        if (!ahora.isAfter(d.horaProceso())) {            // WKS-HRA-HOST-HMSC > WKS-HRA-ARCH
            log.info(" SON LAS {} ARCH={} SE APLICA HASTA LAS {}", ahora, bandera, d.horaProceso());
            return Decision.ESPERAR;                      // se reintenta en la siguiente media hora
        }
        if (d.tipo() != TipoProceso.REMOVER_ARCHIVOS) {
            String pieza = d.tipo() == TipoProceso.LOTE_CON_PARAMETRO
                    ? lotes.tituloLote().orElse(null) : d.nombreProceso();
            if (pieza == null || !esperaResidencia(pieza)) {
                log.warn("PIEZA NO RESIDENTE {} NO EFECTUA PROC-ESPECIAL", pieza);
                return Decision.RECHAZAR;                 // estado 09
            }
        }
        return lanzador.lanza(d) ? Decision.PROCESADO : Decision.RECHAZAR;
    }

    private boolean esperaResidencia(String pieza) {
        for (int i = 0; i <= MAX_INTENTOS; i++) {
            if (archivos.existe(Path.of("(S500)" + pieza))) return true;
            sleepUninterruptibly(ESPERA);
        }
        return false;
    }
}

    private final RemocionArchivos remocion;
    private final P055Properties props;

    /** 50204630-CORRE-ESPECIAL: devuelve true si se lanzó (W77-ARCH-APL = 1). */
    public boolean lanza(DatosBandera d) {
        try {
            return switch (d.tipo()) {
                case WFL_CON_PARAMETRO  -> arranca(d.nombreProceso(), d.parametro());
                case WFL_SIN_PARAMETRO  -> arranca(d.nombreProceso(), null);
                case LOTE_CON_PARAMETRO -> arranca(d.nombreProceso(), d.parametro());
                case REMOVER_ARCHIVOS   -> remocion.remueve(d.nombreProceso());   // valida rutas permitidas
            };
        } catch (NoSuchJobException | JobParametersInvalidException | JobInstanceAlreadyExistsException e) {
            log.warn(" NO EFECTUO PROCESO  ERROR EN PIEZA {}", d.nombreProceso(), e);
            return false;                                    // W77-ERR-ARCH = 1, estado 09
        }
    }

    private boolean arranca(String job, String parametro) throws NoSuchJobException,
            JobParametersInvalidException, JobInstanceAlreadyExistsException {
        registro.getJob(job);                                // lanza NoSuchJobException si no está permitido
        String params = (parametro == null) ? "" : "parametro=" + parametro.strip();
        jobOperator.start(job, params);                      // el ambiente lo da el perfil, no una rama de código
        return true;
    }

Riesgos de migración
Credenciales en el fuente

Las plantillas WFL de desarrollo (líneas 114-115, 186-190, 204-208, 230-234, 252-256 y 684) contienen un ACCESSCODE con contraseña en texto claro (valor ****). No se copiaron a este documento. Deben rotarse y moverse a un gestor de secretos.

Ejecución arbitraria desde el archivo bandera

El nombre del WFL, del lote o del archivo a remover viene del registro de datos sin lista blanca (líneas 1516-1548). Escribir en la ruta de archivos bandera equivale a poder ejecutar o borrar con el usercode del programa.

Procesado no significa exitoso

El estado 01 se graba al lanzar el WFL (líneas 1374-1383); no se verifica el resultado del proceso especial.

Header inválido consume el archivo

Con error de header o archivo vacío el archivo se renombra a APLICADOP055 (líneas 1385-1388) sin ejecución; el proceso especial se pierde sin alerta más allá del listado.

Bloqueo durante la espera de residencia

50204625-VALIDA-PIEZA espera hasta siete veces 20 segundos con interrupciones deshabilitadas, lo que retrasa la atención de los demás archivos y de la orden de fin.

Ambiente por hostname

Un equipo no reconocido se considera producción (líneas 1024-1033); el CSI 10 y 04 por nombre de equipo está compilado.
