Resuelve el título de la librería de control con CTLVERS y lee B01SISDIA; si falla, termina. Con THECALENDAR valida que la fecha recibida sea hábil y obtiene su día de la semana, que se usa como posición del día en la tabla de diez días de B01. Fija fecha de proceso, contable y de proceso S151 con la fecha recibida y llena las demás posiciones con días anteriores retrocediendo en el calendario y saltando sábados y domingos; luego graba B01 con MANTSISDIA. En B03SISMEN busca el año y mes de la fecha entre los ciclos y, si no existe, reemplaza el más antiguo; marca la base de saldos en 1 salvo que esté en 99 y graba con MANTSISMEN. Por último lee B04SISTEM y, si su estatus es 3, invoca CONTROLES de L002 con la función 25.

@Component
@RequiredArgsConstructor
@Slf4j
public class LibControlResolver {

    private final ControlVersionesClient ctlvers;   // DAME_TIT IN CTLVER
    private final P677Properties props;
    // p677.libctl-respaldo: "(S151)S151/OBJECT/L001/CONTROL/02MTP011 ON CMEMP"
    // p677.lib-disp: "(S000)S000/UTILITY/DESPLIEGA/OBJ/LIB ON PACK."

    public String libControl() {
        var r = ctlvers.dameTitulo("S151L001CTL");
        if (r.error() != 0) {
            log.warn("CTLVERS sin título para S151L001CTL ({}); se usa {}", r.error(), props.libctlRespaldo());
            return props.libctlRespaldo();         // el fuente cambia el título sin dejar mensaje
        }
        return r.titulo();
    }
}
@Component
@RequiredArgsConstructor
public class LecturaControlTasklet implements Tasklet {

    private final LibControlClient libctl;   // CONSISDIA IN LIBCTL
    private final CsiProvider csi;           // CSIID OF SOPORTECOMS

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        var p = (AperturaParametros) ctx.getStepContext().getJobExecutionContext().get("parametros");
        var r = libctl.consisdia(csi.numCsi(), p.sistema());        // función 1
        if (r.resultado() > 0) {
            throw new ProcesoAbortadoException("TERMINACION ANORMAL DEL PROGRAMA");   // STATUS -1
        }
        ctx.getStepContext().getStepExecution().getJobExecution()
           .getExecutionContext().put("controlDia", r.controlDia());
        return RepeatStatus.FINISHED;
    }
}

@Service
@RequiredArgsConstructor
public class CalendarioService {

    private final CalendarioBancario calendario;   // THECALENDAR IN LOCSUP, formato 13

    /** 02-00110-DAME-DIA: devuelve el día de la semana (0 domingo ... 6 sábado) de una fecha hábil. */
    public int diaDeLaSemanaHabil(LocalDate fecha) {
        var r18 = calendario.funcion18(fecha);    // ¿es día hábil?
        if (r18.error() != 0 || !"00000001".equals(r18.valor())) {
            throw new ProcesoAbortadoException("LA FECHA %s NO CORRESPONDE A UN DIA HABIL"
                    .formatted(fecha.format(DateTimeFormatter.BASIC_ISO_DATE)));
        }
        var r06 = calendario.funcion6(fecha);     // día de la semana
        if (r06.error() > 0) throw new ProcesoAbortadoException("ERROR EN PARAMETRO FECHA ");
        return Integer.parseInt(r06.valor());     // W77-NUMDIA y WKS-B01-NUMARC
    }
}
@Service
@RequiredArgsConstructor
public class VentanaDiasService {

    private final CalendarioService calendario;
    private final CalendarioBancario lib;   // THECALENDAR función 8 (resta 1 día) y 6

    /** 02-00100-GENERA-B01SISDIA. */
    public ControlDia abreDia(ControlDia b01, LocalDate fecha) {
        int numDia = calendario.diaDeLaSemanaHabil(fecha);
        var dias = new ArrayList<>(b01.dias());
        dias.set(numDia - 1, ControlDia.DiaCiclo.nuevo(fecha));          // posición del día actual

        var cursor = new LocalDate[] { fecha };                           // WKS-FECHA
        // Primero las posiciones posteriores al día actual, hasta la 10...
        llena(dias, numDia + 1, ControlDia.DIAS_VENTANA, cursor);
        // ...y luego, si el día es mayor que 1, las anteriores, siguiendo hacia atrás en el tiempo.
        if (numDia > 1) llena(dias, 1, numDia - 1, cursor);

        return b01.conFechas(fecha, fecha, fecha)                        // FECPRO, FECCON, FECPRO151
                  .conNumArc(numDia).conDias(dias);                      // 02-00105-GRABA-BASE: MANTSISDIA F37
    }

    /** 02-00130-FECHAS: retrocede un día; sólo sábado (6) y domingo (0) se saltan. */
    private void llena(List<ControlDia.DiaCiclo> dias, int desde, int hasta, LocalDate[] cursor) {
        int ind = desde;
        while (ind <= hasta) {
            cursor[0] = lib.restaDias(cursor[0], 1);
            int dow = lib.diaDeLaSemana(cursor[0]);
            if (dow != 0 && dow != 6) {
                dias.set(ind - 1, ControlDia.DiaCiclo.nuevo(cursor[0]));   // el fuente también pone WKS-B01-STAREG en 0
                ind++;
            }
        }
    }

public class CicloMensualTasklet implements Tasklet {

    private static final int STABDSAL_EN_RECUPERACION = 99, STABDSAL_ABIERTA = 1;
    private final LibControlClient libctl;   // CONSISMEN / MANTSISMEN IN LIBCTL

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        var p = (AperturaParametros) ctx.getStepContext().getJobExecutionContext().get("parametros");
        var b03 = libctl.consismen(p.sistema()).orElseThrow(
                () -> new ProcesoAbortadoException("ERROR EN CONSISMEN  "));
        var ciclos = b03.ciclos().subList(0, b03.numCicDia());          // OCCURS 99, vigentes NUMCICDIA
        int aamm = Integer.parseInt(p.cicloMensual().format(DateTimeFormatter.ofPattern("yyyyMM")));

        boolean existe = ciclos.stream().anyMatch(ci -> ci.aamm() == aamm);
        if (!existe && !ciclos.isEmpty()) {
            // 02-00220-BUSCA-MENOR: en empate gana la última posición ("=" o "<")
            int idx = IntStream.range(0, ciclos.size()).boxed()
                    .reduce((a, b) -> ciclos.get(b).aamm() <= ciclos.get(a).aamm() ? b : a).orElseThrow();
            ciclos.set(idx, ciclos.get(idx).conAamm(aamm));               // sólo cambia AAMM
        }
        if (b03.staBdSal() != STABDSAL_EN_RECUPERACION) b03 = b03.conStaBdSal(STABDSAL_ABIERTA);

        if (libctl.mantsismen(b03).resultado() > 0) {                    // F37: se graba siempre
            throw new ProcesoAbortadoException("ERROR EN MANTSISMEN ");
        }
        return RepeatStatus.FINISHED;
    }
}

Riesgos de migración
Cierre de archivos de control comentado

En 01-00400-CIERRA-ARCHIVOS el CALL CONTROLES está comentado (líneas 906-907); el PERFORM VARYING de las líneas 817-819 sólo calcula WKS-DEL-FUNCION diez veces.

Festivos dentro de la ventana de diez días

02-00130-FECHAS sólo salta sábado y domingo (línea 1011); un festivo entre semana ocupa una posición de FECARCMOV como si hubiera sido día de proceso.

Resultados sin validar y respaldos silenciosos

El resultado de B04SISTEM (líneas 1092-1093) no se revisa, y si CTLVERS falla se usan títulos fijos de LIBCTL (02MTP011) y L002 sin mensaje al operador (líneas 846-850 y 879-883).

Renglón de ciclo reutilizado con datos del mes anterior

Al reemplazar el ciclo más antiguo en B03SISMEN sólo se actualiza AAMM (línea 1062); contadores y estatus del mes reemplazado permanecen.

Mensajes de inicio y fin desactivados

Las llamadas a LJ de INICIO PROCESO y FIN NORMAL DE PROCESO están comentadas (líneas 808 y 824-825); sólo se registran los errores.
