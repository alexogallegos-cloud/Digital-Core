public record DiferenciaSaldo(
        int producto,               // SDO-PRD 9(04)
        int instrumento,            // SDO-INS 9(04)
        int moneda,                 // SDO-MON 9(03)
        String contrato,            // SDO-CTO 9(12): String para conservar ceros
        BigDecimal saldoInversion,  // SDO-INV S9(14)V99 -> escala 2, con signo
        BigDecimal saldoHistorico,  // SDO-HIS S9(14)V99
        BigDecimal saldoCifras) {   // SDO-CIF S9(14)V99

    public boolean esTotalProducto() { return CONTRATO_TOTALES.equals(contrato); }

    public DiferenciaSaldo {
        saldoInversion = saldoInversion.setScale(2, RoundingMode.UNNECESSARY);
        saldoHistorico = saldoHistorico.setScale(2, RoundingMode.UNNECESSARY);
        saldoCifras    = saldoCifras.setScale(2, RoundingMode.UNNECESSARY);
    }

@Component
@StepScope
@RequiredArgsConstructor
public class ConciliacionProcessor implements ItemProcessor<Contrato, ResultadoContrato> {

    private static final Set<Pim> EXCLUIDOS_MOTOR = Set.of(        // W88-QPIM-*
            new Pim(66, 90, 1), new Pim(66, 11, 1), new Pim(66, 17, 1), new Pim(500, 6, 1));
    private static final Pim QPIM_0500_0006 = new Pim(500, 6, 1);

    private final HistoricoService historico;
    private final TablaCopias copias;
    @Value("#{jobParameters['copia']}") private int copiaActual;   // WS-COPIAP160

    @Override
    public ResultadoContrato process(Contrato c) {
        if (c.migradoS111()) return null;                          // B03-MIGR-S111 = 1: se ignora
        var his = historico.de(c);                                 // 50213090-LEE-HIS
        var r = new ResultadoContrato(c);

        boolean excluidoMotor = c.status() == 4 || c.numCliente() == 0
                || EXCLUIDOS_MOTOR.contains(c.pim());
        if (copiaActual == 1 && !excluidoMotor) MotorPolicy.registro(c, his, contextoMotor).ifPresent(r::motor);

        if (copias.copiaDe(c.pim()).numero() == copiaActual) {
            boolean excluidoConciliacion = c.pim().equals(QPIM_0500_0006) && Set.of(2, 3, 4).contains(c.status());
            if (!excluidoConciliacion) r.diferencia(concilia(c, his));   // 50210000-OPERO-INV
        }
        return r;
    }
}

    /** 50213090-LEE-HIS: el contrato sin histórico es un caso de negocio; un error de base aborta. */
    public Historico de(Contrato c) {
        try {
            return b06.findByNumContrato(c.numContrato())
                    .orElseGet(() -> {
                        log.info("CTO INEXISTENTE EN B06 , PIM={}, CTO={}, COPIA={}",
                                 c.pim(), c.numContrato(), copiaActual);
                        return Historico.vacio();      // saldos anteriores en 0 y sin acumular días
                    });
        } catch (DataAccessException e) {
            // "ERROR AL HACER FIND  A B06": CALL SYSTEM DMTERMINATE
            throw new ProcesoAbortadoException("ERROR AL HACER FIND A B06 PIM=%s CTO=%s"
                    .formatted(c.pim(), c.numContrato()), e);
        }
    }
}
/** 50200500-ARMA-MOTOR + 50200520-ESC-MOTOR. */
public final class MotorPolicy {

    public static Optional<RegistroMotor> registro(Contrato c, Historico h, ContextoMotor ctx) {
        BigDecimal sdoAct = c.saldoActual().add(c.saldoProtBloq());          // WKS-MOT-SDOACT
        BigDecimal sdoAnt = h.saldoAntAnt().add(h.saldoBloqAntAnt());        // WKS-MOT-SDOANT
        boolean recalculoDistinto = !c.fecRctaOri().equals(c.fecUltMov());

        boolean escribe =
                (ctx.tipoArchivo() == 1 && recalculoDistinto)                 // tipo de archivo 1
             || (c.fecUltMov().equals(ctx.fechaProceso()) && recalculoDistinto) // movimiento del día
             || c.status() == 1                                               // contrato activo
             || sdoAct.compareTo(sdoAnt) != 0;                                // cambió el saldo
        if (!escribe) return Optional.empty();

        return Optional.of(new RegistroMotor(
                c.numCliente(), c.numContrato(), c.producto(), c.instrumento(), c.moneda(),
                c.fecUltMov(),                       // FECHULTM 9(08)
                ctx.tipoArchivo(),                   // TPOMOVTO
                sdoAnt, sdoAct,                      // S9(14)V99, suma exacta de escala 2
                c.status() == 1 ? 1 : 0));           // STACTO
    }
}

/** 50212000-ACUMULO + 50213000-VEOHIS + 50213100-SDO-HIS + 50213110-ACUM-MOV. */
private DiferenciaSaldo concilia(Contrato c, Historico h) {
    // Saldo de inversiones por producto: actual + protegido por orden judicial (sin pendiente).
    totales.acumulaInversion(c.pim(), c.saldoActual().add(c.saldoProtBloq()));

    BigDecimal depositos = BigDecimal.ZERO.setScale(2);
    BigDecimal retiros   = BigDecimal.ZERO.setScale(2);
    if (h.existe()) {                                           // WS-4VEZ = 0
        for (int dia = 1; dia <= 31; dia++) {                   // tope fijo de 31 posiciones
            depositos = depositos.add(h.depositosDia(dia));     // B06-DEPS-DIA(WS-DIAC2)
            retiros   = retiros.add(h.retirosDia(dia));         // B06-RETS-DIA(WS-DIAC2)
        }
    }
    // COMPUTE WS-SUM-SDOCAL = B03-SDO-FINCORTE + DEP - RET  (S9(14)V99, sin redondeo: exacto)
    BigDecimal saldoCalculado = c.saldoFinCorte().add(depositos).subtract(retiros);
    return compara(c, saldoCalculado);
}
private DiferenciaSaldo compara(Contrato c, BigDecimal saldoCalculado) {
    // 50213200-COMPARO: en COBOL este COMPUTE sobrescribe B03-SDO-ACTUAL en el buffer de la base.
    BigDecimal saldoComparable = c.saldoActual()
            .add(c.saldoPendiente())
            .add(c.saldoProtBloq());                     // nunca se modifica la entidad
    if (saldoComparable.compareTo(saldoCalculado) == 0) {
        return null;                                    // cuadra: no se escribe en FILE-SALDOS
    }
    return new DiferenciaSaldo(c.producto(), c.instrumento(), c.moneda(), c.numContrato(),
            saldoComparable, saldoCalculado, BigDecimal.ZERO.setScale(2));   // 50213210-GRABO
}

Riesgos de migración
Saldo de la base modificado en memoria

50213200-COMPARO (líneas 1633-1634) sobrescribe B03-SDO-ACTUAL con la suma de actual, pendiente y protegido. No se encontró un STORE de B03, pero cualquier lógica posterior que lea ese campo en la misma iteración verá un saldo alterado.

Definiciones de saldo inconsistentes

La tabla por producto acumula actual más protegido (líneas 1558-1560) y la comparación por contrato incluye además el pendiente (líneas 1633-1634). Los listados de diferencias y de posición no son comparables entre sí.

Capacidad real de 30 productos

Las búsquedas en las tablas de copias y de cifras (líneas 1465-1471 y 1504-1510) terminan al pasar de 30 y luego exigen índice menor a 31; un producto en la posición 31 aborta el programa.

Acumulación de 31 días sin fecha

50213110-ACUM-MOV (líneas 1618-1624) suma las 31 posiciones del arreglo sin considerar la fecha de corte que menciona el comentario.

Exclusiones compiladas

Los códigos QPIM 0066-0090-01, 0066-0011-01, 0066-0017-01 y 0500-0006-01 y el estatus 4 de cancelado están en condiciones 88 y en IF del fuente (líneas 1377-1382 y 1400-1401). Se trasladan a catálogo o propiedades.
