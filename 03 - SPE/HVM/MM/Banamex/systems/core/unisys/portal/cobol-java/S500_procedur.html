/** Familia 10000xxx-*: resuelve el TITLE vigente de una librería con DAME_TIT IN CTLVERS. */
@Component
@RequiredArgsConstructor
public class LibreriaResolver {

    private final CtlversClient ctlvers;          // 20000100-DAME-TIT-VERS
    private final Map<String, ServicioLigado> ligadas = new ConcurrentHashMap<>();

    public ServicioLigado liga(String libId) {     // p. ej. "S016L422"
        return ligadas.computeIfAbsent(libId, id -> {
            ResultadoCtlvers r = ctlvers.dameTitulo(id);   // GIVING S000-CTR-CVEERROR
            if (r.cveError() != 0) {
                // COBOL solo emite el mensaje y aun así prende WKS-88-S016L422.
                log.error("ERROR EN CALL CTLVERS {} RESULT={}", id, r.cveError());
                return ServicioLigado.noDisponible(id, r.cveError());
            }
            return ServicioLigado.de(id, r.titulo());       // CHANGE ATTRIBUTE TITLE
        });
    }
}

public class RegistroMovimientoS151Service {

    static final int MAX_MOVS = 30;               // WS-S151-01-CVES-I OCCURS 30 TIMES

    private final RuteoContableS151 ruteo;        // 20000151-CARGAMOV1-CTES
    private final AgrupadorMovimientos agrupador; // 20000151-CARGAMOV1-IMP
    private final RegistraS500Client registra;    // 20000151-CARGAMOV1-CALL

    /** 20000151-CARGAMOV1: arma el registro S151 y lo envía en bloques de hasta 5 movimientos. */
    public void cargaMovimientos(EntradaS151 entrada) {
        MovimientoS151 base = ruteo.aplica(MovimientoS151.vacio(), entrada);  // INI + CTES
        var ctx = new ContextoSaldo(entrada.saldoIni());   // SALDO-FIN = SALDO-INI
        var bloque = agrupador.nuevoBloque(base, ctx);
        for (LineaS151 linea : entrada.lineas().subList(0, Math.min(MAX_MOVS, entrada.lineas().size()))) {
            if (linea.cveTran() == 0) break;                // WS-S151-SW = 1
            bloque = agrupador.agrega(bloque, linea, ctx, registra::envia);
        }
        registra.envia(bloque);                             // CALL final, aunque el bloque venga vacío
        // CARGAMOV1-LIMPIA: no aplica, EntradaS151 es inmutable y se descarta.
    }
}

@Component
@RequiredArgsConstructor
public class RuteoContableS151 {

    private final RuteoProperties p;
    private final ModoEjecucion modo;   // $SET OMIT = NOT LINEA

    MovimientoS151 aplica(MovimientoS151 m, EntradaS151 in) {
        int cve = in.lineas().get(0).cveTran();          // WS-S151-0101-CVETRAN-I(1)
        if (modo.esLinea()) {
            m = p.porClave(cve).map(r -> r.aplicaA(m)).orElse(m);   // 4159/4160, 4449
            if (cve == 2136 || cve == 2137 || cve == 2138) m = m.withSucprom(in.sucTran());
            if (cve == in.cveAcNominaPortaPg()) m = p.porClave(4449).orElseThrow().aplicaA(m);
        }
        if (p.depositoVentanilla().claves().contains(cve)) {
            int caja = in.perfilPim().esCuentaMaestraOEquivalente()
                    ? p.depositoVentanilla().cajaPerfilPim()        // 94
                    : p.depositoVentanilla().cajaOtroPerfil();      // 79
            m = m.withCajs028(caja).withSucs028(sucursalPorNodo(m.nodori()));
        }
        // 3027, 3047, 1153 + BIN 554492 y 2534: mismas reglas declarativas de p.porClave()
        return m;
    }

    private int sucursalPorNodo(int nodori) {
        return nodori == 10 ? p.sucursalNodo10() : p.sucursalOtroNodo();
    }
}

@Component
public class AgrupadorMovimientos {

    static final int POSICIONES = 5;   // CVETRAN1..CVETRAN5 del registro S151

    private final NaturalezaMovimientos naturaleza;  // ACUM-MOVS-XNAT + CHECA-SUMA-SDOINI
    private final ClavesCorresponsales corresp;      // 20000151-CLAVES-CORRESP
    private final ModoEjecucion modo;

    BloqueS151 agrega(BloqueS151 b, LineaS151 l, ContextoSaldo ctx, Consumer<BloqueS151> envia) {
        boolean descartable = l.importe().signum() == 0
                && !l.esNoContable()            // 88 CVENOCONT-I: 500-699, 889, 895-904, 914, 942-944
                && !l.esClaveAdicional()
                && l.indice() != 1;
        if (modo.esLinea() && descartable) {
            return b;                            // NEXT SENTENCE: no se coloca ni se acumula
        }
        if (modo.esLinea()) {
            naturaleza.acumula(l);
            l = corresp.heredaLeyenda(l, b.primeraLinea());
        }
        naturaleza.proyectaSaldo(l, ctx);        // solo con BDB17R06
        BloqueS151 r;
        if (b.lineas().isEmpty()) {
            r = b.conSaldoIni(ctx.saldoIni()).mas(l);
        } else if (b.lineas().size() < POSICIONES) {
            r = b.mas(l);
        } else {
            envia.accept(b);                     // CARGAMOV1-CALL con el bloque lleno
            r = b.siguiente()                    // SUBCVE + REFS151 -> REFS151-ANT
                 .conSaldoIni(ctx.saldoIni())
                 .mas(l);
        }
        ctx.cierraMovimiento();                  // SALDO-INI = SALDO-FIN
        return r;
    }
}
@Component
@RequiredArgsConstructor
public class NaturalezaMovimientos {

    private final ClaveTransaccionRepository b17;   // S500B17CVESXFUN, set B17AXCLAVE
    private final ContadoresNaturaleza cont;        // WS-NUM-MOVSCGO / MOVSABO / MOVSNC

    /** 20000151-ACUM-MOVS-XNAT (P10-119 Migración Prepagada). */
    void acumula(LineaS151 l) {
        if (cont.storeB17()) return;                 // WS-88-STORE-B17
        b17.findByClave(l.cveTran()).ifPresent(c -> {
            if (c.afectaS052() == 1) {
                cont.cargo();
            } else if (c.afectaS052() == 2 || esAbonoBeneficencia(c)) {
                cont.abono();
            } else {
                cont.noContable();
            }
        });
    }

    private static boolean esAbonoBeneficencia(ClaveB17 c) {   // R3-2011-BENEII
        return c.afectaS052() == 0 && c.tipoMov() == 2 && c.permiteElim() == 1
            && (c.porApli() == 1 || c.porTerminal(1) == 1 || c.porTerminal(2) == 1);
    }

    /** 20000151-CHECA-SUMA-SDOINI. Importe 9(14)V99 COMP, saldo S9(15)V99: escala 2, suma exacta. */
    void proyectaSaldo(LineaS151 l, ContextoSaldo ctx) {
        if (!ctx.mandaSaldoIni()) return;
        b17.findByClave(l.cveTran()).ifPresent(c -> {
            if (c.afectaS052() == 1)      ctx.restaFin(l.importe());   // cargo
            else if (c.afectaS052() == 2) ctx.sumaFin(l.importe());    // abono
        });
    }
}
@Component
@RequiredArgsConstructor
public class RegistraS500Client {

    private final RegistraS500Port s151;            // CALL "CARGAMOV1 IN REGISTRAS500"
    private final ContingenciaS151Repository conting;  // 00000000-GRABA-CONTING-S151
    private final RechazoS151Repository rechazos;      // 00000000-GRABA-RECHAZOS-S151
    private final EstadoS151 estado;                   // WS-88-SIUSA-S151, BASE-ORI, CONTINGENCIA...
    private final MeterRegistry metrics;               // reemplaza los hooks S500L030 (INIL030/FINL030)

    public int envia(BloqueS151 b) {
        int status = 0;
        boolean llamar = (estado.usaS151() && estado.baseOrigen())
                || (estado.enviaTranControl() && !estado.prebatch());
        if (llamar && !estado.enContingencia()) {
            status = Timer.builder("s500.cargamov1").register(metrics)
                          .recordCallable(() -> s151.cargaMov1(b));   // GIVING WS-S151-0101-STATUS
        }
        if (estado.enContingencia()) {
            conting.save(ContingenciaS151.de(b));   // se reenvía cuando S151 regrese
        }
        if (status > 0) {
            rechazos.save(RechazoS151.de(b, status));
        }
        return status;
    }
}

@Component
public class VentanaSiglo {

    private final int pivote;   // s500.fecha.pivote-siglo: 49  (CRONOS 2000)

    public VentanaSiglo(@Value("${s500.fecha.pivote-siglo:49}") int pivote) {
        this.pivote = pivote;
    }

    /** 90000000-FECHA-2000: completa el siglo de una fecha AAMMDD. */
    public Optional<LocalDate> completa(int siglo, String aammdd) {
        if ("000000".equals(aammdd)) {
            return Optional.empty();                     // WKS-AA1-2000 = 0
        }
        int aa = Integer.parseInt(aammdd, 0, 2, 10);
        int s = (siglo == 19 || siglo == 20) ? siglo     // ya trae siglo válido
              : (aa > pivote ? 19 : 20);                 // 50-99 -> 19xx, 00-49 -> 20xx
        return Optional.of(LocalDate.parse(s + aammdd, DateTimeFormatter.BASIC_ISO_DATE));
    }
}
/** WS-COMICOB-DS: B28..B34 son el mismo registro COMICOB en 7 datasets físicos. */
public final class ParticionComisionCobrada {

    private ParticionComisionCobrada() {}

    /** 90000028-B28COMICOB-NO-DS: (AA * 12 + MM) mod 7, con 0 -> 7. */
    public static int de(int aa, int mm) {        // WS-COMICOB-AA 9(02), WS-COMICOB-MM 9(02)
        int ds = Math.floorMod(aa * 12 + mm, 7);
        return ds == 0 ? 7 : ds;
    }
}

@Table(name = "s500_comision_cobrada")       // una tabla con columna de partición

    @EmbeddedId private ComisionCobradaId id;    // num_contrato + cve_acumula (BnnSXCTOCVE)
    @Column(name = "particion") private int particion;   // 1..7, LIST PARTITION en la BD destino
    // ...
}

@Service
@RequiredArgsConstructor
public class ComisionCobradaStore {
    private final ComisionCobradaRepository repo;

    @Transactional
    public ComisionCobrada crea(ComisionCobrada c, int aa, int mm) {
        int ds = ParticionComisionCobrada.de(aa, mm);
        if (ds < 1 || ds > 7) throw new IllegalStateException("Partición COMICOB inválida " + ds);
        c.setParticion(ds);
        return repo.save(c);
    }
}

Riesgos de migración
Librería marcada como ligada aunque CTLVERS falle

En la familia 10000xxx-* (por ejemplo 10000016-L422, líneas 504-521) la bandera 88 de la librería se pone en 1 fuera del IF de error. Si CTLVERS no devuelve título, el programa sigue y las llamadas posteriores usan el TITLE que la librería tenga declarado por omisión, sin aviso adicional.

Ruteo contable compilado en un copybook

Las sucursales 342, 511, 859, 904, 907 y 7532 y las cajas 40, 55, 60, 79, 92 y 94 de CARGAMOV1-CTES (líneas 3388-3466) están compiladas en cada programa que incluye el copybook. Un cambio contable obliga a recompilar todos. Además, el comentario de las claves 4159 y 4160 contradice el valor (350 frente a 342).

Todo el bloque hereda el ruteo de la primera clave

CARGAMOV1-CTES decide sucursal y caja con WS-S151-0101-CVETRAN-I(1). Los otros cuatro movimientos del registro, y los bloques siguientes, se contabilizan con la sucursal elegida para la primera clave de la entrada.

Descarte silencioso de importes en cero

En línea, CARGAMOV1-IMP salta con NEXT SENTENCE los movimientos de importe cero que no son no contables ni adicionales (líneas 4157-4162), sin bitácora. En batch la misma entrada sí se envía, por lo que el comportamiento depende del tipo de programa.

Escrituras DMSII que pueden no ocurrir sin error

Las cascadas de CREATE y STORE de COMICOB (líneas 10670-10753) no tienen rama final: con WS-COMICOB-DS fuera de 1..7 no se escribe nada y el status queda en 0.

Ventana Y2K con pivote fijo

90000000-FECHA-2000 (líneas 5388-5398) asume que los años 50-99 son del siglo XX. Debe sustituirse por fechas con siglo completo en la base destino.
