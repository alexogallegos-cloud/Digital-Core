@ConfigurationProperties("p184")
public record P184Properties(Listado listado) {
    public record Listado(String titulo, int csiEmisor) {}

    /** IMPREP-FECHA es 9(06): MOVE desde 9(8) deja AAMMDD. IMPREP-MIX pasa a id de ejecución. */
    public String titulo(LocalDate fechaProceso, long jobExecutionId) {
        return listado.titulo()
                .replace("{fecha:yyMMdd}", fechaProceso.format(DateTimeFormatter.ofPattern("yyMMdd")))
                .replace("{instancia}", "%07d".formatted(jobExecutionId));
    }
}

/** Fila de grupo (WLI-DETGPO) y fila de cuenta (WLI-DETCTA). */
public sealed interface LineaInventarioCpe permits LineaInventarioCpe.Grupo, LineaInventarioCpe.Cuenta {

    record Grupo(long grupo, String nombre, String ciudad, String codigoPostal, String representante,
                 String sucursalPago, String cuentaPago,       // B37-MDA-PAGO: 9(04) + 9(08)
                 String domicilio, int tipoPersona, int opcion, int rango, int csiContratoPago,
                 int sucursalContratoPago, String contratoPago,
                 LocalDate fechaVigencia, LocalDate fechaUltimaAct) implements LineaInventarioCpe {}

    record Cuenta(String sucursal, String cuenta,               // B39-NUM-MDA: 9(04) + 9(08)
                  String nombre, int tipoPersona, int rangoPagado, int csi, int status,
                  int sucursalPromotora, String contrato, int subgrupo) implements LineaInventarioCpe {}

    /** WKS-MDA-PAG REDEFINES: los 12 dígitos se parten en sucursal (4) y cuenta (8). */
    static String[] partirMda(long mda) {
        String s = "%012d".formatted(mda);
        return new String[] { s.substring(0, 4), s.substring(4) };
    }
}

@Configuration
@EnableConfigurationProperties(P184Properties.class)
public class InventarioCpeJobConfig {

    /** 000-CONTROLA + 100-INICIA. */
    @Bean
    Job inventarioCpe(JobRepository repo, Step validaVersion, Step fechaProceso,
                      Step generaInventario, CsiEmisorDecider csi) {
        return new JobBuilder("S500P184", repo)
                .start(validaVersion)                      // 100-10-VALIDA-VERSION
                .next(fechaProceso)                        // 100-30 / 100-50: fechas y CSI de B02
                .next(csi).on("EMITE").to(generaInventario)
                .from(csi).on("*").end()                   // otro CSI: cierra la base y termina sin listado
                .end().build();
    }
}

@Component
@RequiredArgsConstructor
class CsiEmisorDecider implements JobExecutionDecider {
    private final P184Properties props;
    public FlowExecutionStatus decide(JobExecution job, StepExecution step) {
        int csi = job.getExecutionContext().getInt("csi");                 // B02-NUM-CSI
        return csi == props.listado().csiEmisor()
                ? new FlowExecutionStatus("EMITE") : FlowExecutionStatus.COMPLETED;
    }
}
@Component
@RequiredArgsConstructor
public class FechaProcesoTasklet implements Tasklet {

    private static final DateTimeFormatter DD_MMM_AAAA =
            DateTimeFormatter.ofPattern("dd/MMM/yyyy", Locale.forLanguageTag("es-MX"));   // WT-MESES ENE..DIC

    private final ControlCaptacionRepository b02;   // S500B02CONTROL, OPEN INQUIRY
    private final CalendarioBancario calendario;    // reemplaza LOCSUP

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        ControlCaptacion control = b02.find()
                .orElseThrow(() -> new ProcesoAbortadoException("ERR FIND B00CTRLPASO"));  // DMTERMINATE
        // LOCSUP función 15, formato 12, desplazamiento 1 sobre B02-FECHA-LINEA
        LocalDate fechaProceso = calendario.retrocede(control.fechaLinea(), 1)
                .orElseThrow(() -> new ProcesoAbortadoException("ERROR FUNCION 15 LOCSUP"));
        var ec = ctx.getStepContext().getStepExecution().getJobExecution().getExecutionContext();
        ec.put("fechaSaldos", fechaProceso);
        ec.put("fechaSaldosEdit", fechaProceso.format(DD_MMM_AAAA).toUpperCase(Locale.ROOT).replace(".", ""));
        ec.putInt("csi", control.numCsi());                                             // B02-NUM-CSI
        return RepeatStatus.FINISHED;
    }
}
/** 200-REPORTA-GRUPOS: un grupo B37 seguido de sus cuentas B39 con estatus 0 o 1. */
@Component
@StepScope
@RequiredArgsConstructor
class InventarioCpeReader implements ItemReader<List<LineaInventarioCpe>> {

    private static final Set<Integer> ESTATUS_LISTADOS = Set.of(0, 1);
    private final GrupoCpeRepository b37;     // SET SXGPO, orden por grupo
    private final CuentaCpeRepository b39;    // SET SXGPOSBGMDA, por grupo, subgrupo y MDA
    private Iterator<GrupoCpe> grupos;

    @Override
    public List<LineaInventarioCpe> read() {
        if (grupos == null) grupos = b37.findAllByOrderByGrupo().iterator();   // 90000037-B37SXGPO-FINDN
        if (!grupos.hasNext()) return null;                                    // FIN-B37-GRUPOS
        GrupoCpe g = grupos.next();
        List<LineaInventarioCpe> lineas = new ArrayList<>();
        lineas.add(Lineas.deGrupo(g));                                         // GENERATE WLI-DETGPO
        b39.findByGrupoOrderBySubgrupoAscNumMdaAsc(g.grupo()).stream()        // FINDFG + FINDN mientras B39-GRUPO = WS-GRUPO
           .filter(c -> ESTATUS_LISTADOS.contains(c.status()))
           .map(Lineas::deCuenta)                                              // GENERATE WLI-DETCTA
           .forEach(lineas::add);
        return lineas;
    }
}

Riesgos de migración
Datos personales en un listado impreso

El reporte imprime nombre, domicilio y representante de los corporativos y nombre de los titulares de cuenta (líneas 194-202 y 223). La versión destino debe definir control de acceso y enmascaramiento del listado.

Contadores y banderas sin inicializar

WS-TOTGPO, WS-TOTCTA y WB-FIN-B37-GRUPOS no tienen VALUE (líneas 39-45) ni se ponen en cero; los totales y el ciclo dependen del estado inicial de la memoria.

Grupo fantasma con B37 vacío

100-INICIA posiciona el primer grupo (línea 264) sin revisar NOTFOUND; si no hay grupos, el ciclo imprime una ficha con el contenido residual del área del registro.

Traslape de columnas en la ficha del grupo

COLUMN 086 PIC 9(04) VALUE "-" (línea 211) se encima con el contrato de pago en la columna 87.

Hardcodes

CSI emisor 10, nodo 10, versión 25MTP004, clave 3084, PYCP JARDINES, C.S.I. MEXICO, MONEDA NACIONAL y TIPO PROCESO MENSUAL están compilados. Se trasladan a configuración. No hay credenciales en el fuente.

Pendiente de revisión SME
  • Los párrafos 90000037-B37SXGPO-FINDN y 90000039-B39SXGPOSBGMDA-FINDFG/FINDN vienen del copybook S500/INC/PRO/25MTP002, que no está en el fuente; se asumió que FINDFG posiciona la primera cuenta del grupo WS-GRUPO. Confirmar.
  • Confirmar si la fecha que proyecta LOCSUP función 15 formato 12 es el día hábil anterior o el día natural anterior a B02-FECHA-LINEA.
  • El encabezado dice TIPO PROCESO: MENSUAL, pero el programa no contiene ninguna validación de calendario; la periodicidad depende del WFL que lo invoca.
  • El inventario no asigna reglas del brain a este componente; el pseudocódigo cita RN-S500-545 para el formato, pero no se puede usar aquí.
