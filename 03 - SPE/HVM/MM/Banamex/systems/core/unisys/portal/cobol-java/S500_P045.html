S500 · P045 a Java 21 Spring Boot

P045 se habilita en COMS y queda en un ciclo que recibe mensajes de terminal y de aplicación hasta que COMS le ordena terminar. Para un donativo de sucursal desarma la pantalla, valida que la sucursal sea de captación de valores, el dígito verificador y el BIN de la tarjeta, la vigencia y el importe, pide la clave de seguridad S041 según un umbral de supervisor y, si el BIN, el carácter internacional o el monto lo exigen, solicita autorización a B24 antes de aplicar. El abono se registra en S500BD06TELETON, en el log de recuperación I02-LOG y en el mayor de S151 mediante L002. Una consola por interrupción (HI) permite al operador activar el monitor, apagar la seguridad, deshabilitar B24 o cambiar el nodo del S036 sin detener el programa.

/**
 * INTERRUPT-BY-HI: el operador escribe un número en TASKVALUE y el programa lo interpreta.
 * En Java es un endpoint de Actuator protegido por rol; cada comando queda auditado.
 */
@Component
@Endpoint(id = "teleton-consola")
@RequiredArgsConstructor
@Slf4j
public class ConsolaOperativaEndpoint {

    private final EstadoLinea estado;            // W77-ACTIVO-MONITOR, W77-SW-SEGURIDAD, W77-SIN-AUTB24, W77-HI-NODOS
    private final RevocacionesL046 l046;
    private final LineaComs linea;

    @WriteOperation
    public String comando(int valor) {           // W77-MY-VALUE PIC 9(06)
        int tipo = valor / 10_000;               // W77-TPO-HI: dos primeros dígitos
        if (tipo == 2) {
            int sucursal = valor % 10_000;       // W77-SUC-HI; 9999 = todas
            return estado.activaMonitor(sucursal)
                    ? "ACTIVO MONITOR PARA SUCURSAL %04d".formatted(sucursal)
                    : "MONITOR YA ESTA ACTIVADO";
        }
        return switch (valor) {
            case 3    -> estado.desactivaMonitor() ? "DESACTIVO MONITOREO DE MENSAJES" : "MONITOR NO ACTIVADO";
            case 4    -> linea.deshabilita();                                  // 20000000-DISABLE-PGM
            case 10   -> { estado.seguridad(false); yield "DESACTIVO SEGURIDAD"; }
            case 11   -> { estado.seguridad(true);  yield "ACTIVO SEGURIDAD"; }
            case 36   -> estado.alternaB24() ? "HABILITO BASE/24." : "DESHABILITO BASE/24.";
            case 1212 -> { l046.olvidaTodo(); yield "RECIBO HI PARA S500/L046/OLVIDATODO"; }
            case 1317 -> { estado.nodoS036(0);  yield "ENVIA MSG DEL S036 A NODO ALTERNOS"; }
            case 1717 -> { estado.nodoS036(17); yield "ENVIA MSG DEL S036 A NODO 17"; }
            case 1313 -> { estado.nodoS036(13); yield "ENVIA MSG DEL S036 A NODO 13"; }
            default   -> "HI INVALIDO %06d".formatted(valor);
        };
    }
}

@Configuration
@RequiredArgsConstructor
public class TeletonIntegrationConfig {

    /**
     * 10000000-CONTROL-PGM: PERFORM 40000000-RECIBE-MENSAJES UNTIL fin de programa.
     * El ciclo de RECEIVE COMS-IN lo hace el adaptador de entrada; cada mensaje
     * se enruta por origen (terminal o aplicación) al flujo correspondiente.
     */
    @Bean
    IntegrationFlow donativosEntrada(MessageProducerSupport comsAdapter,   // reemplaza COMS-IN KEY "ONLINE"
                                     JournalRecuperacion journal) {         // I02-LOG
        return IntegrationFlow.from(comsAdapter)
                .handle(journal, "registra")                                 // header y texto al log antes de procesar
                .<MensajeComs, Origen>route(MensajeComs::origen, r -> r
                        .subFlowMapping(Origen.TERMINAL,   sf -> sf.channel("terminalChannel"))
                        .subFlowMapping(Origen.APLICACION, sf -> sf.channel("aplicacionChannel")))
                .get();
    }

    /** STOP RUN tras DISABLE: el contexto se detiene de forma ordenada y espera los mensajes en curso. */
    @Bean
    SmartLifecycle bajaOrdenada(LineaComs linea) {
        return new LineaLifecycle(linea);          // 20000000-DISABLE-PGM + 20001000-ESPERA-DISABLE
    }
}
@Component
@RequiredArgsConstructor
public class RuteoAplicacion {

    static final int CLAVE_CARGO_TARJETA = 1500;

    private final DatosSucursalS080 s080;
    private final RespuestaB24Handler respuestaB24;       // 60020000-RCV-MSG-S036
    private final DonativoAplicacionHandler donativo;      // 60050800-ABOTARJ-DIALOGO
    private final RechazoAplicacion rechazo;               // 60050700-ARMAMSG-ENVIA

    @ServiceActivator(inputChannel = "aplicacionChannel")
    public void deAplicacion(MensajeComs msg) {
        int error = s080.datosSucursal(msg.sucursal()).ok() ? 0 : 26;
        int clave = msg.claveTransaccion();

        if (error == 0 && msg.vinoDeAplicacion()) {
            if (msg.vinoDeTercerNivel()) {
                error = (clave == CLAVE_CARGO_TARJETA) ? 0 : 2;    // 60050000-VAL-3ERNIVEL
            } else if (msg.esInternet() && clave != CLAVE_CARGO_TARJETA) {
                error = 5;
            }
        }
        if (error == 0) {
            if (msg.esS036())                                 respuestaB24.procesa(msg);
            else if (msg.claveTransaccion() == CLAVE_CARGO_TARJETA) donativo.procesa(msg, msg.operador());
            else                                              error = 56;  // MOVIMIENTO DE APL NO IDENTIFICADO
        }
        if (error > 0) rechazo.envia(msg, error);
    }
}

public class TarjetaValidator {

    private final BinRepository bines;            // S500B04TBINES
    private final TablaBinesMemoria tablaBines;   // WKS-BIN-NUM (hasta 1000 renglones)

    /** 70063000: pesos 2,1,2,1... de izquierda a derecha sobre 16 dígitos con ceros a la izquierda. */
    static boolean luhnValido(String tarjeta16) {
        int suma = 0;
        for (int i = 0; i < 15; i++) {
            int p = (tarjeta16.charAt(i) - '0') * ((i % 2 == 0) ? 2 : 1);
            suma += p / 10 + p % 10;                   // WKS-DG1-PESO + WKS-DG2-PESO
        }
        int digito = (10 - suma % 10) % 10;            // PIC 9 trunca el 10 a 0; aquí se hace explícito
        return digito == tarjeta16.charAt(15) - '0';
    }

    /** 70064000: BIN de 6 dígitos tras los ceros de relleno y clave contable por emisor. */
    ResultadoBin claveDonativo(String tarjeta16) {
        String sinRelleno = tarjeta16.replaceFirst("^0{0,3}", "");
        int bin = Integer.parseInt(sinRelleno.substring(0, 6));
        if (Set.of(34, 35, 37).contains(bin / 10_000)) {
            return ResultadoBin.amex(796, 97);            // W88-BIN-02-AMEX
        }
        return tablaBines.busca(bin)
                .or(() -> bines.findById(bin).map(b -> b.validoTeleton() ? b : null))
                .map(b -> ResultadoBin.nacional(b.emisor() == 2 ? 793 : 794, b.emisor(), b.negocio()))
                .orElseGet(() -> bines.existsById(bin)
                        ? ResultadoBin.error(29)              // BIN no válido para TELETON
                        : ResultadoBin.internacional(795));   // NOTFOUND: tarjeta internacional
    }
}

Riesgos de migración
CVV almacenado

El CVV tecleado se copia a W77-CVV-RECIBIDO (líneas 3203 y 4444) y se graba en B02T-CVV (línea 3355), aunque toda su validación está comentada. Es un incumplimiento directo de PCI DSS que aplica también a los respaldos de S500BD06TELETON e I02-LOG.

Número de tarjeta en claro

El PAN se guarda completo en B02T-CUENTA-TARJ y se imprime en el mensaje de error de la línea 4930. La migración debe tokenizarlo y depurar el histórico.

Seguridad apagable sin control

Los HI 10 y 36 desactivan la seguridad de operador S041 y la autorización B24 en caliente, y el arranque en DESA (líneas 2567-2569) lo hace por nombre de host. No hay registro de quién lo pidió.

Autorización B24 condicional

Solo se consulta B24 para el BIN 602089, tarjetas internacionales e importes iguales o mayores que B00T-IMP-AUTORI (líneas 3282-3298). El resto de los cargos se aplica sin autorización del emisor.

Prefijo 35 tratado como American Express

W88-BIN-02-AMEX (línea 1252) incluye 35, que es rango JCB. Esas tarjetas reciben la clave 796 y el estatus 15 de American Express.

Hardcodes de negocio

Sucursales 0519, 4899, 1037 y 1905; BIN 602089; contrato, cuenta y centro de costo de la fundación (líneas 234-237); nodos B24 17 y 13. Se trasladan a application.yml.
