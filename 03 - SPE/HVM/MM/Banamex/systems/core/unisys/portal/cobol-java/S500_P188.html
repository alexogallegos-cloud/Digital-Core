Recibe en el TASKSTRING sistema-cliente, nodo, secuencia, fecha y registro. Si el sistema-cliente es 999999999999 lo invocó P187 y lee el archivo de control (S500)S500/FILE/P187/nn/IMPDET/aaaammdd/ssss; si no, toma tipo de reporte y buzón de S500B07PARCHIVOS. Con el indicador de reporte (1 todo, 2 aplicados, 3 rechazados, 4 local) filtra los registros de detalle del archivo de salida, los ordena por sucursal y caja con SORT y genera con REPORT WRITER un listado de 40 líneas por página con cortes y totales por sucursal y caja. Al final cierra el listado, ejecuta SYSTEM/FORZA sobre S500/LIST/P187 y, cuando lo invocó P187, purga el archivo de control y copia el archivo de salida a APLXDISP.

/** WKS-WFL-PARAM: 30 posiciones en ATTRIBUTE TASKSTRING OF MYSELF. */
public record PedidoDetalle(String sistemaCliente,   // 9(12): "999999999999" = invocado por P187
                            int nodo,                // 9(02)
                            int secuencia,           // 9(04)
                            LocalDate fechaProceso,  // 9(08) AAAAMMDD
                            int registro) {          // 9(04)

    public static final String INVOCADO_POR_P187 = "999999999999";

    static PedidoDetalle parse(String taskString) {
        var s = String.format("%-30s", taskString);
        return new PedidoDetalle(s.substring(0, 12),
                Integer.parseInt(s.substring(12, 14)),
                Integer.parseInt(s.substring(14, 18)),
                LocalDate.parse(s.substring(18, 26), DateTimeFormatter.BASIC_ISO_DATE),
                Integer.parseInt(s.substring(26, 30)));
    }

    boolean invocadoPorP187() { return INVOCADO_POR_P187.equals(sistemaCliente); }

    JobParameters toJobParameters() {
        return new JobParametersBuilder()
                .addString("sistemaCliente", sistemaCliente)
                .addLong("nodo", (long) nodo)
                .addLocalDate("fechaProceso", fechaProceso)
                .addLong("registro", (long) registro)
                .toJobParameters();
    }
}
public enum TipoReporte {
    TODO(1), APLICADOS(2), RECHAZADOS(3), TODO_LOCAL(4);
    final int codigo;
    TipoReporte(int c) { codigo = c; }

    static TipoReporte of(int c) {
        return Arrays.stream(values()).filter(t -> t.codigo == c).findFirst().orElseGet(() -> {
            log.warn("INDICADOR NO ESPECIFICA TIPO DE REPORTE DE DETALLE: {}; SE ASIGNA TODO", c);
            return TODO;
        });
    }
}

@Component
@RequiredArgsConstructor
public class OrigenPedidoTasklet implements Tasklet {

    private final ArchivoControlP187Repository controlP187;   // (S500)S500/FILE/P187/nn/IMPDET/...
    private final CatalogoArchivosRepository b07;              // S500B07PARCHIVOS

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        var p = pedido(ctx);
        ControlDetalle control = p.invocadoPorP187()
                ? controlP187.busca(p.nodo(), p.fechaProceso(), p.registro())
                      .orElseThrow(() -> new ProcesoAbortadoException(
                          "ERROR: NO EXISTE ARCHIVO DE CONTROL PARA SIST/CTE: " + p.sistemaCliente()))
                : ControlDetalle.desdeCatalogo(b07.busca(p.sistemaCliente(), p.registro(), p.nodo()));
        publica(ctx, control, TipoReporte.of(control.indicador()));
        return RepeatStatus.FINISHED;
    }
}

@Component
@StepScope
public class DetalleReporteProcessor implements ItemProcessor<RegistroSalida, DetalleSalida> {

    @Value("#{jobExecutionContext['tipoReporte']}") private TipoReporte tipo;
    private final CifrasGlobales cifras = new CifrasGlobales();   // WKS-R1-TOTPROC/TOTOK/TOTRECH/TOTFORAN

    @Override
    public DetalleSalida process(RegistroSalida r) {
        if (!(r instanceof DetalleSalida d)) {
            return null;                       // solo A01-S-DET-TREG = 2; header y trailer fuera
        }
        cifras.cuenta(d.status());             // se cuenta todo detalle, se imprima o no
        boolean imprime = switch (tipo) {
            case TODO       -> true;
            case APLICADOS  -> d.status() == 1;
            case RECHAZADOS -> d.status() == 2;
            case TODO_LOCAL -> d.status() == 1 || d.status() == 2;   // excluye foráneos (3)
        };
        return imprime ? d : null;             // RELEASE REG-A01-S-SAL
    }
}
@Bean
@StepScope
JdbcCursorItemReader<RegistroSalida> salidaOrdenada(DataSource ds,
        @Value("#{jobExecutionContext['archivoSalida']}") String archivo) {
    // SORT A01-SALSORT ON ASCENDING KEY TREG, SUCORIG, CAJORIG: el orden lo resuelve la base.
    // Ya no se calculan MEMORY SIZE ni DISK SIZE (70000000-CALCULA-ESPACIO).
    return new JdbcCursorItemReaderBuilder<RegistroSalida>()
            .name("salidaOrdenada")
            .dataSource(ds)
            .sql("""
                 SELECT * FROM salida_aplicacion
                  WHERE archivo = ?
                  ORDER BY tipo_registro, sucursal_origen, caja_origen, secuencia
                 """)
            .preparedStatementSetter(ps -> ps.setString(1, archivo))
            .rowMapper(new RegistroSalidaRowMapper())
            .build();
}
/** 50000100-VAL-DETALLE: deja cada código en 1..3 o en 4 = no reconocido, para indexar las leyendas. */
static DetalleImpreso normaliza(DetalleSalida d) {
    int modo = d.modoAcceso();
    if (modo == 0 || modo == 1 || modo == 2) modo++;          // 0..2 se desplaza a 1..3
    if (modo < 1 || modo > 3) modo = 4;                       // ver riesgo: un 3 original queda en 3
    int tipoMov = (d.tipoMov() < 1 || d.tipoMov() > 3) ? 4 : d.tipoMov();
    int status  = (d.status()  < 1 || d.status()  > 3) ? 4 : d.status();
    return new DetalleImpreso(d, modo, tipoMov, status);
}

@Component
@StepScope
class TotalesSucursalCaja {
    private SucursalCaja actual;
    private final Map<Integer, Long> porStatus = new HashMap<>();
    private long registros;

    void acumula(DetalleImpreso d, ReporteDetalle reporte) {
        var sc = new SucursalCaja(d.sucursal(), d.caja());
        if (actual != null && !sc.equals(actual)) {
            reporte.totalSucursalCaja(actual, registros, porStatus);   // GENERATE R04-SUC-CAJ-TOTAL
            reporte.saltoDePagina();                                   // ADD 41 TO LINE-COUNTER
            registros = 0; porStatus.clear();
        }
        actual = sc;
        registros++;
        porStatus.merge(d.status(), 1L, Long::sum);   // 1 aplicados, 2 rechazados, 3 foráneos
    }
}

@Component
@RequiredArgsConstructor
public class CierreDetalleTasklet implements Tasklet {

    private final ImpresionService impresion;               // RUN SYSTEM/FORZA("S500/LIST/P187")
    private final ArchivoControlP187Repository controlP187;
    private final ApplicationEventPublisher eventos;

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        var p = pedido(ctx);
        if (reporteGenerado(ctx)) {                         // SW-GENERA-REP = 1
            impresion.liberaListados("S500/LIST/P187");      // antes: CLOSE WITH SAVE + WAIT(60) + WFL
            if (p.invocadoPorP187()) {
                controlP187.elimina(p.nodo(), p.fechaProceso(), p.registro());   // CLOSE ... WITH PURGE
            }
        }
        // TASKSTRING := "P187" + WAIT(60): aviso explícito en lugar de esperar a que lo lean
        eventos.publishEvent(new DetalleTerminado(p, "P187"));
        return RepeatStatus.FINISHED;
    }
}

Riesgos de migración
Terminación normal sin reporte

Sin archivo de control de P187, el programa escribe un mensaje y termina con STOP RUN (líneas 809-815 y 841-842); el WFL no ve error.

Señalización por esperas fijas

La coordinación con P187 depende de dos WAIT (60) (líneas 1053 y 1061), de un TASKSTRING y de la purga del archivo de control, sin confirmación de que P187 la haya leído.

Mapeo de modo de acceso

50000100-VAL-DETALLE (líneas 1180-1184) suma 1 a los valores 0 a 2 y deja el 3 sin cambio, por lo que 2 y 3 se imprimen igual.

Cifras de control inconsistentes

TOTNOPROCE se calcula con los totales del archivo de control (línea 1045) antes de que el SORT recuente TOTPROC (líneas 1267-1269 y 1297).

Hardcodes

Plantilla de nombre con fecha fija 20131009 y buzón 50099E01 (líneas 905-906 y 995), nodo 10 o 04 por hostname, pack CAPTACION, prefijo APLXDISP, límites de memoria del SORT y página de 40 líneas están compilados.
