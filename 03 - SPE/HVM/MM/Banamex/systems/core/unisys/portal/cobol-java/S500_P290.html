                               Flow planchado, Flow reactivacion) {
        return new JobBuilder("S500P290", repo)
            .start(contexto)                       // fecha, hora, hostname, LOCSUP día de la semana
            .next(ventana).on("PROCESA").to(planchado)
            .from(ventana).on("*").to(reactivacion)
            .from(planchado).on("*").to(reactivacion)   // 90000050-PROCESO-REACTIVA corre siempre
            .end()
            .build();
    }
}

@Component
public class ContextoTasklet implements Tasklet {
    private final Clock reloj;                         // inyectable para pruebas
    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        // ACCEPT ... FROM DATE trae AAMMDD; el fuente suma 2000 al año (siglo fijo).
        LocalDateTime ahora = LocalDateTime.now(reloj);
        var ec = ctx.getStepContext().getStepExecution().getJobExecution().getExecutionContext();
        ec.put("fechaDia", ahora.toLocalDate());
        ec.put("hora", ahora.toLocalTime());
        ec.put("diaSemana", ahora.getDayOfWeek());   // reemplaza LOCSUP función 6
        return RepeatStatus.FINISHED;
    }
}
/** WKS-TASKVALUE 9(03) redefinido en tres banderas: actualizar, opción y generación. */
public record OpcionesTarea(boolean actualizaBase,      // dígito 1: 0 = sí actualiza, 1 = simula
                            boolean forzarPlanchado,    // dígito 2: 1 = forzar
                            boolean forzarGeneracion) { // dígito 3: 9 = forzar generación

    public static OpcionesTarea de(JobParameters p) {
        return new OpcionesTarea(
                !p.getString("simular", "N").equals("S"),
                p.getString("forzarPlanchado", "N").equals("S"),
                p.getString("forzarGeneracion", "N").equals("S"));
    }

    /** Compatibilidad con el TASKVALUE numérico mientras el scheduler siga enviándolo. */
    public static OpcionesTarea deTaskValue(int v) {
        int act = v / 100, opc = (v / 10) % 10, gen = v % 10;
        boolean valido = (act == 0 || act == 1) && (opc == 0 || opc == 1) && (gen == 0 || gen == 9);
        if (v <= 0 || !valido) return new OpcionesTarea(true, false, false);   // "CORRO SIN TASKVALUE"
        return new OpcionesTarea(act == 0, opc == 1, gen == 9);
    }
}
@Component
@RequiredArgsConstructor
public class VentanaPlanchadoDecider implements JobExecutionDecider {

    private final P290Properties props;

    @Override
    public FlowExecutionStatus decide(JobExecution job, StepExecution step) {
        var ec = job.getExecutionContext();
        LocalDate fecha = (LocalDate) ec.get("fechaDia");
        LocalTime hora = (LocalTime) ec.get("hora");
        boolean forzar = OpcionesTarea.de(job.getJobParameters()).forzarPlanchado();

        boolean domingo = fecha.getDayOfWeek() == DayOfWeek.SUNDAY;
        boolean semana = fecha.getDayOfMonth() > props.ventana().diaDesde()      // > 3
                      && fecha.getDayOfMonth() < props.ventana().diaHasta();     // < 11
        boolean horario = hora.isAfter(props.ventana().horaDesde());             // > 15:59:59

        String motivo;
        boolean procesa;
        if (domingo && semana && horario) { procesa = true;   motivo = "INICIO PROCESO"; }
        else if (forzar)                  { procesa = true;   motivo = "INICIO PROCESO FORZADO"; }
        else                              { procesa = false;  motivo = "NO PROCESO EN ESTA FECHA"; }
        log.info("{} {} {}", motivo, fecha, hora);               // 61000100-ESC-R001 + mensaje
        return new FlowExecutionStatus(procesa ? "PROCESA" : "OMITE");
    }
}

@Component
@RequiredArgsConstructor
public class CatalogoEsquemaComision {

    private final InstrumentoRepository b05;      // S500B05INSTRUMEN
    private final P290Properties props;
    private Map<LlaveProducto, EsquemaProducto> tabla;

    @PostConstruct
    void carga() {
        var m = new LinkedHashMap<LlaveProducto, EsquemaProducto>();
        for (Instrumento i : b05.findAll()) {
            if (m.size() == props.maxProductos())                      // 30
                throw new ProcesoAbortadoException("B05 CON MAS DE 30 PRODUCTOS");
            var llave = new LlaveProducto(i.producto(), i.instrumento(), i.moneda());
            int ind = i.esquemaComision();                            // B05-ESQ-COMIS
            if (props.calificacionCero().contains(llave)) {
                ind = 0;                                              // 0066-0011-001, 0066-0017-001
            }
            if ((i.esquemaComision() == 1 || i.esquemaComision() == 2)
                    && (i.producto() == 1 || props.calificacionTres().contains(llave))) {
                ind = 3;                                              // 0066-0006/0007/0016/0026/0027/0028-001
            }
            m.put(llave, new EsquemaProducto(i.nombre(), ind, i.esquemaComision()));
        }
        tabla = Map.copyOf(m);
    }

    public int indicador(LlaveProducto llave) {
        return Optional.ofNullable(tabla.get(llave)).map(EsquemaProducto::indicador).orElse(0);
    }
}

/** WKS-TIPO-ACT: resultado de clasificar S016 contra B03. Las 88 dicen de dónde sale cada campo. */
public enum TipoActualizacion {
    T01(NO, NO),        T02(NO, S016),     T03(S016, NO),     T04(S016, S016),   T05(NO, S016),
    T06(S016, S016),    T07(S016, NO),     T08(S016, S016),   T09(S016, S016),   T10(NO, DEF),
    T11(S016, DEF),     T12(NO, DEF),      T13(S016, DEF),    T14(S016, DEF),    T15(S016, DEF),
    T16(DEF, NO),       T17(DEF, S016),    T18(DEF, S016),    T19(DEF, NO),
    T20(NINGUNA, S016), // 20 no está en ninguna 88 de esquema: el esquema no se toca
    T21(DEF, S016),     T22(DEF, DEF),     T23(DEF, DEF),     T24(DEF, DEF),     T25(DEF, DEF);

    public enum Fuente { NO, S016, DEF, NINGUNA }
    public final Fuente esquema;          // W88-USA-S016-ESQ / W88-USA-DEF-ESQ / W88-NO-ACT-ESQ
    public final Fuente mediosAcceso;     // W88-USA-S016-MAA / W88-USA-DEF-MAA / W88-NO-ACT-MAA

    TipoActualizacion(Fuente esq, Fuente maa) { this.esquema = esq; this.mediosAcceso = maa; }

    public int codigo() { return ordinal() + 1; }
}
@Component
public class ClasificadorPlanchado {

    private static boolean esqValido(int v) { return v == 1 || v == 2; }
    private static boolean mdaValido(int v) { return v >= 1 && v <= 3; }

    /** 55000000-VAL-OR completo: 25 hojas. Se muestran las que corresponden a las líneas citadas. */
    public TipoActualizacion clasifica(EsquemaS016 s016, ContratoB03 b03) {
        boolean mismoEsq = s016.opcionEsquema() == b03.esquemaTarifario()
                        && s016.resumenMcc() == b03.resumenMcc();
        boolean mismaMda = s016.rangoMedios() == b03.mediosAsignados();

        if (esqValido(s016.opcionEsquema()) && mdaValido(s016.rangoMedios())
                && esqValido(b03.esquemaTarifario()) && mdaValido(b03.mediosAsignados())) {
            if (mismoEsq) return mismaMda ? T01 : T02;     // 1: nada; 2: sólo medios
            return mismaMda ? T03 : T04;                   // 3: sólo esquema; 4: ambos
        }
        return clasificaResto(s016, b03, mismoEsq, mismaMda);   // tipos 5 a 25
    }
}
@Component
@RequiredArgsConstructor
public class PlanchadoProcessor implements ItemProcessor<ContratoB03, ContratoB03> {

    private final ClasificadorPlanchado clasificador;
    private final EsquemaS016Client s016;                 // L422, opción 3
    private final P290Properties props;

    @Override
    public ContratoB03 process(ContratoB03 b03) {
        EsquemaS016 e = s016.consulta(b03.producto(), b03.instrumento(), b03.contrato());
        TipoActualizacion t = clasificador.clasifica(e, b03);
        boolean cambia = false;

        switch (t.esquema) {
            case S016 -> cambia |= b03.cambiaEsquema(e.opcionEsquema(), e.resumenMcc());
            case DEF  -> cambia |= b03.cambiaEsquema(props.defecto().esquema(),       // 1
                                                     props.defecto().resumenMcc());   // 0
            default   -> { }
        }
        switch (t.mediosAcceso) {
            case S016 -> cambia |= b03.cambiaMedios(e.rangoMedios());
            case DEF  -> cambia |= b03.cambiaMedios(props.defecto().mediosAcceso());  // 2
            default   -> { }
        }
        // WS-BAN-B03 = 1 sólo si algún valor cambió; con W88-TASK-NO-ACT se cuenta pero no se graba.
        return cambia && props.actualizaBase() ? b03 : null;
    }
}

// generaArchivos usa un ClassifierCompositeItemWriter que enruta cada pendiente al archivo
// del sistema destino y acumula los totales por producto, sistema y global para los trailers.
@Component
public class GeneracionDominicalDecider implements JobExecutionDecider {

    @Override
    public FlowExecutionStatus decide(JobExecution job, StepExecution step) {
        LocalDate fecha = (LocalDate) job.getExecutionContext().get("fechaDia");
        boolean forzar = OpcionesTarea.de(job.getJobParameters()).forzarGeneracion();   // W88-TASK-FGEN
        boolean domingo = fecha.getDayOfWeek() == DayOfWeek.SUNDAY;
        if (!domingo) {
            log.info("NO ES DOMINGO, {} {}", fecha,
                     forzar ? "PROCESO FORZADO GEN-ARCHIVOS" : "NO PROCESO GEN-ARCHIVOS");
        }
        // W77-NO-PROCESO-GEN = 0 significa SÍ generar; aquí se usa un nombre positivo.
        boolean genera = domingo || forzar;
        return new FlowExecutionStatus(genera ? "GENERA" : "OMITE");
    }
