/** Registros del archivo regulatorio IPAB. Los tres miden 206 posiciones (RECORD CONTAINS 206). */
public sealed interface RegistroIpab permits RegistroIpab.Header, RegistroIpab.Detalle, RegistroIpab.Trailer {

    String toFixed();

    /** Tipo 01. WR00-IPAB-SISO "S500" + fecha de proceso AAAAMMDD + CSI. */
    record Header(String sistemaOrigen, LocalDate fechaProceso, int csi) implements RegistroIpab {
        public String toFixed() {
            return "01" + sistemaOrigen + fechaProceso.format(BASIC_ISO_DATE)
                    + Fixed.num(csi, 2) + " ".repeat(190);
        }
    }

    /** Tipo 02: un registro por contrato de captación. */
    record Detalle(int producto, int instrumento, int moneda,
                   String contrato, String cliente,                  // 9(12): String conserva ceros
                   BigDecimal rendBruto, BigDecimal rendNeto,        // 9(12)V99
                   BigDecimal impuesto,                              // 9(12)V99 (ISR retenido)
                   BigDecimal promedioCiclo,                         // 9(14)V99
                   int diasCiclo,                                    // 9(02)
                   BigDecimal tasaIsr, BigDecimal tasaBruta, BigDecimal tasaNeta,   // 9(04)V9(03)
                   BigDecimal saldoAnterior, BigDecimal saldoActual, // 9(14)V99, sin signo
                   int tipoPersona, int opcion, int rango,           // OPCRGO 9(04) = opción 99 + rango 99
                   int pagaRend, int rendXDia, int diaCorte, int indCalculo) implements RegistroIpab {
        public String toFixed() { return IpabLayout.detalle(this); }   // FILLER X(43) al final
    }

    /** Tipo 99: número de registros de detalle. */
    record Trailer(long numRegistros) implements RegistroIpab {
        public String toFixed() { return "99" + Fixed.num(numRegistros, 10) + " ".repeat(194); }
    }
}

@Bean
@StepScope
FlatFileItemWriter<RegistroIpab> archIpabWriter(P168Properties props,
        @Value("#{jobExecutionContext['fechaProceso']}") LocalDate fecha,
        @Value("${banamex.csi}") int csi) {
    String ruta = props.archivoIpab()
            .replace("{csi}", "%02d".formatted(csi))
            .replace("{fecha}", fecha.format(BASIC_ISO_DATE));                // 9(08) AAAAMMDD
    return new FlatFileItemWriterBuilder<RegistroIpab>()
            .name("archIpab")
            .resource(new FileSystemResource(ruta))
            .lineAggregator(RegistroIpab::toFixed)
            .headerCallback(w -> w.write(new RegistroIpab.Header("S500", fecha, csi).toFixed()))  // 20000030
            .footerCallback(new TrailerIpabCallback())                     // 50117000-TRL-ARCHIPAB
            .build();
}

@Component

public class FiltroContratoIpab {

    private static final Set<Integer> ESTATUS_PROCESABLES = Set.of(0, 1, 5);  // vigente, bloqueado, precancelado
    @Value("#{jobExecutionContext['fechaProceso']}") private LocalDate fechaProceso;
    private final ContadoresIpab cont;     // W77-TOT-B03-*

    /** 50100022-VAL-SI-PROCESO y 50100001-PROCESA-CTO. */
    boolean procesa(Contrato c) {
        if (c.marcaClon() == 2) return false;          // EXIT PERFORM: ni siquiera se cuenta como leído
        cont.leidos++;
        if (c.migradoS111() == 1)                   { cont.migrados++;      return false; }
        if (c.fechaApertura().isAfter(fechaProceso)) { cont.nuevasApert++;   return false; }
        if (c.marcaClon() == 3)                     { cont.extendidos++;    return false; }
        if (!ESTATUS_PROCESABLES.contains(c.status())) { cont.cancelados++; return false; }
        return true;
    }
}

@Component
@StepScope
@RequiredArgsConstructor
public class ContratoIpabProcessor implements ItemProcessor<Contrato, RegistroIpab.Detalle> {

    private final FiltroContratoIpab filtro;
    private final CapitalizacionService capitalizacion;   // include CAPITALIZA: 20500000-CAPITALIZACION
    private final DatosIpabCalculator calculo;

    /** 50110000-ACTUALIZACION. */
    @Override
    public RegistroIpab.Detalle process(Contrato c) {
        if (!filtro.procesa(c)) return null;
        Instrumento inst = c.instrumento();                    // WS-S500B05INSTRUMEN en memoria
        if (inst.indPagoRend() == 0) {                         // no paga rendimiento: no va al archivo
            return null;
        }
        boolean hoyCorta = c.historico().fechaRealCorte().equals(calculo.fechaProceso());
        if (hoyCorta || inst.indRendDia() == 1) {
            return calculo.desdeDatosAlmacenados(c);           // 50116000-ARMA-DTOS-SAL
        }
        // Día sin corte y capitalización mensual: se simula la capitalización del ciclo
        // sobre una copia del histórico (WS-S500B06HISTORICO) y se arma el registro con ese resultado.
        ResultadoCapitalizacion r = capitalizacion.simula(c, c.historico().copia());
        return calculo.desdeSimulacion(c, r);                  // 50116000-ARMA-REGIPAB
    }
}
/** 50116000-ARMA-DTOS-SAL: datos ya almacenados en B03 y B06. */
RegistroIpab.Detalle desdeDatosAlmacenados(Contrato c) {
    Historico h = c.historico();
    BigDecimal saldoActual = c.saldoAnterior();                         // B03-SDO-ANTERIOR
    BigDecimal rendNeto, impuesto, promCiclo; int diasCiclo;
    if (c.instrumento().indRendDia() == 1) {                            // capitalización diaria
        rendNeto  = c.rendDiaIpab();                                    // B03-RENDIA-IPAB
        impuesto  = c.isrDiaIpab();                                     // B03-ISRDIA-IPAB
        promCiclo = h.saldoPromedioIpab();                              // B06-SDOPROM-IPAB
        diasCiclo = c.diasPeriodo();                                    // WS-DIAS-INHAXCTO (1 si abrió hoy)
    } else {                                                            // capitalización mensual
        rendNeto  = c.interesesCapitalizados();                         // B03-INTS-CAPIT
        impuesto  = c.impuestoRetenido();                               // B03-IMPUESTO-RET
        promCiclo = h.promedioCiclo(12);                                // B06-PROM-CICLO(12)
        diasCiclo = h.diasCiclo();                                      // B06-DIAS-CICLO
    }
    BigDecimal saldoAnterior = saldoActual.subtract(rendNeto);          // saldo base sin el interés
    BigDecimal rendBruto = rendNeto.add(impuesto);                      // bruto = neto + ISR retenido
    Tasas tasas = tasas(c.tasaAnterior(), h.tasaBruta(), rendBruto, rendNeto, promCiclo, diasCiclo);
    return arma(c, rendBruto, rendNeto, impuesto, promCiclo, diasCiclo, tasas, saldoAnterior, saldoActual);
}
private static final BigDecimal BASE_360_POR_100 = new BigDecimal("36000");
private static final MathContext INTERMEDIA = MathContext.DECIMAL128;

/** Tasas del registro: por omisión las del contrato; en cierre mensual con capitalización diaria se recalculan. */
Tasas tasas(BigDecimal tasaNetaContrato, BigDecimal tasaBrutaHistorico,
            BigDecimal rendBruto, BigDecimal rendNeto, BigDecimal promCiclo, int diasCiclo) {
    if (!(esMensual && indRendDia == 1)) {
        return new Tasas(tasaBrutaHistorico, tasaNetaContrato);          // B06-TASA-BRUTA, B03-TASA-ANTERIOR
    }
    if (diasCiclo <= 0 || promCiclo.signum() <= 0) {
        return new Tasas(BigDecimal.ZERO, BigDecimal.ZERO);              // evita división entre cero
    }
    return new Tasas(anualiza(rendBruto, promCiclo, diasCiclo), anualiza(rendNeto, promCiclo, diasCiclo));
}

/** rend / promedio * 36000 / días, COMPUTE sin ROUNDED: se trunca a la escala del destino 9(04)V9(03). */
private static BigDecimal anualiza(BigDecimal rend, BigDecimal prom, int dias) {
    return rend.divide(prom, INTERMEDIA)
               .multiply(BASE_360_POR_100, INTERMEDIA)
               .divide(BigDecimal.valueOf(dias), INTERMEDIA)
               .setScale(3, RoundingMode.DOWN);
}
private final IsrProperties isr;   // tasa general (WS-ISR-0-9991-254) y de extranjeros (WS-ISR-EXTR-0-NVO)

/** 50116000-ARMA-REGIPAB, cálculo de tasa ISR y saldos. */
RegistroIpab.Detalle arma(Contrato c, BigDecimal rendBruto, BigDecimal rendNeto, BigDecimal impuesto,
                          BigDecimal promCiclo, int diasCiclo, Tasas tasas,
                          BigDecimal saldoAnterior, BigDecimal saldoActual, boolean datosAlmacenados) {
    BigDecimal tasaIsr = switch (c.tipoPersona().grupoIsr()) {
        case EXTRANJERO     -> isr.extranjeros().multiply(CIEN).setScale(3, RoundingMode.DOWN);
        case MORAL_NO_LUCRO -> BigDecimal.ZERO.setScale(3);
        default             -> isr.general().setScale(3, RoundingMode.DOWN);
    };
    if (!datosAlmacenados) {                               // rama de simulación
        saldoAnterior = c.saldoAnterior();
        saldoActual = saldoAnterior.add(rendNeto);
    }
    if (saldoAnterior.signum() < 0) {
        saldoAnterior = BigDecimal.ZERO.setScale(2);
        saldoActual = rendNeto;
    }
    // INC0106665384: el saldo protegido se suma al actual y el bloqueado del ciclo previo al anterior.
    saldoActual = saldoActual.add(c.saldoProtegidoAnterior());          // B03-SDOPROT-ANT
    saldoAnterior = saldoAnterior.add(c.historico().saldoBloqAntAnt()); // B06-SDOBLO-ANTANT
    return new RegistroIpab.Detalle(/* ... */ tasaIsr, tasas.bruta(), tasas.neta(),
            Fixed.sinSigno(saldoAnterior), Fixed.sinSigno(saldoActual) /* ... */);
}

Riesgos de migración
Importes sin signo en el archivo regulatorio

Los campos de importe de REG-IPAB-DETALLE (líneas 438-447) son PIC 9 sin signo y reciben variables S9(15)V99. Un valor negativo, por ejemplo un saldo actual negativo después del ajuste del incidente, se reportaría como positivo y además se pierde el dígito más alto.

Truncamiento en tasas

Las tasas anualizadas (líneas 1725-1730) se calculan sin ROUNDED y con la división antes de la multiplicación; el resultado depende de la precisión intermedia y se trunca a tres decimales.

Contratos con clonación 2 fuera de las cifras de control

50100022-VAL-SI-PROCESO sale con EXIT PERFORM antes de contar el contrato como leído (línea 1401). El total de leídos del listado no cuadra contra el total de la base.

Dependencia del include CAPITALIZA

La rama de simulación (20500000-CAPITALIZACION, línea 1630) y las definiciones de WS-CAP-* vienen de un include que no está en el fuente; la escala de esos campos no se pudo confirmar.

Hardcodes

La ruta S500/FILE/INTS/50022S01/, el sistema origen "S500" del header y el factor 36000 están compilados. La ruta y el sistema se trasladan a application.yml; el factor se deja como constante con nombre.
