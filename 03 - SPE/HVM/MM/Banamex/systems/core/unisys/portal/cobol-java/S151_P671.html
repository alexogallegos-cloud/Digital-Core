@Component
@RequiredArgsConstructor
public class IntelarArchivo {

    private final P671Properties props;
    private final Clock reloj;

    /** WKS-ARCH-INTELAR: fecha del parámetro (8 dígitos) y hora de la máquina en el nombre. */
    public Path ruta(int csi, Interfaz interfaz, LocalDate fechaParametro) {
        String hhmmss = LocalTime.now(reloj).format(DateTimeFormatter.ofPattern("HHmmss"));
        return Path.of(props.titulo()
                .replace("{dir}", props.directorio())
                .replace("{csi}", "%02d".formatted(csi))
                .replace("{interfaz}", interfaz.nombreArchivo())     // PROTECCOB, ALERTANOT, DOMICILIA
                .replace("{fecha}", fechaParametro.format(DateTimeFormatter.BASIC_ISO_DATE))
                .replace("{hhmmss}", hhmmss));
    }

    /** Se crea con permisos restringidos: el CSV lleva cliente, contrato, cuenta y CLABE. */
    public BufferedWriter abre(Path p) throws IOException {
        Files.createDirectories(p.getParent());
        Files.createFile(p, PosixFilePermissions.asFileAttribute(
                PosixFilePermissions.fromString(props.permisosArchivo())));
        return Files.newBufferedWriter(p, StandardCharsets.UTF_8);
    }
}

@Service
@RequiredArgsConstructor
public class ExtractorIntelarService {

    private final IntelarArchivo archivo;
    private final ProteccionCobroRepository b07;   // S151B07PROTCOB
    private final DomiciliacionRepository b10;     // S151B10DOMI
    private final HostInfo host;                   // CSIID IN SOPORTECOMS

    /** 000000-INIC-PROGRAMA + 00100-PROCESO. Una sola transacción de lectura (OPEN INQUIRY). */
    @Retryable(retryFor = CannotGetJdbcConnectionException.class,
               maxAttempts = 7, backoff = @Backoff(delay = 10_000))      // W77-INTENTO > 6, WAIT(10)
    @Transactional(readOnly = true)
    public Path genera(ParametroIntelar p) throws IOException {
        var caso = p.caso().orElseThrow(() -> new ParametroInvalidoException("PARAMETRO INVALIDO"));
        Path ruta = archivo.ruta(host.csi(), p.interfaz(), p.fecha());
        try (var w = archivo.abre(ruta)) {                                // OPEN OUTPUT REPORTE
            switch (p.interfaz()) {
                case PROTECCION_COBRO, ALERTAS -> exportaB07(w, p, caso);    // 00121-BUSCAB07
                case DOMICILIACION             -> exportaB10(w, p, caso);    // 00130-BUSCAB10
            }
        }                                                                  // CLOSE REPORTE WITH SAVE
        return ruta;
    }

    @Recover
    Path sinBase(CannotGetJdbcConnectionException e, ParametroIntelar p) {
        throw new ProcesoAbortadoException("ERROR AL ABRIR LA S151BD13BIFIN", e);   // STATUS -1
    }
}
public enum Interfaz {
    PROTECCION_COBRO(1, "PROTECCOB"), ALERTAS(2, "ALERTANOT"), DOMICILIACION(3, "DOMICILIA");

    private final int id; private final String nombreArchivo;
    Interfaz(int id, String nombre) { this.id = id; this.nombreArchivo = nombre; }
    public String nombreArchivo() { return nombreArchivo; }

    static Optional<Interfaz> de(int id) {
        return Arrays.stream(values()).filter(i -> i.id == id).findFirst();
    }
}

/** WKSPARAMETRO: id 9(02), fecha 9(08) (solo se usa AAMMDD), contrato 9(16). */
public record ParametroIntelar(int id, int fecha8, String contrato) {

    public enum Caso { POR_FECHA, POR_FECHA_Y_CONTRATO }

    /** 00110-VAL-PARAM: la fecha se valida y se busca con sus 6 dígitos bajos (AAMMDD). */
    public Optional<Caso> caso() {
        int aammdd = fecha8 % 1_000_000;
        if (aammdd <= 0 || Interfaz.de(id).isEmpty()) return Optional.empty();
        return Optional.of(new BigInteger(contrato).signum() == 0 ? Caso.POR_FECHA : Caso.POR_FECHA_Y_CONTRATO);
    }

    public int fechaBusqueda() { return fecha8 % 1_000_000; }        // B07-FEC-CARGO es AAMMDD
    public Interfaz interfaz()  { return Interfaz.de(id).orElseThrow(); }
    public LocalDate fecha()    { return LocalDate.parse("%08d".formatted(fecha8), DateTimeFormatter.BASIC_ISO_DATE); }
}

/** 00121-BUSCAB07: encabezado y una línea por registro. */
void exportaB07(BufferedWriter w, ParametroIntelar p, ParametroIntelar.Caso caso) throws IOException {
    w.write(ProteccionCobroCsv.ENCABEZADO); w.newLine();
    try (Stream<ProteccionCobro> s = (caso == ParametroIntelar.Caso.POR_FECHA)
            ? b07.findByNumIdAndFecCargoOrderByIndiceB07(p.id(), p.fechaBusqueda())
            : b07.findByNumIdAndFecCargoAndContratoOrderByIndiceB07(p.id(), p.fechaBusqueda(), p.contrato())) {
        for (var r : (Iterable<ProteccionCobro>) s::iterator) {
            w.write(ProteccionCobroCsv.de(r).linea()); w.newLine();   // 00122-GRABA-REGISTRO
        }
    }
}

Riesgos de migración
Datos personales en archivo público

El FD REPORTE declara SECURITYTYPE IS PUBLIC (línea 27) para un CSV con número de cliente, contrato, cuenta, CLABE e importes. Cualquier usuario del sistema puede leerlo.

Domiciliación por contrato incompleta

En la interfaz 3 con contrato, la búsqueda se posiciona por fecha y termina en el primer registro de otro contrato (líneas 413-425 y 466-477). El archivo puede salir vacío o parcial sin aviso.

CSV sin escape

WKS-DET-DAT-GEN y WKS-DET2-DAT-GEN son texto libre de 147 posiciones que se insertan sin comillas. Una coma en ese texto desalinea las columnas para INTELAR.
