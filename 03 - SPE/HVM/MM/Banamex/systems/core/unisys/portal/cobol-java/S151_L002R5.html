S151 · L002R5 a Java 21 Spring Boot

Exporta tres procedimientos: CARGAMOV1, CARGASALDOS y CONTROLES. CARGAMOV1 recibe el movimiento como arreglo EBCDIC, lo copia a ALOG1, lo remapea al formato nuevo con MAPEO cuando el sistema que llama coincide con el de la librería, valida el sistema y ubica el día contable (NUMDIAC1) en la tabla FEC_DIA. Después despacha por IDFFUNCION a una de diez funciones (alta, eliminación, inicio, término, rebloqueo, prefinal y final), lanza P015 y P016 para procesos intradía o P025 para consolidación, y devuelve un código RESULT al llamador. Los movimientos rechazados se escriben en ERRORES1.

    @Override
    public ResultadoRegistro cargaMovimiento(byte[] datos) {
        byte[] alog1 = Arrays.copyOf(datos, datos.length);    // REPLACE POINTER(ALOG1) BY DATOS
        byte[] alog2 = Arrays.copyOf(datos, datos.length);    // copia intacta para ERRORES1
        var mov = MovimientoAlog.de(alog1);

        if (mov.sistemaFan() == props.sistema()) {
            mov = mapeador.aFormatoNuevo(mov);                // MAPEO
        }
        if (!props.sistemasSinNormalizarCaja().contains(props.sistema())) {   // 500, 403, 404
            mov = mov.conCampoDigitos(262, 5, 2).conCampoDigitos(267, 5, 2); // offsets en dígitos de 4 bits
        }

        cambiaFecha.readLock().lock();                         // WHILE LOCKSTATUS(CAMBIA_FECHA) ... WAIT
        try {
            if (mov.sistema() != props.sistema()) {
                avisos.postea("SISTEMA NO HABILITADO EN S151, SISTEMA : %04d".formatted(mov.sistema()));
                return registraError(alog2, ResultadoRegistro.SISTEMA_NO_HABILITADO);   // RESULT := 1
            }
            return procesa(mov, alog2);
        } finally {
            cambiaFecha.readLock().unlock();
        }
    }
    /** Devuelve el índice de día (NUMDIAC1) y si la fecha contable es válida (RESULT 0 o 4). */
    DiaContable ubicaDia(LocalDate fecContable) {
        boolean equipoPruebas = props.equiposPruebas().contains(props.nombreEquipo()); // ACYPBETA, VDMBETA
        if (!equipoPruebas && fecContable.equals(control.fechaProceso())) {
            return DiaContable.valido(control.diaSemana());
        }
        if (equipoPruebas || fecContable.isAfter(control.fechaProceso())) {
            List<LocalDate> fecDia = control.fechasDia();          // FEC_DIA[0..9]
            int idx = fecDia.indexOf(fecContable);
            if (idx >= 0) return DiaContable.valido(idx);
        }
        // Fecha anterior a la de proceso o no encontrada: RESULT := 4, se usa el día de la semana
        avisos.postea("FECHA CONTABLE NO VALIDA, FECHA : " + fecContable.format(BASIC_ISO_DATE));
        return DiaContable.invalido(control.diaSemana());
    }
    ResultadoRegistro procesa(MovimientoAlog mov, byte[] alog2) {
        DiaContable dia = ubicaDia(mov.fechaContable());
        int funcion = mov.funcion();                            // FUNCIONVAL
        // RESULT 4 (fecha no válida) NO detiene el despacho: la función se ejecuta sobre el día de la semana.
        Runnable accion = switch (funcion) {
            case 1      -> () -> cargaMemoria(mov, dia);        // CARGAMEMORY
            case 2      -> () -> elimina(mov, dia);             // ELIMINA
            case 11     -> () -> inicia(mov, dia);
            case 12     -> () -> termina(mov, dia);
            case 21     -> () -> eliminaPorProceso(mov, dia);   // ELIMXPROC
            case 22     -> () -> eliminaPorAutorizacion(mov, dia); // ELIMXAUT
            case 31, 32 -> () -> rebloquea(dia);                // REBLOCKADE
            case 97     -> () -> prefinal(dia);
            case 98     -> () -> fin(dia);
            default     -> null;
        };
        if (accion == null) {
            avisos.postea("FUNCION NO VALIDA, FUNCION : %02d".formatted(funcion));
            return registraError(alog2, ResultadoRegistro.FUNCION_NO_VALIDA);   // RESULT := 2, GRABAMOV := FALSE
        }
        accion.run();
        if (dia.valido()) lanzador.evalua(mov, dia, funcion);   // P015, P016, P025
        return cierra(mov, dia.valido() ? ResultadoRegistro.OK : ResultadoRegistro.FECHA_NO_VALIDA, alog2);
    }

@Component
public class MapeadorFormatoNuevo {

    /** Campo destino: origen en ALOG1 (offset en dígitos de 4 bits) o relleno constante. */
    sealed interface Campo permits Copia, Relleno {}
    record Copia(String nombre, int offset, int largo) implements Campo {}
    record Relleno(String nombre, char valor, int largo) implements Campo {}

    /** Orden y tamaños exactos de REPLACE POINTER(BLOG1,4) BY ... en MAPEO. */
    static final List<Campo> FORMATO_NUEVO = List.of(
        new Copia("FUNCION", -1, 2),                 // IDFFUNCION(ALOG1) FOR 2 DIGITS
        new Relleno("FORMATO", '0', 2),
        new Copia("SIST", 2, 4), new Copia("PROD", 6, 4), new Copia("INS", 10, 2),
        new Relleno("SERVICIO", '0', 2), new Relleno("TIPO_OPER", ' ', 2),
        // ... 90 campos más, generados desde el fuente, no a mano ...
        new Relleno("REFCLIENTE", ' ', 32),          // el formato nuevo los trae vacíos
        new Relleno("REF1_CTE", ' ', 70), new Relleno("REF2_CTE", ' ', 70),
        new Relleno("REF3_CTE", ' ', 70), new Relleno("REF4_CTE", ' ', 70),
        new Copia("LYENDA1", 630, 80), new Copia("LYENDA2", 710, 80), new Copia("LYENDA3", 790, 80),
        new Copia("LYENDA4", 870, 80), new Copia("LYENDA5", 950, 80));

    public MovimientoAlog aFormatoNuevo(MovimientoAlog mov) {
        var destino = new NibbleBuffer(165 * 12);    // 165 palabras de 48 bits = 1980 dígitos
        for (Campo c : FORMATO_NUEVO) {
            switch (c) {
                case Copia k   -> destino.append(k.offset() < 0 ? mov.funcionDigitos() : mov.nibbles(k.offset(), k.largo()));
                case Relleno r -> destino.fill(r.valor(), r.largo());
            }
        }
        return MovimientoAlog.de(destino.toBytes());
    }
}

Riesgos de migración
Movimiento con fecha no válida que sí se graba

El despacho de funciones corre con RESULT = 0 o RESULT = 4 (línea 5806). Un movimiento con fecha contable no válida se graba en el LOG del día de la semana y a la vez se escribe en ERRORES1 y se devuelve con error.

Lógica distinta en equipos de prueba

Los nombres ACYPBETA y VDMBETA (línea 5753) activan una búsqueda de fecha que acepta días pasados. Las pruebas en esos equipos no reproducen la regla de producción.

Referencias del cliente vacías en el formato nuevo

MAPEO rellena con espacios REFCLIENTE y REF1 a REF4 y deja en cero NUM. CTE., REF. ORIG. e IN-COMPANY. FEC ULT MOV se toma del offset 322, el mismo de RENG TOTAL (líneas 5407 y 5410).

Errores de control silenciados

El reporte de error de CONSISDIA en ACTNIVEL (líneas 5546 a 5551), MONITOREA (línea 5726) y L002COMMPOST (líneas 5744 y 5827) están comentados.

Longitud de retorno alterada por MAPEO

SIZEDATC1 es una variable global que MAPEO reasigna con SIZE(BLOG1); el REPLACE DATOS de la línea 5916 usa ese valor para devolver el mensaje al llamador.
