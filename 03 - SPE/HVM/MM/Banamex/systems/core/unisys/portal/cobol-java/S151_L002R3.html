        int result = validaEncabezado(mov);                   // 1 sistema distinto, 16 formato > 3
        if (result != 0) return result;

        // El canal es el que tiene la misma fecha contable que el movimiento.
        OptionalInt canal = canales.porFecha(mov.fechaContable());
        boolean sinCanalPermitido = mov.funcion() == 95 || mov.funcion() == 96;
        if (canal.isEmpty() && !sinCanalPermitido) {
            return 4;                                         // RESULT := 4
        }
        var f = funciones.get(mov.funcion());
        if (f == null) return 2;                              // ELSE: RESULT := 2
        return f.ejecuta(mov, canal.orElse(-1));              // CARGAMEMORY, ELIMINA, INICIA...
    }
}

@Component
@RequiredArgsConstructor
public class CanalDiario {

    private static final int FUNCION_CENTINELA = 99;
    private final RegistroProperties props;
    private final ArchivoMovimientos movs;     // acceso por número de registro (75 palabras)
    private final OperadorConsola operador;    // LJ + ACCEPT

    /** ARMALOG: abre el MOV del día y devuelve el nivel del último registro válido. */
    public long arma(int canal, int fecDia /* AAAAMMDD */) {
        // ARC_MOV + fecha de 6 dígitos (FOR 6 DIGITS toma AAMMDD) + " ON " + PK_MOV + "."
        String titulo = "%s%06d ON %s.".formatted(props.arcMov(), fecDia % 1_000_000, props.pkMov());
        var archivo = movs.abre(canal, titulo);
        if (archivo.isEmpty()) return faltaArchivo(canal, titulo, fecDia);

        var f = archivo.get();
        RegistroLog header = f.lee(0);
        long sup = 0;
        if (header.funcion() != FUNCION_CENTINELA) {
            while (header.fecha() != fecDia) {                 // ERROR1: el header es de otro día
                operador.esperaCorreccion("ERROR1 LIB/REGISTRAS %03d, FECHA HEADER FILE/MOVTOS: %s FECHA CORRECTA: %08d"
                        .formatted(props.sistema(), header.fechaTexto(), fecDia));
                header = f.relee(0);
            }
            sup = f.ultimoRegistro();
        }
        // Búsqueda binaria del primer registro con función 99 (centinela).
        long inf = 0;
        while (inf != sup) {
            long mid = (sup - inf) / 2 + inf;                   // INTEGERT trunca
            if (f.lee(mid).funcion() == FUNCION_CENTINELA) sup = mid; else inf = mid;
            if (sup - inf == 1) sup = inf;
        }
        return sup;                                             // NIVLOG: próximo alta en sup + 1
    }
}
    /** Rama ELSE de ARMALOG cuando el MOV no está residente. */
    private long faltaArchivo(int canal, String titulo, int fecDia) {
        boolean requerido = nivelBase(canal) > 0
                && fecDia >= props.fechaProcesoBase()          // FEC_PRO_BASE
                && funcionActual() != 95;                      // 95 = ambientación
        if (!requerido) {
            return 0;                                          // NIVLOG := 0, el canal queda sin archivo
        }
        operador.esperaCorreccion("ERROR : FALTA ARCHIVO %s, FAVOR DE COPIARLO".formatted(titulo));
        return arma(canal, fecDia);                            // GO VERIFICA
    }

                               .orElse(Evento.TIMEOUT);

                case TIMEOUT -> {

Riesgos de migración
Credencial escrita en el fuente

En el evento 7, para CSI 32, el WFL que se envía con ZIP incluye un ACCESSCODE con su contraseña escrita en claro (líneas 9297-9298). No se reproduce en este portal. En la migración debe ir a un almacén de secretos y la credencial actual debe rotarse.

Contador de bloque lleno sin reinicio

N_BLOCK_LLENO sólo se reinicia cuando hay registros en REG_X_BLOCK_DX al llegar a 20 (líneas 9183-9199). Si no los hay, el contador rebasa 20 y el cierre por bloque lleno deja de ocurrir hasta que la librería se reinicie.

Bloqueos esperando al operador

ARMALOG y ARMADES usan ACCEPT tanto para un header con fecha equivocada (ciclo WHILE de la línea 8175) como para un archivo faltante (GO VERIFICA, línea 8235). Mientras el operador no responde, la librería no registra movimientos de ningún sistema.

GRABALOG no depende de la función

Por la puntuación del ALGOL en CARGAMEMORY (líneas 6516-6522), GRABALOG se ejecuta para las funciones 1, 11, 12, 50 y 52; sólo la validación DFVALIDA está restringida al alta.

Valores de operación compilados

Timeout de 30 segundos, umbral de 20 bloques, hora límite 20:00:00 para P016, espera de 2 segundos, identificador S151LOTE, tipos de proceso Teletón 40 a 43 y rutas de archivos bandera están escritos en el código; se trasladan a application.yml.
