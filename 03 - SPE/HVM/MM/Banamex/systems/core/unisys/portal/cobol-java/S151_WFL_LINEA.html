/** Las 43 cadenas que acepta VALIDAPARAM. */
public enum AccionOperativa {
    INICIO, REINICIO,
    P007, P010, P011, P012, P013, P014, P017, P020, P030, P050, P052, P053, P054, P055, P071, P073,
    BAJAP007, BAJAP010, BAJAP011, BAJAP012, BAJAP013, BAJAP014, BAJAP017, BAJAP020, BAJAP030,
    BAJAP050, BAJAP052, BAJAP053, BAJAP054, BAJAP055, BAJAP071, BAJAP073,
    BAJALINEA, BAJAALL, CARGAFILES, CONTINGENCIA, CORRESOLO,
    INSTL002, INSTL011, INSTL030, INSTL040;

    /** HEAD(TAIL(PARAMETRO," "),NOT " "): primer token sin espacios. */
    public static Optional<AccionOperativa> de(String parametro) {
        String token = parametro == null ? "" : parametro.strip().split("\\s+", 2)[0];
        try {
            return Optional.of(valueOf(token));
        } catch (IllegalArgumentException e) {
            return Optional.empty();
        }
    }
}

@RestController
@RequestMapping("/operacion/s151/linea")
@RequiredArgsConstructor
class ConsolaLineaController {
    private final JobLauncher launcher;
    private final Job lineaS151;

    /** El WHILE con ACCEPT se vuelve un 400: la consola ya no pregunta en bucle. */
    @PostMapping
    ResponseEntity<?> ejecuta(@RequestParam String parametro) throws Exception {
        var accion = AccionOperativa.de(parametro)
                .orElseThrow(() -> new ResponseStatusException(BAD_REQUEST,
                        "Parámetro no permitido; valores válidos: " + Arrays.toString(AccionOperativa.values())));
        var params = new JobParametersBuilder()
                .addString("accion", accion.name())
                .addString("parametro", parametro)      // AUX1 y AUX2 para CORRESOLO / P061
                .addLong("ts", System.currentTimeMillis())
                .toJobParameters();
        return ResponseEntity.accepted().body(launcher.run(lineaS151, params).getId());
    }
}

@Component
@RequiredArgsConstructor
public class NodoResolver {

    private final NodosProperties p;

    /** Reproduce el CASE NODO de las líneas 1297-1320, incluido el caso sin ELSE interno. */
    public int resuelve(int serial, String hostname) {
        if (p.serialesConHost().contains(serial)) {
            return p.porHost().getOrDefault(hostname, serial);   // CASE HNAME sin ELSE
        }
        return p.porSerial().getOrDefault(String.valueOf(serial), 0);   // ELSE: NODO := 0 (inválido)
    }
}

        coms.baja("S151P011"); coms.baja("S151P030"); coms.baja("S151P050");

        coms.baja("S151P071"); coms.baja("S151P073");
        coms.baja("S151P054"); coms.baja("S151P055");
        senal.difunde("BLINE");
        if (!teleton.activa()) {
            coms.baja("S151P052");                   // con Teletón, P052 sigue arriba
        }
    }

@Component

public class PasoCargaArchivos implements Tasklet {      // SUB005 / S151P005

    private final ProgramaRunner runner;                 // RUN #S151P005 [T005]
    private final TituloResolver sacatit;                // SACATIT("S151P005CFIL")
    private final DecisionOperadorPort operador;         // reemplaza ACCEPT(MENSAJE)

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        String programa = sacatit.titulo("S151P005CFIL");
        while (true) {
            try {
                runner.run(programa, Map.of("taskfile",
                        "S151/TASKFILE/P005/" + ctx.getStepContext().getJobParameters().get("fechaYyMMdd")));
                return RepeatStatus.FINISHED;
            } catch (TaskFaultException e) {             // ON TASKFAULT
                var d = operador.decide("FALLA DE PASO 5", e);   // ticket o aprobación en consola
                switch (d.decision()) {
                    case TERMINA -> { return RepeatStatus.FINISHED; }   // GO BACKFORZA005
                    case VERSION_DIFERENTE -> programa = d.nuevoPrograma();
                    case REINTENTO -> { /* GO PASO005 */ }
                }
            }
        }
    }
}

public interface ComsControlAdapter {
    void alta(String programa);   // ENABLE PROGRAM
    void baja(String programa);   // DISABLE PROGRAM

/** Implementación destino: escalado del Deployment del servicio en Kubernetes. */
@Component
@RequiredArgsConstructor
class KubernetesControlAdapter implements ComsControlAdapter {

    private final AppsV1Api apps;
    private final ControlProperties props;   // s151.coms.programas.S15107.deployment: s151-p007

    @Override public void alta(String prog) { escala(prog, props.replicas(prog)); }
    @Override public void baja(String prog) { escala(prog, 0); }

    @Retryable(retryFor = ApiException.class, maxAttempts = Integer.MAX_VALUE,
               backoff = @Backoff(delay = 30_000))      // DO ... UNTIL TASKVALUE = 1
    void escala(String prog, int replicas) throws ApiException {
        var d = props.deployment(prog);
        apps.patchNamespacedDeploymentScale(d.nombre(), d.namespace(),
                new V1Patch("{\"spec\":{\"replicas\":" + replicas + "}}"))
            .execute();
        // WAIT ("DAR OK CUANDO *SYSTEM/COMS ESTE ACTIVO",OK) -> alerta a operación, no bloqueo
    }
}

Riesgos de migración
BAJAP030 baja el programa P050

En el despachador primario (línea 1408) la acción BAJAP030 invoca la subrutina BAJAP050. Pedir la baja de P030 deja P030 arriba y baja P050.

Nodos sin rama de instalación

NODO 12 (Corporativo), NODO 0 y un nodo igual al número de serie (host no mapeado en los seriales 3246 o 3251, líneas 1301-1315) no tienen rama en los CASE de instalación de librerías y solo producen un mensaje.

Variable global TIT compartida

SACATIT devuelve el título en la variable global TIT, que sobrescriben ocho llamadas y los manejadores de falla (línea 243). Con subrutinas lanzadas por PROCESS en paralelo, un paso puede ejecutar el título resuelto por otro.

Esperas sin límite

Los ciclos DO ... UNTIL TASKVALUE = 1 de los 16 pares SUBnnn/BAJAPnnn esperan confirmación del operador sin tope de intentos.

Hardcodes de infraestructura

Números de serie, hostnames, familia CMEMP, horarios del LOTE (06:30, 11:30, 16:30) y catálogos de plazas por librería están en el fuente. Se trasladan a application.yml y al planificador.

Dependencia de un include ausente

El $INCLUDE (S151)S151/WFL/INCLUDE/LINEA ON CMEMP (línea 25) define PACK1, MSGHDR, VALUEFECHA, JOBATRIBUTOS y TASKATRIBUTOS, y no está en el corpus.
