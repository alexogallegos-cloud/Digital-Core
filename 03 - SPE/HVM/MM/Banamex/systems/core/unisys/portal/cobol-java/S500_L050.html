S500L050VALRETDEP recibe en un arreglo EBCDIC la función, el cliente, el importe, el indicador de depósito o retiro, el banco emisor, el indicador de movimiento forzado y el tipo de proceso (línea, batch, prelínea, fin de batch). Busca al cliente en B52 de captación o de la base auxiliar según el tipo de proceso y, si el cliente no trae límites propios, los toma de B55 por tipo de persona, tipo de operación y ubicación geográfica. En un depósito que rebasa el límite diario calcula el saldo congelado; en un retiro que rebasa un límite regresa el resultado 10 y en un cliente bloqueado el 9. La actualización de acumulados no se hace en la llamada: el movimiento se deja en un arreglo circular de 500 posiciones y un proceso asíncrono de la propia librería lo aplica en B52 y B53.

    public RespuestaValRetDep valRetDep(SolicitudValRetDep s) {
        Contexto ctx = new Contexto(s);               // reemplaza LIMPIA_VARIABLES_VALIDA

        if (s.funcion() < 1 || s.funcion() > 4) {
            ctx.resultado(ResultadoL050.FUNCION_INCORRECTA);
        } else if (s.funcion() == 4) {
            cambio.procesa(ctx);
        } else if (s.tipoProceso() == null) {         // fuera de 0..3
            ctx.resultado(ResultadoL050.TIPO_PROCESO_INCORRECTO);
        } else if (s.tipoProceso() == TipoProceso.FIN_BATCH && s.funcion() <= 2) {
            estado.cambiaTipoProceso(s.tipoProceso());
            ctx.resultado(ResultadoL050.TIPO_PROCESO_CON_FUNCION_INCORRECTO);
        } else {
            if (estado.primerLlamado() || s.funcion() == 3 || estado.tipoProceso() != s.tipoProceso()) {
                estado.cambiaTipoProceso(s.tipoProceso());    // abre o cambia de base: línea o auxiliar
            }
            if (s.funcion() <= 2) validacion.procesa(ctx);
        }
        return respuesta.arma(ctx);
    }

/** VALIDA_INFORMACION_VALIDA: la primera condición que falla decide el resultado. */
static Optional<ResultadoL050> validaSolicitud(SolicitudValRetDep s) {
    if (new BigInteger(s.cliente()).signum() <= 0)   return Optional.of(ResultadoL050.CLIENTE_INCORRECTO);
    if (s.importe().signum() <= 0)                   return Optional.of(ResultadoL050.IMPORTE_INCORRECTO);
    if (s.indDepRet() != 1 && s.indDepRet() != 2)    return Optional.of(ResultadoL050.IND_DEP_RET_INCORRECTO);
    if (s.movForzado() != 0 && s.movForzado() != 1)  return Optional.of(ResultadoL050.IND_MOV_FORZADO_INCORRECTO);
    return Optional.empty();
}

    /** Solo completa el límite que el cliente no trae en B52 (valor 0). */
    Limites completa(Limites deB52, Contexto ctx) {
        if (deB52.dia().signum() > 0 && deB52.mes().signum() > 0) return deB52;
        Optional<Limites> b55Ug = busca(ctx, ctx.ubicacionGeo());
        Optional<Limites> b55 = b55Ug.isPresent() || ctx.ubicacionGeo() == 0
                ? b55Ug
                : busca(ctx, 0);                     // segundo intento con ubicación 0
        return b55.map(l -> deB52.completaCon(l)).orElse(deB52);   // sin límite si no hay registro
    }

    private Optional<Limites> busca(Contexto ctx, int ug) {
        try {
            return b55.findByFechaAndTipoPersonaAndTipoOperacionAndUbicacionGeo(
                    ctx.fechaLinea(), ctx.tipoPersona(), ctx.indDepRet(), ug).map(B55Limite::limites);
        } catch (DataAccessException e) {
            ctx.resultado(ResultadoL050.ERROR_BD_B52_B55_B53);      // I_RESULT := 15
            throw new ErrorBaseLimites(e);           // corta el segundo intento, como I_RESULT NEQ 15
        }
    }
}
/** PROCESA_DEPOSITO. Importes en BigDecimal escala 2; ALGOL usa DOUBLE. */
void procesaDeposito(Contexto ctx, Limites lim, Acumulados acum) {
    BigDecimal importe = ctx.importe();

    if (lim.dia().signum() > 0) {
        BigDecimal total = acum.dia().add(importe);
        if (total.compareTo(lim.dia()) >= 0) {
            BigDecimal excedente = total.subtract(lim.dia());
            ctx.saldoCongelado(excedente.min(importe));      // nunca más que el propio depósito
            ctx.indSupLim(1);
            verificaAcumSaldo(ctx, OpcionAcum.SALDO_CONGELADO);
        }
    }
    if (lim.mes().signum() > 0) {
        BigDecimal total = acum.mes().add(importe);
        if (total.compareTo(lim.mes()) >= 0) {
            ctx.indSupLim(ctx.indSupLim() == 1 ? 3 : 2);    // solo indicador: no congela
        }
    }
}
/** PROCESA_RETIRO. */
void procesaRetiro(Contexto ctx, Limites lim, Acumulados acum) {
    if (acum.bloqueado()) {                                   // B52-IND-BLOQUEO <> 0
        ctx.resultado(ResultadoL050.CLIENTE_BLOQUEADO_RETIRO);   // 9
        ctx.indIncRet(3);
        ctx.eventoDosACuatro();
        return;
    }
    if (ctx.movForzado() == 1 && estado.tipoProceso() == TipoProceso.PRELINEA) {
        return;                                               // forzado en prelínea: sin validar límites
    }
    if (lim.dia().signum() > 0 && acum.dia().add(ctx.importe()).compareTo(lim.dia()) >= 0) {
        ctx.resultado(ResultadoL050.LIMITE_RETIRO_DIARIO);    // 10
        ctx.indIncRet(1); ctx.indSupLim(1);
        ctx.eventoDosACuatro();
    }
    if (lim.mes().signum() > 0 && acum.mes().add(ctx.importe()).compareTo(lim.mes()) >= 0) {
        if (ctx.indIncRet() == 1) {
            ctx.indIncRet(3); ctx.indSupLim(3);
        } else {
            // El fuente asigna 10 también aquí; RESULT 11 (mensual) solo se usa en PROCESA_FUNCION_4.
            ctx.resultado(props.resultadoMensualCompatible()
                    ? ResultadoL050.LIMITE_RETIRO_DIARIO : ResultadoL050.LIMITE_RETIRO_MENSUAL);
            ctx.indIncRet(2); ctx.indSupLim(2);
        }
        ctx.eventoDosACuatro();
    }
}
/** Tope de los acumulados en B52: DF_MAX_ACUM = @@16 - .01. */
static final BigDecimal MAX_ACUM = new BigDecimal("9999999999999999.99");

/** PROCESA_DEPOSITO_RETIRO: antes de comparar, ajusta los acumulados a la fecha en línea. */
Acumulados vigentes(Contexto ctx, Acumulados acum) {
    if (ctx.evento() == 1) return acum;                       // cliente nuevo: sin acumulados
    YearMonth mesLinea = YearMonth.from(ctx.fechaLinea());
    YearMonth mesAcum  = YearMonth.from(acum.fecha());
    Acumulados a;
    if (!mesLinea.equals(mesAcum)) {
        a = acum.sinMensual().sinDiario();                    // FIN_MES
    } else if (!ctx.fechaLinea().equals(acum.fecha()) || ctx.evento() == 2) {
        a = acum.sinDiario();                                 // FIN_DIA o evento 2
    } else {
        a = acum;
    }
    // VERIFICA_ACUM_SALDO(1): con ambos acumulados saturados el movimiento se valida
    // pero no se guarda (IND_VALIDA 3), salvo tipo de persona 0.
    boolean saturado = a.dia().compareTo(MAX_ACUM) == 0 && a.mes().compareTo(MAX_ACUM) == 0;
    if (saturado && ctx.tipoPersona() != 0) ctx.indValida(3);
    return a;
}

Riesgos de migración
Validación abierta cuando falla la base

Si OPEN INQUIRY de S500BD01CAPTACION falla (ABREIN_BDLINEA, línea 1574) se enciende B_PASA_TODO y PROCESA_MOVIMIENTO responde IND_VALIDA 2 sin validar límites. Ante una falla de base los depósitos y retiros pasan sin control.

Resultado 10 para el límite mensual de retiro

PROCESA_RETIRO asigna 10 al rebasar el límite mensual (línea 3638) aunque el encabezado documenta 11 para ese caso. La versión Java lo deja configurable para no romper a P010.

Acumulados en memoria volátil

Los movimientos validados esperan en un arreglo de 500 posiciones en memoria de la librería. Una caída o una descarga de la librería pierde los pendientes, y LEE_ARREGLO también los descarta si la base de salida no está abierta.

Importes en punto flotante

Importes, límites y acumulados son DOUBLE de ALGOL (DOUBLE(...)/100). La conversión de regreso con * 100 FOR 18 DIGITS puede truncar un centavo. La traducción usa BigDecimal de escala 2.

Movimientos sin límite

Si el cliente no tiene límites en B52 y no existe registro en B55 ni para su ubicación ni para la ubicación 0, el depósito o retiro se acepta sin límite.
