S151 · WFL_LOTE a Java 21 Spring Boot

El job no contiene lógica contable: identifica el nodo por el número de serie del procesador, valida su propia versión contra el catálogo S006, toma el primer token del parámetro y lo traduce en un número de paso que despacha un segundo CASE. Cada paso envuelve un programa COBOL en la misma plantilla de reinicio, falla y listado forzado. Ante una falla el job se detiene esperando que el operador teclee en consola si reintenta, continúa o termina. Casi todo es secuencial; el único paso asíncrono es P199 dentro del proceso por sistema satélite.

/** Prólogo del job: reemplaza el parseo de PARAMETRO, el menú de consola y CAMBIAVER. */
@Component
@RequiredArgsConstructor
public class LoteLauncher {

    private final JobLauncher jobLauncher;
    private final Job loteS151;
    private final VersionGate versionGate;          // VALIDAVER
    private final Clock relojMx;                    // zona America/Mexico_City

    public JobExecution lanza(String parametro) throws Exception {
        // SEPARAPARAM: AUX es el primer token y AUX1..AUX10 los siguientes.
        List<String> t = List.of(parametro.trim().split("\\s+"));
        String aux = t.get(0);
        if (Set.of("", "M", "MENU", "V", "VERSION").contains(aux)) {
            throw new JobParametersInvalidException(
                "El menú interactivo y el cambio de versión por consola no existen en el destino");
        }
        versionGate.verifica("S151LOTE");           // aborta si la versión catalogada difiere
        LocalDate fechaMaquina = LocalDate.now(relojMx);   // FECHAY := TIMEDATE(YYMMDD), sin pivote

        JobParameters params = new JobParametersBuilder()
                .addString("parametro", parametro)
                .addString("comando", aux)
                .addString("aux1", t.size() > 1 ? t.get(1) : "")
                .addString("sistpar", t.size() > 2 ? t.get(2) : "")    // SISTPAR := AUX2
                .addLocalDate("fechaMaquina", fechaMaquina)
                .toJobParameters();
        return jobLauncher.run(loteS151, params);    // JOBSUMMARYTITLE -> nombre e id de la ejecución
    }
}
/** VALIDAVER: compara la versión desplegada del orquestador contra el catálogo S006. */
@Component
@RequiredArgsConstructor
public class VersionGate {

    private final CatalogoVersionesClient catalogo;   // reemplaza RUN #S006P618
    private final BuildProperties build;              // versión del artefacto desplegado

    public void verifica(String componente) {
        String catalogada = catalogo.versionCatalogada(componente);
        if (!catalogada.equals(build.getVersion())) {
            // El WFL hace WAIT(MENSAJE, OK) y GO FIN: espera confirmación del operador y termina.
            throw new VersionNoCatalogadaException(
                "ERROR EN VERSION DEL WFL/LOTE, LA VERSION CATALOGADA ES " + catalogada);
        }
    }
}

/** CASE AUX OF: traduce el comando de arranque en la rama del DAG que se ejecuta. */
@Component
@RequiredArgsConstructor
public class ComandoLoteDecider implements JobExecutionDecider {

    private final LoteProperties props;

    @Override
    public FlowExecutionStatus decide(JobExecution job, StepExecution ultimo) {
        String aux = job.getJobParameters().getString("comando");
        if ("DIARIO".equals(aux))      return new FlowExecutionStatus("DIARIO");   // PASO 1
        if ("DIARIOADMOV".equals(aux)) return new FlowExecutionStatus("ADMOV");    // PASO 50
        if (props.sistemasSatelite().contains(aux)) {
            String sistema = aux.substring(1);                     // SISTPAR := DROP(AUX,1): "084", "500"
            job.getExecutionContext().putString("sistema", sistema);
            return "584".equals(sistema)                           // PASO 72: sistema dado de baja
                    ? new FlowExecutionStatus("BAJA")
                    : new FlowExecutionStatus("SATELITE");         // PASO 72 -> PRODIASNNN
        }
        return new FlowExecutionStatus("INVALIDO");                // PASO := 9999999999
    }
}

@Bean
Flow diarioFlow(LoteSteps s, NodoDecider nodo) {
    Step p000FueraDeLinea = s.p000(0);                             // VALUE000 := 0
    return new FlowBuilder<SimpleFlow>("diario")
            .start(s.p612("DAILY"))                                // aviso de inicio
            .next(s.p000(1))                                       // proyección en línea; TASKVALUE -> fecha
            .next(s.calendario())                                  // FECHAS: DIA30, DIA1MES, FESTIVO, ...
            .next(s.dumpAdSaldo(Respaldo.EN_LINEA))                // B2OLMEN si DIA30, B2OLDIA si no
            .next(s.limpiaListados("P120"))                        // REMUEVEBACKUPS(120)
            .next(nodo)
                .on("VDM").to(s.p120()).next(p000FueraDeLinea)     // SUB120 solo en nodo 10
            .from(nodo)
                .on("*").to(p000FueraDeLinea)
            .next(s.inicializaAdSaldo())                           // INICIALIZASDO
            .next(s.dumpAdSaldo(Respaldo.BATCH))                   // B2OBMEN si DIA30, B2OBDIA si no
            .next(s.limpiaListados("P005"))
            .next(s.p005())                                        // carga de archivos para Tesorería
            .next(s.p612("DDIAR"))                                 // aviso de fin
            .build();
}

/** PROCESS SUB199(SNNN): único paso asíncrono; el job no espera su término. */
@Bean
Step lanzaP199Asincrono(JobRepository repo, PlatformTransactionManager tx,
                        @Qualifier("asyncJobLauncher") JobLauncher async, Job p199Job) {
    return new StepBuilder("lanzaP199Asincrono", repo).tasklet((contrib, ctx) -> {
        String sistema = (String) ctx.getStepContext().getJobExecutionContext().get("sistema");
        Thread.sleep(Duration.ofSeconds(10));                       // WAIT(10) previo
        async.run(p199Job, new JobParametersBuilder()
                .addString("sistema", sistema)
                .addLong("lanzado", System.currentTimeMillis()).toJobParameters());
        Thread.sleep(Duration.ofSeconds(10));                       // WAIT(10) posterior
        return RepeatStatus.FINISHED;
    }, tx).build();
}
/** SUB001 función "00" + FECHAS + SUB178 + archivo de control corporativo del S084. */
@Bean @JobScope
Step proyeccionYControlS084(JobRepository repo, PlatformTransactionManager tx,
                            P001Client p001, P178Client p178, P312Client p312,
                            CalendarioCorporativo calendario,
                            ArchivosMcp fs, LoteProperties props,
                            @Value("#{jobExecutionContext['sistema']}") String sistema) {

        var jctx = contrib.getStepExecution().getJobExecution().getExecutionContext();

        FechasSistema f = p001.proyecta(sistema, "00");      // FANTIER, FAYER, FHOY, FMANANA, ...

        jctx.put("fechas", f); jctx.put("banderas", b); jctx.put("fechaY", fechaY);

        if (b.dia30() && "500".equals(sistema)) {
            String destino = "S151/FILE/S084/%02d/%s".formatted(props.nodo(),

                default -> throw new IllegalStateException("nodo sin rama");   // NODO 32 no existe
            }
        }
        return RepeatStatus.FINISHED;
    }, tx).build();

    }

Riesgos de migración
Bloqueo humano ante cualquier falla

FALLA (líneas 276-301) espera un ACCEPT de consola sin tiempo límite ni política por severidad. Cada falla de paso detiene la ventana batch hasta que un operador responde.

Opción V sin efecto

En FALLA la opción V pide un número de MTP (líneas 286-295) pero el valor no se asigna a ninguna variable de versión; el paso se reintenta con la misma versión.

Ejecución asíncrona sin seguimiento

PROCESS SUB199 (línea 5952) lanza P199 sin esperar su término ni revisar su resultado.

Marca de inicio y fin asimétrica

PRODIASNNN omite la marca de inicio para 804 y 203 (línea 5930) pero ejecuta la de fin para todos (línea 6013).

Identificación de nodo por hardware

El nodo se obtiene del número de serie y del host (líneas 6326-6345). Un valor no previsto deja NODO = 0 sin abortar, y la condición NODO = 32 (línea 5981) es inalcanzable.

Sin guardas de reproceso

Ningún paso consulta un archivo de control antes de ejecutarse; relanzar un comando repite inicializaciones de bases y proyecciones de fecha.

Fecha de máquina de seis dígitos

FECHAY se toma como TIMEDATE(YYMMDD) en la línea 6363; el siglo queda implícito en los títulos de archivo y respaldo.
