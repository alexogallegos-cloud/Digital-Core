S500 · L060 a Java 21 Spring Boot

Al cargarse, la librería identifica en qué plaza corre a partir del hostname, obtiene el designador del servicio de consulta de contratos de la plaza contraria y el de un buffer binario X_OCTET, y se congela como librería compartida. Su única entrada exportada, S500L060CONTRATO, envía el área común del llamador al servicio remoto fuera de transacción, espera la respuesta y devuelve como resultado el código de seis dígitos que trae al inicio, copiando el resto al arreglo de salida. Los errores de transporte se devuelven como 98 si son por tiempo agotado y como 99 en cualquier otro caso.

// ---------------------------------------------------------------------------
public enum Plaza { MONTERREY, VALLE }

@Component
@RequiredArgsConstructor
class PlazaResolver {

    private final L060Properties props;

    /** INICIALIZA: un host fuera de las dos listas termina la librería. */
    Plaza plazaDe(String hostname) {
        if (props.hosts().monterrey().contains(hostname)) return Plaza.MONTERREY;
        if (props.hosts().valle().contains(hostname))     return Plaza.VALLE;
        throw new IllegalStateException("S500/L060/002:E>050110<HOST DESCONOCIDO " + hostname);
    }

    /** Servicio de la plaza contraria a la propia. */
    String servicioRemoto(Plaza propia) {
        return propia == Plaza.MONTERREY ? props.servicioRemoto().monterrey()
                                         : props.servicioRemoto().valle();
    }
}
@Configuration
@EnableConfigurationProperties(L060Properties.class)
public class L060ClientConfig {

    /**
     * GET_DESIGNATOR_USING_NAME + GET_BUFFER_DESIGNATOR("X_OCTET"): en el destino,
     * un cliente HTTP hacia el servicio de consulta de contratos de la otra plaza.
     * Cualquier falla de configuración detiene el arranque (FREEZE no llega a ocurrir).
     */
    @Bean
    RestClient consultaContratoRemota(L060Properties props, PlazaResolver plazas,
                                      @Value("${HOSTNAME}") String hostname) {
        Plaza propia = plazas.plazaDe(hostname);
        String servicio = plazas.servicioRemoto(propia);
        return RestClient.builder()
                .baseUrl(props.endpoint(servicio))           // s500.l060.endpoints.S500_Con_Cto_V: https://...
                .defaultHeader(HttpHeaders.CONTENT_TYPE,
                               MediaType.APPLICATION_OCTET_STREAM_VALUE)   // buffer X_OCTET
                .requestFactory(ClientHttpRequestFactories.get(
                        ClientHttpRequestFactorySettings.DEFAULTS
                            .withConnectTimeout(props.timeoutConexion())
                            .withReadTimeout(props.timeoutRespuesta())))  // TPETIME explícito
                .build();
    }
}

/** Misma interfaz que exportaba la librería: devuelve un código entero y llena la salida. */
public interface ConsultaForaneaContrato {

    int ERROR_REINTENTE    = 98;   // ERROR_REINTENTEV: timeout, el llamador puede reintentar
    int ERROR_NO_REINTENTE = 99;   // ERROR_NO_REINTENTEV

    /**
     * S500L060CONTRATO(AREA_COMUN, EA_O_CTO).
     * @param areaComun petición tal como la arma el programa llamador (bytes EBCDIC)
     * @param salida    buffer del llamador; se llena con la respuesta sin sus 6 bytes de código
     * @return código remoto (6 primeros bytes de la respuesta) o 98 / 99 si falla el transporte
     */
    int consultaContrato(byte[] areaComun, byte[] salida);
}

// LONGITUD_AREA_COMUNV = 180 ya no se usa en el fuente: la longitud real es SIZE(AREA_COMUN).
@Service
@RequiredArgsConstructor
public class ConsultaForaneaContratoClient implements ConsultaForaneaContrato {

    private final RestClient consultaContratoRemota;

    @Override
    public int consultaContrato(byte[] areaComun, byte[] salida) {
        byte[] respuesta;
        try {
            // TP_ACALL con TPNOTRAN: la consulta no participa en la transacción del llamador
            respuesta = consultaContratoRemota.post()
                    .uri("/contratos/consulta")
                    .body(areaComun)                      // LEN := SIZE(AREA_COMUN)
                    .retrieve()
                    .body(byte[].class);
        } catch (ResourceAccessException e) {
            log.error("S500/L060/002:E>050110<TP_ACall error: {}", e.getMessage());
            return e.getCause() instanceof SocketTimeoutException
                    ? ERROR_REINTENTE                     // XOPEN_RSLT = TPETIME
                    : ERROR_NO_REINTENTE;
        } catch (RestClientResponseException e) {
            log.error("S500/L060/002:E>050110<Error [RCode = {}]", e.getStatusCode().value());
            return ERROR_NO_REINTENTE;                    // servicio desconocido o buffer inválido
        }
        return decodifica(respuesta, salida);
    }
    /** Capacidad del buffer temporal de respuesta: EA_O_CTO_TEM[0:1999]. */
    static final int MAX_RESPUESTA = 2000;

    /**
     * TP_GETRPLY con TPTRUNC: si la respuesta excede 2000 bytes, XATMI la trunca y
     * TPETOOSMALL NO se trata como error. Se reproduce ese comportamiento, pero
     * se deja traza para poder medir cuántas respuestas llegan truncadas.
     */
    private byte[] ajustaTamano(byte[] respuesta) {
        if (respuesta.length > MAX_RESPUESTA) {
            log.warn("S500/L060/002: respuesta de {} bytes truncada a {}", respuesta.length, MAX_RESPUESTA);
            return Arrays.copyOf(respuesta, MAX_RESPUESTA);
        }
        return respuesta;
    }
    private static final Charset EBCDIC = Charset.forName("IBM037");   // confirmar página de códigos

    /** Si no hubo error local, el código viene en los 6 primeros bytes y el resto es la salida. */
    private int decodifica(byte[] respuesta, byte[] salida) {
        byte[] r = ajustaTamano(respuesta);
        if (r.length < 6) {
            return ERROR_NO_REINTENTE;                     // el original leería basura del buffer
        }
        int codigo = Integer.parseInt(new String(r, 0, 6, EBCDIC).strip());   // INTEGER(EA_O_CTO_TEM,6)
        int n = Math.min(salida.length, MAX_RESPUESTA - 6);   // nunca más de lo que cabe en la salida
        System.arraycopy(r, 6, salida, 0, Math.min(n, r.length - 6));
        return codigo;
    }
}
Riesgos de migración
Código remoto y código local comparten valores

S500L060CONTRATO devuelve 98 o 99 para errores de transporte y, si no hubo error, el número de los primeros seis bytes de la respuesta (líneas 232-236). Un 98 o 99 enviado por el servicio remoto es indistinguible de un error local.

Truncamiento silencioso

TP_GETRPLY usa TPTRUNC y la condición de las líneas 211-212 excluye TPETOOSMALL, así que una respuesta mayor a 2000 bytes llega cortada como si fuera exitosa.

Restos de respuestas previas

La copia a la salida usa SIZE(EA_O_CTO_TEM) y no la longitud recibida (líneas 237-240). EA_O_CTO_TEM es global y la librería se congela con FREEZE(TEMPORARY), por lo que la puede compartir más de un llamador.

Hostnames compilados

Los nueve hostnames de las líneas 97-107 y los servicios S500_Con_Cto_V y S500_Con_Cto_M están en el fuente; un servidor nuevo cae en HOST DESCONOCIDO.

Inicialización de I_RESULT

I_RESULT solo se asigna en las ramas de error; el camino exitoso depende de que el compilador lo inicialice en cero.
