S151 · P135 a Java 21 Spring Boot

P135 obtiene de SOPORTECOMS el CSI y de S151LIBCONTROL la fecha de proceso y los nombres de los archivos de movimientos y de datos adicionales; un TASKVALUE mayor que cero sustituye la fecha. Exige que el header del archivo de movimientos traiga esa fecha y aborta la tarea si no coincide. Cada movimiento de función 1 y estatus 1 se examina hasta en cinco pares clave de transacción e importe, y según el sistema origen, la clave y el tipo de proceso se copia al archivo de punteo correspondiente acumulando cargos y abonos. Al final escribe trailers con totales, cierra los archivos y los transmite; con sistema 500 genera además salidas para Soriana, MPS, portabilidad, factoraje y EXTNET.

/** A00-R00-HDR-MOV: primer registro del archivo de movimientos. */
public record HeaderMovimientos(String hd, LocalDate fecha, int banDes) {}   // X(02), 9(08), 9(02)

/** A00-R01-REGMOV: campos binarios COMP; se decodifican con un LineMapper binario. */
public record MovimientoContable(
        int funcion,            // 99 = fin lógico del archivo
        int formato,
        int sistema,
        int producto,           // REDEFINES: tipo de crédito o tipo de producto según el sistema
        int instrumento,
        List<ClaveImporte> claves,   // A00-R01-ELE-CVEIMP OCCURS 5 TIMES
        DatosControl control) {      // origen, estatus, tipo de proceso, sucursal, caja, etc.

    /** CVETRAN 9(06) COMP, INDLEY, ESQCON, IMPORTE 9(16)V99 COMP sin signo. */
    public record ClaveImporte(int cveTran, int indLey, int esqCon,
                               BigDecimal importe,      // escala 2, siempre positivo
                               int cveDesvio, int guiaDesvio) {}

    /** El ciclo COBOL corta en la primera clave en cero: "UNTIL W77-IND > 5 OR CVETRAN = 0". */
    public Stream<ClaveImporte> clavesActivas() {
        return claves.stream().takeWhile(c -> c.cveTran() != 0);
    }
}

@Configuration
@EnableConfigurationProperties(P135Properties.class)
public class PunteoJobConfig {

    @Bean
    Job punteoContable(JobRepository repo, Step controlS151, Step ejecutaP110,
                       Step cargaCatalogos, Step generaInterfaces, Step trailersYEnvio,
                       Step listadoCarat) {
        JobExecutionDecider esS500 = (job, step) -> new FlowExecutionStatus(
                job.getJobParameters().getLong("sistema") == 500 ? "S500" : "OTRO");

        return new JobBuilder("S151P135", repo)
                .start(controlS151)                       // CSIID + CONSISDIA
                .next(esS500).on("S500").to(ejecutaP110)  // 000-025-EJECUTA-P110
                .next(cargaCatalogos)
                .from(esS500).on("OTRO").to(cargaCatalogos)
                .next(generaInterfaces)                   // 100 / 120 / 190 / 200
                .next(trailersYEnvio)                     // 300-CREA-TRAILER + 900-CIERRA-ARCHIVOS
                .next(listadoCarat)                       // 1100-SORT + 1200-IMPRIME, si no es solo punteo
                .end().build();
    }
}

@Component
@RequiredArgsConstructor
class ControlS151Tasklet implements Tasklet {
    private final SoporteComsClient soporte;
    private final ControlS151Client control;

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        var job = ctx.getStepContext().getStepExecution().getJobExecution();
        long sistema = job.getJobParameters().getLong("sistema");
        int csi = soporte.csiId().numCsi();
        var dia = control.consultaSistemaDia(sistema, csi)
                .orElseThrow(() -> new ProcesoAbortadoException(
                        "ERROR AL CONSULTAR S151LIBCONTROL PARA OBTENER NOMBRE DE LA BASE DIARIA"));
        long taskValue = job.getJobParameters().getLong("taskValue", 0L);
        LocalDate fecha = taskValue > 0 ? Fechas.aaaammdd(taskValue) : dia.fechaProceso();
        var ec = job.getExecutionContext();
        ec.put("csi", csi);
        ec.put("fechaProceso", fecha);
        ec.put("archivoMovimientos", dia.nombreMovimientos() + " ON " + dia.packMovimientos());
        ec.put("archivoDescripciones", dia.nombreDescripciones() + " ON " + dia.packDescripciones());
        return RepeatStatus.FINISHED;
    }
}
@Component
@StepScope
public class MovimientosReader implements ItemStreamReader<MovimientoContable> {

    @Value("#{jobExecutionContext['archivoMovimientos']}") private String titulo;
    @Value("#{jobExecutionContext['fechaProceso']}")       private LocalDate fechaProceso;
    private BinaryRecordReader delegate;

    @Override
    public void open(ExecutionContext ec) {
        delegate = BinaryRecordReader.open(titulo);           // CHANGE ATTRIBUTE TITLE + OPEN INPUT
        HeaderMovimientos h = delegate.readHeader()
                .orElseThrow(() -> new ArchivoVacioException(titulo));
        if (!h.fecha().equals(fechaProceso)) {
            // SET MYSELF(STATUS) TO -1: el paso termina y ningún archivo de punteo se cierra con SAVE
            throw new ProcesoAbortadoException(
                    " FECHA ERRONEA EN ARCH. MOVTOS. : " + h.fecha());
        }
    }

    @Override
    public MovimientoContable read() {
        MovimientoContable m = delegate.next();
        return (m == null || m.funcion() == 99) ? null : m;   // 190-LECTURA: función 99 es fin
    }
}

    }

    @Value("#{jobParameters['soloPunteo'] == 1}") private boolean soloPunteo;

    @Override
    public List<RegistroPunteo> process(MovimientoContable m) {
        var c = m.control();
        boolean vigente = m.funcion() == 1 && c.status() == 1 && (c.origen() == 1 || c.origen() == 3);
        if (soloPunteo || !vigente) return List.of();

        var adic = adicionales.porApuntador(c.apuDes());     // 400-LEE-DATOS-ADIC
        boolean elegible = SISTEMAS_PUNTEO.contains(c.sistOrig())
                || (c.cveMda() == 2 && c.binTarjeta() == 557874)
                || c.binMovil()
                || (c.sistOrig() == 274 && c.sucTran() == 870 && c.cajaTran() == 76);
        if (!elegible) return List.of();

        return m.clavesActivas()
                .flatMap(k -> reglas.aplica(m, k, adic).stream())   // 210-MUEVE-DATOS por clave
                .toList();
    }
}
/** Fragmento de 210-MUEVE-DATOS: interfaz S253 (tarjetas y cuentas). */
Optional<RegistroPunteo> s253(MovimientoContable m, ClaveImporte k, DatosAdicionales adic) {
    var c = m.control();
    if (!(c.sistOrig() == 253 || c.sistOrig() == 54) || !CLAVES_S253.contains(k.cveTran())) {
        return Optional.empty();
    }
    if (c.tipoProc() >= 11) return Optional.empty();       // solo tipos de proceso menores a 11

    var reg = new RegistroS253(
            c.sucProm(), validaContrato(m),                  // 211-VALIDA-CONTRATO
            k.cveTran(), k.importe(),
            c.sucInic(), c.cajaInic(), c.autoriza(),
            adic.ref23Pos(),
            c.fecVal(),                                      // A2K-CONV-CAMD-TO-AMD: LocalDate, se formatea al escribir
            c.mdaSuc(), c.mdaCta(), m.producto(), m.instrumento(), c.moneda(), c.origen());

    totales.s253(c.cveMda(), k);                             // 215-SUMA-SALDOS-S253
    // 500-ESCRIBE-253: medio 1 va a L03, cualquier otro a L04
    return Optional.of(new RegistroPunteo(c.cveMda() == 1 ? Destino.S253_L03 : Destino.S253_L04, reg));
}

/** W77-IM-253-*: PIC 9(16)V99 sin signo, sin ON SIZE ERROR. */
public final class TotalesPunteo {
    private static final Set<Integer> CARGOS_MEDIO_1 = Set.of(2020, 2100, 2388);
    private static final Set<Integer> ABONOS_MEDIO_1 = Set.of(1001, 1004, 1021, 1193, 1197);
    private static final Set<Integer> CARGOS_OTRO    = Set.of(2020, 2100, 2109, 2111, 2388);
    private static final Set<Integer> ABONOS_OTRO    = Set.of(1001, 1004, 1021, 1070, 1193, 1197);

    private final Total cargoL03 = new Total(), abonoL03 = new Total();
    private final Total cargoL04 = new Total(), abonoL04 = new Total();
    private long regL03, regL04;

    void s253(int cveMda, ClaveImporte k) {
        int cve = k.cveTran();
        if (cveMda == 1) {
            regL03++;
            if (CARGOS_MEDIO_1.contains(cve)) cargoL03.suma(k.importe());
            else if (ABONOS_MEDIO_1.contains(cve)) abonoL03.suma(k.importe());
            // 2109, 2111, 2394, 1070 y 2131 se escriben pero no suman en medio 1
        } else {
            regL04++;
            if (CARGOS_OTRO.contains(cve)) cargoL04.suma(k.importe());
            else if (ABONOS_OTRO.contains(cve)) abonoL04.suma(k.importe());
        }
    }

    /** 300-CREA-TRAILER: el conteo incluye header y trailer (ADD 2). */
    TrailerPunteo trailerL03() { return new TrailerPunteo(regL03 + 2, cargoL03.valor(), abonoL03.valor()); }
    TrailerPunteo trailerL04() { return new TrailerPunteo(regL04 + 2, cargoL04.valor(), abonoL04.valor()); }

    static final class Total {
        private static final BigDecimal TOPE = new BigDecimal("9999999999999999.99");
        private BigDecimal v = BigDecimal.ZERO.setScale(2);
        void suma(BigDecimal x) {
            v = v.add(x);
            if (v.compareTo(TOPE) > 0) throw new DesbordamientoAcumuladorException(v);
        }
        BigDecimal valor() { return v; }
    }
}

Riesgos de migración
Falla de envío silenciosa

030705-ENVIO-POR-INTELAR solo registra el error (líneas 6453-6460) y el proceso termina con FIN NORMAL. Las interfaces de punteo pueden quedar sin transmitir.

Trailers que no cuadran con los detalles

En 215-SUMA-SALDOS-S253 varias claves que se escriben al archivo no suman en ningún total; el receptor no puede conciliar contra la suma de los detalles.

Header leído sin AT END

El READ del header de movimientos en la línea 4365 no tiene cláusula AT END; con un archivo vacío la validación de fecha usa datos residuales.

Hardcodes de negocio y ruteo

Lista de sistemas origen, claves de transacción, BIN 557874, sucursal 870 y caja 76, sucursales 0432, 0859, 0870, 2132, 2864 y 4893, CSI 10 y 32, la clave Soriana por CSI y el literal CORRESOLO P102 S264 con PAGOS PAGOS. Se trasladan a application.yml o a tablas de parámetros.
