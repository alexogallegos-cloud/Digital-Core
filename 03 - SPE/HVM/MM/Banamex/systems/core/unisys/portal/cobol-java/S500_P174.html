S500 · P174 a Java 21 Spring Boot

Recorre S500B03CONTRATOS completo y selecciona los contratos de producto 500 instrumento 1 en moneda 1 y los de producto 66 con instrumento menor a 90. Para cada uno calcula saldo actual (incluye saldo protegido o bloqueado), saldo pendiente y saldo disponible, convirtiendo a pesos con el tipo de cambio de línea el producto 66, instrumento 15, moneda 5. En Monterrey escribe un archivo de 33 bytes con header de control que se transfiere por XRAPID; en Valle de México genera el suyo, espera el de Monterrey con confirmación del operador, valida su header y ordena ambos por CSI, sucursal y contrato. La salida de 342 bytes lleva dos headers (FH, BH), un detalle por contrato con saldos en centavos y dos trailers (BT, FT), y se acompaña de un reporte de cifras de control.

/** 00000000-PRINCIPAL: el CSI de B02 decide si este nodo produce o consolida. */
@Bean
Job saldosTandem(JobRepository repo, Step leeControl, Step extraeSaldosLocales,
                 Step esperaArchivoMty, Step consolidaYEnviaBase24, Step cifrasControl) {
    JobExecutionDecider porCsi = (job, step) ->
            new FlowExecutionStatus(job.getExecutionContext().getInt("csi") == 4 ? "MTY" : "VDM");

    return new JobBuilder("S500P174", repo)
            .start(leeControl)                    // 20003000-FECHA-PRO: B02-NUM-CSI y fecha línea
            .next(extraeSaldosLocales)            // ambos CSI generan su SDOSB24
            .next(porCsi)
                .on("MTY").to(cifrasControl)      // MTY termina; XRAPID transfiere fuera del programa
            .from(porCsi)
                .on("VDM").to(esperaArchivoMty)   // 70005000-RESIDENCIA-ARCHMTY
                .next(consolidaYEnviaBase24)      // SORT ... OUTPUT PROCEDURE 70010000
                .next(cifrasControl)
            .end()
            .build();
}

@Component
@RequiredArgsConstructor
class FiltroElegibles implements ItemProcessor<Contrato, SaldoContrato> {

    private final P174Properties props;
    private final CalculadorSaldos calculador;
    @Value("#{jobExecutionContext['csi']}") private int csi;

    @Override
    public SaldoContrato process(Contrato c) {
        if (!props.esElegible(c.producto(), c.instrumento(), c.moneda())) {
            return null;                                     // se cuenta como leído, no se graba
        }
        Saldos s = calculador.calcula(c);                    // 50003100 + 50003200
        return new SaldoContrato(2, csi, c.sucursalPromotora(), c.contrato(),
                s.actual(), s.pendiente(), s.disponible(), estatusBase24(c.status()));
    }

    /** 50003100: 0 o 5 es cuenta activa ("1"); cualquier otro estatus es "9". */
    static char estatusBase24(int status) {
        return (status == 0 || status == 5) ? '1' : '9';
    }
}
@Component
class CalculadorSaldos {

    /** 50003100-VALIDA-DISP-STATUS + 50003200-CONVIERTE-MONEDA. */
    Saldos calcula(Contrato c, BigDecimal tcLinea) {
        BigDecimal actual = c.saldoActual();
        if (actual.add(c.saldoProtegidoBloqueado()).signum() < 0) {
            actual = BigDecimal.ZERO.setScale(2);           // MOVE ZEROS TO B03-SDO-ACTUAL
        }
        if (c.producto() == 66 && c.instrumento() == 15 && c.moneda() == 5) {
            // COMPUTE sin ROUNDED: se trunca a 2 decimales
            return new Saldos(
                actual.add(c.saldoProtegidoBloqueado()).multiply(tcLinea).setScale(2, RoundingMode.DOWN),
                c.saldoPendiente().multiply(tcLinea).setScale(2, RoundingMode.DOWN),
                actual.multiply(tcLinea).subtract(c.autPendiente().multiply(tcLinea))
                      .setScale(2, RoundingMode.DOWN));
        }
        BigDecimal disponible = actual.subtract(c.autPendiente()).subtract(c.pcobPendiente());
        // El original revisa WS-SDO-DISPONIBLE < 0 ANTES de calcularlo: el piso en cero no opera.
        if (props.aplicarPisoDisponible()) {
            disponible = disponible.max(BigDecimal.ZERO.setScale(2));
        }
        return new Saldos(actual.add(c.saldoProtegidoBloqueado()), c.saldoPendiente(), disponible);
    }
}

@Component
@RequiredArgsConstructor
class EsperaArchivoMtyTasklet implements Tasklet {

    private final TandemProperties props;
    private final Clock clock;

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) throws Exception {
        Path mty = Path.of(props.resuelve(props.archivos().mty(), fechaProceso(ctx)));
        while (!Files.exists(mty)) {                                 // ATTRIBUTE RESIDENT
            Thread.sleep(props.esperaMty().intervalo().toMillis());  // WAIT(300)
            if (LocalTime.now(clock).isAfter(props.esperaMty().horaLimite())) {
                // El original pregunta al operador por consola y, si no responde SI, hace
                // STOP RUN (fin normal). Aquí se falla el step para que el orquestador lo vea.
                throw new ArchivoNoRecibidoException("ESPERANDO ARCHIVO DE MTY", mty);
            }
        }
        return RepeatStatus.FINISHED;
    }
}
/** WS-HDR-SDOB24: tipo 01, sistema 0500, fecha base del día y CSI 04. */
record HeaderControl(int tipo, int sistemaOrigen, LocalDate fechaBase, int csi) {}

void validaHeaderMty(HeaderControl h, LocalDate fechaProceso) {
    boolean ok = h.tipo() == 1
              && h.sistemaOrigen() == 500
              && h.fechaBase().equals(fechaProceso)
              && h.csi() == 4;
    if (!ok) {
        // CHANGE ATTRIBUTE STATUS OF MYSELF TO -1
        throw new HeaderInvalidoException("ARCHIVO %s CON HEADER ERRONEO".formatted(h));
    }
}

/** WS-SDOSB24-DETALLE, segmento 0. Importes en centavos sin signo, 18 dígitos. */
public record DetalleBase24(
        long secuencia,              // WS-DET-SEG0-CNT 9(09)
        String fiid,                 // "CMAE"
        String numeroCuenta,         // PRIKEY-NUM X(19): CSI(2) + sucursal(4) + contrato(12) + 1
        String tipoCuenta,           // "01"
        char estatus,                // '1' activa, '9' inactiva
        long saldoDisponibleCent,    // AVAIL-BAL 9(18)
        long saldoContableCent,      // LEDG-BAL  9(18)
        long montoRetenidoCent,      // AMT-ON-HLD 9(18)
        int moneda) {                // CRNCY-CDE 484 fijo

    static final int LONGITUD_SEG0 = 300;   // WS-DET-SEG0-LGTH

    /** El fuente no tiene signo: un negativo se escribiría como su valor absoluto. */
    static long aCentavos(BigDecimal importe, boolean rechazarNegativos) {
        if (importe.signum() < 0 && rechazarNegativos) {
            throw new SaldoNegativoException(importe);
        }
        return importe.abs().movePointRight(2).longValueExact();   // COMPUTE ... * 100
    }
}
@Component
@StepScope
@RequiredArgsConstructor
class Base24Processor implements ItemProcessor<SaldoContrato, DetalleBase24> {

    private final TandemProperties props;
    private long secuencia = 2;              // WS-SEQ-MTYVDM: los dos headers ocupan 1 y 2
    private BigDecimal totalContable = BigDecimal.ZERO;   // WS-TRL1-AMT

    @Override
    public DetalleBase24 process(SaldoContrato s) {
        if (s.tipo() != 2) return null;      // los headers de control 01 de MTY y VDM se descartan
        secuencia++;
        String cuenta = "%02d%04d%012d ".formatted(s.csi(), s.sucursal(), Long.parseLong(s.contrato()));
        long contable = DetalleBase24.aCentavos(s.actual(), props.rechazarNegativos());
        totalContable = totalContable.add(BigDecimal.valueOf(contable));
        return new DetalleBase24(secuencia, "CMAE", cuenta, "01", s.estatus(),
                DetalleBase24.aCentavos(s.disponible(), props.rechazarNegativos()),
                contable,
                DetalleBase24.aCentavos(s.pendiente(), props.rechazarNegativos()),
                484);
    }
}
@Component
@RequiredArgsConstructor
class Base24Envelope implements FlatFileHeaderCallback, FlatFileFooterCallback {

    private final CalendarioBancario calendario;
    private final Base24Processor detalle;
    private final Clock clock;
    @Value("#{jobExecutionContext['fechaProceso']}") private LocalDate fechaProceso;
    @Value("#{jobExecutionContext['registrosMtyVdm']}") private long registrosMtyVdm;

    @Override
    public void writeHeader(Writer w) throws IOException {
        // 70012000-HABIL-ANT: día hábil anterior como fecha de última extracción ATM y POS
        String ultimaExtraccion = calendario.habilAnterior(fechaProceso)
                .map(d -> d.format(DateTimeFormatter.ofPattern("yyMMdd"))).orElse("      ");
        w.write(Base24Layout.hdr1(fechaProceso, LocalTime.now(clock), ultimaExtraccion)); // FH, cnt 1
        w.write("\n");
        w.write(Base24Layout.hdr2());                                                     // BH, cnt 2
    }

    @Override
    public void writeFooter(Writer w) throws IOException {
        long seq = detalle.secuencia();
        w.write(Base24Layout.trl1(seq + 1, detalle.totalContable(), registrosMtyVdm));   // BT
        w.write("\n");
        w.write(Base24Layout.trl2(seq + 2, registrosMtyVdm));                            // FT
    }
}
Riesgos de migración
Disponible negativo publicado como positivo

El piso en cero de 50003100 (líneas 756-757) se evalúa antes de calcular el disponible en 50003200, así que no opera. El disponible negativo se multiplica por 100 hacia WS-DET-SEG0-AVAIL-BAL, un PIC 9(18) sin signo, y la red de cajeros recibiría un disponible positivo igual al sobregiro. Es el hallazgo de mayor impacto del componente y debe confirmarse con datos reales.

Fin normal cuando no llega el archivo de Monterrey

Si el operador responde que no seguirá esperando (líneas 932-936), el programa termina con STOP RUN sin marcar error y no se genera el archivo de saldos para Tandem.

Conversión de dólares truncada y sin cobros pendientes

Los COMPUTE con B02-TC-LINEA no usan ROUNDED y la rama de dólares no resta B03-PCOBP-PENDTE del disponible, a diferencia de la rama en pesos.

Conteos de trailer derivados de atributos de archivo
