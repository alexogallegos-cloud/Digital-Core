/** WKS-I04-R-00-REGISTRO: header, header de negocio, detalle y trailer en 100 bytes. */
public sealed interface S244Record
        permits S244Record.Header, S244Record.HeaderNegocio, S244Record.Detalle, S244Record.Trailer {

    String linea();

    /** Tipo 3. */
    record Detalle(int numReg,
                   String fechaValorAammdd,        // 9(06): la fecha de máquina pierde el siglo
                   int naturaleza, int tipoTran,
                   String autorizacionB24,         // X(06)
                   BigDecimal importe,             // 9(09)V99 -> escala 2
                   Referencia23 referencia,
                   String tarjeta,                 // X(19)
                   String horaTran) implements S244Record {
        public String linea() { return Fixed.s244Detalle(this); }
    }

    /** WKS-I04-REF-23R: 1 + 6 + 1 + 3 + 11 + 1 posiciones. */
    record Referencia23(int respPago, String binAdquirente, int digitoAnio,
                        int diaJuliano, String datoAdquirente, int digitoVerificador) {
        String sinDigito() {
            return "%d%s%d%03d%s".formatted(respPago, binAdquirente, digitoAnio, diaJuliano, datoAdquirente);
        }
    }

    record Header(String sistemaDestino, String fechaAammdd, String cadena) implements S244Record {
        public String linea() { return Fixed.s244Header(this); }
    }
    record HeaderNegocio(long numNegocio) implements S244Record {
        public String linea() { return Fixed.s244Negocio(this); }
    }
    record Trailer(long cargos, BigDecimal importeCargos, long registros) implements S244Record {
        public String linea() { return Fixed.s244Trailer(this); }
    }
}

@Component
public class MovimientoTeletonClassifier
        implements Classifier<MovimientoTeleton, ItemWriter<? super MovimientoTeleton>> {

    private final ItemWriter<MovimientoTeleton> s244MasPunteo;    // 930-GRABA-I04 + 960-GRABA-I08
    private final ItemWriter<MovimientoTeleton> punteoMasAmex;    // 960-GRABA-I08 + 991-ESCRIBE-AMEXMN
    private final ItemWriter<MovimientoTeleton> ignorado = items -> {};

    @Override
    public ItemWriter<? super MovimientoTeleton> classify(MovimientoTeleton m) {
        return switch (m.staMovto()) {
            case 0  -> s244MasPunteo;      // tarjetas procesadas para S244
            case 15 -> punteoMasAmex;      // American Express
            default -> ignorado;           // cualquier otro estatus no se reporta
        };
    }
}

@Component
@RequiredArgsConstructor
public class Referencia23Builder {

    private final P630Properties props;         // bin-visa-amex: 454061, bin-otros: 543006, dato-adquirente
    private final CalendarioService fechas;     // DAME_DIAJUL2K IN FECHAS

    public S244Record.Detalle detalle(MovimientoTeleton m, int numReg) {
        String aammdd = m.fecMaquina().format(DateTimeFormatter.ofPattern("yyMMdd"));
        boolean sinAutB24 = m.autB24() == null || m.autB24().isBlank() || m.autB24().equals("000000");
        char primerDigito = m.cuentaTarjeta().charAt(0);
        String bin = (primerDigito == '3' || primerDigito == '4') ? props.binVisaAmex() : props.binOtros();
        int diaJuliano = fechas.diaJuliano(m.fecMaquina());
        int digitoAnio = aammdd.charAt(1) - '0';               // WKS-I04-FV-DIGANIO
        var sinDv = new S244Record.Referencia23(7, bin, digitoAnio, diaJuliano,
                                                props.datoAdquirente(), 0).sinDigito();
        var ref = new S244Record.Referencia23(7, bin, digitoAnio, diaJuliano,
                                              props.datoAdquirente(), DigitoVerificador.mod10(sinDv));
        return new S244Record.Detalle(numReg, aammdd, 1, sinAutB24 ? 1 : 0, m.autB24(),
                                      Fixed.trunca(m.importe(), 9, 2), ref, m.cuentaTarjeta(),
                                      m.horaMovto());
    }
}
/** 940-CALCULA-DIGITO: Luhn sobre las primeras 22 posiciones de la referencia. */
public final class DigitoVerificador {

    public static int mod10(String referencia22) {
        int suma = 0;
        for (int i = 0; i < 22; i++) {
            int d = referencia22.charAt(i) - '0';
            int peso = (i % 2 == 0) ? 1 : 2;         // posiciones impares x1, pares x2
            int p = d * peso;
            suma += (p / 10) + (p % 10);              // WKS-DG1-PESO + WKS-DG2-PESO
        }
        int residuo = suma % 10;                      // DIVIDE ... BY 10 REMAINDER
        // 10 - residuo se guarda en PIC 9: con residuo 0 el 10 se trunca a 0.
        return (10 - residuo) % 10;
    }
}
/** 993-BUSCA-IMPORTE + 993-EDITA-IMPORTE sobre WS-IMPORTE-N PIC 9(10)V99. */
public final class AmexImporteFormatter {

    /** Quita ceros a la izquierda e inserta el punto antes de los dos decimales. */
    public static String formatea(BigDecimal importe) {
        String doce = "%012d".formatted(importe.setScale(2, RoundingMode.DOWN).unscaledValue());
        int i = 0;
        while (i < 12 && doce.charAt(i) == '0') i++;   // BUSCA-IMPORTE
        if (i == 12) {
            // En COBOL W77-I sigue avanzando fuera de la tabla OCCURS 12: importe cero no está soportado
            throw new ImporteAmexInvalidoException(importe);
        }
        StringBuilder sb = new StringBuilder();
        for (; i < 12; i++) {
            if (i == 10) sb.append('.');               // W77-I = 11 en base 1
            sb.append(doce.charAt(i));
        }
        return sb.toString();                          // ej. 1500.00 -> "1500.00"; 0.50 -> ".50"
        // 0.05 -> "5": se replica el defecto del fuente hasta que el SME decida (ver riesgo)
    }
}
