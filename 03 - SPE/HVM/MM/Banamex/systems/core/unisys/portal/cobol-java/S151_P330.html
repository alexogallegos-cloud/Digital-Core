/** REG-B72POSCONTA: imagen de S151B72POSCONTA. Llave compuesta de diez campos. */
public record PosicionContableRecord(
        int csi,                  // R72-SDO-KEYCSI   9(04) COMP
        LocalDate fecha,          // R72-SDO-KEYFEC   9(08) COMP, AAAAMMDD
        int banco, int moneda, int fideicomiso, int producto, int instrumento,
        String cuenta,            // R72-SDO-KEYCTA   9(12): String para conservar ceros
        int claveContable,        // R72-SDO-KEYCVEC  9(06)
        int secuencia,            // R72-SDO-KEYSEC   9(02)
        int naturaleza,           // R72-SDO-NATCTA   9(02)
        BigDecimal saldoAnterior, // S9(15)V9(02) -> escala 2, con signo
        BigDecimal cargos,
        BigDecimal abonos,
        BigDecimal saldoActual) {

    /** Invariante que el COBOL no valida: anterior + abonos - cargos = actual (según naturaleza). */
    public boolean cuadra(Naturaleza nat) {
        BigDecimal esperado = nat == Naturaleza.DEUDORA
                ? saldoAnterior.add(cargos).subtract(abonos)
                : saldoAnterior.add(abonos).subtract(cargos);
        return esperado.compareTo(saldoActual) == 0;
    }
}

@Component
@RequiredArgsConstructor
public class AperturaSaldosTasklet implements Tasklet {

    private final LibControlClient control;    // CONSISDIA, B04SISTEM, CONSISMEN en LIB-CONTROL
    private final SaldosDataSourceRouter saldos; // CHANGE ATTRIBUTE TITLE OF SALDOS + OPEN INQUIRY

    @Override
    public RepeatStatus execute(StepContribution sc, ChunkContext ctx) {
        JobParameters p = ctx.getStepContext().getStepExecution().getJobParameters();
        int sistema = p.getLong("sistema").intValue();                 // WKS-PARAM-SIS

        // VALUE OF MYSELF = 0: fecha de LIB-CONTROL; distinto de 0: fecha forzada (reproceso)
        LocalDate fecha = Optional.ofNullable(p.getLocalDate("fechaProceso"))
                .orElseGet(() -> control.fechaProceso(sistema));

        var b04 = control.sistema(sistema);                             // B04SISTEM
        if (b04.resultado() == 0 && b04.estatus() == 3) {
            var b03 = control.sistemaMensual(sistema);                  // CONSISMEN
            if (b03.resultado() == 1) {
                throw new ProcesoAbortadoException("NO EXITE EL NUMERO DE SISTEMA EN BASE MENSUAL");
            } else if (b03.resultado() != 0) {
                throw new ProcesoAbortadoException("ERROR EN BASE DE CONTROL");
            }
            if (!Set.of(1, 2, 5, 6).contains(b03.tipoBase())) {
                throw new ProcesoAbortadoException("EL SISTEMA NO UTILIZA BASE DE SALDOS");
            }
            if (b03.nombreBaseSaldos().isBlank()) {
                throw new ProcesoAbortadoException("NO EXISTE EN NOMB. DE BD SALDOS");
            }
            saldos.abreSoloLectura(b03.nombreBaseSaldos());           // OPEN INQUIRY
        } else {
            log.warn("Sistema {} sin estatus 3 o B04SISTEM con error: no se abre SALDOS", sistema);
        }
        ctx.getStepContext().getStepExecution().getJobExecution()
           .getExecutionContext().put("fechaProceso", fecha);
        return RepeatStatus.FINISHED;
    }
}
/** Copybook CRONOS2K (CPBAS/PD) reducido a java.time. */
public final class FechasA2k {

    /** A2K-BASE-YEAR VALUE 50: AA < 50 -> 20AA, AA >= 50 -> 19AA. */
    static final int PIVOTE = 50;
    private static final DateTimeFormatter AMD = new DateTimeFormatterBuilder()
            .appendValueReduced(ChronoField.YEAR, 2, 2, 1950)   // ventana 1950-2049 = pivote 50
            .appendPattern("MMdd").toFormatter();

    /** A2K-CONV-AMD-TO-CAMD, con los valores centinela del original. */
    public static OptionalInt aCamd(int amd) {
        if (amd == 0) return OptionalInt.empty();                 // 0 -> 0
        if (amd == 999999) return OptionalInt.of(99_999_999);     // fecha máxima
        LocalDate d = LocalDate.parse("%06d".formatted(amd), AMD);
        return OptionalInt.of(Integer.parseInt(d.format(DateTimeFormatter.BASIC_ISO_DATE)));
    }
}

@Bean

JdbcCursorItemReader<SaldoMensualRecord> b20Reader(
        @Qualifier("saldos") DataSource ds, ExtraccionProperties props,
        @Value("#{jobParameters['sistema']}") Long sistema,
        @Value("#{jobExecutionContext['fechaProceso']}") LocalDate fecha) {

    var filtro = props.filtros().get(sistema.intValue());   // p330.filtros.500: {producto: 1, instrumento: 3}
    String sql = "SELECT * FROM s151b20sdomencon"
            + (filtro == null ? "" : " WHERE key_am = ? AND key_prd = ? AND key_ins = ?")
            + " ORDER BY key_am, key_con, key_prd, key_ins, key_mon";   // orden del set B20SXSDOMENCON

    return new JdbcCursorItemReaderBuilder<SaldoMensualRecord>()
            .name("b20")
            .dataSource(ds)
            .sql(sql)
            .preparedStatementSetter(ps -> {
                if (filtro != null) {
                    ps.setInt(1, fecha.getYear() * 100 + fecha.getMonthValue());   // WKS-FECPRO-AAAAMM
                    ps.setInt(2, filtro.producto());
                    ps.setInt(3, filtro.instrumento());
                }
            })
            .rowMapper(SaldoMensualRecord.MAPPER)
            .build();
    // NOTFOUND en el primer FIND = estructura vacía: el step termina sin error con 0 registros.
    // Cualquier otro error de lectura: excepción (CALL SYSTEM DMTERMINATE).
}

@Component
@StepScope
public class CifrasControl implements ItemWriteListener<SaldoMensualRecord>, StepExecutionListener {

    private long registros;                                         // WS-ACUM-B20  9(08)
    private BigDecimal totalOcurrencia2 = BigDecimal.ZERO.setScale(2); // WS-TOTALB20SDOSD2

    @Override
    public void afterWrite(Chunk<? extends SaldoMensualRecord> items) {
        for (SaldoMensualRecord r : items) {
            registros++;
            // ADD B20-SDO-SD(2): en COBOL el total no tiene signo y guarda el valor absoluto
            // del resultado. Se calcula la suma algebraica y, aparte, la cifra compatible.
            totalOcurrencia2 = totalOcurrencia2.add(r.saldos().get(1).saldo());
        }
    }

    @Override
    public ExitStatus afterStep(StepExecution se) {
        se.getExecutionContext().putLong("registros", registros);
        se.getExecutionContext().put("totalAlgebraico", totalOcurrencia2);
        se.getExecutionContext().put("totalCompatibleCobol", Cobol.sinSigno(totalOcurrencia2));
        return se.getExitStatus();
    }
}

Riesgos de migración
Totales de control sin signo

Los acumuladores WS-TOTALB20SDOSD2 a WS-TOTALB72SDOACT son PIC 9(18)V9(02) sin signo (líneas 381-385) y reciben saldos S9(13)V9(02) y S9(15)V9(02) que pueden ser negativos. En COBOL el resultado de cada ADD pierde el signo al guardarse, de modo que el total impreso no es la suma algebraica de los saldos.

El total solo cubre una ocurrencia

Para B20, B21, B70 y B71 el total suma únicamente la ocurrencia 2 del arreglo de saldos (por ejemplo B20-SDO-SD(2), línea 1645); las otras nueve u once no entran a ninguna cifra de control. B80 no tiene total de importe.

Copia de registro por imagen de memoria

MOVE S151B20SDOMENCON TO REG-B20SDOMENCON (línea 1633) y equivalentes copian el registro DMSII completo sobre el registro del archivo. Si la descripción DASDL del data set cambia y el FD no, el archivo plano queda desalineado sin que el programa lo detecte.

Filtros compilados para el sistema 500

Producto 1 e instrumento 3 están en el código (líneas 1579-1582 y 1649-1652), al igual que los títulos S{sis}/FILE/EXTR/... y la ruta de respaldo de LIB-CONTROL ON CMEMP (línea 1442).

Pivote Y2K de dos dígitos

A2K-BASE-YEAR = 50 (línea 364): las fechas AAMMDD con año 00 a 49 se interpretan como 20xx y 50 a 99 como 19xx. Solo afecta la fecha de corrida del listado, que se obtiene de TODAYS-DATE en seis dígitos.
