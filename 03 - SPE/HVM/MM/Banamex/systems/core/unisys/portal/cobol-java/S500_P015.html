P015 lee la estructura de control B02CONTROL de S500BD01CAPTACION para saber qué archivos ya se aplicaron y en qué estado quedó la última dispersión. Si no hay un consolidado pendiente, mete al ORDENADO-CM cada archivo diario de S087, S152 (solo con dispersión CPE habilitada) y Tarjetas desde la última fecha aplicada hasta la fecha de línea, lo ordena y pide autorizaciones a la L010. Después lo dispersa por COMS en tres vueltas, con reintento de pendientes, espera de respuestas retrasadas y confirmación por registro, y marca el avance en B02-STA-DISP (1 al empezar a meter archivos, 2 con el ordenado listo, 3 al terminar el envío). Una rama paralela hace lo mismo con el archivo de saldos pendientes.

/** Sustituye SELECT OPTIONAL ... ACCESS MODE IS RANDOM / ACTUAL KEY. */
public interface RegistroDirecto<T> extends AutoCloseable {
    Optional<T> leer(long llave);          // READ ... INVALID KEY -> Optional.empty()
    void escribir(long llave, T registro); // WRITE ... INVALID KEY -> excepción
    long ultimoRegistro();                 // ATTRIBUTE LASTRECORD
}

@Component
@RequiredArgsConstructor
public class DispersionFiles {

    private final ResourceLoader loader;

    /**
     * OPTIONAL: la ausencia del archivo no aborta el paso; el programa pregunta
     * ATTRIBUTE RESIDENT antes de abrir. Aquí la ausencia es un Optional vacío.
     */
    public Optional<RegistroDirecto<String>> abrirOpcional(String titulo, int longitud) {
        Resource r = loader.getResource(titulo);
        return r.exists() ? Optional.of(new ArchivoLongitudFija(r, longitud)) : Optional.empty();
    }

    /** EAS-HUB, ORDENADO-CM y CTRLARCH no son OPTIONAL en el fuente. */
    public RegistroDirecto<String> abrirObligatorio(String titulo, int longitud) {
        return abrirOpcional(titulo, longitud)
                .orElseThrow(() -> new ArchivoNoResidenteException(titulo));
    }
}

public sealed interface OrdenadoRecord
        permits OrdenadoRecord.Header, OrdenadoRecord.Movimiento, OrdenadoRecord.TotalesArchivo,
                OrdenadoRecord.UltimosArchivos, OrdenadoRecord.Trailer {

    enum EstadoEnvio { NO_ENVIADO, ENVIADO, CONF_OK, CONF_NOK }

/** Niveles 88 de ORDENADO-TB-CLAVE. Los rangos se evalúan en el orden del fuente. */
public enum ClaseClave {
    P015, ELIMINACION, ABONOS, ABONOS_AUT, CARGOS, CARGOS_AUT, AUTORIZACION,
    COMISIONES, FIN_LINEA, PRE_LINEA, FIN_PRE_LINEA, RECHAZOS, NO_CONTABLE, OTRA;

    static ClaseClave de(int c) {
        if (c == 1 || c == 2 || (c >= 10 && c <= 16)) return P015;   // se solapa con 001-002 y 010
        if (c == 534)                    return ELIMINACION;
        if (c >= 1000 && c <= 1999)      return ABONOS;
        if (Set.of(3000, 3001, 3019, 3024).contains(c)) return ABONOS_AUT;
        if (c >= 2000 && c <= 2999)      return CARGOS;
        if (c == 4000 || c == 4009 || (c >= 4053 && c <= 4055)) return CARGOS_AUT;
        if (c == 526 || c == 527)        return AUTORIZACION;
        if (Set.of(584, 585, 613, 615, 700, 701).contains(c)) return COMISIONES;
        if (c == 8)                      return FIN_LINEA;
        if (c == 9)                      return PRE_LINEA;
        if ((c >= 500 && c <= 533) || (c >= 535 && c <= 999)) return NO_CONTABLE;
        return OTRA;
    }
}

@Configuration
@RequiredArgsConstructor
public class DispersadorJobConfig {

    @Bean
    Job dispersador(JobRepository repo, Step verificaVersiones, Step validaInicio,
                    Step generaOrdenado, Step dispersaOrdenado, Step ejecutaP121,
                    Step finPrograma, JobExecutionDecider decideP121) {
        return new JobBuilder("S500P015", repo)
                .start(verificaVersiones)                 // 10000000 + 20000000 + 30000000
                .next(validaInicio)                       // 40000000: falla el job si corre el batch
                .next(generaOrdenado)                     // 50000000 -> 50010000 / 50030000
                .next(dispersaOrdenado)                   // 50050000
                .next(decideP121)
                    .on("EJECUTA").to(ejecutaP121).next(finPrograma)
                .from(decideP121)
                    .on("*").to(finPrograma)              // 80000000-FIN-PGM
                .end()
                .build();
        // DISABLE PROGRAM S500P015 y el drenado de COMS lo hace el adaptador de mensajería al cerrar.
    }

    /** EAS fase 5: P121 solo si el archivo EAS semanal coincide con el de Tarjetas o S111 está decomisionado. */
    @Bean
    JobExecutionDecider decideP121(DispersionProperties props) {
        return (job, step) -> {
            var ctx = job.getExecutionContext();
            boolean coincide = Objects.equals(ctx.get("ultArchApli24"), ctx.get("ultArchApli04"));
            boolean decomS111 = ctx.getInt("decomS111", 0) == 1;
            return new FlowExecutionStatus(props.ejecutaP121() && (coincide || decomS111) ? "EJECUTA" : "OMITE");
        };
    }
}
@Component
@RequiredArgsConstructor
public class ValidaInicioTasklet implements Tasklet {

    private final ControlCaptacionRepository b02;   // S500BD01CAPTACION.B02CONTROL
    private final ControlPasoRepository b00;        // B00CTRLPASO

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        ControlCaptacion control = b02.findControl()
                .orElseThrow(() -> new ProcesoAbortadoException("B02CONTROL VACIO"));

        // HI de la L010 que suma 2 a B02-ELI-MEDACC para regresar a la forma anterior de postear
        boolean usaTipoAut = control.eliMedacc() <= 2;
        int eliMedacc = usaTipoAut ? control.eliMedacc() : control.eliMedacc() - 2;

        if (b00.findTipoProceso() != TipoProceso.LINEA) {          // NOT WS-TP-LINEA
            throw new DispersionNoPermitidaException(                // código 050110, error 2
                "NO PUEDO DISPERSAR.. SE ESTA CORRIENDO BATCH DEL S500, "
              + "FAVOR DE REINTENTAR CUANDO TERMINE PROCESO BATCH");
        }
        var jobCtx = ctx.getStepContext().getStepExecution().getJobExecution().getExecutionContext();
        jobCtx.put("staDisp", control.staDisp());
        jobCtx.put("fechaLinea", control.fechaLinea());
        jobCtx.put("nodo", control.numCsi());
        jobCtx.put("usaTipoAut", usaTipoAut);
        jobCtx.put("eliMedacc", eliMedacc);
        return RepeatStatus.FINISHED;
    }
}
/** 50010000-INI-O-REINI-DISPER: decide si se crea un ordenado nuevo o se retoma el pendiente. */
@Component
public class InicioOReinicioDecider implements JobExecutionDecider {

    @Override
    public FlowExecutionStatus decide(JobExecution job, StepExecution step) {
        var ctx = job.getExecutionContext();
        TipoDispersion tipo = (TipoDispersion) ctx.get("tipoDispersion");   // 1 movimientos, 2 saldo pendiente
        int estado = (tipo == TipoDispersion.MOVIMIENTOS)
                ? ctx.getInt("staDisp")         // B02-STA-DISP
                : ctx.getInt("staDispLsdo");    // B02-STA-DISP-LSDO

        // 2 = el ordenado ya se generó y quedó pendiente de enviar: solo se reabre y se dispersa.
        // Cualquier otro valor (0, 1 o 3) obliga a construirlo de nuevo a partir de las
        // fechas de últimos archivos aplicados que guarda la base.
        return new FlowExecutionStatus(estado == 2 ? "RETOMA_ARCHIVO" : "DESDE_CERO");
    }
}

@Component
@RequiredArgsConstructor
public class IngestaS087 {

    private final DispersionFiles files;
    private final DispersionProperties props;
    private final CalendarioBancario calendario;   // 50031700-PROYECTA-FECHA
    private final OrdenadoWriter ordenado;

    /** 50032000: un archivo por día, desde ultFecha+1 hasta la fecha de línea, sin saltar huecos. */
    public LocalDate procesa(LocalDate ultimaAplicada, LocalDate fechaLinea, int nodo) {
        LocalDate fecha = calendario.siguiente(ultimaAplicada);
        while (!fecha.isAfter(fechaLinea)) {
            String titulo = props.titulo(props.entradas().s087(), nodo, fecha);
            var archivo = files.abrirOpcional(titulo, DISPS087_LONGITUD);
            if (archivo.isEmpty()) {                              // NOT RESIDENT -> alarma y se detiene
                log.warn("SE INTENTA DISPERSAR EL ARCHIVO {} Y NO ESTA RESIDENTE", titulo);
                break;
            }
            try (var a = archivo.get()) {
                Disps087Header h = Disps087Header.parse(a.leer(1)
                        .orElseThrow(() -> new ArchivoVacioException(titulo)));
                if (!h.fechaArchivo().equals(fecha)) {            // fecha interna distinta a la del nombre
                    log.error("EL ARCHIVO DEL S087 CON FECHA {} TRAE INTERNAMENTE LA FECHA {}", fecha, h.fechaArchivo());
                    break;
                }
                ordenado.agregaArchivo(a);                        // 50032300-LEO-ARCH-S087
            }
            ultimaAplicada = fecha;                               // WS-ULT-ARCH-APLI(1), (3) y (5)
            fecha = calendario.siguiente(fecha);
        }
        return ultimaAplicada;
    }
}
@Service

public class EnvioOrdenadoService {

    private final OrdenadoRepository ordenado;           // ORDENADO-CM con acceso por llave
    private final DialogoGateway p010;                   // COMS -> Spring Integration / Kafka
    private final CandadoDispersion candado;
    private final RechazosService rechazos;
    private final DispersionProperties props;

    public void dispersa() {
        long ultimo = ordenado.ultimoRegistro();          // ATTRIBUTE LASTRECORD
        enviaPendientes(Vuelta.PRIMERA, ultimo);
        if (p010.detenido()) return;                      // WS-END-OF-PROGRAM <> 0
        enviaPendientes(Vuelta.SEGUNDA, ultimo);          // reintento de no confirmados
        p010.esperaRespuestas(Duration.ofMinutes(props.minutosEspera()));   // WKS-MINUTOS-A-ESPERAR = 3
        rechazos.arma(ordenado);                          // 002030-LEE-RECHAZOS (RechazosService)
        enviaPendientes(Vuelta.TERCERA, ultimo);          // envía los registros de rechazo
        p010.esperaRespuestas(null);
        if (p010.detenido()) return;                      // queda en 2: el reinicio solo reenvía
        candado.terminaDispersion();                      // STA-DISP = 3 y fechas 4 y 24
    }

    private void enviaPendientes(Vuelta v, long ultimo) {
        for (long llave = 2; llave <= ultimo; llave++) {  // la llave 1 es el header
            var r = ordenado.leer(llave).orElse(null);
            if (r == null || r instanceof OrdenadoRecord.Trailer) break;
            if (r instanceof OrdenadoRecord.Movimiento m && m.estado().ordinal() <= 1) {
                p010.envia(llave, m);                     // 001110-ENVIA-MENSAJE
            }
        }
    }

Riesgos de migración
Cobertura de OPTIONAL distinta a la regla

EAS-HUB (línea 47), ORDENADO-CM y CTRLARCH se declaran sin OPTIONAL; si EAS-HUB no existe al abrirse, el paso sí puede abortar. RN-S500-619 afirma que todos los archivos son OPTIONAL.

Error de CTRLARCH absorbido

En 50010000 (líneas 4824-4826), si falla la escritura del header de CTRLARCH (error 14) el programa pone el error en 0 y purga el archivo, y la dispersión continúa sin archivo de control para P104.

S152 condicionado a dispersión CPE

La línea 4937 solo procesa S152 cuando WS-DISPERSACPE = 1 (cambio Stefanini US-1115); en cualquier otro caso los archivos S152 esperan a la siguiente dispersión.

Actualización asimétrica de fechas aplicadas

Las posiciones 1, 2, 3, 5, 6, 20 y 23 de B02-ULT-ARCH-APLI se graban al marcar STA-DISP = 2; las posiciones 4 y 24 se graban hasta marcar 3 (líneas 7258-7265). Un corte entre ambos momentos deja la base con fechas de distinta época.

Esperas fijas y hardcodes

WAIT (20) segundos alrededor del mensaje con clave 12, el CSI 12 fijo en la ruta de S152, el título de la librería DISP y la versión del programa están compilados.
