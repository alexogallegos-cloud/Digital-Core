Al arrancar determina nodo y ambiente por el nombre del servidor, valida su propia versión y traduce el parámetro (SUBELINEA, BAJALINEA, SUBEPASO, SUBEBASE, INSTALAL020 y otros) a un número de paso. Calcula las banderas de calendario del día (último y primer hábil, fin de semana, quincena y cortes de días fijos) a partir de la utilería de fechas. Según el paso, envía a COMS los comandos para habilitar o deshabilitar los once programas en línea y, si hay Teletón, P045 y P046; corre P005 con el VALUE de la operación; o ejecuta respaldos e instalaciones. Ante la falla de un programa pide al operador reintentar, continuar, terminar o cambiar de versión.

@Component
@RequiredArgsConstructor
public class ArranqueLineaTasklet implements Tasklet {

    private final LineaProperties props;          // ya no se deduce del nombre del servidor
    private final VersionadorWfl versiones;        // *S006/OBJECT/CHECAME ("S500JLINEA")

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        if (props.ambiente() == null) {
            // El WFL asume UAT nodo 10 para un host desconocido sin mostrar el aviso; aquí se falla.
            throw new IllegalStateException("Perfil sin s500.linea.ambiente: no se asume UAT");
        }
        if (versiones.verifica("S500JLINEA") < 0) {
            throw new ProcesoAbortadoException("VERSION DE WFL/LINEA INCORRECTA");   // GO TO FIN
        }
        return RepeatStatus.FINISHED;
    }
}
/** CASE AUX OF: el primer token del parámetro se traduce a un paso numérico. */
public enum ComandoLinea {
    TERMINA(99), SUBELINEA(1), BAJALINEA(2), SUBEPASO(3), BAJAPASO(4), SUBEBASE(5),
    INSTALAL020(6), COPIAGUARD(7), INSTALAL039(8), INSTALAL040(9),
    SUBETELETON(10), BAJATELETON(11), BAJAESCALONADAP010(12), INSTALAL070(13);

    final int paso;
    ComandoLinea(int paso) { this.paso = paso; }

    /** SEPARAPARAM: "SUBEPASO P010" -> comando SUBEPASO, argumento P010 (AUX1). */
    public static Optional<Invocacion> parse(String parametro) {
        String[] t = parametro.trim().split("\\s+");
        return Arrays.stream(values()).filter(c -> c.name().equals(t[0])).findFirst()
                .map(c -> new Invocacion(c, t.length > 1 ? t[1] : ""));
    }
    public record Invocacion(ComandoLinea comando, String argumento) {}
}

// Un parámetro inválido (ELSE: PASO := 9999999999) ya no reabre el menú de consola:

@Bean
JobParametersValidator validaComando() {
    return p -> ComandoLinea.parse(p.getString("parametro", ""))
            .orElseThrow(() -> new JobParametersInvalidException("PARAMETRO INVALIDO"));
}

/** FECHAS(IFECHAHOY): banderas de calendario a partir de S000 FECHAS (día hábil anterior y siguiente). */
public record CalendarioProceso(LocalDate hoy, LocalDate anteriorHabil, LocalDate siguienteHabil,
                                boolean festivo) {

    /** DIA30: el siguiente día hábil ya está en otro mes. */
    public boolean ultimoHabilDelMes()  { return !YearMonth.from(hoy).equals(YearMonth.from(siguienteHabil)); }

    /** DIA1MES: el día hábil anterior estaba en otro mes. */
    public boolean primerHabilDelMes()  { return !YearMonth.from(hoy).equals(YearMonth.from(anteriorHabil)); }

    /** DIAFINSEM: el siguiente hábil cae en un día de la semana menor (la semana "da la vuelta"). */
    public boolean finDeSemana() {
        return siguienteHabil.getDayOfWeek().getValue() % 7 < hoy.getDayOfWeek().getValue() % 7;
    }

    /** DIA15: hoy es 15, es último hábil, o el 15 cae entre hoy y el siguiente hábil (inhábil). */
    public boolean corteQuincenal() {
        int d = hoy.getDayOfMonth(), p = siguienteHabil.getDayOfMonth();
        return d == 15 || ultimoHabilDelMes() || (15 > d && 15 < p);
    }
}
/** DIA5, DIA24, DIA25 y DIA23: el corte de un día fijo se adelanta al último hábil anterior. */
public boolean corteDelDia(int diaObjetivo) {
    int d = hoy.getDayOfMonth(), p = siguienteHabil.getDayOfMonth();
    return d == diaObjetivo || (diaObjetivo > d && diaObjetivo < p);
}

/** DIA5: múltiplos de 5 (5, 10, 15, 20, 25, 30) o fin de mes. */
public boolean corteMultiploDe5() {
    return hoy.getDayOfMonth() % 5 == 0 || ultimoHabilDelMes() || corteQuincenal()
            || IntStream.of(5, 10, 20, 25, 30).anyMatch(this::corteDelDia);    // el WFL omite el 20
}

public boolean corteDia24() { return corteDelDia(24); }
public boolean corteDia25() { return corteDelDia(25); }
/** DIA23: solo si no aplica el corte del 24 el mismo día. */
public boolean corteDia23() { return corteDelDia(23) && !corteDia24(); }

// ---------------------------------------------------------------------------
@Component
@RequiredArgsConstructor
public class HabilitaLineaTasklet implements Tasklet {

    private final LineaProperties props;
    private final AdministradorEnLinea coms;            // (S100)COMS/UTILITY/SENDTOMCS
    private final ArchivoMarcador marcadores;

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        RetryTemplate.builder()
                .infiniteRetry()                          // DO ... UNTIL TCOMS(TASKVALUE) = 1
                .fixedBackoff(Duration.ofSeconds(30))     // en el WFL el operador responde OK
                .retryOn(ComsNoDisponibleException.class)
                .build()
                .execute(r -> { coms.habilita(props.programas()); return null; });

        if (marcadores.existe(props.marcadorTeleton())) {          // Teletón activo
            coms.habilita(props.programasTeleton());               // P045 y P046 aparte
        }
        return RepeatStatus.FINISHED;
    }
}
@Bean
Job lineaS500(JobRepository repo, Step arranque, JobExecutionDecider porComando,
              Step habilitaLinea, Step p005Subida, Step p005Bajada, Step deshabilitaLinea,
              Flow subePaso, Flow bajaPaso, Flow respaldoBase, Flow instalaLibrerias /* ... */) {
    return new JobBuilder("S500-WFL-LINEA", repo)
            .start(arranque)
            .next(porComando)                                  // CASE PASO OF
              .on("SUBELINEA").to(habilitaLinea).next(p005Subida)          // (1): SUBETODOS; SUB005(26)
            .from(porComando)
              .on("BAJALINEA").to(p005Bajada).next(deshabilitaLinea)       // (2): SUB005(24); BAJATODOS
            .from(porComando).on("SUBEPASO").to(subePaso)
            .from(porComando).on("BAJAPASO").to(bajaPaso)
            .from(porComando).on("SUBEBASE").to(respaldoBase)
            .from(porComando).on("INSTALAL0*").to(instalaLibrerias)
            .from(porComando).on("TERMINA").end()                         // (99)
            .end()
            .build();
}

/** FALLA(PROGRAMA, IDD): R reintenta, C continúa, T termina, V reintenta con otra versión. */
public enum DecisionFalla { REINTENTO, CONTINUA, TERMINA, VERSION_DIFERENTE }

@RestController
@RequestMapping("/operacion/linea")
@RequiredArgsConstructor
public class OperacionLineaController {

    private final JobOperator operador;
    private final JobExplorer explorador;

    /** Sustituye a ACCEPT(MENSAJE): el operador responde por API o consola de operación. */
    @PostMapping("/{ejecucion}/decision")
    public void decide(@PathVariable long ejecucion, @RequestParam DecisionFalla decision,
                       @RequestParam(required = false) String version) throws Exception {
        switch (decision) {
            case REINTENTO -> operador.restart(ejecucion);
            case CONTINUA  -> operador.abandon(ejecucion);          // y se lanza el siguiente comando
            case TERMINA   -> operador.stop(ejecucion);
            case VERSION_DIFERENTE -> {
                versiones.cambia("S500", Objects.requireNonNull(version));   // S006/WFL/CAMBIAVER
                operador.restart(ejecucion);
            }
        }
    }
}

Riesgos de migración
Corte del día 20 omitido

La bandera DIA5 (líneas 306-312) no adelanta el corte del día 20 cuando es inhábil, aunque el comentario lo incluye. Los procesos condicionados a DIA5 no corren para ese corte en los meses en que el 20 cae en fin de semana o festivo.

Host desconocido tratado como UAT sin aviso

La rama ELSE del CASE HNAME (líneas 1677-1684) asigna nodo 10 y UAT y arma un mensaje que nunca se despliega. Un servidor nuevo de producción quedaría operando con banderas de UAT.

Dependencia del operador

Las esperas WAIT(MENSAJE,OK) cuando COMS no está activo y el menú de FALLA detienen el flujo hasta que alguien responda en consola; no hay tiempo máximo ni escalamiento.

Comparación de meses por un solo dígito

DIA30 y DIA1MES comparan solo el último dígito del mes (MOD 10); el resultado es correcto para fechas contiguas, pero la regla es frágil ante cualquier cambio.

Hardcodes de servidores, programas y archivos

Nueve nombres de servidor, la lista de once programas en línea, P045 y P046 de Teletón, el archivo S500BD06TELETON/CONTROL, los VALUE 24, 26 y 50 de P005 y la versión 24MTP005 están compilados en el WFL.
