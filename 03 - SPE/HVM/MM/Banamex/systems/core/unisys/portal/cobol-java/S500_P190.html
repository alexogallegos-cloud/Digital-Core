S500 · P190 a Java 21 Spring Boot

P190 toma por TASKVALUE el tipo de proceso (línea, batch o ambos), y opcionalmente fecha y CSI; si no vienen, los lee de S500B02CONTROL. Concentra hasta tres archivos de captación de 60 bytes en uno solo, ordena ese archivo y el de S151 por la misma llave y los recorre en paralelo como un merge: un movimiento que solo existe de un lado se lista como faltante en S500 o en S151, salvo importes en cero, no contables y migración. Al final imprime por cargos, abonos y no contables el número e importe de cada sistema y la diferencia S500 menos S151. Con proceso 0 repite el punteo para batch después del de línea.

/** TASKVALUE: proceso 0 = línea y después batch, 1 = línea (PLINEA), 2 = batch (PBATCH). */
public record PunteoParametros(int proceso, Optional<LocalDate> fecha, OptionalInt csi) {

    /** csiValidos viene de punteo.csi-validos en application.yml (hoy 04 y 10). */
    public static PunteoParametros de(int proceso, Optional<LocalDate> fecha, OptionalInt csi,
                                      Set<Integer> csiValidos) {
        if (proceso < 0 || proceso > 2) {
            throw new ParametroInvalidoException("ERR TPO PROC EN VALUE PROCESO=" + proceso);
        }
        if (fecha.isPresent() && csi.isPresent()) {                // 00004000-VALIDA-FEC-TASKVALUE
            boolean ok = fecha.get().getYear() > 1994 && csiValidos.contains(csi.getAsInt());
            if (!ok) throw new ParametroInvalidoException("ERR FECHA-CSI EN VALUE");
        }
        return new PunteoParametros(proceso, fecha, csi);
    }

    /** Sin fecha o sin CSI en el TASKVALUE se toman de S500B02CONTROL (W77-VEO-BASE = 1). */
    public boolean leeControl() { return fecha.isEmpty() || csi.isEmpty(); }

    public List<String> pasos() {
        return switch (proceso) {
            case 0  -> List.of("PLINEA", "PBATCH");   // 30000000-SEGDO-PUNTEO repite con batch
            case 1  -> List.of("PLINEA");
            default -> List.of("PBATCH");
        };
    }
}

        } catch (DataAccessException e) {
            log.warn("ERR B17CVESXFUN-FINDN USO NIVEIL 88 CVES PROT/COBRO", e);
            usaRangosFijos = true;                                     // no aborta; conserva lo ya cargado
        }
    }

    public boolean esProtCob(int clave) {
        Boolean enCatalogo = claves.get(clave);
        if (enCatalogo != null) return enCatalogo;
        return usaRangosFijos && props.rangosProtCob().stream().anyMatch(r -> r.contiene(clave));
    }

public enum Categoria { CARGO, ABONO, NO_CONTABLE }

/** WKS-I01-R01-DATOS / WKS-E02-R01-DATOS: misma estructura para S500 y S151. */
public record MovimientoPunteo(int tipoRegistro,           // 11, 12, 13 = migración
                               int sucursal, int caja, int autorizacion,   // LLAVE 9(12) COMP redefinida
                               String contrato,           // 9(12)
                               int producto, int instrumento, int moneda,
                               int clave,
                               BigDecimal importe,        // 9(10)V99 COMP -> escala 2
                               long referencia, long autS151) {

    public long llave() { return sucursal * 100_000_000L + caja * 1_000_000L + autorizacion; }

    public boolean esMigracion()  { return tipoRegistro >= 11 && tipoRegistro <= 13; }
    public boolean esContable()   { return clave >= 1000 && clave <= 4999; }
    public boolean esNoContable() { return clave >= 500 && clave <= 999; }
    /** W88-...-NC-TESO: 858-871, 875-884, 887-888, inyectados desde punteo.rangos-tesoreria. */
    public boolean esTesoreria(List<RangoClave> tesoreria) {
        return tesoreria.stream().anyMatch(r -> r.contiene(clave));
    }

    /** El orden de los IF del COBOL: abono, luego cargo, y todo lo demás es no contable. */
    public Categoria categoria() {
        if ((clave >= 1000 && clave <= 1999) || (clave >= 3000 && clave <= 3999)) return Categoria.ABONO;
        if ((clave >= 2000 && clave <= 2999) || (clave >= 4000 && clave <= 4999)) return Categoria.CARGO;
        return Categoria.NO_CONTABLE;
    }
}

public final class TotalesPunteo {

    private final Map<Categoria, Long> numero = new EnumMap<>(Categoria.class);
    private final Map<Categoria, BigDecimal> importe = new EnumMap<>(Categoria.class);

    /** 50003110-SUMA-S500 / 50003210-SUMA-S151. */
    public void suma(MovimientoPunteo m) {
        if (m.importe().signum() == 0 && m.esContable()) {
            return;                                    // contable en cero: no cuenta ni suma
        }
        numero.merge(m.categoria(), 1L, Long::sum);
        importe.merge(m.categoria(), m.importe(), BigDecimal::add);   // suma exacta, escala 2
    }

    public long numero(Categoria c)        { return numero.getOrDefault(c, 0L); }
    public BigDecimal importe(Categoria c) { return importe.getOrDefault(c, BigDecimal.ZERO.setScale(2)); }
}

/** Orden del merge: tipo de registro, clave, importe, contrato (los mismos del SORT). */
static final Comparator<MovimientoPunteo> ORDEN =
        Comparator.comparingInt(MovimientoPunteo::tipoRegistro)
                  .thenComparingInt(MovimientoPunteo::clave)
                  .thenComparing(MovimientoPunteo::importe)
                  .thenComparing(MovimientoPunteo::contrato);

public void puntea(PeekableItemReader<MovimientoPunteo> s500, PeekableItemReader<MovimientoPunteo> s151) {
    MovimientoPunteo a = siguiente(s500, totales500), b = siguiente(s151, totales151);
    while (a != null || b != null) {                           // UNTIL W77-EOF-ARCHS = 1
        int cmp = (a == null) ? 1 : (b == null) ? -1 : comparador.compara(a, b);
        if (cmp > 0) {                                         // W77-SW-1VEZ = 1: S151 va atrás
            faltaEnS151(b); b = siguiente(s151, totales151);
        } else if (cmp < 0) {                                  // W77-SW-1VEZ = 2: S500 va atrás
            faltaEnS500(a); a = siguiente(s500, totales500);
        } else {                                               // pareados: avanzan los dos
            a = siguiente(s500, totales500); b = siguiente(s151, totales151);
        }
    }
}

@Component
@RequiredArgsConstructor
public class ComparadorPunteo {

    private final CatalogoClavesProtCob protCob;
    private final PunteoProperties props;
    @Value("#{jobExecutionContext['proceso']}") private String proceso;   // PLINEA o PBATCH

    int compara(MovimientoPunteo a, MovimientoPunteo b) {
        int c = ORDEN.compare(a, b);
        if (c != 0) return c;
        int porLlave = Long.compare(a.llave(), b.llave());               // 50003052-VAL-AUTORIZ
        boolean esProtCob = "PLINEA".equals(proceso) && protCob.esProtCob(a.clave());
        return (porLlave != 0 && esProtCob)
                ? Long.compare(a.autS151(), b.autS151())                 // protección y cobranza: autorización S151
                : porLlave;
    }

    /** El faltante solo se reporta si es tesorería o si es contable, con importe y no es migración. */
    boolean seReporta(MovimientoPunteo m) {
        if (m.esTesoreria(props.rangosTesoreria())) return true;
        return !(m.importe().signum() == 0 || m.esNoContable() || m.esMigracion());
    }
}

                         long numS151, BigDecimal impS151) {

    public long difNumero()        { return numS500 - numS151; }
    public BigDecimal difImporte() { return impS500.subtract(impS151); }   // escala 2, exacta
    public boolean cuadra()        { return difNumero() == 0 && difImporte().signum() == 0; }
}

                totales151.numero(c), totales151.importe(c)))
        .toList();

Diferencia de totales sin control de cuadre

50006000-TOTALES-REP (líneas 1584-1621) calcula S500 menos S151 por categoría y la imprime, pero el programa no cambia su estado ni emite mensaje si la diferencia es distinta de cero. El descuadre solo se detecta si alguien lee el reporte.

Exclusiones del punteo que no aparecen en ningún lado

Un movimiento sin pareja con importe cero, clave no contable (500 a 999, salvo tesorería) o tipo de registro de migración (11, 12, 13) se descarta en silencio (líneas 1366-1386): no se lista ni se suma a las diferencias por faltante. Además, en los totales se omiten los movimientos contables con importe cero (líneas 1433-1435).
