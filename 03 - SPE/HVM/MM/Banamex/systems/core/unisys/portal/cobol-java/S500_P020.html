P020 (PROGRAM-ID LINCOMS) es la gemela secundaria de P010. Cada instancia se identifica por número de copia (1 a 5) y por host, y con eso decide qué actividades atiende, qué tipo de proceso reporta a S151 y si puede abrir el archivo TEF, que solo se procesa fuera de Monterrey. El bucle principal recibe mensajes de COMS y los enruta; además, el bloque DECLARATIVES atiende interrupciones del operador por TASKVALUE para monitor, trace, deshabilitar COMS, activar S151 y otros eventos. Para el posting ordena el archivo S02 por clave de transacción y clasifica cada clave en cargo, abono o no contable mediante rangos fijos.

/** WS-GEN-CLAVE-TRANS PIC 9(004) y sus niveles 88 (líneas 303-315). */
public enum ClaseClaveTransaccion {
    ELIM, ELIM_DEVOL, NO_CONTABLE, ABONO, CARGO, OTRA;

    public static ClaseClaveTransaccion de(int cve) {
        if (cve == 534) return ELIM;                                   // WS-88-ELIM
        if (cve == 575) return ELIM_DEVOL;                             // WS-88-ELIM-DEVOL
        if (enRango(cve, 1000, 1999) || enRango(cve, 3000, 3999)) return ABONO;   // WS-88-ABONO
        if (enRango(cve, 2000, 2999)) return CARGO;                    // WS-88-CARGO
        if (enRango(cve, 500, 999)) return NO_CONTABLE;                // WS-88-NOCONT: 500-999 menos 534 y 575
        return OTRA;                                                   // 0-499 y 4000-9999: ningún 88 aplica
    }

    /** Subconjunto de asignación TEF (759-762), independiente de la clase contable. */
    public static boolean esAsignaTef(int cve)        { return enRango(cve, 759, 762); }   // WS-88-ASIGNA-TEF
    public static boolean esAsignaTefMasivo(int cve)  { return cve == 759 || cve == 760; }
    public static boolean esAsignaTefSimple(int cve)  { return cve == 761 || cve == 762; }

    private static boolean enRango(int v, int a, int b) { return v >= a && v <= b; }
}
/** 00000000-ARMA-SA0-SA2-TERM: código de destino del encabezado SA0. */
static int codigoDestinoSa0(int claveTransaccion) {

    };

@Bean
Step ordenaPosting(JobRepository repo, PlatformTransactionManager tx,
                   ItemReader<PostingDetalle> postingEntrada,      // 70000000-IN-POST-SEC (I08-POSTING-I)
                   ItemWriter<PostingDetalle> postingSalida) {      // 70000000-OU-POST-SEC
    return new StepBuilder("ordenaPosting", repo)
            .<PostingDetalle, PostingDetalle>chunk(500, tx)
            .reader(new SortingItemReader<>(postingEntrada, ORDEN_S02))
            .writer(postingSalida)
            .build();
}

/** Llave S02 en el orden del SORT; todas DESCENDING. Importe como BigDecimal (9(14)V99). */
static final Comparator<PostingDetalle> ORDEN_S02 =
        Comparator.comparingInt(PostingDetalle::claveTransaccion)
                  .thenComparing(PostingDetalle::sucursal)
                  .thenComparing(PostingDetalle::contrato)      // String de 12 posiciones con ceros
                  .thenComparing(PostingDetalle::importe)
                  .thenComparingLong(PostingDetalle::secuencia)
                  .reversed();
