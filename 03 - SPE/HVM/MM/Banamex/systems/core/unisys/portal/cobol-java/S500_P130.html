@Component
@RequiredArgsConstructor
public class SecuenciaPasosValidator {

    private static final int PASO_PROPIO = 130;
    private final ControlPasoRepository ctrlPaso;     // S500B00CTRLPASO, lock al arranque
    private final ControlCaptacionRepository control; // S500B02CONTROL

    /** Devuelve la copia (1, 2 o 3) que debe correr esta instancia y marca su contador en 0. */
    @Transactional
    public CopiaAsignada asigna() {
        var b02 = control.findControl().orElseThrow();
        LocalDate fecBase = b02.fechaLote();          // B02-FECHA-LOTE
        LocalDate fecProx = b02.fechaLinea();         // B02-FECHA-LINEA
        boolean mensual = fecProx.getMonthValue() != fecBase.getMonthValue();   // WKS-ES-MENSUAL
        int pasoAnterior = mensual ? 305 : 127;       // W77-PASOANT
        int base = mensual ? 3 : 0;                   // W77-ID-NUM-COPIA-MAPLI inicial

        var b00 = ctrlPaso.lockForUpdate();           // CALL SYSTEM DMTERMINATE si falla el LOCK
        int p1 = b00.paso(1), p2 = b00.paso(2), p3 = b00.paso(3);
        boolean libre1 = p1 == 0 || p1 == PASO_PROPIO;
        boolean libre2 = p2 == 0 || p2 == PASO_PROPIO;

        int copia;
        if (p1 == pasoAnterior && p2 == pasoAnterior && p3 == pasoAnterior) copia = 1;
        else if (libre1 && p2 == pasoAnterior && p3 == pasoAnterior)         copia = 2;
        else if (libre1 && libre2 && p3 == pasoAnterior)                     copia = 3;
        else throw new ProcesoAbortadoException(
                "ERROR SEQ PROCESOS B00-NUM-PASO-01=%d B00-NUM-PASO-02=%d B00-NUM-PASO-03=%d TPO-PROCESO=%d"
                        .formatted(p1, p2, p3, b00.tipoProceso()));

        b00.ponPaso(copia, 0);                        // MOVE 0 TO B00-NUM-PASO-0n, STORE en la misma transacción
        return new CopiaAsignada(copia, base + copia, mensual, fecBase, fecProx);
    }
}

/**
 * 50100000-LEE-BASE: si la copia trae contrato de reinicio, lee desde él hasta el fin
 * y luego da una segunda vuelta desde el inicio hasta llegar otra vez a ese contrato.
 */
@Component
@StepScope
public class ContratoReader extends AbstractItemCountingItemStreamItemReader<Contrato> {

    private final ContratoRepository contratos;          // S500B03CONTRATOS, orden físico del dataset
    private final Long contratoInicial;                  // W77-CTO-INICIAL (B00-NUM-CTO-CPY)
    private final int copia;
    private Iterator<Contrato> it;
    private int vuelta = 1;                              // W77-Q-VUELTA

    @Override protected void doOpen() {
        it = (copia < 3 && contratoInicial != null)
                ? contratos.streamDesde(contratoInicial).iterator()
                : contratos.streamTodos().iterator();
    }

    @Override protected Contrato doRead() {
        if (!it.hasNext() && vuelta == 1 && copia < 3 && contratoInicial != null) {
            vuelta = 2;                                  // INICIO SEGUNDA VUELTA
            it = contratos.streamTodos().iterator();
        }
        if (!it.hasNext()) return null;
        Contrato c = it.next();
        if (vuelta == 2 && c.numContrato().equals(contratoInicial)) return null;   // W77-EOF-B03 = 1
        return c;
    }

    @Override protected void doClose() { it = null; }
}
@Component
@StepScope
@RequiredArgsConstructor
public class ParticionCopiaProcessor implements ItemProcessor<Contrato, ContratoEnProceso> {

    private final CatalogoInstrumentos instrumentos;   // TB-B05-* en memoria (50104000-DAME-INSTRUMENTO)
    @Value("#{jobExecutionContext['copia']}")   private int copia;
    @Value("#{jobExecutionContext['fecBase']}") private LocalDate fecBase;
    private final ContadoresBarrido contadores;        // W77-TOT-B03-MIGR, W77-TOT-B03-YAPROC

    @Override
    public ContratoEnProceso process(Contrato b03) {
        if (b03.migradoS111()) {                       // B03-MIGR-S111 = 1
            contadores.migrado();
            return null;
        }
        if (fecBase.equals(b03.fechaActBatch())) {     // B03-FECACT-BATCH = WKS-FEC-BASE: ya se cortó hoy
            contadores.yaProcesado();
            return null;
        }
        var inst = instrumentos.de(b03.producto(), b03.instrumento(), b03.moneda());
        int copiaAsignada = inst.procesaCopia();       // WS05-PROCESA-COPIA
        boolean esMio = (copiaAsignada >= 1 && copiaAsignada <= 3)
                ? copiaAsignada == copia
                : copia == 1;                          // instrumentos sin copia asignada: los toma la copia 1
        return esMio ? new ContratoEnProceso(b03, inst) : null;
    }
}

public class TarifasS080Client {

    private final S080TarifasPort l700;   // reemplaza la librería S080L700 (PERFORM 20000080-L700-TARIFAS)
    private final P130Properties props;  // tarifa-uma: 252, producto-uma: 9991, instrumento-uma: 2

    /** 4011000-CARGA-UMA-FEB-DIC: UMA diario vigente a la fecha base. */
    public Optional<BigDecimal> umaDiario(int paso, LocalDate fechaVigencia) {
        var consulta = TarifaConsulta.builder()
                .paso(paso)
                .fechaVigencia(fechaVigencia)              // WKS-FEC-BASE-06 (AAMMDD)
                .tarifa(props.uma().tarifa())              // 252
                .producto(props.uma().producto())          // 9991
                .instrumento(props.uma().instrumento())    // 2
                .moneda(0)
                .numRegistros(1)                           // WS-S080-0201-NUMREGS
                .build();
        var r = l700.consulta(consulta);
        if (r.resultado() != 0 && r.resultado() != 1) {
            return Optional.empty();                       // el fuente deja el UMA en 0 sin abortar
        }
        // COMPUTE sin ROUNDED: entero / 10^decimales, truncado a 2 decimales (PIC 9(06)V9(02))
        BigDecimal uma = new BigDecimal(r.campoEntero(1))
                .movePointLeft(r.campoDecimales(1))
                .setScale(2, RoundingMode.DOWN);
        return Optional.of(uma);
    }

@Component
@RequiredArgsConstructor
public class TraspasoBeneficenciaRule {

    /**
     * 50113600-TRASP-BENEF. Solo se evalúa el primer viernes del año, en moneda nacional
     * y con estado de beneficencia 3 u 8 (bloque de las líneas 6601-6607).
     */
    public boolean aplicaTraspaso(ContratoEnProceso c, ParametrosBeneficencia p) {
        if (!(p.esPrimerViernesAnio() && c.moneda() == 1
              && (c.staBenef() == 3 || c.staBenef() == 8)
              && p.mesesTraspaso() > 0 && p.saldoMaximo().signum() > 0)) {
            return false;
        }
        // WKS-IND-SDO-BENEF se fuerza a 0 al cargar parámetros (línea 5056): siempre se toma el saldo total.
        BigDecimal saldoTraspaso = (p.indicadorSaldo() == 0)
                ? c.saldoActualTotal()                       // W77-SDOACT-TOTAL  S9(13)V99
                : c.saldoCongeladoBenef();                   // B03-SDO-CONGBENE
        if (saldoTraspaso.signum() <= 0) return false;
        if (c.fechaBenef().isAfter(p.fechaTraspaso())) return false;      // B03-FECH-BENEF NOT > fecha
        if (c.fechaBenef().getYear() != p.anioValidacion()) return false; // año inmediato anterior

        BigDecimal tope = (c.fechaBenef().getMonthValue() > 1)
                ? p.saldoMaximo()                            // W77-SDO-MAX-BENF      (febrero a diciembre)
                : p.saldoMaximoEnero();                      // W77-SDO-MAX-BENF-ENE  (UMA del año anterior)
        return c.saldoCongeladoBenef().compareTo(tope) <= 0; // W77-TRP-CTA-BENEF = 1
    }
}

public class AsientosRendimientoService {

    private static final int CVE_REND_NETO  = 3000;   // claves comunes compiladas; candidatas a application.yml
    private static final int CVE_ISR        = 4009;
    private static final int CVE_REND_BRUTO = 809;
    private final MovimientosS151Buffer movimientos;   // 50116700-LLENA-MOVS-S151

    @Transactional(propagation = Propagation.MANDATORY)
    public void asienta(ContratoEnProceso c, Capitalizacion cap, BigDecimal tcVenta,  // W77-TCAMBIO-VTA 9(04)V9(04)
                        CifrasControl cifras) {
        boolean dolares = c.moneda() == 5;
        if (cap.rendNeto().signum() > 0) {
            c.marcaPagoRendimiento();                                    // W77-HAY-PAGO-REND = 1
            movimientos.agrega(c, MovimientoS151.de(CVE_REND_NETO, cap.rendNeto(), c.esquemaContable()));
            if (dolares) {
                cifras.cargoMn(aPesos(cap.rendNeto(), tcVenta));         // WKS-VAL-CGOMN
                cifras.cargoDl(cap.rendNeto());                          // WKS-VAL-CGODLS
            }
        }
        if (cap.impuesto().signum() > 0) {
            movimientos.agrega(c, MovimientoS151.de(CVE_ISR, cap.impuesto(), c.esquemaContable()));
            if (dolares) {
                BigDecimal mn = aPesos(cap.impuesto(), tcVenta);
                cifras.cargoMn(mn); cifras.abonoMn(mn); c.historico().sumaIsrValorizadoMn(mn);   // B06-IMPTO-VALMN
                cifras.cargoDl(cap.impuesto()); cifras.abonoDl(cap.impuesto());
            }
        }
        if (cap.rendBruto().signum() > 0) {
            movimientos.agrega(c, MovimientoS151.de(CVE_REND_BRUTO, cap.rendBruto(), c.esquemaContable()));
        }
    }

    /** COMPUTE W77-VAL-PUENTE ROUNDED = importe * tipo de cambio: redondeo a 2 decimales HALF_UP. */
    static BigDecimal aPesos(BigDecimal usd, BigDecimal tc) {
        return usd.multiply(tc).setScale(2, RoundingMode.HALF_UP);
    }
}

    /** 50116400-SUMA-COMISIONES para una posición WS-IND-CVE de la tabla de comisiones calculadas. */
    @Transactional(propagation = Propagation.MANDATORY)
    public void aplica(ContratoEnProceso c, ComisionCalculada com) {

            c.cargaSaldo(com.conIva());             // SUBTRACT WS-CAP-COMICONIVA FROM B03-SDO-ACTUAL

                mov = mov.conLeyenda("COMISION POR DEPOSITO EXTRA MENSUAL", /*sustituye*/ true);

@Component

public class MovimientosS151Buffer {

    private static final int TAMANO_LOTE = 30;           // OCCURS de WS-S151-0101-*-I
    private final S151RegistraPort s151;                 // librería S151REGISTRA
    private final List<MovimientoS151> lote = new ArrayList<>(TAMANO_LOTE);

    /** 50116700-LLENA-MOVS-S151: al llenar el arreglo se envía y se reinicia. */
    public void agrega(ContratoEnProceso c, MovimientoS151 mov) {
        if (lote.size() == TAMANO_LOTE) {
            s151.envia(List.copyOf(lote));               // 50122000-ENVIA-MOVSS151
            lote.clear();
        }
        // INDLEY: 2 = leyenda adicional, 3 = sustituye la leyenda de la clave (66-14)
        lote.add(mov.leyenda() == null ? mov
                : mov.conIndicadorLeyenda(mov.sustituyeLeyenda() ? 3 : 2));
        c.generaPunteo(mov);                             // 50117700-GENERA-PUNTEO
        c.acumulaClave(mov);                             // 50116800-ACUMULA-CLAVES
    }

    /** El remanente se envía al cerrar el contrato o el step. */
    public void vacia() {
        if (!lote.isEmpty()) { s151.envia(List.copyOf(lote)); lote.clear(); }
    }

Bypass contable controlado por nombre de equipo

Las banderas de TASKVALUE (líneas 2895-2903) permiten correr sin las librerías S050, S016, S408, S151 o S500. Fuera de equipos PBA, SIT o UAT el programa aborta (líneas 3155-3166), pero esa clasificación depende del hostname. Con WKS-SIN-LBS151 = 1 (línea 5313) se actualizan saldos sin asientos en S151.

Indicador de saldo de Beneficencia forzado a cero

En la línea 5056 se reemplazó la carga del indicador desde L710 por un MOVE 0 (cambio EVM-2019). La rama que traspasa solo el saldo congelado (líneas 7166-7169) nunca se ejecuta.

Tope de Beneficencia asignado fuera del fuente

W77-SDO-MAX-BENF y W77-SDO-MAX-BENF-ENE se leen pero no se asignan en P130; su cálculo debe estar en un copybook no incluido. Si quedaran en cero, el traspaso no se evalúa (condición de la línea 6606).

Validación de fechas de control comentada

El chequeo de que B02-FECHA-LOTE sea menor que B02-FECHA-LINEA (líneas 3013-3023) está comentado; el modo mensual se decide solo comparando meses.

Partición dependiente del catálogo

La copia que procesa cada contrato depende de WS05-PROCESA-COPIA del catálogo de instrumentos; valores fuera de 1 a 3 caen en la copia 1 sin aviso, y las tres copias leen el maestro completo.
