/** Las etiquetas de archivo y los títulos de reporte se resuelven desde aquí. */
record ContextoProceso(LocalDate fecha, int csi) {
    String etiqueta(String plantilla) {
        return plantilla.replace("{csi}", "%02d".formatted(csi))
                        .replace("{fecha}", fecha.format(DateTimeFormatter.BASIC_ISO_DATE));
    }

/** REG-CANCEL1: 43 bytes, tipo de registro 9(02) y dos vistas REDEFINES. */
sealed interface CancelacionCopia permits CancelacionCopia.Header, CancelacionCopia.Detalle {
    record Header(int sistema, LocalDate fecProceso) implements CancelacionCopia {}
    record Detalle(int producto, int instrumento, int moneda, int sucursal,
                   String contrato,                 // 9(12): String para conservar ceros
                   BigDecimal importe)              // 9(12)V99 -> escala 2
            implements CancelacionCopia {}
}

@Component
@StepScope
class UnificaCopiaProcessor implements ItemProcessor<CancelacionCopia, Cancelacion> {

    @Value("#{jobExecutionContext['proceso']}") private ContextoProceso proceso;
    private boolean headerLeido;

    @Override
    public Cancelacion process(CancelacionCopia r) {
        if (!headerLeido) {                     // el primer registro siempre se trata como header
            headerLeido = true;
            var h = (CancelacionCopia.Header) r;
            if (!h.fecProceso().equals(proceso.fecha())) {
                throw new ProcesoAbortadoException("FEC.HEAD.C1 DIFERENTE A PRO");
            }
            return null;
        }
        var d = (CancelacionCopia.Detalle) r;   // todo lo demás es detalle, sin revisar el tipo
        return new Cancelacion(d.producto(), d.instrumento(), d.moneda(),
                               d.sucursal(), d.contrato(), d.importe());
    }
}

    private final ClavesPorFuncionRepository b17;   // S500B17CVESXFUN
    private Map<Integer, Integer> indicador;         // WS-B17-IND-ACUM-VAL(clave)

    @PostConstruct
    void carga() {
        indicador = b17.findAll().stream()
                .filter(r -> r.claveTrans() > 0 && r.claveTrans() < 9999)   // misma guarda del fuente
                .collect(Collectors.toMap(ClavePorFuncion::claveTrans,
                                          ClavePorFuncion::indAcumVal,
                                          (a, b) -> b));   // si se repite la clave gana la última
    }

    boolean acumula(int clave) { return indicador.getOrDefault(clave, 0) == 1; }
}

@Component
@RequiredArgsConstructor
class TotalesS151Client {

    private final TotXCveTraClient s151;   // TOTXCVETRAS500, FUNCION 06

    /**
     * En el fuente un resultado distinto de 0, 3, 4, 5 o 99 detiene el paso y espera
     * un ACCEPT del operador: "C" continúa sin los totales, "R" recarga la librería y reintenta.
     * Aquí el reintento es automático y la continuación es una política explícita.
     */
    @Retryable(retryFor = S151NoDisponibleException.class,
               maxAttemptsExpression = "${p110.s151.reintentos:3}",
               backoff = @Backoff(delay = 2000))       // WAIT (2)
    ConsultaTotales consulta(ClaveConsulta k) {
        return s151.totalesPorClave(k);
    }

    @Recover
    ConsultaTotales sinS151(S151NoDisponibleException e, ClaveConsulta k) {
        // Equivale a responder "C": se sigue sin totales de S151 y el cuadre mostrará diferencias.
        log.error("S151 no respondió para {}; se continúa sin sus totales", k);
        return ConsultaTotales.vacia();
    }
}
@Component
@StepScope
@RequiredArgsConstructor
class TotalesS151Processor implements ItemProcessor<ConsultaTotales, List<TotalPorClave>> {

    private static final int IDEN_S151 = 151;
    private static final int MAX_CLAVES = 100;            // WS-S151-0301-CLAVE-S OCCURS 100
    private final CifrasControl cifras;                    // 50004330-ACUM-151

    @Override
    public List<TotalPorClave> process(ConsultaTotales c) {
        var salida = new ArrayList<TotalPorClave>();
        for (var r : c.renglones().subList(0, Math.min(MAX_CLAVES, c.renglones().size()))) {
            if (r.clave() == 0) break;                     // fin de claves
            if (r.paso() >= 11 && r.paso() != 90) continue;   // solo pasos < 11 o 90
            cifras.acumula151(r);
            // Consulta reetiquetada: 66/15 se pidió a S151 en moneda 1 y se registra como moneda 5.
            int moneda = c.reetiquetada() ? 5 : c.moneda();
            salida.add(new TotalPorClave(c.bancos(), c.producto(), c.instrumento(), moneda,
                    r.clave(), IDEN_S151, r.numMovs(),
                    r.impMovs(), BigDecimal.ZERO, BigDecimal.ZERO));   // TCVE-IMPFE y TCVE-IMPFR en 0
        }
        return salida;
    }
}
    /**
     * TOTCVE se ordena por banco, producto, instrumento, moneda, clave e identificador,
     * así que para una misma clave llega primero 151 (S151) y después 500 (S500).
     */
    private Diferencia compara(TotalPorClave s151, TotalPorClave s500) {
        BigDecimal impS500 = s500.imp().add(s500.impFe()).add(s500.impFr());   // WS-IMPPASO2
        BigDecimal impS151 = s151.imp().add(s151.impFe()).add(s151.impFr());   // WS-IMPPASO1
        long numDif = (long) s500.num() - s151.num();
        BigDecimal impDif = impS500.subtract(impS151);   // con signo y sin truncar

        // Comportamiento del fuente, documentado para la prueba en paralelo:
        // WS-TCVE-IDIF es PIC 9(12)V99 sin signo y recibe una resta de dos 9(14)V99.
        BigDecimal impDifLegado = impDif.abs().remainder(new BigDecimal("1000000000000"));

        return new Diferencia(s500.clave(), numDif, impDif, impDifLegado);
    }

@RequiredArgsConstructor
class IncidenteWriter implements ItemWriter<Diferencia>, StepExecutionListener {

    private final FlatFileItemWriter<String> i06;      // I06-ARCH-INCIDENTE
    private final List<Diferencia> diferencias = new ArrayList<>();

    @Override
    public void write(Chunk<? extends Diferencia> chunk) {
        chunk.forEach(d -> {
            if (d.clave() > 999 && d.impDif().signum() != 0) {   // solo claves contables
                diferencias.add(d);                          // el fuente conserva solo la última
            }
        });
    }

    @Override
    public ExitStatus afterStep(StepExecution se) {
        diferencias.forEach(d -> i06.write(Chunk.of(
            " DIFERENCIA ENCONTRADA EN LA CLAVE %d POR UN IMPORTE %s".formatted(d.clave(), d.impDif()))));
        i06.write(Chunk.of(Trailer.incidentes(diferencias.size())));   // WKS-I06-05-TOTREG
        if (!diferencias.isEmpty()) {
            se.getJobExecution().getExecutionContext().putInt("parResult", 99);
        }
        return se.getExitStatus();
    }
}

Riesgos de migración
Incidentes posiblemente nunca registrados

90009999-ARCH-INC-DIF (líneas 10245-10247) compara WS-CVE-IMP3, un campo numérico editado, contra 0. Según ANSI 85 la comparación es alfanumérica y los espacios de relleno la hacen falsa para importes menores a cien mil millones. Debe confirmarse con el compilador COBOL85 de MCP o con el histórico de archivos I06.

Solo la última diferencia llega al archivo de incidentes

WKS-INC-CVE-CVE y WKS-INC-CVE-IMP son campos únicos que se sobrescriben con cada diferencia; 90009999-ESCRIBE-INCIDENTE se ejecuta una vez al cierre (línea 10011). El trailer cuenta todas las diferencias, pero el detalle muestra solo una.

Diferencia sin signo y con posible truncamiento

WS-TCVE-IDIF es PIC 9(12)V99 sin signo y recibe la resta de dos PIC 9(14)V99. El listado no distingue si S500 está por arriba o por debajo de S151.

Intervención manual del operador

Ante un error de S151 el paso queda en ACCEPT esperando C o R (50004302-DECIDE). Con C el cuadre continúa sin los totales de S151.

Detalle de copias sin validar tipo de registro

20000720-VAL-DETC1 y sus equivalentes copian cualquier registro posterior al header sin revisar WS-CAN-TR1C1.

Hardcodes

Paso previo 105, tope de 30 productos, rango de claves 1 a 9998, reetiquetado del producto 66, instrumento 15 y moneda 5, rangos de claves 1000 a 4999 para las cifras de control y pasos menores a 11 o 90 de S151. Se trasladan a application.yml.
