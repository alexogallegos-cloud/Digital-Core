/** E_CARGADEF: un valor fuera de rango no se rechaza, se sustituye por el default. */
@ConfigurationProperties("p810")
public record P810Properties(int hhmmCorte, int hhmmProIni, int hhmmProFin, int delayPro,
                             String dirFam, String logFam, String fwdrName,
                             boolean sndToSplk, boolean grabaLog, String libCtlVersion,
                             List<Integer> sistemas) {

    public int corteIni()   { return enRango(hhmmCorte, 1, 1500, 5) * 100; }        // HORA_INI
    public int corteFin()   { return corteIni() + 30000; }                           // + 3 horas
    public int ventanaIni() { return enRango(hhmmProIni, 1, 1500, 800) * 100; }
    public int ventanaFin() { return enRango(hhmmProFin, 1500, 2359, 1900) * 100 + 1000; } // +10 min implícitos

    private static int enRango(int v, int min, int max, int porDefecto) {
        return (v < min || v > max) ? porDefecto : v;
    }
}

/** CONTROLA: latido cada DELAY_PRO minutos (RSLT_WAIT = 1, por timeout). */
@Component
@RequiredArgsConstructor
public class MonitorTotales {

    private final DisparadorTotales disparador;
    private final AtomicBoolean activo = new AtomicBoolean(true);   // CONTROLA_PROCESO

    @Scheduled(fixedDelayString = "#{${p810.delay-pro} * 60 * 1000}")
    void latido() {
        if (!activo.get()) return;
        try {
            disparador.cambioHora(false);   // CAMBIO_HORA(FALSE)
            disparador.finDia();            // FIN_DIA
        } catch (RuntimeException e) {      // ON ANYFAULT ... GO CONT_CONTROLA
            log.error("FALLA EN PROC.: CONTROLA", e);   // sin reactivar 'activo' (ver riesgo)
        }
    }

    void detiene() { activo.set(false); }
}

/** RSLT_WAIT = 2 (por HI): los comandos de consola pasan a un endpoint de Actuator. */
@Component
@Endpoint(id = "p810")
@RequiredArgsConstructor
public class ConsolaP810Endpoint {

    private final MonitorTotales monitor;
    private final DisparadorTotales disparador;
    private final Bitacora bitacora;
    private final Parametros parametros;

    @WriteOperation
    public String hi(int valor) {
        return switch (valor) {
            case 1  -> AYUDA;
            case 2  -> { bitacora.cierraYReabre(); yield "EL LOG FUE CERRADO Y REABIERTO..."; }
            case 3  -> { disparador.cambioHora(true); yield "SE FORZA PROCESAMIENTO ..."; }
            case 4  -> { monitor.detiene(); yield "RECIBI 'HI 4' PARA TERMINAR !"; }
            case 10 -> { parametros.desligaLibrerias(); yield "ME DESLIGUE DE LIBRERIAS"; }
            case 21 -> { parametros.recargaDefinfo(); yield "SE CARGA ARCHIVO DEFINFO !"; }
            case 22 -> { parametros.recargaSucursales(); yield "SE CARGA ARCHIVO TABLASUCS !"; }
            default -> "";                   // cualquier otro valor se ignora
        };
    }
}
@Component
@RequiredArgsConstructor
public class DisparadorTotales {

    private final P810Properties props;
    private final Clock reloj;
    private final JobLauncher lanzador;
    private final Job procesaTot;
    private final ControlDelDia control;
    private final Bitacora bitacora;

    private LocalDateTime horaActual;     // DT_ACT.[47:30]: fecha y hora, sin minutos
    private LocalDate fechaProc;          // FECHA_PROC

    /** CAMBIO_HORA(FORZA_PROCESA_TOT). */
    public synchronized void cambioHora(boolean forzado) throws Exception {
        LocalDateTime ahora = LocalDateTime.now(reloj);
        LocalDateTime hora = ahora.truncatedTo(ChronoUnit.HOURS);
        if (horaActual == null) horaActual = hora;                       // AL INICIO
        int min = ahora.getMinute();
        if (!hora.equals(horaActual) || min == 15 || min == 30 || min == 45 || forzado) {
            horaActual = hora;
            int hhmmss = Hhmmss.de(ahora.toLocalTime());
            if (hhmmss >= props.ventanaIni() && hhmmss <= props.ventanaFin()) {
                bitacora.inf(forzado ? "--> SE FORZA RUTINA DE PROCESAMIENTO !" : "--> HAY CAMBIO DE HORA !");
                lanzador.run(procesaTot, new JobParametersBuilder()
                        .addLocalDateTime("disparo", ahora).toJobParameters());   // PROCESA_TOT
            } else if (forzado) {
                bitacora.inf("--> SE FORZO PROCESO DE TOTALES, PERO NO ESTAMOS EN HORARIO !");
            }
        }
    }

    /** FIN_DIA: solo dentro de la ventana de corte y solo si cambió la fecha. */
    public synchronized void finDia() {
        LocalDateTime ahora = LocalDateTime.now(reloj);
        int hhmmss = Hhmmss.de(ahora.toLocalTime());
        LocalDate hoy = ahora.toLocalDate();   // el ALGOL compara YYMMDD; sin pivote porque solo es igualdad
        if (hhmmss > props.corteIni() && hhmmss < props.corteFin() && !hoy.equals(fechaProc)) {
            fechaProc = hoy;
            control.inicializa(hoy);           // INICIALIZA_CTL
            bitacora.rota(props.grabaLog());   // CLOSE(LOG,LOCK) + INICIALIZA_LOG
        }
    }
}
@Configuration
public class ProcesaTotJobConfig {

    /** PROCESA_TOT: los pasos no revisan el resultado del anterior; se replica con on("*"). */
    @Bean
    Job procesaTot(JobRepository repo,
                   Step cargaSistemasDelDia,   // CARGA_B01SISDIA -> CONSISDIA de S151L001CTL
                   Step cargaConceptos,        // CARGA_B09CONXSIS: 30 renglones, PRODUCTO/INSTRUMENTO "0000"
                   Step totales264,            // BUSCA_B10MOVDIA264 + GENERA_ARCH
                   Step totales500) {          // BUSCA_B10MOVDIA500 + GENERA_ARCH
        return new JobBuilder("S151P810-PROCESA-TOT", repo)
                .start(cargaSistemasDelDia).on("*").to(cargaConceptos)
                .from(cargaConceptos).on("*").to(totales264)
                .from(totales264).on("*").to(totales500)
                .end()
                .build();
        // DELINKLIBRARY(LIB_S151L001CTL) en cada corrida: el cliente de la librería
        // de control se crea por ejecución (@JobScope) para tomar la versión vigente.
    }
}

/** INICIO: parámetros, sucursales, control, bitácora y dos prerrequisitos duros. */
@Component
@RequiredArgsConstructor
public class ArranqueP810 implements ApplicationRunner {

    private final Parametros parametros;
    private final SplunkForwarder splunk;            // LINKFWDR de OMNISPLUNK_LIB
    private final LibreriaControlClient libCtl;      // INI_LIBCTL
    private final ControlDelDia control;
    private final Bitacora bitacora;
    private final Clock reloj;

    @Override
    public void run(ApplicationArguments args) {
        parametros.recargaDefinfo();                 // CARGADEF
        parametros.recargaSucursales();              // CARGASUCS
        control.inicializa(LocalDate.now(reloj));    // INICIALIZA_CTL
        if (parametros.grabaLog()) bitacora.abre();  // INICIALIZA_LOG
        if (!splunk.liga()) throw new IllegalStateException("NO SE PUDO LIGAR CON SPLUNK");
        if (!libCtl.inicializa()) throw new IllegalStateException("NO SE PUDO INICIALIZAR LA LIB S151L001CTL");
        // a partir de aquí MonitorTotales empieza a latir
    }
}

@Repository
@RequiredArgsConstructor
public class MovDiaRepository {

    private final NamedParameterJdbcTemplate jdbc;

    /** CASE NUMARC OF: el dataset depende del día de la semana, no de la fecha de proceso. */
    static Optional<String> tablaDelDia(int numArc) {
        int n = (numArc > 5) ? numArc - 5 : numArc;      // GAEC 251008: 6..10 se reducen a 1..5
        return switch (n) {
            case 1 -> Optional.of("csisuccaj_b03");
            case 2 -> Optional.of("csisuccaj_b13");
            case 3 -> Optional.of("csisuccaj_b23");
            case 4 -> Optional.of("csisuccaj_b33");
            case 5 -> Optional.of("csisuccaj_b43");
            default -> Optional.empty();                 // ELSE: vacío, sin error (ver riesgo)
        };
    }

    /** FIND FIRST ... AT (Bnn-MOV-SUCINI = suc) + IF de igualdad = WHERE por sucursal. */
    public void recorre(int sistema, int numArc, List<Integer> sucursales,
                        Consumer<RenglonCsiSucCaj> destino) {
        String tabla = tablaDelDia(numArc).orElse(null);
        if (tabla == null) return;
        for (int suc : sucursales) {                    // FOR POS_SUC ... UNTIL ULT_SUCS
            jdbc.query(("select mov_sucini, mov_cajini, mov_moneda, mov_bancos, mov_secren,"
                      + " num_cr_01, imp_cr_01, num_ab_01, imp_ab_01 /* ... hasta _10 */"
                      + " from movdia%d.%s where mov_sucini = :suc").formatted(sistema, tabla),
                    Map.of("suc", suc),
                    (RowCallbackHandler) rs -> destino.accept(RenglonCsiSucCaj.de(rs)));
        }
    }

@Component
@JobScope
public class TablaScbm {

    static final int MAX_SCBM = 3000;
    private final SortedMap<ClaveScbm, AcumuladorScbm> tabla = new TreeMap<>();

    public void registra(RenglonCsiSucCaj r) {
        var clave = new ClaveScbm(r.sucursal(), r.caja(), r.bancos(), r.moneda());
        var acc = tabla.computeIfAbsent(clave, k -> {
            if (tabla.size() >= MAX_SCBM) {             // el ALGOL no valida: falla por índice
                throw new CapacidadScbmExcedidaException(MAX_SCBM);
            }
            return new AcumuladorScbm();
        });
        acc.cuentaPagina();                             // SCBM_CTL_PAG + 1
        int secren = r.secren();
        if (secren > AcumuladorScbm.MAX_SCBM_PAG) {
            return;                                     // SECREN > 8: se descartan los 10 conceptos
        }
        for (int i = 1; i <= AcumuladorScbm.MAX_SCBM_CON_PAG; i++) {
            int pos = secren * AcumuladorScbm.MAX_SCBM_CON_PAG + i;
            acc.asigna(pos, r.numCr(i), r.impCr(i), r.numAb(i), r.impAb(i));   // := reemplaza
            acc.sumaTotales(r.numCr(i), r.impCr(i), r.numAb(i), r.impAb(i));   // + acumula
        }
    }

    public SortedMap<ClaveScbm, AcumuladorScbm> vista() { return Collections.unmodifiableSortedMap(tabla); }
}

            }
            emiteUno(linea("3", e, r.siguiente(), k, "99", "TOTALES SCBM   ",
                           acc.totNumCr(), acc.totImpCr(), acc.totNumAb(), acc.totImpAb()), r);
        });
        emiteUno(trailer(e, r.siguiente()), r);
        return r;                     // a diferencia del ALGOL, el error se propaga al Step
    }

        archivo.escribe(csv);                         // WRITE(CON, MR_CON, E_ARY_STS)
        if (splunk.envia(csv)) r.enviado();           // RSLT_IQ = 1
        else r.error();                               // cuenta todos los fallos, no solo el último código
    }

    /** Importes con escala 2 y punto decimal fijo; el ALGOL los imprime con ST(REAL,*). */
    private static String imp(BigDecimal v) { return v.setScale(2, RoundingMode.UNNECESSARY).toPlainString(); }

Riesgos de migración
Importes en punto flotante

B99_IMP_CR, B99_IMP_AB y los acumuladores de TAB_SCBM son REAL (líneas 2610-2611 y 740-747). Los totales de importe por llave se acumulan en punto flotante y se publican con ST(...,*), lo que puede introducir diferencias de redondeo en sumas grandes. La versión Java usa BigDecimal.

Banco truncado en la tabla

SCBM_CTL_BAN ocupa 24 bits (línea 734) mientras la llave usa 10 dígitos de banco. Un valor mayor a 16,777,215 rompe la búsqueda de la llave y duplica renglones.

Reporte vacío sin señal

Sin TABSUCS, con NUMARC fuera de 1 a 10 o con falla de CONSISDIA, el programa envía a Splunk solo encabezado y trailer, o datos con sistema y CSI en cero, sin registrar error.

Errores de envío que no se propagan

GENERA_ARCH siempre devuelve 0 (línea 2397) y el código de error de Splunk se sobrescribe; el operador no puede saber cuántos registros fallaron.

Control de instancia desactivado

La llamada a CHECA_VERSION que impedía una segunda instancia activa está comentada (líneas 4097-4107), por lo que nada evita dos monitores enviando los mismos totales.

Hardcodes

Sistemas 264 y 500 en VARY_SIS con MAX_SIS = 2, títulos de base S151BD10MOVDIA264 y S151BD10MOVDIA500, nombres de dataset B03 a B43, minutos 15, 30 y 45, más 10 minutos implícitos al fin de ventana y 3 horas de ventana de corte. Se trasladan a application.yml.
