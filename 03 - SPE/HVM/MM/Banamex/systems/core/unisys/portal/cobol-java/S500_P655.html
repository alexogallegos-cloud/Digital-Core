@Component
@RequiredArgsConstructor
class AmbienteGuard implements JobExecutionListener {

    private final AmbienteProperties props;
    private final HostInfo host;                       // ATTRIBUTE HOSTNAME OF MYSELF

    @Override
    public void beforeJob(JobExecution je) {
        String h = host.nombre().trim();
        if (props.hostsProduccion().contains(h)) {
            // El fuente solo registra el mensaje y pone STATUS en -1; el proceso continúa.
            // Aquí se detiene: enmascarar una base productiva es irreversible.
            throw new EjecucionEnProduccionException("NO CORRE EN PRODUCCION; HOST: " + h);
        }
        if (!props.hostsPrueba().contains(h)) {
            // Host desconocido: el fuente lo trata como prueba sin aviso (falla abierto).
            throw new AmbienteDesconocidoException(h);
        }
        je.getExecutionContext().putString("host", h);
    }
}
@Component
@RequiredArgsConstructor
class TamanoBloque {

    private static final int MAX_BLOQUE = 1800;            // WKS-TB-NOM-PREF OCCURS 1800
    private final AmbienteProperties props;
    private final Clock clock;

    /** 1800 (o 200 en prueba interna) menos hora, minuto y segundo; si es impar se resta 1. */
    int calcula(String host) {
        LocalTime t = LocalTime.now(clock);
        int base = host.equals(props.hostPruebaInterna()) ? 200 : MAX_BLOQUE;
        int bloque = base - t.getHour() - t.getMinute() - t.getSecond();
        return (bloque % 2 == 0) ? bloque : bloque - 1;   // DIVIDE ... REMAINDER y SUBTRACT 1
    }
}

@Component
@RequiredArgsConstructor
class ReinicioTasklet implements Tasklet {

    private final ContratoRepository b03;                  // S500B03CONTRATOS

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        ExecutionContext job = ctx.getStepContext().getStepExecution().getJobExecution().getExecutionContext();
        Control actual = (Control) job.get("control");     // CSI y fecha de B02CONTROL
        Optional<CheckpointScrambling> previo = Optional.ofNullable((CheckpointScrambling) job.get("checkpoint"));

        CheckpointScrambling cp = previo
                .filter(p -> p.tipo() == 1 && p.csi() == actual.csi()
                          && p.fecha().equals(actual.fechaLote()) && p.ultimoContrato() > 0)
                .filter(p -> b03.existsById(p.ultimoContrato()))   // si no existe: desde el inicio
                .orElseGet(() -> CheckpointScrambling.nuevo(actual, (int) job.getInt("bloque")));

        job.putInt("bloque", cp.bloque());                  // WKS-I99-HEAD-BLQ sustituye al recién calculado
        job.put("checkpoint", cp);
        return RepeatStatus.FINISHED;
    }
}

@Bean
Step enmascaraContratos(JobRepository repo, PlatformTransactionManager tx,
                        @Value("#{jobExecutionContext['bloque']}") int bloque,

/** 50002000-SUBE-TABLA: nombre truncado a 29 posiciones más " PRUEBA" (X(29) + X(07)). */
static String nombreConSufijo(String nombreB03) {
    String base = nombreB03.length() > 29 ? nombreB03.substring(0, 29) : String.format("%-29s", nombreB03);
    return base + " PRUEBA";
}
/** Comportamiento del fuente, aislado para poder reemplazarlo. */
interface EstrategiaEnmascaramiento {
    List<String> nombresPara(List<String> nombresDelBloque, boolean ultimoBloque, String nombreReserva);
}

@Component("inversionLegada")
class InversionLegada implements EstrategiaEnmascaramiento {

    /** 50002000-MODIF-NOMBRE: el contrato k recibe el nombre de la posición n + 1 - k. */
    @Override
    public List<String> nombresPara(List<String> tabla, boolean ultimo, String reserva) {
        int n = tabla.size();
        int medio = (ultimo && n % 2 != 0) ? n / 2 + 1 : 0;           // 50002060-VAL-REGS
        var salida = new ArrayList<String>(n);
        for (int k = 1; k <= n; k++) {
            int origen = n + 1 - k;                                   // W77-IND-BXX desciende
            salida.add(origen == medio ? reserva : tabla.get(origen - 1));  // W77-NOMBRE-PTE
        }
        return salida;
    }
}
@Component
@RequiredArgsConstructor
class BloqueNombresWriter implements ItemWriter<Contrato> {

    private final ContratoRepository b03;
    private final EstrategiaEnmascaramiento estrategia;
    private final CheckpointScrambling checkpoint;

    /** El chunk completo es una transacción: el bloque se aplica entero o no se aplica. */
    @Override
    public void write(Chunk<? extends Contrato> bloque) {
        var nombres = estrategia.nombresPara(bloque.getItems().stream()
                .map(c -> nombreConSufijo(c.getNombre())).toList(), esUltimo(bloque), checkpoint.reserva());
        for (int i = 0; i < bloque.size(); i++) {
            bloque.getItems().get(i).setNombre(nombres.get(i));   // STORE S500B03CONTRATOSW
        }
        b03.saveAll(bloque.getItems());
        checkpoint.avanza(bloque.getItems().getLast().getNumContrato());   // WKS-I99-HEAD-CTO
    }
}

RN-S500-027Al arrancar, el programa determina si se está ejecutando en un servidor productivo o de prueba del banco, obteniendo el nombre del servidor (ATTRIBUTE HOSTNAME OF MYSELF, WKS-MY-HOST) y comparándolo contra dos listas cerradas de nombres exactos: "VDMALFA." o "MONBETA." se clasifican como producción (W77-ES-PRODUCCION); "VDMBETA.", "ACYPGAMA.", "ACYPBETA.", "MONALFA." o "ACYPOMEGA." se clasifican como prueba (W77-ES-PRUEBA), y específicamente "ACYPBETA." se marca además como prueba interna (W77-ES-PBA-INTERNA — el nombre de campo es "PBA-INTERNA", de PRUEBA, no "desarrollo interno"). Esta clasificación es la que gobierna el intento de bloqueo de producción evaluado en la siguiente regla (línea 304).Clasificación por hostname y bloqueo de producción que no detiene
RN-S500-028El programa detecta si se ejecuta en un servidor productivo del banco (indicador W77-ES-PRODUCCION) y, de ser así, arma el mensaje de error "NO CORRE EN PRODUCCION; HOST: <hostname>", lo registra (70000050-MENSAJE) y marca su estatus de terminación como fallido (CHANGE ATTRIBUTE STATUS OF MYSELF = -1). Sin embargo, esta señal de error NO detiene la ejecución: la instrucción no incluye STOP RUN ni GOBACK, y el párrafo no tiene una ruta de salida anticipada, por lo que el programa continúa incondicionalmente, abre la base de datos productiva y ejecuta el enmascaramiento de nombres y domicilios de titulares (B03CONTRATOS/B37GRUPOCPE/B39CTASCPE) igual que en cualquier otro ambiente. En la práctica, el único control real de "no correr en producción" es que el job/WFL no programe este proceso en los servidores productivos (VDMALFA o MONBETA) — el código en sí no tiene un corte de ejecución que lo impida.Clasificación por hostname y bloqueo de producción que no detiene
RN-S500-029El tamaño del bloque de contratos que se intercambiarán entre sí en cada ciclo de enmascaramiento no es fijo: se calcula a partir de la hora exacta de arranque del proceso batch (ACCEPT WKS-TIME-MAQ FROM TIME, línea 286), precisamente para que el tamaño de bloque no sea predecible entre corridas. Fórmula exacta: si el host es "ACYPBETA." (ambiente de prueba interna), `W77-CONTA-01 = 200 - WKS-TIME-HH - WKS-TIME-MM - WKS-TIME-SS`; en cualquier otro host (VDMBETA, ACYPGAMA, MONALFA, ACYPOMEGA, o cualquier hostname no reconocido — la rama ELSE no tiene condición adicional), `W77-CONTA-01 = 1800 - WKS-TIME-HH - WKS-TIME-MM - WKS-TIME-SS`, donde WKS-TIME-HH/MM/SS son la hora, minuto y segundo del reloj del sistema al arrancar el batch. El resultado (W77-CONTA-01, PIC 9(06)) se ajusta después a un número par (ver regla línea 327): si el residuo de dividirlo entre 2 no es cero, se le resta 1.Tamaño de bloque par derivado de la hora de arranque
RN-S500-030El tamaño de bloque calculado (fórmula de la regla línea 321) debe ser siempre un número par, para que el intercambio de nombres entre titulares (50002000-MODIF-NOMBRE) sea simétrico y ningún contrato quede sin nombre asignado dentro del bloque. Para garantizarlo, el programa divide W77-CONTA-01 entre 2 (DIVIDE W77-CONTA-01 BY 2 GIVING W77-TOT-DIV REMAINDER W77-DIV-RESTO); si el residuo no es cero (tamaño impar), le resta 1 (SUBTRACT 1 FROM W77-CONTA-01).Tamaño de bloque par derivado de la hora de arranque
RN-S500-031El proceso de enmascaramiento puede reanudarse desde donde quedó si se interrumpió una corrida anterior, en vez de reiniciar desde cero cada vez. Para decidir esto, el programa verifica si ya existe en disco un archivo de control/checkpoint específico de esta corrida ("S500/FILE/SCRBLING/<CSI>/<fecha>.", con CSI = W77-MY-CSI tomado de B02-NUM-CSI, y fecha = B02-FECHA-LOTE). Si existe, lo abre en modo I-O, lee su registro de encabezado (20000400-LEE-ARCH) y evalúa si corresponde a una reanudación válida (20000500-REINICIO, ver regla línea 441). Si no existe, lo crea desde cero: escribe un encabezado inicial con el contador en cero (WKS-I99-HEAD-CTO = ZERO, párrafo 20000300-GEN-ARCH), lo cierra y reabre en modo I-O, y continúa el proceso como una corrida nueva, sin punto de reanudación.Validación del encabezado y reposicionamiento en el contrato
RN-S500-032Para decidir si la reanudación desde el checkpoint es válida, el programa compara el encabezado leído del archivo de control (tipo de registro, identificador de nodo/CSI, fecha de proceso y contador de contrato — WKS-I99-HEAD-TPO, WKS-I99-HEAD-CSI, WKS-I99-HEAD-FEC, WKS-I99-HEAD-CTO) contra los valores de la corrida actual (CSI del nodo — W77-MY-CSI, tomado de B02-NUM-CSI — y fecha de proceso — WKS-FECHA-PROCESO, tomada de B02-FECHA-LOTE). Si coinciden y el contador de contrato es mayor a cero, el proceso retoma desde ese número de contrato (WS-NUM-CONTRATO/WS-MSG-CTO) y restaura el tamaño de bloque calculado en la corrida anterior (WKS-I99-HEAD-BLQ sobreescribe el W77-CONTA-01 recién calculado al arrancar, ver regla línea 321), buscando el punto exacto de reanudación (20000600-BUSCA-REINICIO). Si el nodo, la fecha o el contrato no coinciden, se descarta el encabezado existente y se regenera desde cero (20000300-GEN-ARCH), como corrida nueva. Adicionalmente, si al buscar el punto de reanudación el contrato correspondiente ya no se encuentra o no puede bloquearse en la tabla de contratos (B03CONTRATOS), el mecanismo de reanudación se abandona: el contador (WKS-I99-HEAD-CTO) se reinicia a cero y ambos cursores de lectura (S500B03CONTRATOS y S500B03CONTRATOSW) regresan al inicio del archivo (BEGINNING) — es decir, el proceso reanuda desde el principio en vez de fallar.Validación del encabezado y reposicionamiento en el contrato
RN-S500-033Como el enmascaramiento de nombres funciona intercambiando el nombre de un contrato por el de otro dentro del mismo bloque, un bloque con número impar de contratos dejaría uno sin pareja. Esta lógica (50002060-VAL-REGS) resuelve ese caso, y solo aplica al ÚLTIMO bloque de la corrida — el que cierra por fin de archivo del cursor de contratos (W77-EOF-B03 = 1 en 50000200-ACT-B03) — ya que todos los bloques "completos" anteriores tienen por construcción un tamaño par (ver regla línea 327) y nunca la necesitan. En ese último bloque parcial, si el conteo de contratos leídos es impar (DIVIDE W77-LEIDOS BY 2 GIVING W77-TOT-DIV REMAINDER W77-DIV-RESTO, residuo distinto de cero), se calcula la posición central (COMPUTE W77-REG-MEDIO = W77-TOT-DIV + 1). Enmascaramiento reversible con nombres reales

50002000-MODIF-NOMBRE invierte el orden de los nombres dentro de cada bloque. Los nombres reales siguen en la base y, conociendo el tamaño de bloque que se publica en bitácora y en SCRBLING, la asignación original puede reconstruirse. Ningún otro campo de B03 se altera y la razón social de B37 tampoco. Es un riesgo de protección de datos personales para el uso de estas bases fuera de producción.

Lista cerrada de hosts que falla en modo abierto

Cualquier host que no esté en las dos listas de las líneas 581-603 no se clasifica como producción ni como prueba y el proceso corre sin advertencia.

Reinicio a mitad de bloque

El checkpoint se escribe por bloque y la confirmación es por contrato; un reinicio vuelve a alterar los contratos del bloque interrumpido.
