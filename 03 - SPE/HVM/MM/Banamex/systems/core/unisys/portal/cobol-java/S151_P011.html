Al iniciar abre S151BD13BIFIN, obtiene la fecha contable y el consecutivo de transacciones del registro global, y se registra en COMS.     private static final Logger MTV = LoggerFactory.getLogger("bifin.monitor");   // appender propio
    private volatile boolean activo;                                         // W77-ACTIVO-MONITOR

    public void registra(String texto, String mensaje) {   // E00-MTV-NUM / -TXT / -MSG
        if (activo) {
            MTV.info("{} {} {}", LocalTime.now().format(DateTimeFormatter.ofPattern("HHmmss")),
                     texto, mensaje.length() > 2350 ? mensaje.substring(0, 2350) : mensaje);
        }
    }
    public void enciende() { activo = true; }
    public void apaga()    { activo = false; }   // CLOSE E00-MTV WITH SAVE
}

@Component
@RequiredArgsConstructor
public class GestorAlarmasRunner implements ApplicationRunner {

    private final InicioBifin inicio;            // 100000-INICIO
    private final JobLauncher launcher;
    private final Job cicloBifin;                // 200000-PROCESO como Flow
    private final EstadoCiclo estado;

    @Override
    public void run(ApplicationArguments args) throws Exception {
        inicio.ejecuta();                        // DISALLOW / ALLOW INTERRUPT alrededor
        Thread.sleep(Duration.ofSeconds(60));    // WAIT(60) antes del primer ciclo
        while (!estado.fin) {                    // PERFORM 200000-PROCESO UNTIL W77-SW-FIN = 9
            launcher.run(cicloBifin, new JobParametersBuilder()
                    .addLong("ciclo", System.nanoTime()).toJobParameters());
        }
        // 300000-FIN: CLOSE S151BD13BIFIN lo hace el pool al apagar el contexto.
    }
}
@Component
@RequiredArgsConstructor
public class InicioBifin {

    private final ControlVersiones ctrlVers;     // DAME_TIT IN CTLVERS
    private final LibControlClient libControl;   // CONSISDIA IN S151L001
    private final ReporteOperador operador;      // LJ IN LIBDESPLIEGA
    private final BifinProperties props;

    void ejecuta() {
        var tit = ctrlVers.dameTitulo("S151L001CTL ");
        if (tit.estatus() < 0) {                 // el fuente revisa W77-ESTATUS, no WKS-CTRV-STATUS
            operador.error(10201, "S151/P011:ERROR EN VERSION S151L001 ");
        } else {
            libControl.usaTitulo(tit.titulo());
        }
        // Sistema 500 compilado: la fecha de proceso de P011 es la de S500, no la de S151.
        var dia = libControl.consisdia(1, props.sistemaFecha());     // bifin.sistema-fecha: 500
        if (dia.resultado() == 0) {
            fechas.fijaBase(dia.fecCon());       // WKS-FECHA-MAQ, WKS-FECHA-BASE, WKS-TASK-FECHA
        }                                        // si falla, sigue con la fecha de la máquina
    }
}
@Component
@RequiredArgsConstructor
public class ConsecutivoBifin {

    private final GlobalesBifinRepository globales;    // registro global B00 de S151BD13BIFIN
    private final BifinProperties props;               // inicio-por-csi: {4: 40000}

    /** Número de transacción SA2: 6 dígitos de consecutivo + 2 dígitos del día. */
    @Transactional
    public long inicializa(LocalDate fechaBase, int csi) {
        var g = globales.findForUpdate();              // 500100-LOCK-GLOBALES
        if (!g.fecProc().equals(fechaBase)) {          // cambio de día: se reinicia
            g.setFecProc(fechaBase);
            long inicio = props.inicioPorCsi().getOrDefault(csi, 0L);
            g.setConsBifin(inicio == 0 ? 0 : inicio * 100 + fechaBase.getDayOfMonth());
        }
        return g.getConsBifin();                       // BEGIN/END-TRANSACTION NO-AUDIT
    }

    /** 250100-ARMA-MSG: ADD 1 TO W77-NUM-T. */
    static long siguiente(long actual, LocalDate fecha) {
        long consecutivo = actual / 100 + 1;
        if (consecutivo > 999_999) throw new IllegalStateException("W77-NUM-T 9(06) agotado");
        return consecutivo * 100 + fecha.getDayOfMonth();
    }
}
@Bean
Job cicloBifin(JobRepository repo, Step enviaAlarmas, Step enviaTotales,
               Step enviaBatch, Step recibeRespuestas, VentanaBatchDecider ventana) {
    Flow ciclo = new FlowBuilder<SimpleFlow>("cicloBifin")
            .start(enviaAlarmas)                 // 501000 + 210000: hasta 17 alarmas por mensaje
            .next(enviaTotales)                  // 503000 + 220000: hasta 16 totales por mensaje
            .next(ventana)
                .on("BATCH").to(enviaBatch)      // 06:00 < hora < 07:00
                .from(ventana).on("*").to(recibeRespuestas)
            .from(enviaBatch).next(recibeRespuestas)
            .build();
    return new JobBuilder("S151P011", repo).start(ciclo).end().build();
}

@Component
@RequiredArgsConstructor
public class VentanaBatchDecider implements JobExecutionDecider {
    private final EstadoCiclo estado;
    private final Clock reloj;

    @Override
    public FlowExecutionStatus decide(JobExecution je, StepExecution se) {
        LocalTime t = LocalTime.now(reloj);
        if (t.isAfter(LocalTime.of(6, 0)) && t.isBefore(LocalTime.of(7, 0)) && !estado.envioSuspendido) {
            return new FlowExecutionStatus("BATCH");
        }
        if (t.isAfter(LocalTime.of(7, 0))) estado.batch = EstadoCiclo.Barrido.TERMINADO;
        return FlowExecutionStatus.COMPLETED;
    }
}
// Cada step de envío revisa estado.envioSuspendido (W77-ACTIVO) antes de leer la base.
@Component
@Endpoint(id = "bifin")
@RequiredArgsConstructor
public class BifinOperadorEndpoint {

    private final EstadoCiclo estado;
    private final GlobalesBifinRepository globales;
    private final EnvioBatch envioBatch;
    private final BifinDestino destino;          // nodo 15/16 en memoria, inicial de application.yml

    @WriteOperation
    @Transactional
    public String hi(int valor) {
        return switch (valor) {
            case 11 -> { estado.batch = Barrido.PENDIENTE; estado.envioSuspendido = true;
                         globales.marcaEstado(1); yield "SUSPENDE ENVIO"; }   // B00-GLO-STAPROC = 1
            case 12 -> { estado.batch = Barrido.PENDIENTE; estado.envioSuspendido = false;
                         globales.marcaEstado(0); yield "ACTIVA ENVIO"; }
            case 13 -> { estado.batch = Barrido.PENDIENTE; envioBatch.ejecuta();
                         yield "REENVIO BATCH"; }                             // fuera de la ventana
            case 14 -> "NODO DESTINO CAMBIADO A " + destino.alterna();       // 15 <-> 16
            default -> "HI no soportado: " + valor;
        };
    }
}

@Component
@StepScope
@RequiredArgsConstructor
public class EmpacadorAlarmas implements ItemProcessor<Alarma, Alarma> {

    private static final int MAX_RENGLONES = 17;           // W77-IND > 16 corta el mensaje
    private final MensajeBifin mensaje;                     // WKS-POR-ALARMAS + WKS-REG-B06
    @Value("#{jobExecutionContext['fechaBase']}") private LocalDate fechaBase;

    @Override
    public Alarma process(Alarma a) {
        // Solo alarmas no enviadas cuya fecha de información es la fecha de proceso.
        if (a.staEnvio() != 0 || !a.fecInf().equals(fechaBase)) return null;
        mensaje.agrega(RenglonAlarma.builder()
                .csiOrigen(a.csi()).sistema(a.sistema()).producto(a.producto())
                .instrumento(a.instServ()).moneda(a.moneda())
                .autOrigen(a.autOrig()).autS151(a.autS151()).aut3Niv(a.aut3Niv())
                .sucursal(a.suc()).cuenta(a.cta())                       // String, 4 y 12 dígitos
                .transmision(a.transSuc(), a.transCja(), a.transHor(), a.transCve())
                .indContable(a.indCont())
                .importe(a.importe().setScale(2, RoundingMode.DOWN))     // 9(16)V99 sin signo
                .claveAlarma(a.cveAlarm()).numOcurr(a.numOcurr())
                .build());
        if (mensaje.renglones() >= MAX_RENGLONES) mensaje.cierra();      // 250000-ENVIA-MENSAJE
        return a;
    }
}

    private final ControlEnvioRepository b06;     // S151B06CTLENVIO por B06SXNUMENS
    private final AlarmaRepository b03;           // B03SXALARMA

    @ServiceActivator(inputChannel = "bifinRespuestas")
    @Transactional
    public void maneja(MensajeSa2 r) {
        if (r.status() == 99) { estado.fin = true; return; }               // DISABLE de COMS
        if (r.tipoResultado() >= 2 || !ORIGENES.contains(r.acaOrigen())) return;   // 243000-RECHAZADA vacío
        b06.findByNumMens(r.numTran()).ifPresent(ctl -> {
            int max = "A".equals(ctl.tipoMens()) ? 17 : "B".equals(ctl.tipoMens()) ? 16 : 0;
            for (var occ : ctl.ocurrencias().subList(0, max)) {
                boolean marcado = "A".equals(ctl.tipoMens())
                        ? b03.marcaEnviada(ctl.csi(), ctl.fecInf(), occ) == 1     // STA-ENVIO = 1
                        : b05.marcaEnviado(ctl.csi(), ctl.fecInf(), occ) == 1;
                if (!marcado) break;                                       // MOVE 17 TO W77-IND

        });

Riesgos de migración
Validación de versión que nunca se cumple

En 100000-INICIO (línea 1447) se evalúa IF W77-ESTATUS < 0, pero el resultado de DAME_TIT queda en WKS-CTRV-STATUS y W77-ESTATUS es un PIC 9(11) sin signo que nadie asigna. El título de S151L001 se cambia aunque CTLVERS falle.

Límite horario sin efecto

W77-HORA-LIMITE = 200000 (línea 124) no se lee en ningún párrafo. La ventana operativa real del programa es solo la de envío batch, de 06:00 a 07:00 (líneas 1579-1584).

Rechazos sin tratamiento

243000-RECHAZADA (línea 1997) está vacío y 241000-ANALIZA-RESP ignora respuestas con resultado 2 o mayor. Las alarmas rechazadas quedan con STA-ENVIO en cero y se reenvían en el siguiente ciclo sin registro del rechazo.

STORE sin manejo de excepción

511000-UPDATE-ALARMAS y 514000-UPDATE-BATCH hacen STORE sin ON EXCEPTION; un error deja la alarma sin marcar y no se informa.

Consecutivo de seis dígitos sin control

W77-NUM-T PIC 9(06) se incrementa sin ON SIZE ERROR (línea 2060). El rango de CSI 10 (desde 0) puede alcanzar el de CSI 4 (desde 40000) y duplicar números de transacción.

Mensajes de error copiados

504000-NEXT-TOTPROD y 506000-NEXT-TOTBATCH informan 'ERROR-FIND FIRST' aunque hacen FIND NEXT, y los de B05 nombran S151B05POSICION en lugar de S151B05BATCH.

Hardcodes

TRANCODE 015111, destino 013702, nodo 16, estación 342/42, LSN 1141, sistema 500 para la fecha, CSI 4 con consecutivo 40000 y la ventana 06:00-07:00 están compilados.

Pendiente de revisión SME
  • RN-S151-1083 afirma que el límite de 20:00 es una regla de negocio transversal; en P011 el valor existe pero no se usa. Revisar la regla con el SME y con P016.
  • Confirmar por qué P011 consulta la fecha contable del sistema 500 (línea 1456) y no la de S151.
  • Confirmar si B03-ALA-IMPORTE, B01-TOT-IMPCARGOS y B05-TOT-IMPCARGOS pueden traer signo; el puente de formato no lo conserva.
  • Confirmar si las respuestas de totales por producto (tipo T) deben aplicarse; hoy los totales se reenvían completos en cada ciclo.
  • Los layouts de B00, B01, B03, B05 y B06 de S151BD13BIFIN no vienen en el fuente; los nombres de campo Java se tomaron de los MOVE del programa.
  • Definir si el nodo destino elegido con HI14 debe persistirse; hoy se pierde al reiniciar.
