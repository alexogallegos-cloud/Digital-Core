S151 · P090 a Java 21 Spring Boot

P090 llama a la librería L030 de cada sistema (S087 solo en Monterrey, S264 y S500 en ambas plazas) pidiendo la pantalla 16, toma el renglón 8 de totales y convierte sus cifras editadas (con comas y punto decimal) a números para escribir un detalle por sistema en un archivo de 60 bytes con header, detalles y trailer. En el CSI 04 (Monterrey) el archivo se copia con XFER al nodo par. En el CSI 10 (Valle de México) el programa espera a que llegue el archivo de Monterrey, revisando cada minuto, y lo une al propio en un archivo consolidado con un trailer recalculado.

/** A00-R00/R01/R02-S502: un registro de 60 bytes, discriminado por TIPREG (1, 2, 3). */
public sealed interface LideRecord permits LideRecord.Header, LideRecord.Detalle, LideRecord.Trailer {

    /** TIPREG 1. FECPRO 9(06) AAMMDD y HORAPRO 9(06) en COMP (dígitos de 4 bits). */
    record Header(LocalDate fecPro, LocalTime horaPro) implements LideRecord {}

    /** TIPREG 2: un renglón por sistema. */
    record Detalle(int sistema,              // 9(04) COMP: 87, 264, 500
                   int cci,                  // 9(02) COMP: 04 MTY, 10 VDM
                   String cobertura,         // X(03): NAL, VDM, MTY
                   String nombreRenglon,     // X(15)
                   long numCargos,           // 9(10) COMP
                   BigDecimal impCargos,     // 9(18)V99 COMP -> escala 2
                   long numAbonos,
                   BigDecimal impAbonos) implements LideRecord {}

    /** TIPREG 3. */
    record Trailer(long totReg,              // 9(08) COMP
                   long totNumCar, BigDecimal totImpCar,
                   long totNumAbo, BigDecimal totImpAbo) implements LideRecord {}
}
/** WKS-SAL-P16DATOS: 12 renglones de pantalla; P090 lee siempre el renglón 8. */
@Component
class Pantalla16Parser {

    static final int RENGLON_TOTALES = 8;     // WKS-SAL-P16DAT(8), compilado

    record Totales(String concepto, long numCargos, BigDecimal impCargos,
                   long numAbonos, BigDecimal impAbonos) {}

    /** NUCA PIC ZZZ,ZZZB(01) e IMCA PIC ZZZ,ZZZ,ZZZ,ZZZ,ZZ9.99B(2): se quitan comas y espacios. */
    Totales renglonTotales(Pantalla16 p) {
        var r = p.renglon(RENGLON_TOTALES);
        return new Totales(r.concepto(),                       // P16CON X(17), se trunca a X(15)
                entero(r.nuca()), importe(r.imca()),
                entero(r.nuab()), importe(r.imab()));
    }

    private static long entero(String editado) {
        String d = editado.replace(",", "").strip();
        return d.isEmpty() ? 0 : Long.parseLong(d);           // espacios equivalen a cero
    }

    private static BigDecimal importe(String editado) {
        String d = editado.replace(",", "").strip();
        return d.isEmpty() ? BigDecimal.ZERO.setScale(2) : new BigDecimal(d).setScale(2);
    }
}

@Configuration
@RequiredArgsConstructor
public class ReclideJobConfig {

    @Bean
    Job reclide(JobRepository repo, Step generaLide, PlazaDecider plaza,
                Step enviaXfer, Step esperaMty, Step concentra) {
        return new JobBuilder("S151P090", repo)
                .start(generaLide)                       // 020000-INICIO + 030100-PROCESO
                .next(plaza)
                    .on("MTY").to(enviaXfer)             // CSI 04: 070100 + 080100 (XFER)
                .from(plaza)
                    .on("VDM").to(esperaMty).next(concentra)   // CSI 10: 050100 + 060100
                .from(plaza)
                    .on("*").end()                       // otro CSI: solo genera su archivo
                .end()
                .build();
    }
}

@Component
class PlazaDecider implements JobExecutionDecider {
    private final HostInfo host;                         // CSIID OF SOPORTECOMS
    PlazaDecider(HostInfo host) { this.host = host; }

    @Override
    public FlowExecutionStatus decide(JobExecution je, StepExecution se) {
        int csi = host.numCsi() == 32 ? 4 : host.numCsi();   // IF WKS-NUMCSI-HOST = 32 MOVE 04
        return switch (csi) {
            case 4  -> new FlowExecutionStatus("MTY");
            case 10 -> new FlowExecutionStatus("VDM");
            default -> new FlowExecutionStatus("OTRO");
        };
    }
}
@Component
@RequiredArgsConstructor
class ConsultaL030 {

    private final Map<Integer, L030Client> clientesPorSistema;   // 87, 264, 500
    private final HostInfo host;
    private final Pantalla16Parser parser;

    /**
     * En MCP cada sistema usa su propia copia de la librería L030 ("NNNN" se sustituye por
     * 0087, 0264 o 0500) que se crea con BEGIN JOB; REMOVE; COPY. En Java basta un cliente
     * por sistema configurado con el mismo código.
     */
    List<Integer> sistemasDelHost() {
        return host.numCsi() == 4 ? List.of(87, 264, 500) : List.of(264, 500);
    }

    /** 030364-CALL-L030-Snnn: resultados 0, 3 y 4 son aceptables; cualquier otro deja ceros. */
    Pantalla16Parser.Totales totales(int sistema, LocalDate fecha) {
        var resp = clientesPorSistema.get(sistema).pantalla16(sistema, fecha, /*bcoOri*/ 2, /*bcoDes*/ 2);
        return switch (resp.resultado()) {
            case 0, 3, 4 -> parser.renglonTotales(resp.pantalla());
            default      -> Pantalla16Parser.Totales.ceros();      // MOVE 0 TO P16NUCA(8) ...
        };
    }
}

class DetalleLideProcessor {

    private final ConsultaL030 l030;
    private final HostInfo host;
    private Acumulado total = Acumulado.vacio();     // W77-TOT-REG, W77-TOT-NUM-*, W77-TOT-IMP-*

    LideRecord.Detalle detalle(int sistema, LocalDate fecha) {
        var t = l030.totales(sistema, fecha);
        String cobertura = (sistema == 87) ? "NAL" : (host.numCsi() == 10 ? "VDM" : "MTY");
        var d = new LideRecord.Detalle(sistema, host.numCsi(), cobertura,
                Fixed.ajusta(t.concepto(), 15),              // X(17) -> X(15)
                t.numCargos(), t.impCargos(), t.numAbonos(), t.impAbonos());
        total = total.mas(d);                           // ADD ... TO W77-TOT-*: suma exacta, escala 2
        return d;
    }

    LideRecord.Trailer trailer() {                      // 030370-GRABA-TRAILER
        return new LideRecord.Trailer(total.registros(), total.numCar(), total.impCar(),
                                      total.numAbo(), total.impAbo());
    }
}

class EsperaArchivoMty implements Tasklet {

    private final ReclideProperties props;
    private final ResourceLoader rl;
    private final OperadorNotifier operador;     // LJ IN LIB-DISP, código 010401

    /** 060100-VALIDA-RESIDENCIA UNTIL W77-UNE-LIDE = 1, con WAIT(60) entre revisiones. */
    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) throws InterruptedException {
        Resource mty = rl.getResource(props.resuelve(props.archivos().mty(), 4, LocalDate.now(clock)));
        Instant limite = Instant.now(clock).plus(props.esperaMtyMaxima());
        while (!mty.exists()) {
            if (Instant.now(clock).isAfter(limite)) {
                throw new ArchivoNoRecibidoException(mty.getDescription());   // el fuente espera sin fin
            }
            operador.aviso("NO SE ENCUENTRA RESIDENTE EL ARCHIVO " + mty.getDescription()
                         + " ESPERO 1 MIN. PARA VOLVER A VERIFICAR RESIDENCIA");
            Thread.sleep(props.esperaMty().toMillis());
        }
        return RepeatStatus.FINISHED;
    }
}

@Component
@StepScope
class ConcentradorLideProcessor implements ItemProcessor<LideRecord, LideRecord> {

    private Acumulado total = Acumulado.vacio();
    private boolean headerEscrito;
    private int trailersVistos;

    @Override
    public LideRecord process(LideRecord r) {
        return switch (r) {
            case LideRecord.Header h -> {
                if (headerEscrito) yield null;          // el header de MTY se descarta
                headerEscrito = true;
                yield new LideRecord.Header(h.fecPro(), LocalTime.now(clock));   // hora nueva
            }
            case LideRecord.Detalle d -> {
                total = total.mas(d);
                yield d;
            }
            case LideRecord.Trailer t -> {
                // Los trailers de entrada no se validan contra lo leído.
                if (trailersVistos++ == 0) yield null;  // trailer de VDM: fin del primer archivo
                yield new LideRecord.Trailer(total.registros(), total.numCar(), total.impCar(),
                                             total.numAbo(), total.impAbo());
            }
        };
    }
}

Riesgos de migración
Totales obtenidos de una pantalla

Las cifras se leen del renglón 8 de la pantalla 16 de L030 como texto editado (líneas 365-374 y 647-671). Un cambio de formato en la pantalla o un volumen mayor a 999,999 movimientos altera los totales sin error.

Relleno con ceros sin efecto

030361-LLENA-CEROS se ejecuta después de mover los valores al registro de salida en los tres párrafos 030360 (por ejemplo líneas 826-846 frente a 855-860), por lo que no corrige el registro escrito.

Acumuladores de importe sin inicializar

W77-TOT-IMP-CAR y W77-TOT-IMP-ABO no tienen VALUE (líneas 71 y 73) y el primer archivo los acumula sin ponerlos en cero.

Espera indefinida en Valle de México

060100-VALIDA-RESIDENCIA repite WAIT(60) sin tope mientras no exista el archivo de Monterrey (líneas 922-941). Una falla de XFER en Monterrey solo genera un aviso para copiar a mano (líneas 1068-1076).

Errores de L030 convertidos en ceros

Un resultado de L030 distinto de 0, 3 o 4 pone en cero los totales del sistema (líneas 726-730, 808-812 y 890-894) y el archivo se envía igual.

Hardcodes

Rutas y usuarios (S151) y (S502), packs CMEMP e IMPUESTOS, CSI 04 y 10, mapeo del CSI 32 al 04, renglón 8 de la pantalla 16, bancos 2 y 2, pares de nodos ACYPGAMA, MONALFA, VDMALFA, MONBETA, ACYPBETA y VDMBETA (líneas 1035-1057). Se trasladan a application.yml.
