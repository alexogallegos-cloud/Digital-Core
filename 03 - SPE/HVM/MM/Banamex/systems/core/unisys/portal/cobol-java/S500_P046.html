/** CTLVERS: catálogo de versiones que traduce un identificador lógico al título del objeto vigente. */
public interface CatalogoVersiones {

    boolean checaVersion(String id, String version);  // CHECAME: declarado en P046 pero no se usa

    @Bean MessageChannel comsEntrada() { return new QueueChannel(); }        // COMSINHDR
    @Bean MessageChannel comsSalida()  { return new DirectChannel(); }       // COMSOUTHDR

@Component
@RequiredArgsConstructor
class ResolucionLibreria {

    private final CatalogoVersiones ctlvers;
    private final P046Properties props;

    /** IF DAME_TIT(...) < 0 THEN título fijo ELSE título del catálogo (72 caracteres). */
    String tituloLibreria() {
        var lib = props.libreriaRevocaciones();
        return ctlvers.titulo(lib.idLogico())
                .map(t -> t.length() > 72 ? t.substring(0, 72) : t)
                .orElseGet(() -> {
                    log.warn("CTLVERS sin título para {}; se usa respaldo {}", lib.idLogico(), lib.tituloRespaldo());
                    return lib.tituloRespaldo();
                });
    }
}

@SpringBootApplication
@EnableConfigurationProperties(P046Properties.class)

    /** ENABLE(COMSINHDR,"ONLINE") y llamada a ENVIA_CANCELACIONES; al regresar, END. */
    @Bean
    ApplicationRunner arranque(CancelacionesPort cancelaciones, ComsAdapter coms,
                               MessageChannel comsEntrada, MessageChannel comsSalida,
                               ConfigurableApplicationContext ctx) {
        return args -> {
            coms.habilita("ONLINE");
            cancelaciones.enviaCancelaciones(comsEntrada, comsSalida);   // bloquea hasta la orden de fin
            ctx.close();
        };
    }
}
