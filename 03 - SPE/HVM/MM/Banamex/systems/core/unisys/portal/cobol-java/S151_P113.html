S151 · P113 a Java 21 Spring Boot

P113 lee del archivo de control CORP la fecha de proceso y la lista de sistemas marcados para proceso foráneo. Para cada sistema carga desde la librería L710 el catálogo que da el sentido contable (AFC) de cada clave de transacción y corre el punteo dos veces, de 04 a 10 y de 10 a 04. En cada pasada ordena los archivos PFOTR y PFORE, valida que sus headers traigan la fecha de proceso y los recorre a la par por una llave que incluye el importe. Reporta cada movimiento transmitido sin recibir o recibido sin transmitir y acumula cargos y abonos correctos y con diferencia por moneda.

        LlavePunteo llave,              // A00-F01-KEY X(29) con su REDEFINES
        int producto, int instrumento, int sucursalPromotora,
        long autorizacionS151, int sucursalAdicional, int cajaAdicional,

    /** La llave completa decide el punteo: incluye el importe, no solo la operación. */
    public record LlavePunteo(int tipo, int moneda, long autoriza, int sucursalInicial,
                              int cajaInicial, String contrato,       // 9(16) como String
                              int claveTransaccion,                   // CVETRAN1
                              BigDecimal importe)                     // 9(14)V99 -> escala 2
            implements Comparable<LlavePunteo> {

        static final Comparator<LlavePunteo> ORDEN = Comparator
                .comparingInt(LlavePunteo::tipo).thenComparingInt(LlavePunteo::moneda)
                .thenComparingLong(LlavePunteo::autoriza).thenComparingInt(LlavePunteo::sucursalInicial)
                .thenComparingInt(LlavePunteo::cajaInicial).thenComparing(LlavePunteo::contrato)
                .thenComparingInt(LlavePunteo::claveTransaccion).thenComparing(LlavePunteo::importe);

        /** TIPO = 99 con el resto en ceros es el centinela de fin de archivo (5000/6000-LEE-*). */
        static final LlavePunteo FIN = new LlavePunteo(99, 0, 0, 0, 0, "0000000000000000", 0, BigDecimal.ZERO);

        @Override public int compareTo(LlavePunteo o) { return ORDEN.compare(this, o); }
    }
}

    Map<Integer, Integer> carga(int sistema, int formatoCatalogo, int numCatalogo) {

                log.info("CVE.TRANSACCION > 9999 ");            // el fuente lo informa y lo omite
                continue;
            }
            afcPorClave.put(indice, r.dato(15));                 // WKS-CVE-AFC(W77-IND)
        }
        return Map.copyOf(afcPorClave);
    }

@Component
@RequiredArgsConstructor
class ControlCorpReader {
    private final P113Properties props;

    /** 1000-INICIO + 1010-LEE-CONTROL: header (CSI, fecha) y detalle con INDPF = 1. */
    ControlForaneos lee(Resource control) throws IOException {
        try (var r = new ControlCorpFile(control)) {
            HeaderControl h = r.header();
            List<Integer> sistemas = new ArrayList<>();
            for (DetalleControl d; (d = r.next()) != null; ) {
                if (d.indProcesoForaneo() == 1) sistemas.add(d.sistema());
                if (d.sistema() == props.sistemaFinCatalogo()) break;
            }
            if (sistemas.size() > props.maxSistemas()) {
                throw new IllegalStateException("Más sistemas foráneos que la tabla de P113: " + sistemas.size());
            }
            return new ControlForaneos(h.csi(), h.fechaProceso(), List.copyOf(sistemas));
        }
    }
}
@Configuration
@RequiredArgsConstructor
class PunteoForaneosJobConfig {

    /** 2000-PROCESO / 2100-RECORRE-SISTEMAS como PartitionedStep: una partición por sistema y sentido. */
    @Bean
    Step punteoForaneos(JobRepository repo, Step punteoPorParticion, Partitioner porSistemaYSentido) {
        return new StepBuilder("punteoForaneos", repo)
                .partitioner("punteoPorParticion", porSistemaYSentido)
                .step(punteoPorParticion)
                .gridSize(4)
                .build();
    }

    @Bean
    @JobScope
    Partitioner porSistemaYSentido(ControlForaneos control, P113Properties props) {
        return grid -> {
            Map<String, ExecutionContext> p = new LinkedHashMap<>();
            for (int sistema : control.sistemas()) {
                for (Sentido s : props.sentidos()) {             // [04 -> 10] y luego [10 -> 04]
                    var ctx = new ExecutionContext();
                    ctx.putInt("sistema", sistema);
                    ctx.putInt("csiTra", s.transmisor());
                    ctx.putInt("csiRec", s.receptor());
                    p.put("S%03d-%02d-%02d".formatted(sistema, s.transmisor(), s.receptor()), ctx);
                }
            }
            return p;
        };
    }
}

@Component

class ValidaHeadersForaneos implements StepExecutionListener {
    @Value("#{jobExecutionContext['fechaProceso']}") private LocalDate fechaProceso;

    /** 2400-VALIDA-HEADERS: un archivo de otro día aborta el job (SET MYSELF(STATUS) TO -1). */
    @Override
    public void beforeStep(StepExecution se) {
        valida(lector("MOVTRA").header(), "MOVTRA");
        valida(lector("MOVREC").header(), "MOVREC");
    }

    private void valida(Optional<HeaderForaneo> h, String archivo) {
        h.filter(x -> !x.fecha().equals(fechaProceso))
         .ifPresent(x -> {
             throw new ProcesoAbortadoException(
                     "FECHA DE PROCESO DIF. A FECHA HEADER ARCHIVO : " + archivo);   // LJ código 010105
         });
        // Archivo vacío (sin header): el fuente lo deja pasar y luego informa "NO EXISTEN MOVTOS..."
    }
}

/** 4000-GENERA-REPORTE: match-merge de dos archivos ordenados por la misma llave. */
@Component
@StepScope
@RequiredArgsConstructor
class PunteoTasklet implements Tasklet {
    private final PeekableItemReader<MovimientoForaneo> transmitidos;   // MOVTRA ordenado (SORT)
    private final PeekableItemReader<MovimientoForaneo> recibidos;      // MOVREC ordenado
    private final AcumuladorPunteo acum;                                 // OK, TRA y REC por moneda
    private final ReportePunteo reporte;

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext cc) throws Exception {
        MovimientoForaneo t = transmitidos.read(), r = recibidos.read();
        while (t != null || r != null) {
            int cmp = llave(t).compareTo(llave(r));      // null -> LlavePunteo.FIN (TIPO 99)
            if (cmp == 0) {
                acum.ok(r);                              // 4030-ADD-OK
                t = transmitidos.read(); r = recibidos.read();
            } else if (cmp > 0) {
                reporte.soloRecibido(acum.errorRec(r));  // 4010-SUCREC-ERROR
                r = recibidos.read();
            } else {
                reporte.soloTransmitido(acum.errorTra(t)); // 4020-SUCTRA-ERROR
                t = transmitidos.read();
            }
        }
        reporte.corteMoneda(acum);                       // 9020-CORTE-MONEDA final
        return RepeatStatus.FINISHED;
    }

    private static MovimientoForaneo.LlavePunteo llave(MovimientoForaneo m) {
        return m == null ? MovimientoForaneo.LlavePunteo.FIN : m.llave();
    }
}
@Component
@StepScope
@RequiredArgsConstructor
class AcumuladorPunteo {
    private final Map<Integer, Integer> afcPorClave;     // CatalogoClavesAfc
    private final ReportePunteo reporte;
    private Integer monedaAnterior;                      // W77-MON-ANT
    private TotalesMoneda moneda = TotalesMoneda.vacio(); // WKS-IMP-*
    private TotalesMoneda pasada = TotalesMoneda.vacio(); // WKS-TOT-*

    /** 4010-SUCREC-ERROR (4020 es simétrico con el lado transmitido). */
    RenglonDiferencia errorRec(MovimientoForaneo m) {
        cortaSiCambiaMoneda(m.llave().moneda());
        BigDecimal imp = m.llave().importe();
        if (imp.signum() > 0) {                                  // solo importes mayores a cero
            switch (afcPorClave.getOrDefault(m.llave().claveTransaccion(), 0)) {
                case 1 -> { moneda = moneda.masAbonoRec(imp); pasada = pasada.masAbonoRec(imp); }
                case 2 -> { moneda = moneda.masCargoRec(imp); pasada = pasada.masCargoRec(imp); }
                default -> log.warn("Clave {} sin AFC: no se acumula", m.llave().claveTransaccion());
            }
        }
        monedaAnterior = m.llave().moneda();
        String leyenda = (sistema == 500 && m.indSubcuenta() == 1)
                ? Leyendas.recibidoSubcuenta(csiRec, csiTra)     // "... NO TRANS EN nn. SC"
                : Leyendas.recibido(csiRec, csiTra);             // "RECIBIDO  EN nn NO TRANSMIT. EN nn"
        return RenglonDiferencia.de(m, leyenda);
    }
}
    /** 4030-ADD-OK: se toma el lado recibido; las llaves son idénticas, importe incluido. */
    void ok(MovimientoForaneo m) {
        cortaSiCambiaMoneda(m.llave().moneda());
        BigDecimal imp = m.llave().importe();
        if (imp.signum() > 0) {
            switch (afcPorClave.getOrDefault(m.llave().claveTransaccion(), 0)) {
                case 1 -> { moneda = moneda.masAbonoOk(imp); pasada = pasada.masAbonoOk(imp); }
                case 2 -> { moneda = moneda.masCargoOk(imp); pasada = pasada.masCargoOk(imp); }
                default -> { }
            }
        }
        monedaAnterior = m.llave().moneda();
        contadores.movimientoOk();                               // W77-MOVS-OK
    }

    /** 9020-CORTE-MONEDA: renglón WRS-D2 con OK, error TRA y error REC; reinicia la moneda. */
    private void cortaSiCambiaMoneda(int nueva) {
        if (monedaAnterior != null && monedaAnterior != nueva) {
            reporte.totalesMoneda(monedaAnterior, moneda);      // "sin diferencias" si TRA y REC son cero
            moneda = TotalesMoneda.vacio();
            reporte.forzarSaltoDePagina();                       // MOVE 60 TO W77-CONT-LIN
        }
    }
}

Riesgos de migración
Catálogo AFC arrastrado entre sistemas

WKS-CVE-SIS (líneas 894-897) se llena por posición en 0520-CARGA-PARAMETROS y nunca se limpia; un sistema puede heredar el AFC de claves del sistema anterior.

Importes fuera de totales

En 4010, 4020 y 4030 un AFC distinto de 1 o 2 no acumula y no deja aviso; las cifras del corte por moneda pueden no cuadrar con el detalle impreso.

Índice del catálogo sin validar

WKS-CVE-AFC se indexa con CVETRAN1 (líneas 1570, 1616, 1662) sin comprobar que esté entre 1 y 10000.

Tabla de sistemas por renglón y no por sistema

W77-I avanza con cada detalle del control aunque INDPF no sea 1 (líneas 1279-1299); más de diez renglones antes del 999 desbordan WKS-TABLA-SISTEMA.

Importe dentro de la llave

Como IMPORTE1 forma parte de A00-F01-KEY, una misma operación con importe distinto aparece como dos diferencias (una de cada lado) y no como una diferencia de importe.

Hardcodes

CSI 04 y 10, paso P113, MTP 001, catálogo 523, sistema 151 programa 30 en L710, rutas de control y reporte. Se trasladan a configuración.
