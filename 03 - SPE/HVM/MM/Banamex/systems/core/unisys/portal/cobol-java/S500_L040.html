    private final ControlVersiones ctlvers;        // LIBRARY CTLVER BYFUNCTION "CTLVERS"
    private final Environment env;

    /** CHECA_VERSION: la librería debe estar registrada como S500L040LIGA versión 25MTP002. */
    void checaVersion() {
        boolean exento = props.hostsSinValidacion()                   // [ACYPBETA]
                .contains(env.getRequiredProperty("ligas.host"));
        if (!exento && ctlvers.checame("S500L040LIGA", "25MTP002") < 0) {
            lj.error("E>050110<", "ERROR MI NOMBRE NO CORRESP. AL DE CTLVER");
            throw new IllegalStateException("Versión no registrada en CTLVERS");   // MYSELF.STATUS := -1
        }
    }
    /** PROC_LEVANTA: valida versión, enciende la liga, resuelve el título de L039 y la levanta. */
    @PostConstruct
    void levanta() {
        checaVersion();
        ligaActiva.set(true);                                 // B_LLAMAL039 := TRUE
        resuelveDestinoL039();                                // PROC_DAME_TIT_S500L039
        l039.levanta();                                       // S500L040_99_LEVANTA_C
    }

    private void resuelveDestinoL039() {
        ctlvers.dameTitulo("S500L039BD04")
               .ifPresentOrElse(l039::apuntaA,                // REPLACE LIGACALL.TITLE BY EA_TIT FOR 72
                    () -> {
                        lj.aviso("L>020901<", "ERROR CALL CTLVER S500L039BD04");
                        ligaActiva.set(false);                // nuevo: no dejar la liga encendida sin destino
                    });
    }
@Component
@Endpoint(id = "ligas")
@RequiredArgsConstructor
public class LigasAdminEndpoint {

    private final LigasFacade fachada;

    @ReadOperation                                         // HI 3: STATUS ACCESO A L039
    public String estado() {
        return "STATUS ACCESO A L039: " + (fachada.ligaActiva() ? "CON" : "SIN");
    }

    @WriteOperation
    @PreAuthorize("hasRole('OPERADOR_S500')")
    public String hi(@Selector int valor) {
        return switch (valor) {
            case 1 -> { fachada.suspendeLiga();  yield "SE SUSPENDE LIGA A L039"; }  // DELINKLIBRARY
            case 2 -> { fachada.restableceLiga(); yield "SE HACE LIGA A L039"; }      // título + LEVANTA_C
            case 4 -> { fachada.baja();           yield "DESABILITA S500L040LIGA"; }  // GOINGAWAY
            default -> " 1: SUSPENDE LIGA, 2: REALIZA LIGA, 3: ESTATUS, 4: DESABILITA, 99: LISTA";
        };
    }
}

Riesgos de migración
Liga encendida sin destino resuelto

PROC_LEVANTA enciende B_LLAMAL039 antes de resolver el título de L039 (líneas 274-275); si DAME_TIT falla (línea 259) la liga queda activa con el título por omisión de LIGACALL.

Excepción de versión por host

CHECA_VERSION omite la validación con CTLVERS cuando el host es ACYPBETA (líneas 128-130).

Código 99 ambiguo

Los nueve procedimientos devuelven 99 cuando la liga está suspendida; el consumidor no puede distinguirlo de un resultado de la propia L039.

Control por instancia privada

Con SHARING = PRIVATE (línea 8) cada consumidor tiene su propia instancia y su propio interruptor; el HI afecta solo a la instancia que lo recibe.

Pendiente de revisión SME

  • Confirmar qué título usa LIGACALL cuando CTLVERS no resuelve S500L039BD04; el fuente no declara TITLE en la librería importada.
  • Confirmar si 99 es un código que S500L039BD04 también puede devolver, para decidir si se cambia el código de liga suspendida en la versión Java.
  • Confirmar el motivo de exceptuar ACYPBETA de la validación de versión.
