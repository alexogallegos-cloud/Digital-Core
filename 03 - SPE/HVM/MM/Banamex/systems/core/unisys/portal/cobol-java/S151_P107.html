S151 · P107 a Java 21 Spring Boot

Lee el log LOG151 y su complemento, valida cada clave de transacción, fecha contable, sucursal, libro, sucursal promotora y tipo de crédito contra los catálogos de S080, y en lugar de rechazar sustituye los valores inválidos por los defaults del catálogo de desvíos, avisando por ESENDAUTO y dejando el registro en DESVIOS. Los movimientos válidos se graban en MOVIMIENTOS (para la bitácora) y en CVETRAN (para los totales). Después ordena e imprime la bitácora clasificando cada movimiento como cargo, abono, no contable o eliminado según su naturaleza, acumula por caja, sucursal y CSI, y finalmente ordena CVETRAN para producir los totales por clave, el reporte por proceso y el archivo hacia Mercator o INTELAR.

/** REG-MOVIMIENTOS: archivo de trabajo que alimenta la bitácora impresa y el S028. */
public record MovimientoBitacora(
        int nodoImpresion,           // RM-NODO-IMP 9(04)
        int sucursalIniciativa,      // RM-SUC-INIC
        int cajaIniciativa,
        int tipoCredito, int fondo, int sucursalPromotora,
        String contrato,             // RM-CONTRATO 9(16): String, conserva ceros
        int cveTrans, int tm,
        BigDecimal importe,          // RM-IMPORTE 9(14)V99 COMP: escala 2, sin signo
        long autAplicativa,          // RM-AUT-APLIC 9(08)
        long autTercerNivel,         // RM-AUT-3NIVEL 9(08): 0 = no hay
        long autS151,
        int moneda,
        String referencia,           // RM-REFERENCIA 9(20): excede long, se guarda como String
        int origen,                  // 1 = local; otro = foráneo ("FOR")
        LocalTime hora,              // RM-HORA 9(06) con REDEFINES HH/MM/SS
        LocalDate fechaReal, LocalDate fechaValor,   // 9(08) AAAAMMDD
        long nominaOperador,
        int cveMda, int mdaSucursal, String mdaCuenta,
        int estatus,                 // 1 = vivo; otro = eliminado ("ELI")
        int tipoProceso, int libro,
        Naturaleza naturaleza,       // RM-NATS28
        String tipoCreditoAlfa) {

    public enum Naturaleza { CARGO, ABONO, NO_CONTABLE_9, OTRA;
        public static Naturaleza de(int nats28) {
            return switch (nats28) { case 1 -> CARGO; case 2 -> ABONO; case 9 -> NO_CONTABLE_9; default -> OTRA; };
        }
    }

    /** RN-S151-751: la autorización de tercer nivel prevalece sobre la aplicativa. */
    public long autorizacionReportada() {
        return autTercerNivel == 0 ? autAplicativa : autTercerNivel;
    }
}

@Configuration
@EnableConfigurationProperties({P107Properties.class, P107Rutas.class})
public class BitacoraJobConfig {

    @Bean
    Job bitacoraMovimientos(JobRepository repo,
                            Step inicio,             // 10000-INICIO-PROGRAMA
                            Step cargaCatalogos,     // 11000-CARGA-ESTRUCTURA + 12000-CARGA-CATALOGOS + 12900-OBTEN-TIT940
                            Step procesaMovimientos, // 20000: valida, aplica defaults, graba MOVIMIENTOS y CVETRAN
                            Step generaBitacora,     // 30000: SORT + impresión + MOVSXTERM (S028)
                            Step totalesPorClave,    // 35000: SORT CVETRAN + TOTCVETRAN + Mercator/INTELAR
                            Step rechazos) {         // 60000-GENERA-RECHAZOS
        return new JobBuilder("S151P107", repo)
                .start(inicio)
                .next(cargaCatalogos)
                .next(procesaMovimientos)
                .next(generaBitacora)
                .next(totalesPorClave)
                .next(rechazos)
                .build();
    }
}

@Component
@RequiredArgsConstructor
@Slf4j
public class LibreriaControlResolver {

    private final ControlVersionesClient ctlvers;   // DAME_TIT IN CTRLVERS
    private final P107Rutas rutas;                   // p107.libcontrol-respaldo

    /** Devuelve el destino del servicio de control de día (S151LIBCONTROL). */
    public String resuelve() {
        var r = ctlvers.dameTitulo("S151L001CTL");
        if (r.status() < 0) {
            log.warn("ERROR EN CTLVERS, USO LICBRERIA DE CONTROL SIN RELEASE NI MTP.");   // LJ IN LIB-DISP, código 020101
            return rutas.libcontrolRespaldo();       // "(S151)S151/OBJECT/L001/CONTROL ON CMEMP."
        }
        return r.titulo();
    }
}

public class ValidadorMovimiento implements ItemProcessor<MovimientoLog, MovimientoValidado> {

    private final CatalogoTransacciones catalogo;   // WKS-PT-EXISTE(cve, tm)
    private final CatalogoDesvios desvios;          // WKS-CVETRA-DESVIO, WKS-TM-DESVIO (S080)
    @Value("#{jobExecutionContext['fechaProceso']}") private LocalDate fechaProceso;

    @Override
    public MovimientoValidado process(MovimientoLog m) {
        var d = new Desvios();                       // banderas A..G
        int cve = m.cveTrans(), tm = m.tm();
        boolean tmValido = (tm == 1 || tm == 2);

        if (cve == desvios.cveTrans() || cve < 1 || cve > 10000) {
            cve = desvios.cveTrans(); d.marca('D');           // la propia clave de desvío también se reporta
            if (!tmValido) { tm = desvios.tm(); d.marca('G'); }
        } else if (tmValido) {
            if (!catalogo.existe(cve, tm)) { cve = desvios.cveTrans(); d.marca('D'); }
        } else {
            tm = desvios.tm(); d.marca('G');
            if (!catalogo.existe(cve, tm)) { cve = desvios.cveTrans(); d.marca('D'); }
        }

        LocalDate fechaContable;
        if (m.fechaContable().equals(fechaProceso)) {
            fechaContable = (m.fechaValor() != null) ? m.fechaValor() : m.fechaContable();
        } else {
            fechaContable = fechaProceso; d.marca('A');        // fecha errónea: se fuerza la de proceso
        }
        // ... sucursal (B), sucursal promotora (C), tipo de crédito (E), libro (F)
        return new MovimientoValidado(m, cve, tm, fechaContable, d);   // nunca devuelve null: no rechaza
    }
}
    /** Valida la sucursal promotora y aplica el centralizador contable del libro (crédito y 600). */
    int sucursalPromotora(MovimientoLog m, PerfilSistema p, int indLibro, Desvios d) {
        int prom = m.sucursalPromotora();
        int def  = props.sucursalPromotoraDefault();          // p107.sucursal-promotora-default: 907
        if (p.esCredito() || p.esSistema600()) {             // W88-SIST-CRED OR W88-SIST-CS
            if (catalogo.indS254(m.cveTrans(), m.tm()) != 2) return prom;   // NEXT SENTENCE: sin cambio
            int centralizador = d.tiene('F') ? 0 : libros.centralizador(indLibro);
            if (prom > 0) {
                return centralizador > 0 ? centralizador : prom;
            }
            d.marca('C');
            return centralizador > 0 ? centralizador : def;
        }
        return prom > 0 ? prom : def;                         // otros sistemas: sin bandera C
    }

    public LineaBitacora mapea(MovimientoBitacora m) {
        String localFor = (m.origen() == 1) ? "" : "FOR";
        long autorizacion = m.autorizacionReportada();       // RN-S151-751

        Concepto concepto;
        Montos montos = Montos.CERO;
        if (m.estatus() != 1) {
            concepto = Concepto.ELI;                          // eliminado: se imprime y no se acumula
        } else {
            concepto = switch (m.naturaleza()) {
                case CARGO -> { montos = Montos.cargo(m.importe());  yield Concepto.CAR; }
                case ABONO -> { montos = Montos.abono(m.importe());  yield Concepto.ABO; }
                default    -> { montos = Montos.noContable(m.importe()); yield Concepto.NC; }
            };
        }
        return new LineaBitacora(localFor, m.tipoCredito(), m.fondo(), autorizacion,
                m.moneda(), m.importe(), "CTO", m.sucursalPromotora(), m.contrato(),
                m.hora(), m.nominaOperador(), m.cveTrans(), m.referencia(), concepto, montos);
    }

    public enum Concepto { CAR, ABO, NC, ELI }   // "N/C" en el reporte
}

public class AcumuladorBitacora {

    private static final int MONEDA_OTRAS = 100;          // RM-MONEDA fuera de 1..99
    private final Map<Corte, Totales> caja = new HashMap<>(), sucursal = new HashMap<>(), csi = new HashMap<>();

    /** 31110-ACUMULA: sólo movimientos vivos con naturaleza 1, 2 o 9. */
    public void acumula(MovimientoBitacora m, LineaBitacora l, int indLibro) {
        int idxMoneda = (m.moneda() > 0 && m.moneda() < 100) ? m.moneda() : MONEDA_OTRAS;
        if (m.naturaleza() == Naturaleza.OTRA) {
            return;                                           // se imprimió como N/C pero no se suma
        }
        var k = new Corte(indLibro, idxMoneda);
        for (var mapa : List.of(caja, sucursal, csi)) {
            mapa.merge(k, Totales.de(l.montos()), Totales::mas);   // BigDecimal.add: exacto, escala 2
        }
    }
}

@Component
public class TotalesPorClaveReader implements ItemReader<TotalClave> {

    private final PeekableItemReader<RegCveTran> cvetran;   // CVETRAN ya ordenado por el step anterior
    private static final BigDecimal TOPE_9_16_V99 = new BigDecimal("9999999999999999.99");

    @Override
    public TotalClave read() throws Exception {
        RegCveTran primero = cvetran.read();
        if (primero == null) return null;
        var llave = LlaveClave.de(primero);
        long num = 1;
        BigDecimal imp = primero.importe();
        while (cvetran.peek() != null && LlaveClave.de(cvetran.peek()).equals(llave)) {
            imp = imp.add(cvetran.read().importe());
            num++;
        }
        if (imp.compareTo(TOPE_9_16_V99) > 0) throw new DesbordamientoAcumuladorException(llave, imp);
        return new TotalClave(llave, num, imp);   // el processor agrega leyenda, CAR/ABO/N/C y afecta S115
    }
}

Riesgos de migración
Cuadre contable sin validación de partida doble

CUADRECON está declarado (SELECT en la línea 54, FD en la 277) pero no se abre ni se escribe en ninguna parte del fuente, y WKS-CC-CARGOS y WKS-CC-ABONOS (líneas 4888 y 4890) no se usan en la PROCEDURE DIVISION. El programa reporta totales por clave pero no compara cargos contra abonos.

Datos erróneos se corrigen en silencio

Clave de transacción, TM, fecha contable, libro, tipo de crédito y sucursal promotora se reemplazan por defaults del catálogo de desvíos; la fecha contable distinta de la de proceso se fuerza a la de proceso (desvío A). La sucursal promotora en cero se contabiliza en la 907 sin aviso para sistemas que no son de crédito ni 600.

Diferencias entre bitácora y totales

Los movimientos con sucursal de iniciativa 0 no se escriben en MOVIMIENTOS pero sí en CVETRAN (líneas 6747-6760 y 6717-6726), y los N/C con naturaleza distinta de 9 se imprimen pero no se acumulan (línea 7994). Bitácora impresa, S028 y totales por clave no cuadran entre sí en esos casos.

Hardcodes de sistemas y rangos

Sistemas 252, 402, 403, 404 y 600; rangos de sucursal válidos 1-999 y 4000-4899 (líneas 6840-6841); sucursal 907; título fijo de LIB-DISP y de la librería de control de respaldo. Se trasladan a application.yml.

Desbordamiento silencioso de acumuladores

W77-IMPMOV es PIC 9(16)V99 y los acumuladores de corte no tienen ON SIZE ERROR. La versión Java falla de forma explícita si se excede la capacidad.
