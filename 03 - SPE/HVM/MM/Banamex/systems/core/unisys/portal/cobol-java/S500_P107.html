Al iniciar, P107 valida el TASKVALUE, identifica la plaza por el nombre del host para fijar CSI y sucursal, toma la fecha de lote y la fecha de línea de S500B02CONTROL y decide si la corrida es mensual. En el proceso EPP carga de S500B17CVESXFUN los meses de vigencia por clave de cobro y su fecha proyectada con LOCSUP, recorre B49 marcando como pagadas (estatus 4) las comisiones con saldo cero y como vencidas (estatus 3) las que tienen saldo y una fecha de línea anterior a la proyectada, refuerza vencimientos por asistencia y genera el archivo COMPENDEPP con lo actualizado en el día o, si es fin de mes, con todo B49. En el proceso TESOFE compara los saldos de cada cuenta control contra la suma de sus subcuentas, reporta y avisa a S151 las diferencias, y reinicia los campos TESOFE de las cuentas que lo tienen pendiente.

public enum ModoP107 {
    EPP(1), TESOFE(2), TODO(3);
    final int valor;
    ModoP107(int v) { valor = v; }
    boolean incluyeEpp()    { return this == EPP || this == TODO; }
    boolean incluyeTesofe() { return this == TESOFE || this == TODO; }

    /** W88-TASK-VAL-OK: cualquier otro valor aborta (CHANGE ATTRIBUTE STATUS OF MYSELF TO -1). */
    static ModoP107 de(long taskValue) {
        return Arrays.stream(values()).filter(m -> m.valor == taskValue).findFirst()
                .orElseThrow(() -> new JobParametersInvalidException("ERROR TASK VALUE " + taskValue));
    }
}

@Component
@RequiredArgsConstructor
public class TablaVigenciaEpp {

    private final ClavesPorFuncionRepository b17;   // S500B17CVESXFUN
    private final CalendarioBancario locsup;        // S006LOCSUP

    public record Vigencia(int mesesVigencia, int indicadorVencimiento, LocalDate fechaProyectada) {}
    private final Map<Integer, Vigencia> porClave = new HashMap<>();   // WS-EPP-TABLA OCCURS 10000

    public void carga(LocalDate fecBase) {
        Map<Integer, LocalDate> proyectadaPorMeses = new HashMap<>();
        for (var r : b17.findAll()) {
            int clave = r.claveTransaccion();
            if (r.mesesVigencia() <= 0 || clave <= 0 || clave >= 10_000) continue;
            // LOCSUP función 22, formato 13: fecha base desplazada N meses ("PROY-FECH-ATRAS")
            LocalDate proy = proyectadaPorMeses.computeIfAbsent(r.mesesVigencia(),
                    m -> locsup.funcion22(fecBase, m)
                               .orElseThrow(() -> new IllegalStateException("LOCSUP sin fecha para " + m)));
            porClave.put(clave, new Vigencia(r.mesesVigencia(), r.indicadorVencimiento(), proy));
        }
    }

    public Optional<Vigencia> de(int claveCobro) { return Optional.ofNullable(porClave.get(claveCobro)); }
}

@Component
@StepScope
@RequiredArgsConstructor
public class RevisaComisionProcessor implements ItemProcessor<ComisionEpp, ComisionEpp> {

    static final int ACTIVA = 0, VENCIDA = 3, PAGADA = 4;

    private final TablaVigenciaEpp vigencias;
    @Value("#{jobExecutionContext['fecBase']}") private LocalDate fecBase;

    @Override
    public ComisionEpp process(ComisionEpp c) {
        if (c.getStatus() != ACTIVA) return null;                      // solo comisiones activas
        if (c.getSaldo().signum() == 0) {                              // 50001300-COMISION-PAGADA
            return c.marca(PAGADA, fecBase);
        }
        return vigencias.de(c.getClaveCobro())                         // 50001620-VALIDA-CLAVE
                .filter(v -> v.mesesVigencia() > 0 && v.indicadorVencimiento() == 0)
                .filter(v -> c.getFechaLinea().isBefore(v.fechaProyectada()))   // 50001630
                .map(v -> c.marca(VENCIDA, fecBase))
                .orElse(null);                                         // sin cambio: no se reescribe
    }
}

/** 50004400-600-VAL-SAL-BLO: decide si se reinicia el árbol TESOFE de la cuenta control. */
@Component
public class ReinicioTesofePolicy {

    static final int BLOQUEO_TESOFE = 14;

    public boolean reinicia(ContratoCaptacion b03) {
        boolean sinBloqueoTesofe = b03.impSolBloTes().signum() == 0;
        int st = b03.status(), tipo = b03.tipoBloqueo();
        boolean bloqueoGeneral = b03.impSolBloq().signum() > 0;

        if (sinBloqueoTesofe && st == 1 && tipo != BLOQUEO_TESOFE)              return true;
        if (sinBloqueoTesofe && st == 1 && tipo == BLOQUEO_TESOFE && bloqueoGeneral)  return true;
        if (sinBloqueoTesofe && st == 0 && tipo == 0 && bloqueoGeneral)         return true;
        if (sinBloqueoTesofe && st == 1 && tipo == BLOQUEO_TESOFE && !bloqueoGeneral) return false;
        if (sinBloqueoTesofe && st < 2)                                         return true;
        return false;   // IMPSOLBLO-TES > 0 o STATUS > 1: solo se limpia el indicador de reinicio
    }
}
