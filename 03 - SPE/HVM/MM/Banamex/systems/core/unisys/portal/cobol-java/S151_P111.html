S151 · P111 a Java 21 Spring Boot

P111 obtiene de LIB-CONTROL la fecha de proceso del sistema y el nombre de su base semanal, o toma la fecha del TASKVALUE si viene informada. Abre la base en actualización con reintentos, borra de B50 todos los totales con fecha menor o igual a la de proceso y lee el archivo de movimientos. Por cada movimiento con función 1 y estatus 1 escribe un registro de trabajo por cada uno de sus hasta cinco conceptos con clave de transacción, aplicando reglas propias de S500 (normalización del tipo de proceso, bancos y exclusión Teletón). Ordena el archivo de trabajo, corta por la llave de 42 posiciones y graba en B50 el número de movimientos y el importe de cada llave.

@Component
@StepScope
@RequiredArgsConstructor
class FechaProcesoTasklet implements Tasklet {

    private final ControlSistemaClient control;          // CONSISDIA IN LIB-CONTROL, función 1
    @Value("#{jobParameters['sistema']}")      private Integer sistema;        // WKS-PARAM-SIS
    @Value("#{jobParameters['fechaProceso']}") private LocalDate fechaParam;   // TASKVALUE

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        ControlDiario b01 = switch (control.consultaDia(sistema)) {
            case ControlDiario.Ok ok        -> ok.valor();
            case ControlDiario.NoExiste n   -> throw new ProcesoAbortadoException("SISTEMA NO DADO DE ALTA");
            case ControlDiario.Error e      -> throw new ProcesoAbortadoException("ERROR EN BASE DE CONTROL");
        };
        LocalDate fecha = (fechaParam != null) ? fechaParam : b01.fechaProceso();   // B01-FECPRO
        var ec = c.getStepExecution().getJobExecution().getExecutionContext();
        ec.put("fechaProceso", fecha);
        ec.put("baseSemanal", b01.nombreBaseSemanal());                            // B01-NOMBDSEM
        ec.put("estatusBaseSemanal", b01.estatusBaseSemanal());                    // B01-STABDSEM
        return RepeatStatus.FINISHED;
    }
}

@Component
@RequiredArgsConstructor
class AperturaBaseSemanal {

    private final BaseSemanalDataSource base;
    private final P111Properties props;

    /** 000700-ABRE-BASE. Reintenta mientras la base esté en uso exclusivo (DMERRORTYPE 39). */
    void abre(int estatusBaseSemanal) throws InterruptedException {
        if (estatusBaseSemanal != 0) {
            throw new ProcesoAbortadoException("ESTADO INVALIDO DE LA BASE DE MOVTOS");
        }
        for (int intento = 1; ; intento++) {
            try {
                base.abreParaActualizar();
                return;
            } catch (BaseEnUsoExclusivoException e) {
                if (intento % props.apertura().avisoCada() == 0) {
                    log.warn("ERROR EN BASE BD10 : ARCHIVO DE CONTROL EN USO EN FORMA EXCLUSIVA");
                }
                Thread.sleep(props.apertura().espera().toMillis());
            } catch (DataAccessException e) {
                throw new ProcesoAbortadoException("ERROR AL ABRIR LA BASE SEMANAL", e);
            }
        }
    }
}

/** Lector del archivo de movimientos con el centinela de función 99. */
@Component
@StepScope
@RequiredArgsConstructor
class MovimientoReader implements ItemReader<Movimiento> {

    private final FlatFileItemReader<Movimiento> archivo;     // L01-DATOS, el header ya se consumió
    private final FiltroTeleton teleton;
    private boolean fin;

    @Override
    public Movimiento read() throws Exception {
        while (!fin) {
            Movimiento m = archivo.read();
            if (m == null || m.funcion() == 99) {             // AT END o trailer lógico
                fin = true;
                return null;
            }
            if (!teleton.excluye(m)) {
                return m;
            }
        }
        return null;
    }
}
@Component
@StepScope
@RequiredArgsConstructor
class ConceptosProcessor implements ItemProcessor<Movimiento, List<RegistroTotal>> {

    @Value("#{jobParameters['sistema']}") private Integer sistema;
    private final P111Properties props;   // p111.s500.claves-reclasifica: [3009, 3036]
                                          // p111.sistemas-sin-banco-default: [403, 404]

    /** 001000-ARMA-ARCHIVO. */
    @Override
    public List<RegistroTotal> process(Movimiento m) {
        int producto = (sistema == 87) ? 87 : m.producto();
        int tipoProceso = m.tipoProceso();
        if (sistema == 500) {
            boolean reclasifica = (tipoProceso >= 15 && tipoProceso <= 17)
                    && props.s500().clavesReclasifica().contains(m.claveTransaccion(1));
            tipoProceso = (tipoProceso < 15 || reclasifica) ? 1 : 11;
        }
        if (m.funcion() != 1 || m.estatus() != 1) {
            return null;                                  // no se escribe nada
        }
        Bancos bancos = switch (sistema) {
            case 500 -> new Bancos(m.bancoOrigen(), m.bancoDestino());
            case 403, 404 -> Bancos.SIN_ASIGNAR;           // COBOL: NEXT SENTENCE, ver riesgo
            default -> new Bancos(2, 2);
        };
        var base = new LlaveTotal(m.fechaContable(), tipoProceso, producto,
                                  m.instrumento(), m.moneda(), bancos, 0);
        return conceptos(base, m);                         // 001100-ARMA-REGISTRO x 5
    }
}
@Component
@RequiredArgsConstructor
class FiltroTeleton {
    private final P111Properties props;   // p111.s500.tipos-teleton: [40, 41, 42, 43]
    @Value("#{jobParameters['sistema']}") private Integer sistema;

    /** <<INICIO-IBM>> MGGR EXCLUSION DE MOVIMIENTOS TELETON. */
    boolean excluye(Movimiento m) {
        return sistema == 500 && props.s500().tiposTeleton().contains(m.tipoProceso());
    }
}

/** 001100-ARMA-REGISTRO VARYING W77-IND FROM 1 BY 1 UNTIL W77-IND > 5. */
static List<RegistroTotal> conceptos(LlaveTotal base, Movimiento m) {
    return IntStream.rangeClosed(1, 5)
            .mapToObj(m::concepto)
            .filter(c -> c.claveTransaccion() > 0)
            .map(c -> new RegistroTotal(base.con(c.claveTransaccion(), c.esquemaContable()),
                                        c.importe()))          // 9(16)V99 -> escala 2
            .toList();
}

Riesgos de migración
Filtro de fecha contable documentado pero no programado

El encabezado (líneas 18-20) y RN-S151-812 dicen que solo se toman movimientos con FECCONT menor o igual a la fecha de proceso; en 001000-ARMA-ARCHIVO solo se evalúan función y estatus (líneas 2164 y 2183) y FECCONT se copia a la llave sin compararse.

Primer movimiento sin exclusión Teletón

La lectura del primer detalle en el párrafo principal (líneas 1922-1929) no aplica el filtro de tipos 40 a 43 de S500 que sí tiene 001000-LEE-REGISTRO.

Bancos sin asignar para 403 y 404

La rama NEXT SENTENCE (líneas 2177-2178) deja WKS-RS-BCOORI y WKS-RS-BCODES con el contenido inicial del área de trabajo, que forma parte de la llave de B50.

Borrado y carga sin auditoría ni atomicidad

Cada DELETE y cada STORE corre en su propia transacción NO-AUDIT sobre S151B99REINICIO (líneas 2023-2036 y 2310-2330). Una falla a mitad deja B50 parcialmente borrado o cargado y los errores de DMSII solo se registran.

Reintento infinito al abrir la base

000700-ABRE-BASE reintenta cada 20 segundos sin tope mientras la base esté en uso exclusivo (líneas 1960-2001).

Hardcodes por sistema

Los sistemas 87, 403, 404 y 500, las claves 3009 y 3036, los tipos Teletón 40 a 43, el banco por omisión 2 y el título de LIB-CONTROL de respaldo (línea 1810) están compilados. Se trasladan a configuración.
