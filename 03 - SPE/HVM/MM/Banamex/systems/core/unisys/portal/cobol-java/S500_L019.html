@Component
public class SaldoDisponibleCalculator {

    public record Saldos(BigDecimal actual, BigDecimal disponible, BigDecimal pendiente) {}

    /** SALDO_DISP. Importes en pesos con escala 2; el fuente los lleva en centavos enteros. */
    public Saldos calcula(SaldoContrato b03, BaseCaptacion base, TipoProceso proceso, LocalDate fechaLinea) {
        BigDecimal actual = b03.saldoActual(), pendiente = b03.saldoPendiente(), protegido = b03.saldoProtegido();

        // (a) corte no aplicado en batch sobre captación
        if (base == BaseCaptacion.CAPTACION && proceso == TipoProceso.BATCH
                && !fechaLinea.equals(b03.fechaSaldoPendiente())) {
            actual = actual.add(pendiente);
            pendiente = b03.saldoSbc().add(b03.sbcUnDia());
            protegido = BigDecimal.ZERO;
        }
        // (b) y (c)
        BigDecimal disponible = actual;
        actual = actual.add(pendiente);
        if (b03.indicadorSaldoPendiente() == 1) {
            disponible = disponible.add(pendiente.subtract(protegido));
            pendiente = protegido;
        }
        // (d) el pago por cobrar caduca en línea fuera de la ventana ciega
        BigDecimal pagoPorCobrar = (proceso == TipoProceso.LINEA
                && !fechaLinea.equals(b03.fechaVentanaCiega())) ? BigDecimal.ZERO : b03.pagoPorCobrar();
        // (e) cinco retenciones y piso en cero
        disponible = disponible
                .subtract(b03.saldoTesDisponible())
                .subtract(b03.cargoPendienteSabado())
                .subtract(b03.autorizacionesPendientes())
                .subtract(b03.saldoBloqueadoGce())
                .subtract(pagoPorCobrar)
                .max(BigDecimal.ZERO);
        return new Saldos(actual, disponible, pendiente);
    }
}

@Component
@RequiredArgsConstructor
public class ConversionDolares {

    private final SaldosProperties props;

    /**
     * SALDO_ACT_CTAMAE_DLS_MN y SALDO_ACT_CTAPER_DLS_MN. Los tipos de cambio llegan como
     * BigDecimal con su escala real, así que no hace falta el factor 10000 del fuente.
     */
    public Optional<BigDecimal> enPesos(ProductoKey p, BigDecimal saldoDolares, TiposCambio tc) {
        return props.dolares().stream()
                .filter(d -> d.coincide(p))
                .findFirst()
                .map(d -> saldoDolares.multiply(tc.de(d.tipoCambio()))
                                      .setScale(2, RoundingMode.DOWN));   // DIV entera del fuente: trunca
    }
}

public record AcumuladosDepRet(BigDecimal depDia, BigDecimal depMes, BigDecimal retDia,
                               BigDecimal retMes, BigDecimal saldoConsolidadoDep) {

    static final AcumuladosDepRet CERO = new AcumuladosDepRet(

Sin fuente para verificar

Escala de la Cuenta Maestra en dólares

El pseudocódigo reporta que la división entre 10 000 de SALDO_ACT_CTAPER_DLS_MN y SALDO_DISP_CTAPER_DLS_MN queda fuera del IF y subvalúa el saldo en pesos de la Cuenta Maestra en dólares (hallazgos A-1 y A-2).
