S151 · P167 a Java 21 Spring Boot

P167 recibe como parámetro la secuencia de ejecución y el sistema. Valida que el header del archivo SDOS traiga la fecha de proceso obtenida de LIB-CONTROL (o del TASKVALUE de la tarea) y recorre los registros tipo 81. Con secuencia 2 reparte los saldos en SDO0001 (cheques) y SDO0066 (cuenta maestra) para S050 y acumula cifras de control de la clave de saldo 2. Con secuencia 1 arma el registro ACCI de 360 bytes para Citi, lo enriquece con datos de cliente vía S016L422, ajusta el saldo restando la línea de crédito consultada en S408 (CONSULTASALDO), escribe el archivo final con un registro END OF FILE y lo envía con un WFL cuando el CSI es 04.

                 String sucursal, String contrato,          // String: conserva ceros a la izquierda
                 int edoCta, LocalDate fecPro, LocalDate fecIni, LocalDate fecFin,
                 LocalDate fecUltMov, int status,

            implements SaldoSdosRecord {

        /** Primer importe con la clave indicada; P167 usa 2 (saldo actual) y 354 (disponible). */
        Optional<BigDecimal> importe(int cveSdo) {
            return importes.stream().filter(c -> c.cveSdo() == cveSdo)
                           .map(ClaveImporte::importe).findFirst();
        }
    }

    record Otro(int tipoReg, byte[] raw) implements SaldoSdosRecord {}

@Component

public class HeaderSdosValidator implements StepExecutionListener {

    @Value("#{jobExecutionContext['fechaProceso']}") private LocalDate fechaProceso;

    /**
     * 150-007-VALIDA-HEADER. En el mainframe un header erróneo detiene la tarea con STOP
     * y espera a que el operador copie el archivo correcto; al continuar, el PERFORM UNTIL
     * vuelve a abrir y leer. Aquí el step falla y el reinicio del job hace el mismo papel.

@Component
@StepScope
@RequiredArgsConstructor
public class RuteoSaldosProcessor implements ItemProcessor<SaldoSdosRecord, Ruteado> {

    @Value("#{jobParameters['secuencia']}") private int secuencia;

    @Override
    public Ruteado process(SaldoSdosRecord rec) {
        if (!(rec instanceof SaldoSdosRecord.Saldo s) || s.tipoReg() != 81) {
            return null;                          // solo los tipo 81 producen salida
        }
        if (secuencia == 1) {
            return enviaACiti(s) ? new Ruteado(Destino.CITI, s) : null;
        }
        if (secuencia == 2) {
            boolean cuentaMaestra = s.producto() == 66
                    || (s.producto() == 500 && s.instrumento() == 1);
            if (cuentaMaestra)        return new Ruteado(Destino.SDO0066, s);
            if (s.producto() != 500)  return new Ruteado(Destino.SDO0001, s);   // 150-040
            return null;              // producto 500 con instrumento distinto de 1: se descarta
        }
        return null;
    }

    /** Precedencia literal del IF de las líneas 7132-7137: cheques 11-15 O indicador 1/2. */
    private static boolean enviaACiti(SaldoSdosRecord.Saldo s) {
        boolean chequesInstr11a15 = s.producto() == 1 && s.instrumento() > 10 && s.instrumento() < 16;
        boolean indicador = s.s500().indEnvCtd() == 1 || s.s500().indEnvCtd() == 2;
        return chequesInstr11a15 || indicador;
    }
}
// El writer es un ClassifierCompositeItemWriter que manda Destino.SDO0001 a
// S500/FILE/S050/{csi}/{csi}/SDO0001/{aammdd} y Destino.SDO0066 al archivo SDO0066.
/** W77-NUMREG-*, W77-NUMCTO-*, W77-SDOACT-* (S9(15)V99 COMP) por archivo de salida. */
public final class CifrasControl {
    private long registros;
    private long ocurrenciasCve2;           // W77-NUMCTO: cuenta claves 2, no contratos
    private BigDecimal saldoActual = BigDecimal.ZERO.setScale(2);
    private static final BigDecimal TOPE = new BigDecimal("999999999999999.99");

    /** 150-030 / 150-040 + 150-035 / 150-045: recorre las 30 ocurrencias. */
    void acumula(SaldoSdosRecord.Saldo s) {
        registros++;
        for (var ci : s.importes()) {
            if (ci.cveSdo() == 2) {
                ocurrenciasCve2++;
                saldoActual = saldoActual.add(ci.importe());      // escala 2, suma exacta
                if (saldoActual.abs().compareTo(TOPE) > 0) {
                    throw new DesbordamientoAcumuladorException(saldoActual);
                }
            }
        }
    }

    /** Trailer tipo 99 del archivo y renglón WKS-DETALLE-1 del reporte R01-CIFRAS. */
    TrailerS050 trailer() { return new TrailerS050(99, registros, ocurrenciasCve2, saldoActual); }
}

@Component
@RequiredArgsConstructor
public class AcciMapper {

    private final P167Properties props;       // p167.citi.*: valores fijos del layout
    private final ClienteS016Adapter clientes;

    public AcciRecord map(SaldoSdosRecord.Saldo s) {
        boolean usd = s.s500().moneda() == 5;
        char estatus = switch (s.status()) { case 0 -> ' '; case 1 -> 'B'; default -> 'C'; };

        // 190-CVESDO-CBII: recorre las 30 ocurrencias; si una clave se repite gana la última.
        var actual = AcciRecord.SignedAmount.of(BigDecimal.ZERO.setScale(2));
        var disponible = actual;
        for (var ci : s.importes()) {
            if (ci.cveSdo() == 2)   actual = AcciRecord.SignedAmount.of(ci.importe());
            if (ci.cveSdo() == 354) disponible = AcciRecord.SignedAmount.of(ci.importe());
        }

        var datosCliente = clientes.consulta(s);      // 150-050-LLAMA-CTES
        return new AcciRecord("ACI", props.citi().branch(),            // 484
                datosCliente.genCust(),
                usd ? 126 : 0, datosCliente.acctNbr(), datosCliente.liaNbr(),
                usd ? "USD" : "MXN", datosCliente.nombreCorto(), estatus,
                props.citi().catCode(),                                   // 1
                Fechas.formatoCiti(s.fecUltMov()),                       // 9(07): formato por confirmar
                actual, actual,                                           // misma cifra en moneda local
                disponible, s.contrato());
    }
}

@Component

public class AjusteLineaS408Processor implements ItemProcessor<AcciRecord, AcciRecord> {

    private final CreditoS408Client s408;   // CONSULTASALDO IN CONL020, función 02
    @Value("#{jobExecutionContext['fechaProceso']}") private LocalDate fechaProceso;

    @Override
    @Retryable(retryFor = S408OcupadoException.class,      // resultado 96: WAIT(300)
               maxAttempts = Integer.MAX_VALUE,
               backoff = @Backoff(delay = 300_000))
    public AcciRecord process(AcciRecord a) {
        var r = s408.consultaSaldo(a.contrato(), fechaProceso);
        if (r.resultado() == 96) throw new S408OcupadoException();
        if (r.resultado() != 0) {
            log.error("ERROR AL CONSULTAR S408 CON CTO. : {} RESULT : {}", a.contrato(), r.resultado());
            return a;                                         // se envía sin ajuste
        }
        if (r.saldoLinea().signum() <= 0) return a;
        // COMPUTE sobre ACCI-ACCT-BAL-AMT-T, que es magnitud sin signo: se respeta tal cual.
        BigDecimal ajustado = a.acctBal().magnitud().subtract(r.saldoLinea());
        var saldo = new AcciRecord.SignedAmount(ajustado.abs(), ajustado.signum() > 0 ? '+' : '-');
        return a.conSaldo(saldo, saldo).conFechaUltMov(fechaProceso);
    }

Riesgos de migración
Reintento por S408 solo en el primer registro

300-LLAMA-S408 se repite hasta W88-S408-OK solo para el primer registro (línea 7449). Los siguientes pasan por 310-LLAMA-S408 una sola vez desde 320-LEE-CBII-T: si S408 responde 96, el programa espera 300 segundos pero no reintenta y el registro siguiente se escribe sin ajuste de línea.

Fallback de librerías sobre LIB-CONTROL

Cuando CTLVERS no resuelve S016L422 o S408L020, el código cambia el TITLE de LIB-CONTROL en lugar del de la librería que falló (líneas 6918-6919 y 6934-6935).

Precedencia del filtro hacia Citi

La condición de las líneas 7098-7103 y 7132-7137 envía a Citi cualquier registro con indicador de envío 1 o 2 sin importar el producto, además de cheques con instrumento 11 a 15.

Saldos con signo perdido en el ajuste

El ajuste por línea de S408 resta sobre la magnitud del saldo sin su signo, de modo que saldos deudores pueden cambiar de signo de forma incorrecta.

Hardcodes de rutas y envío

Rutas S500/FILE/S050/, (S151)S151/FILE/E01/S500ACC/ y ACCMONITOR, el nodo /10/, la familia CMEMP, el título de respaldo de LIB-CONTROL y el WFL CORRESOLO P102 que se arma en WKS-COPY-L030 y solo se dispara con CSI 04 (41230-ENVIA-ACC). Se trasladan a application.yml.
