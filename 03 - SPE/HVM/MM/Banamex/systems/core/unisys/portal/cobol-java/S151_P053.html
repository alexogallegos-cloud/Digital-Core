S151 · P053 a Java 21 Spring Boot

Al arrancar carga las tablas de sistemas, sucursales, librerías de consulta y el catálogo de leyendas TAX-ID, y luego queda en un ciclo que espera eventos de COMS con ESPERACOMS. Cada mensaje recibido se clasifica por el código de transacción de la cabecera SA0, se validan sistema, sucursal, cuenta y fecha con JUSTIFIER, se decide si el mensaje se reenvía a otro CSI y, si se atiende localmente, se piden hasta 20 movimientos a PLATINO de S151L011 o a la librería de consulta del sistema. La respuesta se devuelve por SEND COMS y la operación se registra en bitácora con GRABA_BITACORA. Opcionalmente, con HI 2, cada mensaje de entrada y salida se copia a un archivo de monitor protegido.

@Configuration
@RequiredArgsConstructor
public class ConsultaMovimientosFlowConfig {

    private final EnrutadorConsultas enrutador;   // 300000-PROCESA-MSG / 300000-PROCESA-TRANSACC
    private final MonitorComs monitor;            // ARCH-MONITOR

    /**
     * 000000-INIPGM: PERFORM 200000-PROCESO UNTIL W77-FIN = 1.
     * El ciclo explícito lo sustituye el contenedor del adaptador de entrada,
     * que entrega un mensaje por invocación y se detiene con el ciclo de vida de Spring.
     */
    @Bean
    IntegrationFlow consultaMovimientos(ConnectionFactory cf) {
        return IntegrationFlow
                .from(Jms.inboundGateway(cf).requestDestination("S151.P053.REQ"))   // RECEIVE CS-COMS-IN
                .<MensajeComs>filter(MensajeComs::esDeAplicacion,                     // W88-MSGRES-APL (2)
                        f -> f.discardFlow(d -> d.handle(this::rechazoCabecera)))
                .wireTap(monitor.entrada())                                          // "RECV COMS: "
                .handle(MensajeComs.class, (m, h) -> enrutador.procesa(m))
                .wireTap(monitor.salida())                                           // "ENV. COMS: "
                .get();                                                              // SEND CS-COMS-OUT
    }

    /** Rama ELSE de 200000-PROCESO: cabecera que no es de aplicación, error 31 con texto de 195. */
    private MensajeComs rechazoCabecera(MensajeComs m, MessageHeaders h) {
        return m.respuestaError(31, 195);
    }
}

@Component
@Endpoint(id = "p053")
@RequiredArgsConstructor
public class P053AdminEndpoint {

    private final MonitorComs monitor;
    private final ApplicationContext ctx;

    @WriteOperation                                   // HI 2 / HI 3
    public String monitor(@Selector Accion accion) {
        return switch (accion) {
            case ENCIENDE -> monitor.enciende()       // CHANGE ATTRIBUTE TITLE + OPEN OUTPUT
                    ? "PRENDE MONITOR" : "MONITOR YA ENCENDIDO";
            case APAGA    -> monitor.apaga()          // CLOSE ARCH-MONITOR WITH SAVE
                    ? "APAGA MONITOR" : "MONITOR YA APAGADO";
        };
    }

    @DeleteOperation                                  // HI 4: W77-FIN = 1, termina tras el mensaje en curso
    public String finNormal() {
        new Thread(() -> SpringApplication.exit(ctx, () -> 0)).start();
        return "FIN NORMAL POR OPERADOR";
    }

    public enum Accion { ENCIENDE, APAGA }
}
@Component
@Slf4j
public class RecepcionComs {

    /** CS-COMS-IN-STATUS-KEY aceptados: 0, 89, 92, 93. 91 descarta, 99 termina. */
    private static final Set<Integer> ACEPTADOS = Set.of(0, 89, 92, 93);

    public Optional<MensajeComs> evalua(int statusKey, MensajeComs m) {
        if (statusKey == 99) {
            log.error("FIN DEL PASO P053 STAUS {}", statusKey);   // en el fuente el texto nunca se despliega
            throw new FinDeServicioException(statusKey);            // STOP RUN
        }
        if (statusKey == 91) {
            return Optional.empty();                                // W77-CONTROL = 1, sin aviso
        }
        if (!ACEPTADOS.contains(statusKey)) {
            log.warn(" >> ERROR RECEIVE MSG >>> status={}", statusKey);
            return Optional.empty();
        }
        // W88-MSGRES-RED / APL / ERR / MAS / TER: cualquier otra cabecera se ignora
        return m.cabeceraConocida() ? Optional.of(m) : Optional.empty();
    }
}
@Component
@RequiredArgsConstructor
public class EnvioComs {

    private static final int LONG_CABECERA = 36;      // ADD 36 TO W77-LONG-MSG
    private final MonitorComs monitor;
    private final ComsGateway gateway;                 // SEND CS-COMS-OUT FROM WKS-MSG

    public void responde(MensajeComs m, int longTexto) {
        if (m.resultadoCabecera() == 2) {              // VA PARA OTRO CSI
            envia(m, 200);                             // sin traza de monitor
            return;
        }
        MensajeComs salida = m.esDeAplicacion()
                ? m.conOrigenDestinoInvertidos()      // 6000210 / 6000230 / 6000220
                   .conLongitud(m.sa2Resultado() == 2 ? 195 : longTexto)

public class ConsultaDoceService {

    private final P053Properties props;          // p053.consulta12.sistema-permitido: 500
    private final CatalogoSistemas sistemas;     // WKS-APU-SIS, WKS-TAB-CSIRES
    private final ValidadorCampos validador;     // 420000 / 420030 / 420040
    private final PlatinoClient platino;         // PLATINO IN S151L011
    private final Bitacora bitacora;

    public MensajeComs atiende(MensajeComs m) {
        var e = m.entradaTrans12();
        // W77-RES-MSG: el primer error detectado se conserva y las validaciones siguientes se saltan
        OptionalInt err = (e.sistema() == props.consulta12().sistemaPermitido())
                ? validador.sistema(e.sistemaAlfa())                    // 420000: 42
                : OptionalInt.of(42);
        if (err.isEmpty()) {
            var sis = sistemas.porNumero(e.sistema());
            if (sis.isEmpty()) {
                err = OptionalInt.of(40);
            } else if (sis.get().csiResidente() > 0 && !m.yaCambioCsi()
                       && sis.get().csiResidente() != m.csiDestino()) {
                return m.reenviaA(sis.get().csiResidente());          // EXIT PERFORM: sin bitácora
            }
        }
        if (err.isEmpty()) err = validador.sucursal(e.sucursalAlfa(), m);   // 420030: puede marcar reenvío
        if (m.esReenvio()) return m;                                  // WKS-MSGHDR-RES = 2
        if (err.isEmpty() && e.cuentaEsCero()) err = OptionalInt.of(46);    // única validación efectiva de cuenta

        MensajeComs r;
        if (err.isPresent()) {
            r = m.respuestaError(err.getAsInt(), 190);
        } else {
            var resp = platino.consulta(TipoConsulta.CON_012, m.versionOpcion(),
                                        e.aPmc(ConsultaMovimientos.MAX_MOVS));

                case 0       -> m.respuestaMovimientos(resp, 2486);
                case 1, 2, 7 -> m.respuestaError(39, 190);              // 310012-0090-TIPO-ERROR
                case 4       -> m.respuestaError(19, 190);
                default      -> m.respuestaError(resp.resultado(), 190);
            };
        }
        if (m.csiHost() != props.csiSinBitacora()) bitacora.graba(m, r);   // p053.csi-sin-bitacora: 32
        return r;
    }

Riesgos de migración
Validación de cuenta invertida en la consulta 12

En 310012-TRANS-12 (líneas 11451-11453) 420040-VALIDA-NUM-CTA sólo se ejecuta si la cuenta es cero, y después se fuerza el error 46 de todos modos. Las cuentas distintas de cero no se validan como numéricas antes de llamar a PLATINO.

Terminación por status 99 sin aviso ni cierre

En 600000-RECEIVE-MSG (líneas 12355-12359) el texto de fin se arma pero no se despliega y se ejecuta STOP RUN directo: no se cierra ARCH-MONITOR con SAVE ni se ejecuta 999999-FIN, que deshabilita el programa en COMS.

Hardcodes de nodo y sistema

El CSI 32 condiciona la carga de sucursales, la bitácora y el ruteo; el sistema 500 es el único aceptado en la consulta 12 (línea 11426) y tiene una librería de consulta fija LIB-CONS0500 (línea 12130); los sistemas 1, 66, 501 y 264 tienen excepciones de ruteo en 420030-VALIDA-NUM-SUC. Se trasladan a application.yml.

Errores de envío sólo en consola

600100-SEND-COMS despliega el status cuando SEND falla (distinto de 0 y 96) y continúa; la terminal no recibe respuesta y no hay reintento ni registro en bitácora del fallo.

Saldo sin signo en el contrato con L011

PMC-MOV-SALDO es PIC 9(14)V99 COMP sin signo en las tres variantes del área de diálogo, por lo que la consulta no puede devolver saldos negativos a la pantalla.
