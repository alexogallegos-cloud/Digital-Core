Recorre S500B03CONTRATOS, descarta clones, contratos dados de baja, migrados a S111 y cuentas de ahorro, y calcula si el contrato corta entre la fecha de batch y la siguiente. Para los que cortan y tienen saldos integrados o comisión por consulta, bloquea su registro B06, llama a S050 para traer saldos de fondos, banca de inversión, plazo y mesa de dinero o los contadores de consultas, marca B06-ID-S050 y graba el histórico. Al terminar enciende B02-STA-OTRINV para que una nueva corrida en el mismo día no repita la carga.

@Component
@RequiredArgsConstructor
public class CatalogoInstrumentos {

    private static final int MAX_INSTRUMENTOS = 30;   // WKS-TB-B05INSTRUMEN OCCURS 30
    private static final List<Integer> OCCURS_COMISION_CONSULTA = List.of(6, 24, 25);
    private final InstrumentoRepository b05;

    private Map<Pim, Instrumento> porClave;

    @PostConstruct
    void carga() {
        var todos = b05.findAll();
        if (todos.size() > MAX_INSTRUMENTOS) {
            throw new ProcesoAbortadoException("MAS DE 30 INST. EN LA BASE");   // DMTERMINATE
        }
        porClave = todos.stream().collect(toMap(Instrumento::pim, identity()));
    }

    /** WKS-TB-B05-S050L190: el producto cobra consulta si algún occurs 06, 24 o 25 trae tarifa y comisión. */
    static boolean usaTarifasL190(Instrumento i) {
        return OCCURS_COMISION_CONSULTA.stream()
                .anyMatch(n -> i.claveTarifa(n) > 0 && i.claveComision(n) > 0);
    }

    Optional<Instrumento> busca(Pim pim) { return Optional.ofNullable(porClave.get(pim)); }
}

@Component
@StepScope
@RequiredArgsConstructor
public class ContratoCorteFilter implements ItemProcessor<Contrato, ContratoACargar> {

    private final CatalogoInstrumentos catalogo;
    private final P199Properties props;
    private final FechaCorte fechaCorte;

    @Override
    public ContratoACargar process(Contrato c) {
        if (c.marcaClon() == 3) return null;                               // extended: solo se cuenta
        boolean vigente = (c.status() < 2 || c.status() == 5) && c.migradoS111() != 1;
        if (!vigente || props.esAhorros(c.pim())) return null;

        int persona = c.tipoPersona().esFisica() ? 1 : 2;                 // W77-IND-PERS
        if (!fechaCorte.cortaEnVentana(c)) return null;                   // 50003160-VALIDA-CORTE

        var inst = catalogo.busca(c.pim()).orElse(null);
        if (inst == null) {
            log.warn("NO EXISTE EN B05 {} {}", c.pim(), c.numContrato());
            return null;
        }
        boolean saldos = inst.saldosIntegrados(persona);                  // WKS-TB-B05-SI-SDOSINTEG
        boolean consultas = CatalogoInstrumentos.usaTarifasL190(inst);
        return (saldos || consultas)
                ? new ContratoACargar(c, inst, persona, saldos, consultas) : null;
    }
}
@Component
@StepScope
public class FechaCorte {

    @Value("#{jobExecutionContext['fechaBatch']}")     private LocalDate fechaBatch;
    @Value("#{jobExecutionContext['fechaSiguiente']}") private LocalDate fechaSiguiente;   // WKS-FEC-PROX

    /** Corta si la fecha de corte es la del batch o cae después y antes del siguiente día de proceso. */
    boolean cortaEnVentana(Contrato c) {
        LocalDate nuevo = proyecta(c.diaCorte());
        LocalDate efectiva = (c.diaCorteAnterior() > 0)
                ? min(proyecta(c.diaCorteAnterior()), nuevo)                // cambio de día de corte
                : nuevo;
        return efectiva.equals(fechaBatch)
                || (efectiva.isAfter(fechaBatch) && efectiva.isBefore(fechaSiguiente));
    }

    /** 50003160-ARMA-FEC-CORTE: día 30 significa último día del mes; si ya pasó, mes siguiente. */
    LocalDate proyecta(int diaCorte) {
        YearMonth mes = YearMonth.from(fechaBatch);
        int dia = (diaCorte == 30) ? mes.lengthOfMonth() : diaCorte;
        if (dia < fechaBatch.getDayOfMonth()) {
            mes = mes.plusMonths(1);
        }
        return mes.atDay(Math.min(dia, mes.lengthOfMonth()));   // ver riesgo: el COBOL no acota
    }
}

@Component
@RequiredArgsConstructor
public class CargaOtrasInversionesProcessor
        implements ItemProcessor<ContratoACargar, Historico> {

    private final SaldosOtrasInversionesClient s050;   // S050/L301 vía 20000050-SALDOS
    private final HistoricoRepository b06;
    private final MeterRegistry metricas;

    @Override
    public Historico process(ContratoACargar c) {
        Historico h = b06.findByContratoForUpdate(c.numContrato())       // 50003200: LOCK B06
                .orElseThrow(() -> new ProcesoAbortadoException("FALTA HISTORICO " + c.numContrato()));
        if (h.idS050() != 0) {
            return null;                                                 // ya cargado en corrida previa
        }
        if (c.saldos()) {
            var r = s050.saldos(c.pim(), c.numContrato());
            h.marcaCargadoS050();                                        // B06-ID-S050 = 1
            boolean alguno = false;
            if (c.instrumento().usaInversionIntegral(c.persona()) && r.fondos().ok()) {
                h.setSaldoFondos(r.fondos().importe());                  // BigDecimal, escala del PIC destino
                alguno = true;
            }
            // ... banca de inversión, plazo y mesa de dinero con el mismo patrón
            r.fondos().errorFatal().ifPresent(e -> { throw new S050NoDisponibleException(e); });
            metricas.counter(alguno ? "p199.si.s050" : "p199.no.s050").increment();
        }
        return h;
    }

Riesgos de migración
Bandera de carga sin dato

B06-ID-S050 se enciende en 50004050 (línea 1282) y en 50005200 (líneas 1467-1469) aunque S050 no devuelva saldos ni contadores. P130 deja de consultar S050 para esos contratos.

Fechas de corte inexistentes

50003160-ARMA-FEC-CORTE (líneas 1208-1221) pasa al mes siguiente sin ajustar el día: con día de corte 29 y batch a fin de enero de un año no bisiesto genera 29 de febrero. El año bisiesto se calcula solo con divisibilidad entre 4 (líneas 884-888).

Tope de 30 instrumentos

La tabla WKS-TB-B05INSTRUMEN tiene 30 posiciones (línea 164); el instrumento 31 aborta el paso (líneas 706-711).

Modo de pruebas en el mismo ejecutable

Los TASKVALUE 01 a 03 apagan las librerías S050 (líneas 209-214); la única protección en producción es la validación de las líneas 637-643.

Espera e intervención manual

Con S050 no disponible el programa espera 60 segundos por intento y al quinto se detiene esperando al operador (líneas 1444-1461); el fin forzado también se pide por consola con ACCEPT (líneas 1054-1069).
