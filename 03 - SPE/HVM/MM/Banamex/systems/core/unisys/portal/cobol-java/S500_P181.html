S500 · P181 a Java 21 Spring Boot

P181 abre S500BD01CAPTACION en consulta, obtiene la fecha de proceso del control B02 o del TASKVALUE de la tarea y carga a memoria hasta 30 combinaciones de producto, instrumento y moneda de B05, preparando una tabla de totales para cheques (producto 1) y otra para Cuenta Maestra (producto 66, o 500 con instrumento 1). Recorre B03 y, para cada contrato que no sea de ahorro, no esté migrado a S111 y ya esté abierto a la fecha, completa datos de B06, calcula días y saldo promedio anual cuando el corte cae en la fecha de proceso, y escribe un registro de 360 posiciones en INVEPTE. Ordena INVEPTE por nodo, sucursal, producto, instrumento, moneda y contrato y lo relee: el producto 1 va a CHESDOS y su adicional para Teradata, y todo lo demás a CMASDOS y su adicional. Cierra cada archivo con renglones de totales por producto y un trailer con el número de registros.

@Component
@RequiredArgsConstructor
public class FechaProcesoTasklet implements Tasklet {

    private final ControlCaptacionRepository b02;   // S500B02CONTROL
    private final CalendarioBancario calendario;     // LOCSUP

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        var params = ctx.getStepContext().getJobParameters();
        var control = b02.find().orElseThrow(() ->
                new ProcesoAbortadoException("ERROR REG. DE CONTROL VACIO"));   // DMTERMINATE

        LocalDate fechaPro, fechaProx;
        if (params.get("fecha") instanceof String f && !f.isBlank()) {          // WS-VAL-FEC > 0
            fechaPro  = LocalDate.parse(f, DateTimeFormatter.BASIC_ISO_DATE);
            fechaProx = calendario.siguienteHabil(fechaPro);                 // función 13, formato 13
        } else {
            fechaPro  = control.fechaLote();
            fechaProx = control.fechaLinea();
        }
        boolean esMensual = fechaPro.getMonth() != fechaProx.getMonth();
        boolean esSemanal = calendario.diaSemana(fechaPro) > calendario.diaSemana(fechaProx);

        var ec = ctx.getStepContext().getStepExecution().getJobExecution().getExecutionContext();
        ec.put("fechaProceso", fechaPro);
        ec.put("csi", control.numCsi());
        ec.put("esMensual", esMensual);    // calculado y publicado, aunque P181 no lo consulta (ver riesgo)
        ec.put("esSemanal", esSemanal);
        return RepeatStatus.FINISHED;
    }
}

@Component
@StepScope
@RequiredArgsConstructor
public class SeleccionContratoProcessor implements ItemProcessor<ContratoB03, InventarioRecord> {

    @Value("#{jobExecutionContext['fechaProceso']}") private LocalDate fechaProceso;
    private final HistoricoB06Repository b06;
    private final InventarioMapper mapper;            // 52230000-LLENO-SORT

    @Override
    public InventarioRecord process(ContratoB03 c) {
        var c2 = c.pimProductoASor() ? c.sinFechaCambioProducto() : c;   // WS03-88-PIM-PROD-A-SOR
        if (c2.pimAhorros()                                   // WS03-88-PIM-AHORROS
                || c2.fechaApertura().isAfter(fechaProceso)   // aún no abierto a la fecha
                || c2.migradoS111()) {                        // B03-MIGR-S111 = 1
            return null;                                      // NEXT SENTENCE: no entra al inventario
        }
        var hist = b06.findByContrato(c2.numContrato());      // 52240000-VE-B06
        var calculos = hist.fechaRealCorte().equals(fechaProceso)
                ? CalculosCiclo.de(c2, hist, fechaProceso)    // 52220050-CAL-CAMPOS
                : CalculosCiclo.ninguno();
        return mapper.toInventario(c2, hist, calculos);
    }
}
public record CalculosCiclo(int diasAno, BigDecimal saldoPromedioAnual, boolean hoyEsCorte) {

    /** 52220060 + 52220070. WS-SDO-PROMANUAL es S9(14)V99: COMPUTE sin ROUNDED trunca. */
    static CalculosCiclo de(ContratoB03 c, HistoricoB06 h, LocalDate fechaProceso, CalendarioBancario cal) {
        LocalDate inicioAno = fechaProceso.withDayOfYear(1);
        LocalDate desde = c.fechaApertura().isBefore(inicioAno) ? inicioAno : c.fechaApertura();
        int dias = cal.diasEntre(h.fechaCicloAnterior(), desde);     // LOCSUP función 02, formato 12
        if (!Objects.equals(c.fechaCambioProducto(), h.fechaRealCorte())) {
            dias += 1;
        }
        BigDecimal promedio = dias > 0
                ? h.acumuladoPromedioAnual().divide(BigDecimal.valueOf(dias), 2, RoundingMode.DOWN)
                : BigDecimal.ZERO.setScale(2);
        return new CalculosCiclo(dias, promedio, true);
    }

    static CalculosCiclo ninguno() { return new CalculosCiclo(0, BigDecimal.ZERO.setScale(2), false); }
}

static String bicaDetalle(InventarioRecord r) {
    return Fixed.bica02(r.producto(), r.instrumento(), r.moneda(), r.contrato(), r.numCliente(),
                        r.status(), r.saldoActual(), r.saldoAnterior() /* ... */);
}
/** Totales por PIM (WS-TBTOT-* para cheques, WS-NUMCMA-* para Cuenta Maestra). */
public final class TotalProducto {
    long mayores, iguales, menores;                       // MACE, IGCE, MECE
    BigDecimal saldoActual = BigDecimal.ZERO.setScale(2); // SDOA, S9(14)V99 con signo

    void suma(BigDecimal sdoAct) {
        saldoActual = saldoActual.add(sdoAct);
        switch (sdoAct.signum()) {
            case 1  -> mayores++;
            case 0  -> iguales++;
            default -> menores++;
        }
    }
}

@Component
@StepScope
@RequiredArgsConstructor
public class TotalesProductoListener implements ItemWriteListener<InventarioRecord> {
    private final CatalogoProductos catalogo;
    private final Map<ProductoB05Key, TotalProducto> totales = new LinkedHashMap<>();

    @Override
    public void afterWrite(Chunk<? extends InventarioRecord> items) {
        for (var r : items) {
            var k = ProductoB05Key.of(r);
            if (!catalogo.tieneTotales(k)) {
                log.warn("PIM {} sin renglón de totales en su familia", k);   // ver riesgo
            }
            totales.computeIfAbsent(k, x -> new TotalProducto()).suma(r.saldoActual());
        }
    }
}
// Al cerrar, un renglón tipo 03 por PIM con totales, como 53443100-LEO-CHE.
@Component
@StepScope
public class SaldosFooterCallback implements FlatFileFooterCallback {

    private final Familia familia;
    private final TotalesProductoListener totales;
    private final AtomicLong registros;   // WS-NUMREG-CHE / -CMA: el writer de detalle lo incrementa por registro

    public SaldosFooterCallback(Familia f, TotalesProductoListener t, AtomicLong contador) {
        familia = f; totales = t; registros = contador;
    }

    @Override
    public void writeFooter(Writer w) throws IOException {
        for (var linea : totales.renglones(familia)) {      // tipo 03 por PIM
            w.write(linea);
            w.write(System.lineSeparator());
            registros.incrementAndGet();
        }
        // Tipo 99: cuenta detalles y renglones de totales (no el header), como en el original.
        w.write(Fixed.bicaTrailer(registros.get()));
    }
}

Riesgos de migración
Indicadores de fin de mes y de semana sin uso

WS-ES-MENSUAL (línea 844) y WS-ES-SEMANAL (línea 873) se calculan pero no se consultan en este fuente. RN-S500-389 afirma que gobiernan totales y envíos; en P181 no es así.

Productos sin renglón de totales

Todo producto distinto de 1 va a CMASDOS, pero solo 66 y 500/1 tienen renglón de totales de Cuenta Maestra (líneas 1055-1063). Los demás productos aparecen en el detalle y no en los totales.

Pérdida de signo y dígitos en saldo promedio anual

WS-SDO-PROMANUAL S9(14)V99 se mueve a INV-SDOPROMANO-S 9(12)V99 sin signo.

Límite fijo de 30 productos

50001600-CARGO aborta con DMTERMINATE si B05 tiene más de 30 combinaciones de producto, instrumento y moneda. Un producto nuevo en el catálogo detiene el inventario nocturno.

Hardcodes

Productos 1, 66 y 500/1, el límite de 30, las rutas INTS/50003S02, S691 e INVEPTE y las constantes de dimensionamiento del SORT están compiladas. Se trasladan a application.yml.
