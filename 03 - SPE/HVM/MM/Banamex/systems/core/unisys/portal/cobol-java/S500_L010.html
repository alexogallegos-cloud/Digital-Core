/** Estado de copias: MIXES y CONTRATOS. Índice 0..copiasP010, igual que el arreglo ALGOL. */
@Component
public class EstadoCopias {
    final long[] mezcla;              // MIXES[0:DF_NUMCOPIAS_P010]
    final String[][] contratos;       // DOUBLE ARRAY CONTRATOS: se usa String para no perder ceros
    final Condition[] liberacion;     // EVENT ARRAY COPIAEVENT
    final Condition[] solicitando;    // EVENT ARRAY E_SOLCTO
    final ReentrantLock usoContrato = new ReentrantLock(true);   // INTERLOCK LK_USO_CTO

    EstadoCopias(ControlProperties p) {
        int n = p.copiasP010() + 1;
        mezcla = new long[n];
        contratos = new String[n][p.contratosPorCopia()];
        liberacion = new Condition[n];
        solicitando = new Condition[n];
        for (int i = 0; i < n; i++) {
            liberacion[i] = usoContrato.newCondition();
            solicitando[i] = usoContrato.newCondition();
        }
    }
}
/** CIERRA_CANDADO / ABRE_CANDADO / ABRE_CANDADO_POR_EXCEPCION en un solo patrón. */
public final class Candados {

    private Candados() {}

    /**
     * LOCKSTATUS(C).[47:24] = PROCESSID evita volver a tomar un candado propio;
     * ReentrantLock ya es reentrante. La EXCEPTION PROCEDURE que libera al abortar
     * la tarea se traduce en try/finally.
     */
    public static <T> T conCandado(ReentrantLock candado, Supplier<T> cuerpo) {
        candado.lock();                                   // CIERRA_CANDADO
        try {
            return cuerpo.get();
        } finally {
            if (candado.isHeldByCurrentThread()) {        // ABRE_CANDADO
                candado.unlock();
            }
        }
    }
}

/** VALIDA_MIX: devuelve la ranura libre de la copia o el RESULT de error. */
private Validacion validaCopia(SolicitudContrato in) {
    if (in.contrato().chars().allMatch(c -> c == '0')) {
        return Validacion.error(ResultadoControl.CONTRATO_EN_CEROS);           // 4
    }
    int copia = in.copia();
    if (copia < 1 || copia > props.copiasP010() + 1) {
        return Validacion.error(ResultadoControl.COPIA_INVALIDA);              // 9
    }
    if (estado.mezcla[copia - 1] != in.mezcla()) {
        return Validacion.error(ResultadoControl.COPIA_NO_DADA_DE_ALTA);       // 5
    }
    String[] ranuras = estado.contratos[copia - 1];
    for (int r = 0; r < ranuras.length; r++) {
        if (ranuras[r] == null) {
            return Validacion.ok(r);                                           // INDLIBRE
        }
    }
    return Validacion.error(ResultadoControl.COPIA_CON_MAS_DE_4_CONTRATOS);    // 6
}
/** VALIDA_USOCTO. Se ejecuta con estado.usoContrato tomado (LOCK(LK_USO_CTO) en el llamador). */
private ResultadoControl validaUsoContrato(SolicitudContrato in, int ranuraLibre)
        throws InterruptedException {

    barrerasDeFase.esperaSiCambioAPrelinea(in.copia());   // WAIT(E_ESP_SOL_CTO_PRELIN), copias 3 a 14
    barrerasDeFase.esperaFinDeInicio();                   // WAIT(E_ESP_SOL_CTO_INICIO)
    barrerasDeFase.esperaFinDeRecuperacion();             // WAIT(E_ESP_SOL_CTO_RECUP)

    ResultadoControl result = ResultadoControl.OK;
    buscar:
    for (int c = 0; c <= props.copiasP010(); c++) {
        for (int r = 0; r < props.contratosPorCopia(); r++) {
            if (!in.contrato().equals(estado.contratos[c][r])) continue;
            if (c == in.copia() - 1) {
                result = ResultadoControl.OCUPADO_POR_LA_MISMA_COPIA;           // 7, el ciclo sigue
                continue;
            }
            // WAIT((I_TIMEOUT_SOLCTO), COPIAEVENT[c], E_SOLCTO[c])
            Evento ev = esperaPrimero(props.timeoutSolicitaContrato(),
                                      estado.liberacion[c], estado.solicitando[c]);
            result = switch (ev) {
                case TIMEOUT   -> ResultadoControl.OCUPADO_REINTENTAR;             // 8
                case LIBERADO  -> ResultadoControl.OK;                             // 2: se concede sin re-verificar
                case SOLICITA  -> ResultadoControl.OCUPADO_POR_OTRA_SOLICITUD;     // 37
            };
            break buscar;
        }
    }
    if (result == ResultadoControl.OK) {
        estado.contratos[in.copia() - 1][ranuraLibre] = in.contrato();   // el bloqueo efectivo
    }
    return result;
}
@Override
public RespuestaContrato solicitaContrato(SolicitudContrato in) {
    Validacion v = validaCopia(in);                                   // VALIDA_MIX
    ResultadoControl result = v.resultado();
    if (result == ResultadoControl.OK) {
        estado.usoContrato.lock();                                    // LOCK(LK_USO_CTO)
        try {
            estado.solicitando[in.copia() - 1].signalAll();           // CAUSE(E_SOLCTO[copia-1])
            result = validaUsoContrato(in, v.ranura());
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            result = ResultadoControl.OCUPADO_REINTENTAR;
        } finally {
            estado.usoContrato.unlock();                              // UNLOCK + RESET(E_SOLCTO)
        }
    }
    // La salida se arma siempre, aun con error.
    var fase = estado.fase();
    return new RespuestaContrato(
            result,
            fase.indicadorSaldoPendiente(),                           // I_B02_IND_SDOPEN
            (in.copia() <= 2) ? fase.proceso() : fase.procesoTemporal(),
            fase.tipoCambioLinea().setScale(4, RoundingMode.UNNECESSARY), // 9(4)V9(4)
            fase.indicadorDepositosRetiros(),                         // I_B02_IND_FUNC_DYR
            fase.indicadorGbnp());                                    // I_B02_IND_GBNP
}
/** S500L010LIBERACTO: libera las cuatro ranuras de la copia y despierta a quien espere. */
@Override
public ResultadoControl liberaContrato(LiberaContrato in) {
    int copia = in.copia();
    if (copia < 1 || copia > props.copiasP010() + 1) {
        return ResultadoControl.COPIA_INVALIDA;                  // 9
    }
    if (estado.mezcla[copia - 1] != in.mezcla()) {
        return ResultadoControl.COPIA_NO_DADA_DE_ALTA;           // 5
    }
    estado.usoContrato.lock();      // ALGOL no toma candado aquí; en Java es obligatorio para signalAll
    try {
        Arrays.fill(estado.contratos[copia - 1], null);
        estado.liberacion[copia - 1].signalAll();                // CAUSE(COPIAEVENT[copia-1])
    } finally {
        estado.usoContrato.unlock();
    }
    return ResultadoControl.OK;
}

private static final long TOPE_SEQ_DIALOGO = 99_999_900L;   // margen de 100 antes de 9(8)

/** ASIGNAAUTDIA, ejecutado bajo LK_AUTDIA. */
private RespuestaAutorizacion asignaAutorizacion(SolicitudAutorizacion in) {
    if (!fase.dialogoForaneoActivo()                                   // B_DIALOGO_FOR
            && (in.pideSecuencialForaneo() || in.pideSecuencialInterbancario())) {
        return RespuestaAutorizacion.error(ResultadoControl.SIN_DIALOGO_FORANEO);   // 16
    }
    long autoriz = in.pideAutorizacion() ? contadores.ultAutoriz.incrementAndGet() : 0;
    long seqDial = 0, seqObco = 0;
    if (in.pideSecuencialForaneo()) {
        seqDial = contadores.seqB46Dialog.incrementAndGet();
        if (seqDial > TOPE_SEQ_DIALOGO) {
            // Se rechaza antes de que el campo de 8 dígitos dé la vuelta y el P010 falle por DUPLICATES.
            return RespuestaAutorizacion.error(ResultadoControl.SIN_DIALOGO_FORANEO);   // 16
        }
    }
    if (in.pideSecuencialInterbancario()) {                            // solo si el área > 6 bytes
        seqObco = contadores.seqB21Dialog.incrementAndGet();          // sin tope equivalente
    }
    return RespuestaAutorizacion.ok(autoriz, seqDial, seqObco);       // cada uno FOR 8 DIGITS
}
/** ASIGNASECDIA, ejecutado bajo LK_SECDIA. */
private RespuestaSecuencial asignaSecuencial() {
    if (contadores.ultSecuencial.get() == 99_999_999L) {
        return RespuestaSecuencial.error(ResultadoControl.MAXIMO_SECUENCIALES);   // 50
    }
    long secuencial = contadores.ultSecuencial.incrementAndGet();
    if (++pendientesDePersistir > 99 || secuencial == 99_999_999L) {
        try {
            persistencia.guardaSecuencial(props.copiaSecuencial(), secuencial);
        } catch (DataAccessException e) {
            // El original pone RESULT 12 y luego lo pisa con RESULT := 0 (línea 7614).
            log.error("No se pudo persistir el secuencial {} en B01AXCOPIA", secuencial, e);
            if (props.propagarErrorPersistencia()) {
                return RespuestaSecuencial.error(ResultadoControl.ERROR_BASE_DE_DATOS);  // 12
            }
        }
        pendientesDePersistir = 0;
    }
    return RespuestaSecuencial.ok(secuencial);                       // FOR 8 DIGITS
}

    }

Riesgos de migración
Credenciales en claro dentro de la librería

Las cadenas que arman los WFL lanzados por HI (líneas 2051, 3190, 3320 y 3508) contienen ACCESSCODE = ****/**** con usuario y contraseña escritos en el fuente. No se copian en esta demostración. En el destino deben venir de un gestor de secretos y la credencial actual debe rotarse.

Error de base enmascarado en el secuencial diario

En ASIGNASECDIA la línea 7614 fija RESULT := 0 después de intentar persistir en S500B01CONTROL, de modo que un error DMSII nunca llega al llamador. Si la librería cae después de un fallo de persistencia silencioso, el margen de 99 del arranque podría no cubrir los secuenciales ya entregados.

Contención global durante la espera de contrato

VALIDA_USOCTO espera hasta I_TIMEOUT_SOLCTO segundos con LK_USO_CTO tomado. Mientras una copia espera un contrato ocupado, todas las demás solicitudes de contrato del sistema quedan detenidas en LOCK(LK_USO_CTO).

Concesión sin re-verificación

Si la espera termina por COPIAEVENT, SOLCTO concede el contrato sin volver a buscarlo en la tabla. Ese evento también se dispara en otros caminos (limpieza de copias), por lo que podría haber dos copias con el mismo contrato.

Estado de negocio solo en memoria

La tabla de contratos, las mezclas por copia y los contadores de autorización y folio de Fraud Manager viven en el espacio de la librería residente. Una migración a varias instancias exige un almacén compartido; un port uno a uno a un singleton solo es válido con una única instancia activa.

Códigos de RESULT sin catálogo

El código 6 que devuelve VALIDA_MIX no está en la cabecera de resultados y el 38 aparece dos veces con significados distintos (líneas 51 y 52). El comentario del 8 menciona dos segundos fijos, cuando el timeout es configurable.
