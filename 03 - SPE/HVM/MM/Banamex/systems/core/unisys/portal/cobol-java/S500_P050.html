class VentanaEnvio {
    private final P050Properties props;
    private final CalendarioBancario calendario;      // LOCSUP funciones 18 y 06

    /** Límites exclusivos, como en el fuente (> 70000 y < 200000). */
    boolean permite(LocalDateTime t) {
        LocalDate d = t.toLocalDate();
        int hms = t.getHour() * 10000 + t.getMinute() * 100 + t.getSecond();
        if (calendario.esHabil(d) && entre(hms, props.ventana().habil()))     return true;
        if (calendario.esSabado(d) && entre(hms, props.ventana().sabado()))   return true;
        return calendario.esDomingo(d) && entre(hms, props.ventana().domingo());
    }

    private static boolean entre(int hms, Rango r) { return hms > r.desdeHms() && hms < r.hastaHms(); }
}
@Service

class CalendarioBancario {
    private final DiaInhabilRepository inhabiles;   // catálogo que hoy consulta LOCSUP

    /** LOCSUP función 18, formato 13: FECHA1 = 1 si el día es hábil. */
    boolean esHabil(LocalDate d) {
        DayOfWeek dow = d.getDayOfWeek();
        return dow != DayOfWeek.SATURDAY && dow != DayOfWeek.SUNDAY && !inhabiles.existsByFecha(d);
    }

    /** LOCSUP función 06: el fuente obtiene el día de la semana para las ventanas de sábado y domingo. */
    boolean esSabado(LocalDate d)  { return d.getDayOfWeek() == DayOfWeek.SATURDAY; }
    boolean esDomingo(LocalDate d) { return d.getDayOfWeek() == DayOfWeek.SUNDAY; }

class ProcesadorAvisos {
    private final ColaAvisosClient l010;        // S500L010USAGENMDA función 2
    private final AvisoS016Service s016;
    private final BitacoraProcesados bitacora;  // I04-PROCMED
    private final VentanaEnvio ventana;
    private final ReporteActivamed reporte;
    private boolean s016EnContingencia;

    void procesaPendientes(LocalDateTime inicio) {
        while (ventana.permite(LocalDateTime.now())) {        // se reevalúa por registro
            LecturaAviso l = l010.siguienteAviso();
            switch (l.resultado()) {
                case 0  -> {
                    s016EnContingencia = false;
                    ResultadoS016 r = s016.envia(l.aviso());
                    bitacora.graba(l.aviso(), r);
                }
                case 41 -> { log.warn("LA LIBRERIA S016L428 SE ENCUENTRA EN CONTINGENCIA");
                             s016EnContingencia = true; return; }
                case 42 -> { log.info("SE CONCLUYO LECTURA DE UN ARCHIVO : ACTIVAMED/{}", l.archivo());
                             reporte.imprimeYRota(l.archivo()); return; }
                case 43 -> { return; }                          // no hay más avisos por ahora
                default -> throw new ProcesoAbortadoException("L010 RESULT " + l.resultado());
            }
        }
    }
