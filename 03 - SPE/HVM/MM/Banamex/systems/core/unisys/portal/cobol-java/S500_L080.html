S500 · L080 a Java 21 Spring Boot

Al cargarse, L080 actualiza el control de BD07, abre la bitácora, lanza la tarea que atiende los HI del operador y fija en dos el número de copias del P080. Cada copia se registra al arrancar y recibe un número de copia y la fecha de línea. En el corte, S500L080INIBATCHP080 copia a BD07 la fecha de línea y de lote de BD01, envía el HI 4030 a todas las copias y espera hasta dos minutos a que cada una se detenga en S500L080ESBATCHP080; si todas confirman, cierra la bitácora, lanza los WFL de envío y libera a las copias con la nueva fecha de línea. La baja ordenada sigue el mismo esquema con los HI 4038 y 4040.

@Service
@RequiredArgsConstructor
public class CorteLineaBatchService {

    private final ValidadorMixes mixes;            // VALIDA_IN_MIX
    private final ControlBd07Service control;      // ACTUALIZA_BD07
    private final BarreraCopias barrera;           // VALIDA_INIBATCH
    private final EnvioArchivos envio;             // ENVIO_ARCHIVOS
    private final EstadoCorteRepository estado;

    /** S500L080INIBATCHP080, bloque externo y TERMINAINIBATCHP080. */
    public ResultadoL080 iniciaBatch() {
        ResultadoL080 r = mixes.valida();
        if (r.ok()) r = control.copiaFechasABd07();
        if (r.ok()) r = barrera.esperaConfirmaciones(Duration.ofMinutes(2));

        var e = estado.actual();
        if (r.ok()) {
            envio.cierraBitacoraYEnvia();          // trailer, WFL 1 y 2, nueva bitácora
            e.publicaFechaLinea(e.fechaLineaB02()); // I_B07_FECHA_LINEA := I_B02_FECHA_LINEA
            e.fase(FaseCorte.CORTE_OK);
        } else {
            e.fase(FaseCorte.CORTE_FALLIDO);
        }
        barrera.liberaCopias(e);                   // CAUSE(E_ESPERA_INIBATCHP080), en cualquier caso
        return r;
    }
}

@Component
@RequiredArgsConstructor
class BarreraCopias {

    private final ComandosP080 comandos;          // ENVIO_HI
    private final EstadoCorteRepository estado;

    /** VALIDA_INIBATCH: HI 4030 a todas las copias y espera a que todas confirmen. */
    ResultadoL080 esperaConfirmaciones(Duration limite) {
        var e = estado.actual();
        e.iniciaCorte();                           // B_INIBATCHP080 := TRUE, OK := FALSE, RESET evento
        estado.save(e);
        comandos.envia(Hi.CAMBIO_A_BATCH);         // HI 4030

        Instant fin = Instant.now().plus(limite);  // 12 x WAITANDRESET((10)) = 120 s
        while (Instant.now().isBefore(fin)) {
            if (estado.actual().todasConfirmaronCorte()) {
                estado.actual().limpiaConfirmacionesCorte();   // BA_COP_INIBATCH := FALSE
                return ResultadoL080.OK;
            }
            esperaEvento(Duration.ofSeconds(10));  // notificación de confirmación o timeout
        }
        estado.actual().limpiaConfirmacionesCorte();
        return ResultadoL080.COPIA_SIN_CONFIRMAR_CAMBIO;     // 7
    }
}
@Service
@RequiredArgsConstructor
class EsperaBatchService {

    private final EstadoCorteRepository estado;
    private final ValidadorCopias validador;           // VALIDA_PMC
    private final L080Properties props;

    /** S500L080ESBATCHP080: la copia confirma y queda en espera hasta el fin del corte. */
    @Transactional
    public ResultadoL080 esperaBatch(CopiaRef copia, SalidaEspera salida) {
        var e = estado.findForUpdate();                // LOCK(LK_ESB_P080)
        if (e.fase() != FaseCorte.CORTE_EN_CURSO) {
            return ResultadoL080.AUN_NO_INICIA_BATCH;  // 8
        }
        ResultadoL080 r = validador.valida(copia.paso(), copia.mix(), copia.numero());
        if (!r.ok()) return r;
        e.confirmaCorte(copia.numero());               // BA_COP_INIBATCH[..] := TRUE y CAUSE(E_INIBATCHP080)
        return ResultadoL080.PENDIENTE;                // la respuesta final llega por mensaje (ver racional)
    }

    /** Al recibir la liberación: WAIT (E_ESPERA_INIBATCHP080) ya no bloquea un hilo. */
    @KafkaListener(topics = "p080.corte.liberacion")
    void alLiberar(LiberacionCorte msg, CopiaLocal copiaLocal) {
        if (msg.fase() == FaseCorte.CORTE_OK) copiaLocal.usaFechaLinea(msg.fechaLinea());
        else copiaLocal.marcaError(ResultadoL080.ERROR_EN_BATCH);   // 10
    }
}

@Component
@RequiredArgsConstructor
class ComandosP080 {

    private final KafkaTemplate<String, ComandoCopia> kafka;
    private final EstadoCorteRepository estado;

    /** ENVIO_HI: arma "mix mix ... HI nnnn" y lo teclea en la consola del MCP con DCKEYIN. */
    void envia(Hi hi) {
        var copias = estado.actual().copiasActivas();     // IA_MIXES_XPGM[..] > 0
        if (copias.isEmpty()) return;                    // OFFSET(P_EA_RSPO) = 0: no se envía nada
        copias.forEach(c -> kafka.send("p080.comandos", String.valueOf(c.mixId()),
                                        new ComandoCopia(c.mixId(), hi.codigo)));
    }
}
@Service
@RequiredArgsConstructor
class BajaOrdenadaService {

    private static final Duration ESPERA_POR_HI = Duration.ofMinutes(2);   // (I_WAIT_TIMES * 5) > 119
    private final ComandosP080 comandos;
    private final EstadoCorteRepository estado;

    /** AVISO_BAJAP080: solo escala a 4040 si con 4038 no confirmaron todas. */
    ResultadoL080 avisoBaja() {
        ResultadoL080 r = ResultadoL080.OK;
        for (Hi hi : List.of(Hi.AVISO_BAJA, Hi.BAJA_FORZADA)) {
            comandos.envia(hi);
            if (esperaConfirmacionesBaja(ESPERA_POR_HI)) { r = ResultadoL080.OK; break; }
            r = ResultadoL080.COPIA_SIN_CONFIRMAR_BAJA;      // 12
        }
        estado.actual().limpiaConfirmacionesBaja();          // BA_COP_AVISBAJA := FALSE, B_BAJA_P080 := FALSE
        return r;
    }

    private boolean esperaConfirmacionesBaja(Duration limite) {
        Instant fin = Instant.now().plus(limite);
        while (Instant.now().isBefore(fin)) {
            if (estado.actual().todasConfirmaronBaja()) return true;
            esperaEvento(Duration.ofSeconds(5));             // WAITANDRESET((5),E_AVISO_P080)
        }
        return false;
    }
}

Riesgos de migración
Credencial en el código fuente

CALL_SYSTEM_WFL (línea 320) incluye un ACCESSCODE con usuario y contraseña en claro (se muestra como ****/****) para los hosts ACYPBETA y CACYPBET. Debe rotarse y retirarse del fuente; no se copió en este documento.

Desalineación de fecha de línea si falla el corte

ACTUALIZA_BD07 escribe la nueva fecha en BD07 antes de esperar a las copias (líneas 1130-1131). Si el corte termina con resultado 7, BD07 tiene la fecha nueva y las copias conservan la anterior.

Espera sin límite de las copias

S500L080ESBATCHP080 bloquea cada copia en WAIT (E_ESPERA_INIBATCHP080) sin tiempo máximo (línea 1165). Una falla del coordinador deja detenido el servicio en línea del P080.

Transacciones sin verificación

En ACTUALIZA_BD07 el estatus de BEGINTRANSACTION y ENDTRANSACTION no se revisa (líneas 991, 1008, 1015 y 1029).

Número de mix truncado en el HI

ENVIO_HI formatea el mix con 4 dígitos (línea 510) aunque se recibe con 6; mixes mayores a 9999 recibirían un comando equivocado.

Valores fijos

Número de copias del P080 (2), máximo de copias (10), tamaño de bitácora (150), títulos y packs de BD01 y BD07, nombres de host de prueba y nombres de los WFL de envío están compilados. Se trasladan a application.yml.
