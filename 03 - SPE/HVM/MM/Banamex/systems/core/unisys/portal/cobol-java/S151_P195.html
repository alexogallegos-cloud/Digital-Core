@ConfigurationProperties("p195")
public record P195Properties(String usuario, String familia, String patron, List<String> datasets) {

    /** Un solo patrón sustituye los seis grupos 01 WKS-NOMARCH-B0x. */
    public String titulo(String dataset, FechaRotulado f) {
        return patron.replace("{usuario}", usuario)
                     .replace("{dataset}", dataset)
                     .replace("{fecha}", f.aammdd())
                     .replace("{familia}", familia);
    }
}
/** Fecha que rotula los seis archivos (AAMMDD). Ceros si B01SISDIA está vacío. */
public record FechaRotulado(LocalDate fecha) {

    private static final DateTimeFormatter AAMMDD = DateTimeFormatter.ofPattern("yyMMdd");

    public String aammdd() { return fecha == null ? "000000" : fecha.format(AAMMDD); }

    /**
     * 02-00100-GENERA-B01SISDIA, rama NOT OLDCODE: B01-SIS-FECPRO viene en CCAAMMDD
     * y A2K-CONV-CAMD-TO-AMD descarta el siglo. Se toma del PRIMER registro de B01,
     * sea cual sea su sistema.
     */
    public static FechaRotulado dePrimerRegistro(Optional<SistemaDiario> primero) {
        return new FechaRotulado(primero
                .map(b01 -> LocalDate.parse(String.format("%08d", b01.fecPro()),
                                            DateTimeFormatter.BASIC_ISO_DATE))
                .orElse(null));
    }
}

@Component
@RequiredArgsConstructor
public class ControlEjecucionDecider implements JobExecutionDecider {

    private static final int SISTEMA = 151, NUM_PROCESO = 195;
    private final ProcesosClient b05;            // "B05PROCESOS IN S151-L001": func 01, 19, 37

    @Override
    public FlowExecutionStatus decide(JobExecution je, StepExecution se) {
        if (!je.getJobParameters().getString("altaDisponibilidad", "N").equals("S")) {
            return FlowExecutionStatus.COMPLETED;              // W77-VALUE distinto de 1
        }
        LocalDate fecha = FechaProceso.de(je);                // CONSISDIA del sistema 151
        return b05.consulta(SISTEMA, fecha, NUM_PROCESO)       // función 01
                .map(r -> r.staPgm() == 0
                        ? FlowExecutionStatus.COMPLETED        // existe pendiente: regenera
                        : new FlowExecutionStatus("YA_TERMINADO"))
                .orElseGet(() -> {
                    b05.alta(SISTEMA, fecha, NUM_PROCESO);     // función 19; si falla, lanza excepción
                    return FlowExecutionStatus.COMPLETED;
                });
    }
}

@Component
@RequiredArgsConstructor
public class VolcadoDatasetStepFactory {

    private final JobRepository repo;
    private final PlatformTransactionManager tx;
    private final DataSource controlS151;          // réplica relacional de S151BD99CONTROL (solo lectura)
    private final P195Properties props;

    /** Un step por dataset: recorre todo y escribe el registro tal cual (MOVE dataset TO registro). */
    public <T> Step volcado(String dataset, RowMapper<T> mapper, LineAggregator<T> formato,
                            FechaRotulado fecha) {
        var lector = new JdbcCursorItemReaderBuilder<T>()
                .name(dataset).dataSource(controlS151)
                .sql("SELECT * FROM " + dataset + " ORDER BY ID")   // orden físico del dataset
                .rowMapper(mapper)
                .build();
        var escritor = new FlatFileItemWriterBuilder<T>()
                .name(dataset + "Writer")
                .resource(new FileSystemResource(props.titulo(dataset, fecha)))
                .lineAggregator(formato)                            // layout fijo, sin separadores
                .shouldDeleteIfEmpty(false)                         // el fuente deja el archivo vacío
                .build();
        return new StepBuilder("volcado" + dataset, repo)
                .<T, T>chunk(500, tx).reader(lector).writer(escritor)
                .build();
    }
}
