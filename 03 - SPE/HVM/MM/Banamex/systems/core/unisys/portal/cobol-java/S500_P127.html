Abre la base de captación en modo de actualización, valida el header del archivo E01-MOVS408 (origen S408, destino S500 y fecha igual a la fecha base) y el header del archivo de punteo que deja P120, y ordena el archivo del S408 por tipo de registro, contrato y clave. Por cada detalle localiza el contrato en B03; según la clave de sobregiro marca línea de crédito (88 y 89), genera un aviso de próxima cancelación con clave 503 (01) o, si todos los saldos del contrato están en cero, lo bloquea con tipo 15 y genera la clave 504 (02). Los movimientos se envían al S151 en grupos de hasta 30 claves, se agregan al punteo y el listado R04-CANCS408 cierra con totales por tipo de tratamiento.

/** Registro del S408. Tipo 00 = header, 99 = trailer, cualquier otro = detalle. */
public sealed interface MovimientoS408 permits MovimientoS408.Header, MovimientoS408.Detalle, MovimientoS408.Trailer {

    /** WKS-E01-R00: fecha AAMMDD de seis dígitos. */
    record Header(LocalDate fechaProceso, String sistemaOrigen, String sistemaDestino)
            implements MovimientoS408 {}

    record Detalle(int tipoRegistro,
                   ClaveSobregiro claveSobregiro,     // 9(02): 88, 89, 01, 02 u otra
                   int prefijoCuenta,
                   String contrato,                   // 9(12) COMP: String para conservar ceros
                   int sucursalPromotora,
                   BigDecimal importe)                // 9(10)V99 COMP -> escala 2
            implements MovimientoS408 {}

    record Trailer() implements MovimientoS408 {}

    enum ClaveSobregiro {
        LINEA_CREDITO(88), LINEA_CREDITO_LINEA(89), AVISO_PROXIMA(1), CANCELACION(2), OTRA(-1);
        final int codigo;
        ClaveSobregiro(int c) { codigo = c; }
        static ClaveSobregiro de(int c) {
            return Arrays.stream(values()).filter(v -> v.codigo == c).findFirst().orElse(OTRA);
        }
    }

    /** Pivote de siglo para AAMMDD: el fuente compara seis dígitos y no lo necesita; Java sí. */
    DateTimeFormatter AAMMDD = new DateTimeFormatterBuilder()
            .appendValueReduced(ChronoField.YEAR, 2, 2, 2000).appendPattern("MMdd").toFormatter();
}
@Component
@StepScope
@RequiredArgsConstructor
public class ValidacionS408Tasklet implements Tasklet {

    private final ListadoMensajes mensajes;             // R05-MENSAJES
    private final P127Properties props;
    @Value("#{jobExecutionContext['fechaBase']}") private LocalDate fechaBase;

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) throws IOException {
        boolean hayS408 = validar(Path.of(props.archivoS408()));
        ctx.getStepContext().getStepExecution().getJobExecution()
           .getExecutionContext().put("hayS408", hayS408);     // W77-NOHAY-S408 = 0
        return RepeatStatus.FINISHED;
    }

    private boolean validar(Path archivo) throws IOException {
        Optional<MovimientoS408.Header> h = S408Reader.primerRegistro(archivo);
        if (h.isEmpty()) {                                      // AT END en la primera lectura
            mensajes.escribe("ARCHIVO S408 VACIO ");
            return false;                                       // se tolera: no aborta
        }
        if (!h.get().fechaProceso().equals(fechaBase)) {
            mensajes.escribe("ERR FECHA E01-MOVS408 " + h.get().fechaProceso().format(MovimientoS408.AAMMDD));
            return false;
        }
        if (!"S408".equals(h.get().sistemaOrigen()) || !"S500".equals(h.get().sistemaDestino())) {
            mensajes.escribe("ERR E01-MOVS408 NUM-SISTEMAS  SISORIG =" + h.get().sistemaOrigen()
                    + " SISDEST =" + h.get().sistemaDestino());
            return false;
        }
        mensajes.escribe("HEADER ARCH  S408 MOVNOCONT OK ");
        mensajes.escribe("SE APLICO ARCH: " + archivo);
        return true;
    }
}

/** 20004720-VALIDA-PUNTEO: sin punteo válido no hay corrida (DMTERMINATE). */
@Component
@RequiredArgsConstructor
class ValidadorPunteo {

    private final P127Properties.Punteo cfg;
    private final Clock reloj;

    PunteoHeader validarYReetiquetar(Optional<PunteoHeader> leido, LocalDate fechaBase, int miCsi) {
        PunteoHeader h = leido.orElseThrow(() -> new ProcesoAbortadoException("ARCH PUNTEO VACIO "));
        boolean valido = h.tipoRegistro() == 0
                && h.sistemaOrigen() == 500 && h.sistemaDestino() == 500
                && h.fechaMovimiento().equals(fechaBase)
                && h.csiOrigen() == miCsi
                && cfg.pasoAnterior().equals(h.paso())
                && h.copia() == cfg.copia();
        if (!valido) throw new ProcesoAbortadoException("ERR HEADER PUNTEO " + h);
        return h.conMarcaDeMaquina(LocalDateTime.now(reloj))
                .conPaso(cfg.pasoPropio(), cfg.copia());
    }
}
/**
 * 20004800-CLASI-S408: SORT USING/GIVING sobre el mismo archivo, por tipo de registro,
 * contrato y clave. El cálculo de MEMORY SIZE y DISK SIZE es propio del MCP y no se migra.
 */
@Bean
Step ordenaS408(JobRepository repo, PlatformTransactionManager tx, P127Properties props) {
    return new StepBuilder("ordenaS408", repo)
            .tasklet((contribution, ctx) -> {
                Path archivo = Path.of(props.archivoS408());
                List<String> lineas;
                try (var s = Files.lines(archivo, StandardCharsets.ISO_8859_1)) {
                    lineas = s.sorted(Comparator
                                    .comparing(S408Layout::tipoRegistro)   // 00 primero, 99 al final
                                    .thenComparing(S408Layout::contrato)
                                    .thenComparing(S408Layout::clave))
                              .toList();
                }
                Path tmp = Files.createTempFile(archivo.getParent(), "s408", ".sort");
                Files.write(tmp, lineas, StandardCharsets.ISO_8859_1);
                Files.move(tmp, archivo, StandardCopyOption.REPLACE_EXISTING, StandardCopyOption.ATOMIC_MOVE);
                return RepeatStatus.FINISHED;
            }, tx)
            .build();
}

@Component
@StepScope
@RequiredArgsConstructor
public class CancelacionProcessor implements ItemProcessor<MovimientoS408, ResultadoCancelacion> {

    private final ContratoCaptacionRepository contratos;   // S500B03CONTRATOS
    private final ReglasCancelacion reglas;
    private final Autorizaciones autorizaciones;           // W77-MY-AUTORIZ

    @Override
    public ResultadoCancelacion process(MovimientoS408 m) {
        if (!(m instanceof MovimientoS408.Detalle d)) return null;     // tipos 00 y 99 se saltan

        // LOCK B03SXCTO: lectura con bloqueo pesimista dentro de la transacción del chunk
        return contratos.findForUpdate(d.contrato())
                .map(cto -> {
                    autorizaciones.siguiente();                         // ADD 1 TO W77-MY-AUTORIZ
                    return reglas.aplica(d, cto);                       // 50003320 + STORE
                })
                .orElseGet(() -> ResultadoCancelacion.rechazo(d, TipoCancelacion.CONTRATO_INEXISTENTE)); // 5
    }
}

@Component
@RequiredArgsConstructor
class ReglasCancelacion {

    private final AvisoCancelacion aviso;            // 50003321 / 50003322
    private final MovimientosS151 s151;              // 50008190-ENVIA-MOVSS151
    private final Totales totales;

    ResultadoCancelacion aplica(MovimientoS408.Detalle d, ContratoCaptacion cto) {
        ResultadoCancelacion r = switch (d.claveSobregiro()) {
            case LINEA_CREDITO -> {                                     // 88
                cto.setIndLineaCredito(1);
                totales.sumaAperturaLc();
                yield ResultadoCancelacion.reporta(d, TipoCancelacion.LINEA_CREDITO);        // 2
            }
            case LINEA_CREDITO_LINEA -> {                               // 89
                cto.setIndLineaCredito(3);
                totales.sumaAperturaLcLinea();
                yield ResultadoCancelacion.reporta(d, TipoCancelacion.LINEA_CREDITO_LINEA);  // 8
            }
            case AVISO_PROXIMA -> aviso.proximaCancelacion(d, cto);                          // 3, clave 503
            case CANCELACION -> {
                if (cto.esSocioFuncionario())
                    yield ResultadoCancelacion.rechazo(d, TipoCancelacion.SOCIO_FUNCIONARIO);  // 4
                if (cto.bloqueadoCon(66) || cto.bloqueadoCon(14))
                    yield ResultadoCancelacion.omitido(d, TipoCancelacion.YA_BLOQUEADO);       // 9: sin renglón
                yield aviso.cancelacion(d, cto);                                             // 1 o 7
            }
            case OTRA -> ResultadoCancelacion.rechazo(d, TipoCancelacion.CLAVE_INVALIDA);    // 6
        };
        s151.enviarPendientes(cto);                  // IF WKS-IND-MOV > 0
        return r;
    }
}
@Component
@RequiredArgsConstructor
class AvisoCancelacion {

    private final HistoricoRepository historico;   // S500B06HISTORICO
    private final MovimientosS151 s151;
    private final Totales totales;
    private final P127Properties props;
    private final FechaProceso fecha;              // fecha base del job (WKS-FEC-BASE-08)

    /** 50003322-AVISO-CANCELACION. */
    ResultadoCancelacion cancelacion(MovimientoS408.Detalle d, ContratoCaptacion cto) {
        boolean sinSaldos = Stream.of(cto.getSdoActual(), cto.getSdoProtBloq(), cto.getSdoPendiente(),
                                      cto.getSdoSbc(), cto.getSbc1Dia())
                                  .allMatch(s -> s.signum() == 0);
        if (!sinSaldos) {
            return ResultadoCancelacion.rechazo(d, TipoCancelacion.CON_SALDO);        // 7
        }
        var hist = historico.findById(cto.getNumContrato()).orElseThrow();          // DAME-BDHISTORICO
        s151.agrega(cto, props.claveCancelacion(), d.importe());                     // 504
        cto.bloquear(props.tipoBloqueoCancelacion());                                // STATUS 1, TIPO 15
        cto.setIndLcCancel(1);
        hist.setFecBloqueo(fecha.base());                                            // B06-FEC-BLOQUEO
        totales.sumaCancelado();
        return ResultadoCancelacion.reporta(d, TipoCancelacion.CANCELADO);           // 1
    }
}

     * 50090000-STORE-BASE: BEGIN-TRANSACTION, STORE B03 y B06 solo si cambiaron, END-TRANSACTION.

        }
    }
