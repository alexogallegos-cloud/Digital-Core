/**
 * WKS-TASKVALUE-PBA PIC 9(02) REDEFINES: decena = función, unidad = WKS-SIN-LBS408.
 * En Java son dos parámetros de job con nombre propio.
 */
public record P115Parametros(Funcion funcion, boolean sinLibreriaS408) {

    public enum Funcion {
        LISTADO(0),
        GENERA_I10(1), GENERA_I10PS(4), GENERA_I10PS2(7),          // por plaza, actualizan B03
        CONSOLIDA_I10(2), CONSOLIDA_I10PS(5), CONSOLIDA_I10PS2(8),  // VDM + MTY
        SOLO_VDM_I10(3), SOLO_VDM_I10PS(6), SOLO_VDM_I10PS2(9);

        final int codigo;
        Funcion(int c) { codigo = c; }

        static Funcion de(int c) {
            return Arrays.stream(values()).filter(f -> f.codigo == c).findFirst()
                    .orElseThrow(() -> new JobParametersInvalidException("ERROR FUNCION INVALIDA" + c));
        }
        boolean actualizaBase() { return this == GENERA_I10 || this == GENERA_I10PS || this == GENERA_I10PS2; }
    }

    /** Compatibilidad con el WFL actual mientras conviven ambos: 11 = función 1 sin S408. */
    static P115Parametros desdeTaskValue(int taskValue) {
        return new P115Parametros(Funcion.de(taskValue / 10), taskValue % 10 > 0);
    }
}

    public String plaza(String corte, int nodo, LocalDate fecha) {

@Component
@RequiredArgsConstructor
class ArchivosPorFuncion {

    private final P115Properties props;

    record Plan(List<Path> entradas, Path salida) {}

    Plan plan(P115Parametros.Funcion f, int miCsi, LocalDate fecha) {
        String corte = switch (f) {
            case GENERA_I10, CONSOLIDA_I10, SOLO_VDM_I10 -> "I10";
            case GENERA_I10PS, CONSOLIDA_I10PS, SOLO_VDM_I10PS -> "I10PS";
            case GENERA_I10PS2, CONSOLIDA_I10PS2, SOLO_VDM_I10PS2 -> "I10PS2";
            case LISTADO -> throw new IllegalArgumentException("La función 0 no genera archivos");
        };
        Path vdm = Path.of(props.plaza(corte, 10, fecha));
        Path mty = Path.of(props.plaza(corte, 4, fecha));
        Path consolidado = Path.of(props.archivos().consolidado().get(corte).replace("{fecha}",
                fecha.format(DateTimeFormatter.BASIC_ISO_DATE)));

        return switch (f) {
            case GENERA_I10, GENERA_I10PS, GENERA_I10PS2 -> new Plan(List.of(), miCsi == 10 ? vdm : mty);
            case CONSOLIDA_I10, CONSOLIDA_I10PS, CONSOLIDA_I10PS2 -> new Plan(List.of(vdm, mty), consolidado);
            default -> new Plan(List.of(vdm), consolidado);       // 3, 6, 9: solo Valle de México
        };
    }
}
@Component

public class ContratoGceProcessor implements ItemProcessor<ContratoGce, SaldoI10> {

    private final ContratoCaptacionRepository contratos;   // S500B03CONTRATOS
    private final LineaCreditoClient s408;
    @Value("#{jobParameters['funcion']}") private String funcion;
    @Value("#{jobExecutionContext['sinLibreriaS408'] ?: false}") private boolean sinS408;

    /** 30000205-LEE-B48 (solo B48-STATUS = 1) + 30001070-BUSCA-B03. */
    @Override
    public SaldoI10 process(ContratoGce b48) {
        if (b48.status() != 1) return null;

        Optional<ContratoCaptacion> b03 = esI10PS2()
                ? contratos.findById(b48.numContrato())             // FIND: función 7 no actualiza
                : contratos.findForUpdate(b48.numContrato());       // LOCK: funciones 1 y 4
        if (b03.isEmpty()) return null;                             // W77-NO-B03 = 1
        var cto = b03.get();
        if (cto.getMarcaClon() == 2) return null;                   // contratos clon no se reportan

        BigDecimal lineaCredito = (cto.getIndLineaCredito() > 0 && !sinS408)
                ? s408.saldoLinea(cto)                              // 30001080-CONS-SDO-LC
                : BigDecimal.ZERO.setScale(2);

        return cto.getStatus() == 0 ? SaldoI10.de(cto, lineaCredito) : null;   // solo contratos activos
    }

    private boolean esI10PS2() { return "GENERA_I10PS2".equals(funcion); }
}
@Component
@RequiredArgsConstructor
public class LineaCreditoClient {

    private static final int NO_HABILITADA = 96;
    private final S408LinCredGateway lincred;      // 20000408-LINCRED, función 11, paso 10

    /** W77-SDO-LC = IMPORTE + IMPAUT + IMP-EXCEDIDO + SALDO - IMPDEVOL (escala 2). */
    @Retryable(retryFor = S408NoHabilitadaException.class,
               maxAttempts = 5, backoff = @Backoff(delay = 5000))   // WAIT(05), hasta 4 fallas
    public BigDecimal saldoLinea(ContratoCaptacion cto) {
        var r = lincred.consulta(new ConsultaS408(500, 11, 10, cto.getNumContrato(),
                cto.getSucPromotora(), cto.getNumProducto(), cto.getNumInstrum(), cto.getMoneda()));
        return switch (r.status()) {
            case 0 -> r.importe().add(r.impAut()).add(r.impExcedido()).add(r.saldo())
                         .subtract(r.impDevol());           // suma exacta: no hay redondeo
            case NO_HABILITADA -> throw new S408NoHabilitadaException(r.status());
            default -> throw new S408ErrorException(r.status());   // el fuente lo trata como línea 0
        };
    }

    @Recover
    BigDecimal agotado(S408NoHabilitadaException e, ContratoCaptacion cto) {
        // El fuente hace STOP y espera al operador indefinidamente; aquí falla el step.
        throw new ProcesoAbortadoException("ERR CALL(11) S408 RS=96: S408L011LCRD no disponible");
    }
}
/** Detalle I10 tipo "485". Los tres saldos (DISP, VD, LB) llevan el mismo valor. */
public record SaldoI10(String ctaCheques, int sucChequera, BigDecimal saldoDisponible,
                       String monedaSwift) {

    /** W77-TOT-SDO-DISP = B03-SDO-ACTUAL - B03-AUT-PENDIENTE - W77-SDO-LC. */
    static SaldoI10 de(ContratoCaptacion cto, BigDecimal lineaCredito) {
        BigDecimal disp = cto.getSdoActual()
                .subtract(cto.getAutPendiente())
                .subtract(lineaCredito)
                .setScale(3, RoundingMode.UNNECESSARY);     // S9(15)V9(03): solo se agrega un cero
        String moneda = (cto.getMoneda() == 5) ? "USD" : "MXN";   // cualquier otra moneda sale como MXN
        return new SaldoI10(cto.getCtaCheques(), cto.getSucChequera(), disp, moneda);
    }

    /** Formato del I10: signo aparte, entero 9(15), punto y decimales 9(03) en milésimas. */
    String importeI10() {
        BigDecimal abs = saldoDisponible.abs();
        String signo = saldoDisponible.signum() < 0 ? "-" : " ";
        String entero = "%015d".formatted(abs.toBigInteger());
        String milesimas = "%03d".formatted(abs.remainder(BigDecimal.ONE).movePointRight(3).intValueExact());
        return signo + entero + "." + milesimas;
    }
}

        BigDecimal mxn = BigDecimal.ZERO.setScale(3), usd = BigDecimal.ZERO.setScale(3);
        for (String r : (Iterable<String>) registros::iterator) {
            if (!I10Layout.tipo(r).equals("485")) continue;          // header "***" y trailer "---" fuera
            BigDecimal valor = new BigDecimal(I10Layout.sdoDisp(r))  // 9(15)
                    .add(new BigDecimal(I10Layout.decSdoDisp(r)).movePointLeft(3));   // milésimas / 1000
            if ("-".equals(I10Layout.signoSdoDisp(r))) valor = valor.negate();
            if ("MXN".equals(I10Layout.monedaSwift(r))) { nMxn++; mxn = mxn.add(valor); }
            else                                         { nUsd++; usd = usd.add(valor); }   // todo lo demás
        }
        return new TotalesI10(nMxn, mxn, nUsd, usd);
    }

Riesgos de migración
I10PS e I10PS2 escritos con el título de I10

En 30001010-ABRE-ARCH las funciones 4 y 7 asignan WKS-TIT-I10V y WKS-TIT-I10M a los archivos I10PS e I10PS2 de cada plaza (líneas 1773, 1777, 1783 y 1787), mientras que la función 0 y los FD usan WKS-TIT-I10PSV, WKS-TIT-I10PS2V y sus equivalentes de Monterrey. Si nada más los renombra, la función 4 sobrescribe el I10 del día y las funciones 5 y 8 consolidan archivos que P115 no generó.

Error del S408 tratado como línea en cero

30001080-CONS-SDO-LC (líneas 2033-2034) da por terminada la consulta con cualquier estatus distinto de 0 y 96, sin mensaje y con W77-SDO-LC = 0. El saldo disponible enviado queda sobrestimado para contratos con línea de crédito.

Pausa de operador en batch

Con estatus 96 repetido, el fuente detiene la tarea con STOP (línea 2028) hasta que un operador responda, y después reintenta sin límite. En desarrollo también hay un STOP interactivo si falla XATMI (línea 842).

Función inválida sin error de terminación

El IF de 00000000-CONTROL-PROGRAMA solo escribe ERROR FUNCION INVALIDA y continúa a STOP RUN (líneas 765-774); el WFL no detecta el problema.

Espacios dentro del título de I10 consolidado

WKS-TIT-I10 reserva 32 posiciones para GCE/MX1/485/BALANCES/, que mide 21 (líneas 429-430). El título resultante lleva 11 espacios antes de la fecha.

Hardcodes

Sucursal 485, interfaz 10, versión 01, país MX, sucursal MX1, referencia 0000606 y formato 0000153 del I7, número de registros 2 en el trailer del I7, CSI 10 como Valle de México, moneda 5 como USD y los diez títulos de archivo. Se trasladan a application.yml.
