INTEGER PROCEDURE GUARDAMSG(FOLIO,TIEMPO,MSG_SZ,EDO,MSG);               00006400

    BEGIN                                                               00006800          
    INTEGER J;                                                          00006900          
    REAL KEY;                                                           00007000          
    EPILOG PROCEDURE ADIOS;                                             00007100          
       IF CAND(J>0,LOCKSTATUS(LK_S500).[47:24]=PROCESSID) THEN          00007200          
           J:=UNLOCK(LK_S500);                                          00007300          
                                                                        00007400          
    KEY:=FOLIO;                                                         00007500          
    IF COR(COR(MSG_SZ > MAX_SZ,MSG_SZ > SIZE(MSG)),                     00007600          
           COR(TIEMPO < NSEG  ,FOLIO  <         1) ) THEN               00007700          
        BEGIN                                                           00007800          
        IF MSG_SZ > MAX_SZ    THEN GUARDAMSG:=3 ELSE                    00007900          
        IF MSG_SZ > SIZE(MSG) THEN GUARDAMSG:=4 ELSE                    00008000          
        IF TIEMPO < NSEG      THEN GUARDAMSG:=5 ELSE                    00008100          
                                   GUARDAMSG:=6;                        00008200          
        END                                                             00008300          
    ELSE                                                                00008400          
    IF J:=MASKSEARCH(KEY,MASK,MEM) GEQ 0 THEN           % DUPLICADO     00008500          
        GUARDAMSG:=2                                                    00008600          
    ELSE                                                                00008700          
    IF COR(LOCK(LK_S500,5)=2,J:=MASKSEARCH(CERO,MASK,MEM) < 0) THEN     00008800          
        GUARDAMSG:=1                                                    00008900          
    ELSE                                                                00009000          
        BEGIN                                           % HAY LUGAR     00009100          
        SZ      [J]:=MSG_SZ;                                            00009200          
        MEM     [J]:=FOLIO & EDO [47:1];                                00009300          
        TIME_OUT[J]:=TIEMPO;                                            00009400          
        VENCE   [J]:=TIME(1)/60;                                        00009500          
        IF EDO=0 THEN VENCE[J]:=*+TIEMPO;               % EL UNO VENCIO 00009600          
        REP CONTEXTO[J,0] BY MSG FOR MSG_SZ;                            00009700          
        P:=*+1;                                                         00009800          
        END;                                                            00009900          
    END  GUARDAMSG;                                                     00010000

PROCEDURE INFORMA(MSG);                                                 00014600          
EBCDIC ARRAY MSG[0];                                                    00014800          
    BEGIN                                                               00014900          
    INTEGER K, L;                                                       00015000          
    REAL AHORA;                                                         00015100          
    ARRAY A_MSG[0]=MSG;                                                 00015200          
    POINTER PT;                                                         00015300          
                                                                        00015400          
    IF SIZE(MSG)<2000 THEN RESIZE(MSG,2000,RETAIN);                     00015500          
    REP MSG[23] BY "01";                                                00015600          
    PT:=POINTER(A_MSG[7]);                                              00015700          
    REP PT:PT BY 48"1E276B","010077INFO",48"0D25",                      00015800          
        "INFORMACION DE LA LIBRERIA ASINCRONA",                         00015900          
        48"0D250D25","LUGARES OCUPADOS ",P FOR 4 NUMERIC,", LIBRES ",   00016000          
        (2048-P) FOR 4 NUMERIC,48"0D25",                                00016100          
        "EL ASINCRONO HA ENVIADO ",ENVIOS FOR 6 NUMERIC,                00016200          
        " MENSAJES EN ",VUELTAS FOR 6 NUMERIC," REVISIONES";            00016300          
                                                                        00016400          
    IF P>0 THEN                                                         00016500          
        BEGIN                                                           00016600          
        AHORA:=TIME(1)/60;                                              00016700          
        REP PT:PT BY 48"0D250D25","LUGAR     FOLIO ESTADO ",            00016800          
            "ENVIO_EN  A_LAS... TIME-OUT MSG_SZ",48"0D25";              00016900          
        FOR K:=2047 STEP -1 UNTIL 0 DO                                  00017000          
            BEGIN                                                       00017100          
            IF MEM[K] NEQ 0 THEN                                        00017200          
                BEGIN                                                   00017300          
                IF MEM[K].[47:1]>0 THEN                                 00017400          
                    REP DESC BY "VENCIO" ELSE REP DESC BY "      ";     00017500          
                REP PT:PT BY K FOR 5 NUMERIC,                           00017600          
                    MEM[K].[46:47] FOR 10 NUMERIC," ",DESC FOR 6,       00017700          
                    "    ",VENCE[K]-AHORA FOR 5 NUMERIC," ",            00017800          
                    (VENCE[K] DIV 3600) FOR 2 NUMERIC,":",              00017900          
                    ((VENCE[K] MOD 3600) DIV 60) FOR 2 DIGITS,":",      00018000          
                    (VENCE[K] MOD 60) FOR 2 DIGITS," ",                 00018100          
                    TIME_OUT[K] FOR 9 NUMERIC,SZ[K] FOR 7 NUMERIC,      00018200          
                    48"0D25";                                           00018300          
                    IF (L:=*+1)>MIN(15,P) THEN K:=-1;                   00018400          
                END;                                                    00018500          
            END;                                                        00018600          
        END;                                                            00018700          
    HDR[0]             :=1;                     % DESTCOUNT             00018800          
    HDR[11]            :=A_OUTPUT;              % AGENDA                00018900          
    HDR[1]             :=OFFSET(PT);            % TEXTLENGTH            00019000          
    HDR[2]             :=0;                     % SATUSVALUE            00019100          
    HDR[4]             :=T_DUMMY;               % DESTINATIONDESG       00019200          
    K:=SEND(COMSOUT,HDR[1],MSG);                                        00019300          
    END INFORMA;                                                        00019400

PROCEDURE ENVIA_CANCELACIONES(COMS_IN,COMS_OUT,IN_EV,TSK_EV);           00019600          
INPUTHEADER  COMS_IN;                                                   00019700          
OUTPUTHEADER COMS_OUT;                                                  00019800          
EVENT        IN_EV, TSK_EV;                                             00019900          
    BEGIN                                                               00020000          
    REAL DELTA, LAST_WAKE_UP;                                           00020100          
    BOOLEAN ENDING, CAMBIO_DIA;                                         00020200          
    EBCDIC ARRAY MSG[0:99];                                             00020300          
    DEFINE UN_DIA = 86400 #;                                            00020400          
                                                                        00020500          
PROCEDURE ENVIA;                                                        00020600          
IF P=0 THEN DELTA:=TIME_OUT_DEF ELSE                                    00020700          
    BEGIN                                                               00020800          
    INTEGER J, K;                                                       00020900          
    EPILOG PROCEDURE ADIOS;                                             00021000          
        IF  CAND(J>0,LOCKSTATUS(LK_MEM[J]).[47:24]=PROCESSID) THEN      00021100          
            K:=UNLOCK(LK_MEM[J]);                                       00021200          
                                                                        00021300          
    VUELTAS:=*+1;                                                       00021400          
    DELTA:=NSEG;                        % POR SI HAY EXCEPCION          00021500          
    MENOR:=1@6;                                                         00021600          
    CAMBIO_DIA:=(LIMITE:=TIME(1)/60) < LAST_WAKE_UP;                    00021700          
    LAST_WAKE_UP:=LIMITE;                                               00021800          
    FOR J:=N STEP -1 UNTIL 0 DO                                         00021900          
        IF CAND(MEM[J] NEQ 0,LOCK(LK_MEM[J],5)=1) THEN                  00022000          
            BEGIN                                                       00022100          
            IF MEM[J] NEQ 0 THEN                % REVISA "AFTER LOCKED" 00022200          
                BEGIN                                                   00022300          
                IF VENCE[J]>UN_DIA THEN VENCE[J]:=*-UN_DIA;             00022400          
                IF VENCE[J]<LIMITE THEN                                 00022500          
                    BEGIN                                               00022600          
                    MEM[J].[47:1]:=1;               % CANCELADO         00022700          
                    VENCE[J]:=LIMITE+TIME_OUT[J];                       00022800          
                    COMS_OUT.DESTCOUNT      :=1;                        00022900          
                    COMS_OUT.AGENDA         :=A_OUTPUT;                 00023000          
                    COMS_OUT.TEXTLENGTH     :=SZ[J];                    00023100          
                    COMS_OUT.STATUSVALUE    :=0;                        00023200          
                    COMS_OUT.DESTINATIONDESG:=T_DUMMY;                  00023300          
                    K:=SEND(COMS_OUT,COMS_OUT.TEXTLENGTH,CONTEXTO[J,*]);00023400          
                    ENVIOS:=*+1;                                        00023500          
                    END                                                 00023600          
                ELSE                                                    00023700          
                    IF VENCE[J]<MENOR THEN MENOR:=VENCE[J];             00023800          
                END;                                                    00023900          
            K:=UNLOCK(LK_MEM[J]);                                       00024000          
            END;                                                        00024100          
    IF MENOR=1@6 THEN MENOR:=LIMITE+TIME_OUT_DEF;                       00024200          
    DELTA:=MAX(NSEG,MENOR-TIME(1)/60);                                  00024300          
    END ENVIA;                                                          00024400          
                                                                        00024500          
    LAST_WAKE_UP:=TIME(1)/60;                                           00024600          
    WHILE NOT ENDING DO                                                 00024700          
        IF WAIT((DELTA),IN_EV,TSK_EV) = 1 THEN ENVIA                    00024800          
        ELSE ENDING:=RECEIVE(COMS_IN[DONTWAIT],MSG)=99;                 00024900          
    END ENVIA_CANCELACIONES;                                            00025000
