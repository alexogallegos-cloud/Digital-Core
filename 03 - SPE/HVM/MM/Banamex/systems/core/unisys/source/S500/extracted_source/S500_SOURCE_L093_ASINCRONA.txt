INTEGER PROCEDURE GUARDAMSG(FOLIO,MSG_SZ,EDO,MSG);                      00007400          
INTEGER FOLIO, MSG_SZ, EDO;                                             00007500          
EBCDIC ARRAY MSG[0];                                                    00007600          
    BEGIN                                                               00007700          
    INTEGER J;                                                          00007800          
    REAL KEY;                                                           00007900          
    EXCEPTION PROCEDURE ADIOS;                                          00008000          
       IF LOCKSTATUS(LK_S500).[47:24]=PROCESSID THEN J:=UNLOCK(LK_S500);00008100          
                                                                        00008200          
    KEY:=FOLIO;                                                         00008300          
    IF COR(COR(MSG_SZ > MAX_SZ,MSG_SZ > SIZE(MSG)),FOLIO < 1) THEN      00008400          
        BEGIN                                                           00008500          
        IF MSG_SZ > MAX_SZ    THEN GUARDAMSG:=3 ELSE                    00008600          
        IF MSG_SZ > SIZE(MSG) THEN GUARDAMSG:=4 ELSE                    00008700          
                                   GUARDAMSG:=5;                        00008800          
        END                                                             00008900          
    ELSE                                                                00009000          
    IF J:=MASKSEARCH(KEY,MASK,MEM) GEQ 0 THEN           % DUPLICADO     00009100          
        GUARDAMSG:=2                                                    00009200          
    ELSE                                                                00009300          
    IF LOCK(LK_S500,3)=2 THEN GUARDAMSG:=1 ELSE                         00009400          
        BEGIN                                                           00009500          
        IF J:=MASKSEARCH(CERO,MASK,MEM) < 0 THEN GUARDAMSG:=1 ELSE      00009600          
            BEGIN                                       % HAY LUGAR     00009700          
            KEY        :=TIME(60);                                      00009800          
            MEM     [J]:=FOLIO & EDO [47:1];            % YA MARQUE MEM 00009900          
            SUMM    (J):=1;                                             00010000          
            SZ      [J]:=MSG_SZ;                                        00010100          
            VENCE   [J]:=0 & (KEY.[39:16]        ) [35:16]              00010200          
                           & (KEY.[23:24] DIV 100) [19:20];             00010300          
            OLVIDA  [J]:=VENCE[J];                      % SE CONSERVA   00010400          
            STS   [0,0]:=*+1;                           % UN INSERT MAS 00010500          
            REP CONTEXTO[J,0] BY MSG FOR MSG_SZ;                        00010600          
            END;                                                        00010700          
        J:=UNLOCK(LK_S500);                                             00010800          
        END;                                                            00010900          
    END  GUARDAMSG;                                                     00011000

INTEGER PROCEDURE RESUELVEMSG(FOLIO,ESTADO);                            00011200          
INTEGER FOLIO, ESTADO;                                                  00011300          
    BEGIN                                                               00011400          
    INTEGER J;                                                          00011500          
    REAL KEY;                                                           00011600          
    EXCEPTION PROCEDURE ADIOS;                                          00011700          
       IF CAND(J GEQ 0,LOCKSTATUS(LK_MEM[J]).[47:24]=PROCESSID) THEN    00011800          
           J:=UNLOCK(LK_MEM[J]);                                        00011900          
                                                                        00012000          
    IF FOLIO>0 THEN KEY:=FOLIO ELSE KEY:=48"FFFFFFFFFFFF";              00012100          
    IF J:=MASKSEARCH(KEY,MASK,MEM) < 0 THEN RESUELVEMSG:=2 ELSE         00012200          
    IF LOCK(LK_MEM[J],3)=2             THEN RESUELVEMSG:=1 ELSE         00012300          
        BEGIN                                                           00012400          
        IF MEM[J].[47:1] NEQ ESTADO    THEN RESUELVEMSG:=1 ELSE         00012500          
            BEGIN                                                       00012600          
            KEY:=TIME(1) DIV 60 - OLVIDA[J].[19:20];                    00012700          
            IF KEY < 0 THEN KEY:=*+UN_DIA;                              00012800          
            STS[1,0]:=*+1;                                              00012900          
            STS[2,0]:=*+KEY;                                            00013000          
            SUMM(J) :=0;                                                00013100          
            MEM[J]  :=OLVIDA[J]:=VENCE[J]:=0;                           00013200          
            END;                                                        00013300          
        UNLOCK(LK_MEM[J]);                                              00013400          
        END;                                                            00013500          
    END RESUELVEMSG;                                                    00013600

     ELSE                                                               00015800

             PROTECTION=PROTECTED);                                     00017400

PROCEDURE INFORMA;                                                      00019100          
    BEGIN                                                               00019200          
    LABEL FIN;                                                          00019300          
    REP MSG[23] BY "01";                                                00019400          
    PT:=MSG[42];                                                        00019500          
    REP MSG[100] BY MSG[36] FOR 12;                                     00019600          
    IF PT = "INFO #" THEN                                               00019700          
        BEGIN                                                           00019800          
        H:=J:=MIN(N,INTEGER(MSG[48],4));                                00019900          
        WHILE CAND(J GEQ 0,MEM[J] = 0) DO J:=*-1;                       00020000          
        IF J<0 THEN                                                     00020100          
            REP PT:PT BY 48"1E276B",MSG[100] FOR 12,CRLF,CRLF,          00020200          
                "NO HAY LUGARES OCUPADOS DEBAJO DE ",                   00020300          
                H FOR * DIGITS,CRLF                                     00020400          
        ELSE                                                            00020500          
            BEGIN                                                       00020600          
            AHORA:=VENCE[J].[19:20]+TIEMPO;                             00020700          
            IF T_60:=OLVIDA[J].[19:20]+HORAS_VENCE>UN_DIA THEN          00020800          
                T_60:=*-UN_DIA;                                         00020900          
            IF MEM[J].[47:1]>0 THEN                                     00021000          
                REP DESC BY "VENCIO" ELSE REP DESC BY "ACTIVO";         00021100          
            REP PT:PT BY 48"1E276B",MSG[100] FOR 12,                    00021200          
                (J-1) FOR 4 DIGITS,CRLF,CRLF,                           00021300          
                "LUGAR        :", J FOR 9 NUMERIC,CRLF,                 00021400          
                "FOLIO        :", MEM[J].[23:24] FOR 9 NUMERIC,CRLF,    00021500          
                "LONGITUD     :",SZ[J] FOR 9 NUMERIC,CRLF,              00021600          
                "ESTADO       :   ", DESC FOR 6,CRLF,                   00021700          
                "HORA DE ENVIO: ",                                      00021800          
                (AHORA DIV 3600) FOR 2 NUMERIC,":",                     00021900          
                ((AHORA MOD 3600) DIV 60) FOR 2 DIGITS,":",             00022000          
                (AHORA MOD 60) FOR 2 DIGITS,CRLF,                       00022100          
                "OLVIDO A LAS : ",(T_60 DIV 3600) FOR 2 NUMERIC,":",    00022200          
                ((T_60 MOD 3600) DIV 60) FOR 2 DIGITS,":",              00022300          
                (T_60 MOD 60) FOR 2 DIGITS;                             00022400          
            END;                                                        00022500          
GO TO FIN;                                                              00022600          
        END;                                                            00022700          
    H:=L:=0;                                                            00022800          
    IF AHORA:=ENTIER((TIME(1)-STA_TIME)/60)+120<0 THEN AHORA:=*+UN_DIA; 00022900          
    FOR J:=0 STEP 1 UNTIL M DO H:=*+ONES(RESUMEN[J]);                   00023000          
    REP PT:PT BY 48"1E276B",MSG[100] FOR 6,CRLF,CRLF,                   00023100          
        "INFORMACION DEL PROCESO DE ENVIO AUTOMATICO DE CANCELACIONES", 00023200          
        CRLF,CRLF,"UTILIZACION DE MEMORIA:",CRLF,                       00023300          
        "HORAS DE VIGENCIA: ",(HORAS_VENCE/3600) FOR 6 NUMERIC,CRLF,    00023400          
        "TIME OUT (SEGS)  : ",TIEMPO FOR 6 NUMERIC,CRLF,                00023500          
        "LONGITUD MENSAJES: ",MAX_SZ FOR 6 NUMERIC,CRLF,                00023600          
        "LUGARES  OCUPADOS: ",H FOR 6 NUMERIC,CRLF,                     00023700          
        "         LIBRES  : ",(N+1-H) FOR 6 NUMERIC,CRLF,               00023800          
        "USUARIOS         : ",MAX(0,CUANTOS_ME_USAN-1) FOR 6 NUMERIC,   00023900          
        CRLF,CRLF,"ACTIVIDAD DEL ASINCRONO:",CRLF,                      00024000          
        "NUMERO REVISIONES: ",VUELTAS FOR 6 NUMERIC,CRLF,               00024100          
        "MENSAJES ENVIADOS: ",ENVIOS  FOR 6 NUMERIC,                    00024200          
        CRLF,CRLF,"ESTADISTICAS:",CRLF,                                 00024300          
        "             EN EL DIA     EN LOS ULTIMOS ",                   00024400          
        AHORA FOR * NUMERIC," SEGUNDOS:",                               00024500          
        CRLF,"INSERCIONES: ",STS[0,0] FOR 5 NUMERIC,"     ",            00024600          
                        STS[0,0]-STS[0,2] FOR 5 NUMERIC,CRLF,           00024700          
        "RESPUESTAS : ",STS[1,0] FOR 5 NUMERIC,"     ",                 00024800          
        STS[1,0]-STS[1,2] FOR 5 NUMERIC,CRLF,"T.RESPUESTA: ";           00024900          
        IF STS[1,0]>0 THEN                                              00025000          
            BEGIN                                                       00025100          
            T_60:=STS[2,0]/STS[1,0];                                    00025200          
            REP PT:PT BY ENTIER(T_60) FOR 5 NUMERIC,".",                00025300          
                (100*T_60) MOD 100 FOR 2 DIGITS;                        00025400          
            IF (STS[1,0]-STS[1,2])>0 THEN                               00025500          
                BEGIN                                                   00025600          
                T_60:=(STS[2,0]-STS[2,2])/(STS[1,0]-STS[1,2]);          00025700          
                REP PT:PT BY "  ",ENTIER(T_60) FOR 5 NUMERIC,".",       00025800          
                  (100*T_60) MOD 100 FOR 2 DIGITS;                      00025900          
                END;                                                    00026000          
            END;                                                        00026100          
        REP PT:PT BY CRLF,"DESHECHADOS: ",STS[3,0] FOR 5 NUMERIC,       00026200          
            "     ",STS[3,0]-STS[3,2] FOR 5 NUMERIC,CRLF,48"1E";        00026300          
                                                                        00026400          
FIN:                                                                    00026500          
    COMS_OUT.DESTCOUNT      :=1;                                        00026600          
    COMS_OUT.AGENDA         :=A_OUTPUT;                                 00026700          
    COMS_OUT.TEXTLENGTH     :=OFFSET(PT);                               00026800          
    COMS_OUT.STATUSVALUE    :=0;                                        00026900          
    COMS_OUT.DESTINATIONDESG:=T_DUMMY;                                  00027000          
%   K:=SEND(COMS_OUT,COMS_OUT.TEXTLENGTH,MSG);                          00027100          
    END INFORMA;                                                        00027200          
                                                                        00027300          
PROCEDURE ENVIA;                                                        00027400          
    IF VENCE[J]<LIMITE THEN                                             00027500          
        BEGIN                                                           00027600          
        MEM[J].[47:1]:=1;                       % MARCA COMO CANCELADO  00027700          
        VENCE[J]:=LIMITE;                       % HORA ULTIMO ENVIO     00027800          
        COMS_OUT.DESTCOUNT      :=1;                                    00027900          
        COMS_OUT.AGENDA         :=A_OUTPUT;                             00028000          
        COMS_OUT.TEXTLENGTH     :=SZ[J];                                00028100          
        COMS_OUT.STATUSVALUE    :=0;                                    00028200          
        COMS_OUT.DESTINATIONDESG:=T_DUMMY;                              00028300          
        K:=SEND(COMS_OUT,COMS_OUT.TEXTLENGTH,CONTEXTO[J,*]);            00028400          
        ENVIOS:=*+1;                                                    00028500          
        IF OLVIDA[J]<T_60 THEN                                          00028600          
            BEGIN                                                       00028700          
            STS[3,0]:=*+1;                                              00028800          
            SUMM(J) :=0;                                                00028900          
            REP A_MSG BY "<" FOR 30 WORDS;                              00029000          
            REP A_MSG BY TIME(9),MEM[J].[23:24] FOR 6 DIGITS,           00029100          
                CONTEXTO[J,0] FOR MIN(168,SZ[J]);                       00029200          
            WRITE(LOG,30,A_MSG);                                        00029300          
            LOG_W:=*+1;                                                 00029400          
            MEM[J]:=OLVIDA[J]:=VENCE[J]:=0;                             00029500          
            END;                                                        00029600          
        END                                                             00029700          
    ELSE                                                                00029800          
        IF VENCE[J]<MENOR THEN MENOR:=VENCE[J];         % END ENVIA     00029900          
                                                                        00030000          
PROCEDURE REVISA;                                                       00030100          
    BEGIN                                                               00030200          
    EXCEPTION PROCEDURE ADIOS;                                          00030300          
       IF CAND(J GEQ 0,LOCKSTATUS(LK_MEM[J]).[47:24]=PROCESSID) THEN    00030400          
            K:=UNLOCK(LK_MEM[J]);                                       00030500          
                                                                        00030600          
    J:=-1;                                                              00030700          
    L:=TIME(60);                                                        00030800          
    AHORA:=0 & (L.[39:16]) [35:16] & (L.[23:24] DIV 100) [19:20];       00030900          
    IF (L:=AHORA.[19:20] - HORAS_VENCE) < 0 THEN                        00031000          
        T_60:= 0 & (AHORA.[35:16]-1) [35:16] & (L+UN_DIA) [19:20]       00031100          
    ELSE                                                                00031200          
        T_60:= 0 & (AHORA.[35:16]  ) [35:16] & (L       ) [19:20];      00031300          
    IF (L:=AHORA.[19:20] - TIEMPO     ) < 0 THEN                        00031400          
        LIMITE:= 0 & (AHORA.[35:16]-1) [35:16] & (L+UN_DIA) [19:20]     00031500          
    ELSE                                                                00031600          
        LIMITE:= 0 & (AHORA.[35:16]  ) [35:16] & (L       ) [19:20];    00031700          
    VUELTAS:=*+1;                                                       00031800          
    MENOR:=549755813887;                                                00031900          
                                                                        00032000          
    IF CUANTOS_ME_USAN>1 THEN                                           00032100          
        FOR H:=0 STEP 1 UNTIL M DO                                      00032200          
            IF RESUMEN[H] ISNT 0 THEN                                   00032300          
                BEGIN                                                   00032400          
                L:=48*H;                                                00032500          
                FOR J:=L STEP 1 UNTIL L+47 DO                           00032600          
                    IF CAND(MEM[J] NEQ 0,LOCK(LK_MEM[J],2)=1) THEN      00032700          
                        BEGIN                                           00032800          
                        IF MEM[J] NEQ 0 THEN ENVIA;                     00032900          
                        K:=UNLOCK(LK_MEM[J]);                           00033000          
                        END;                                            00033100          
                END;                                                    00033200          
                                                                        00033300          
    IF MENOR=549755813887 THEN                                          00033400          
        IF (L:=AHORA.[19:20] + TIEMPO) > UN_DIA THEN                    00033500          
            MENOR:=0 & (AHORA.[35:16]+1) [35:16] & (L-UN_DIA) [19:20]   00033600          
        ELSE                                                            00033700          
            MENOR:=0 & (AHORA.[35:16])   [35:16] & (L       ) [19:20];  00033800          
    NEXT_TIME:=MENOR.[19:20]+5;                                         00033900          
    END REVISA;                                                         00034000          
                                                                        00034100          
    REP MSG BY "DEFAULTSTATION",0;                                      00034200          
    J:=GET_DESIGNATOR_USING_NAME(MSG,1,T_DUMMY);                        00034300          
    REP MSG BY "OUTPUT OF LINEA",0;                                     00034400          
    J:=GET_DESIGNATOR_USING_NAME(MSG,3,A_OUTPUT);                       00034500          
    REP POINTER(TITULO)+23 BY MYSELF.STACKNO FOR 7 DIGITS;              00034600          
    REP LOG.TITLE BY TITULO;                                            00034700          
    J:=DELINKLIBRARY(COMS);                                             00034800          
    ACTIVO:=TRUE;                                                       00034900          
    NEXT_TIME:=TIME(1)/60+TIME_OUT_DEF;                                 00035000          
                                                                        00035100          
    WHILE NOT ENDING DO                                                 00035200          
        BEGIN                                                           00035300          
        DELTA:=MIN(TIEMPO,ABS(NEXT_TIME-TIME(1)/60));                   00035400          
        CASE WAIT((DELTA),CIERRA_LOG,IN_EV,TSK_EV) OF                   00035500          
            BEGIN                                                       00035600          
        01: REVISA;                                                     00035700          
        02: IF LOG_W>0 THEN                                             00035800          
                BEGIN                                                   00035900          
                REP POINTER(TITULO)+14 BY "RECHAZOS";                   00035920          
                REP LOG.TITLE BY TITULO;                                00035940          
                CLOSE(LOG,LOCK);                                        00036000          
                REP POINTER(TITULO)+14 BY "RECHOPEN";                   00036050
