CORE-GL · P197 a Java 21 Spring Boot

Recibe el número de sistema y devuelve los nombres de la base de saldos, de la base de movimientos vigente y de la base que debe inicializarse. Lee la fecha de proceso de la base de control diario, calcula la siguiente fecha hábil y consulta el registro de control mensual con la política de ciclos. Todos los días limpia los niveles diarios y depura datasets de posición; si mañana cambia el mes, además arrastra el saldo de cierre de cada contrato al mes siguiente, borra la información fuera de la ventana de retención y rota el ciclo de bases de movimientos. Al final marca el estatus de la base de saldos y graba la fecha en la base de control.

@Component
@RequiredArgsConstructor
public class RotaBaseDecider implements JobExecutionDecider {

    private final ControlSistemaMensualRepository control;

    /** 01-0000-INICIO. */
    @Override
    public FlowExecutionStatus decide(JobExecution job, StepExecution last) {
        int sistema = job.getJobParameters().getLong("sistema", 0L).intValue();
        if (sistema <= 0) {
            // TIPO-LJ "E", CODIGO-LJ 051001; CHANGE ATTRIBUTE STATUS OF MYSELF TO -1
            throw new ParametroInvalidoException("EL NUMERO DE SISTEMA DEBE DE SER MAYOR A CERO");
        }
        var ctx = job.getExecutionContext();
        YearMonth hoy = YearMonth.from((LocalDate) ctx.get("fechaHoy"));        // WKS-FECHOY-AAMM
        YearMonth manana = YearMonth.from((LocalDate) ctx.get("fechaManana"));  // WKS-FECMAN-AAMM
        if (hoy.equals(manana)) {
            return new FlowExecutionStatus("DIARIO");
        }
        int estatus = control.findById(sistema).orElseThrow().getEstatusBaseSaldos();
        return (estatus == 99 || estatus == 1)
                ? new FlowExecutionStatus("CIERRE_SIN_MENSUAL")   // diario + limpia cuenta
                : new FlowExecutionStatus("CIERRE_MENSUAL");      // diario + limpia cuenta + mensual
    }
}

@Bean
Job rotaBase(JobRepository repo, Step proyectaFechas, RotaBaseDecider decider,
             Step diario, Step limpiaCuenta, Step mensual, Step cierre) {
    return new JobBuilder("CORE-GLP197", repo)
            .start(proyectaFechas)               // 01-00300, 01-0190, 01-0110, 01-0200, 01-00200
            .next(decider)
                .on("DIARIO").to(diario).next(cierre)
            .from(decider).on("CIERRE_SIN_MENSUAL").to(diario).next(limpiaCuenta).next(cierre)
            .from(decider).on("CIERRE_MENSUAL").to(diario).next(limpiaCuenta).next(mensual).next(cierre)
            .end().build();
    // cierre: 01-0230 cierra bases, 01-0160 marca STABDSAL = 1, 01-0150 graba fecha, 01-00220 B05, 01-0170 L002
}
/** 01-0160-GRABA-STABDSAL: CONSISMEN función 9 marca la base de saldos como procesada (STABDSAL = 1). */
@Transactional
public void marcaBaseSaldosProcesada(int sistema) {
    int filas = control.actualizaEstatusBaseSaldos(sistema, 1);
    if (filas != 1) {
        // El COBOL arma el mensaje con W77-RES-LIBCTL (resultado de otra llamada) en lugar de
        // W77-RESULT-LIBCON; aquí se informa el resultado real de esta operación.
        throw new ControlSistemaException(
                "ERROR %d AL GRABAR EL ESTATUS EN BASE DE CONTROL".formatted(filas), sistema);
    }
}

    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        ControlSistemaMensual cfg = Ctx.control(ctx);
        LocalDate fechaProceso = Ctx.fechaProceso(ctx);              // WKS-B01-FECPRO
        boolean baseSaldosAbierta = Ctx.baseSaldosAbierta(ctx);      // W77-ABRE-SDO = 1

        if (cfg.getNumCiclosDiarios() > 0 && TIPOS_CON_DIARIOS.contains(cfg.getTipoBd().ordinal())) {
            LocalDate fechaBorrado = calendario.retrocedeCiclos(fechaProceso, cfg.getNumCiclosDiarios());
            if (baseSaldosAbierta) saldosDiarios.borraHasta(fechaBorrado);   // B72, B70 y B80
        }
        if (cfg.getNumCiclosMensuales() > 0 && TIPOS_CON_MENSUALES.contains(cfg.getTipoBd().ordinal())) {
            LocalDate fechaBorradoMensual = calendario.retrocedeCiclos(fechaProceso, cfg.getNumCiclosMensuales());
            Ctx.put(ctx, "fechaMensualBorrar", fechaBorradoMensual);         // WKS-FECMEN-BORRAR
        }
        return RepeatStatus.FINISHED;
    }
}
@Component

public class ArrastreSaldoMensualProcessor implements ItemProcessor<SaldoMensual, SaldoMensual> {

    private final ArrastreProperties props;   // sistema 84, instrumentos 15 y 17
    @Value("#{jobExecutionContext['mesSiguiente']}") private YearMonth mesSiguiente;   // WKS-FECMAN-AAMM

    /** 01-0554-FN-B20MENCON + 01-0556-GRABA-MES sobre CORE-GLB20SDOMENCON. */
    @Override
    public SaldoMensual process(SaldoMensual actual) {
        BigDecimal cierre = actual.saldo(2);                       // WKS-B20-SD(2), S9(13)V9(02)
        boolean excepcion84 = props.sistema() == actual.sistema()
                && props.instrumentos().contains(actual.instrumento());
        if (cierre.signum() == 0 && !excepcion84 && actual.apuImp() == 0) {
            return null;                                           // saldo cero: no se arrastra
        }
        // LOCK ... ON EXCEPTION NOTFOUND -> CREATE; luego MOVE y STORE: upsert por llave completa.
        return actual.paraMes(mesSiguiente)
                     .conSaldo(1, cierre)                          // apertura del mes siguiente
                     .conSaldo(2, cierre);                         // saldo actual = apertura
    }
}

@Service
@RequiredArgsConstructor
public class RotacionBasesService {

    private final ControlSistemaMensualRepository control;

    /** 01-0590-ROTABASES. Devuelve el nombre de la base a inicializar (WKS-SAL-NOMBD12-INI). */
    @Transactional
    public Optional<String> rota(int sistema, YearMonth mesNuevo) {
        ControlSistemaMensual cfg = control.findById(sistema).orElseThrow();
        int n = cfg.getNumCiclosDiarios();
        if (n <= 0) return Optional.empty();

        List<CicloBase> ciclos = cfg.getCiclos().subList(0, n);
        int masAntiguo = IntStream.range(0, n)                     // 01-0592-BUSCA-INDICE
                .boxed().min(Comparator.comparing(i -> ciclos.get(i).anioMes())).orElseThrow();

        // Función 15: el ciclo más antiguo pasa a representar el mes nuevo.
        CicloBase reciclado = ciclos.get(masAntiguo);
        ciclos.set(masAntiguo, reciclado.paraMes(mesNuevo).sinContadores());

        // Función 17: todos los números de base avanzan uno y regresan a 1 al pasar de n.
        for (int i = 0; i < n; i++) {
            CicloBase c = ciclos.get(i);
            ciclos.set(i, c.conNumeroBase(c.numeroBase() + 1 > n ? 1 : c.numeroBase() + 1));
        }
        return Optional.of(reciclado.nombreBaseMov());   // la WFL que llama a P197 inicializa esa base
    }
}
/** 01-0350-BORRA-NIVELES: pone en cero los contadores del día en el registro global de cada base. */
@Transactional
public void limpiaNiveles(LocalDate fechaMovtos, CalendarioBancario cal) {
    int dia1 = fechaMovtos.getDayOfMonth();                     // W77-BORRA-DIA1
    int dia2 = cal.avanza(fechaMovtos, 1).getDayOfMonth();      // W77-BORRA-DIA2
    int dia3 = cal.siguiente(fechaMovtos, 2).getDayOfMonth();   // W77-BORRA-DIA3 (ver pendientes)

    NivelesBase saldos = globalSaldos.lock();                   // LOCK BASESDO
    saldos.limpia(dia1, dia2, dia3);                            // LECARCDIA, MOVCICDIA, SDOCICDIA
    globalSaldos.store(saldos);                                 // STORE BASESDO

    NivelesBase movs = globalMovs.lock();                       // LOCK BASEMOV
    movs.limpia(dia1, dia2, dia3);   // el COBOL limpia DIA3 sobre B00-SDO-* y no sobre B00-MOV-*
    globalMovs.store(movs);                                     // STORE BASEMOV
}

Riesgos de migración
Limpieza incompleta en la base de movimientos

En 01-0350-BORRA-NIVELES (líneas 2084-2086) el tercer día se limpia sobre los campos B00-SDO-* después de haber grabado la base de saldos, en lugar de B00-MOV-*. Los contadores de ese día en BASEMOV quedan con el valor anterior.

Rotación no atómica

Las funciones 15 y 17 de CONSISMEN se llaman por separado y sus errores solo se registran (líneas 2739-2750 y 2781-2792); el programa termina normalmente aunque el control quede a medio rotar.

Código de error equivocado en bitácora

01-0160-GRABA-STABDSAL evalúa W77-RESULT-LIBCON pero reporta W77-RES-LIBCTL (líneas 1628 y 1632). En 01-00300-PROYECTA-FECHAS (línea 1909) el mensaje usa W77-ERR-LIBCTL sin haberle asignado el resultado.

Borrado de movimientos sin invocar

Los párrafos 01-0320-FF-MOVTOS y 01-0330-BORRA-MOV (líneas 1974-2013), que borran movimientos con sus importes y datos adicionales, no se invocan desde ningún PERFORM. No se migran como lógica activa.

Casos particulares compilados

El sistema 84 y sus instrumentos 15 y 17 tienen tratamiento especial en el arrastre de saldos y en la retención mensual (líneas 1399, 2138 y 2420); los tipos de base válidos para cada borrado también están en el código. Se trasladan a application.yml.
