/** 010-0100-CHECA-OPCION: en Java la inicialización es del contenedor, no de la primera llamada. */
@Override
public PantallaResponse consulta(PantallaRequest req) {
    return switch (req) {                                    // 010-0300-VALIDA
        case ConsultaProteccionRequest p    -> proteccion(p, 1);          // pantalla 27, servicio 1
        case ConsultaAlertasRequest a       -> a.servicio() > 1
                                               ? proteccion(a.comoProteccion(), a.servicio())
                                               : PantallaResponse.error(0, "NUMERO DE SERVICIO INVALIDO");
        case ConsultaMigracionRequest m     -> migracion(m);             // pantalla 29
        case ConsultaDomiciliacionRequest d -> domiciliacion(d);         // pantalla 30
    };
}

// 010-0200-INICIA-LIBRERIA: los MOVE a SISTEMA-LJ/PASO-LJ/MTP-LJ y el título de LIBLJ
// pasan a application.yml (bitácora) y LOCSUP se inyecta como CalendarioBancario.
@Configuration
@EnableRetry
class BifinDataSourceConfig {

    /** 090-ABRE-BASE: hasta 6 intentos (WS-VECES > 5) con WAIT(10). */
    @Bean
    RetryTemplate aperturaBifin() {
        return RetryTemplate.builder()
                .maxAttempts(6)
                .fixedBackoff(Duration.ofSeconds(10))
                .retryOn(CannotGetJdbcConnectionException.class)
                .withListener(new RetryListener() {
                    @Override public <T, E extends Throwable> void onError(RetryContext c, RetryCallback<T, E> cb, Throwable t) {
                        log.warn("BASE NO ABIERTA,ESPERO 10 SEG (intento {})", c.getRetryCount());   // CALL LJ IN LIBLJ
                    }
                })
                .build();
    }
}
// En lugar de bloquear al cajero hasta 60 segundos, el pool se valida al arrancar
// y un HealthIndicator marca el servicio fuera de línea si BIFIN no responde.
/** 010-0300-PANTALLA27. */
private PantallaResponse proteccion(ConsultaProteccionRequest p, int servicio) {
    // IF NOT(CONTRATO > ZEROS OR CONTRATO = ZEROS): para un PIC 9(16) sin signo la condición
    // solo es verdadera si el campo trae datos no numéricos. Se valida el formato explícitamente.
    if (!p.contrato().chars().allMatch(Character::isDigit)) {
        return PantallaResponse.error(46, "CONTRATO INVALIDO");
    }
    if (!p.autorizacion().chars().allMatch(Character::isDigit)) {
        return PantallaResponse.error(68, "AUTORIZACION INVALIDA");
    }
    return consultaProteccion(p, servicio);                  // 010-0400-VALIDA-COND
}

/** 010-0300-VALIDA-FECHA. LOCSUP función 20 = fecha válida; función 18 = día hábil (0 si es inhábil). */
private Optional<PantallaResponse> validaFecha(PantallaRequest req, LocalDate fecha) {
    if (!calendario.esValida(fecha)) {                         // CALL THECALENDAR, función 20, formato 13
        return Optional.of(PantallaResponse.error(19, "FECHA INVALIDA"));
    }
    if (req instanceof ConsultaMigracionRequest) {
        return Optional.empty();                               // pantalla 29: no exige día hábil
    }
    if (!calendario.esHabil(fecha)) {                          // función 18
        return Optional.of(PantallaResponse.error(76, "FECHA INHABIL"));
    }
    if (req instanceof ConsultaDomiciliacionRequest d) {       // 010-0030-VALIDA-FECHA-FIN
        LocalDate fin = d.actual().fechaFin();
        if (!calendario.esValida(fin)) return Optional.of(PantallaResponse.error(19, "FECHA INVALIDA"));
        if (!calendario.esHabil(fin))  return Optional.of(PantallaResponse.error(76, "FECHA INHABIL"));
        if (d.actual().fechaIni().isAfter(fin))
            return Optional.of(PantallaResponse.error(97, "RANGO DE FECHA INVALIDO"));
    }
    return Optional.empty();
}

/** W77-OPC-BUS-MIG con sus niveles 88 W88-X-*. */
public enum CriterioMigracion { FECHA, FECHA_SUC_CONTRATO, FECHA_SUC_CUENTA, FECHA_TARJETA, FECHA_BIN, BIN }

/** 010-2900-PANTALLA29: el criterio se deduce de qué campos vienen en cero. */
static Optional<CriterioMigracion> criterio(ConsultaMigracionRequest m) {
    boolean fec = m.fecha() != null, suc = m.sucursal() > 0, cto = !esCero(m.contratoCuenta()),
            tar = !esCero(m.tarjeta()), bin = !esCero(m.bin());
    int mda = m.medioAcceso();

    if (fec && mda == 0 && !suc && !cto && !tar && !bin)                  return Optional.of(CriterioMigracion.FECHA);
    if (fec && (mda == 0 || mda == 1) && suc && cto && !tar && !bin)
        return Optional.of(mda == 0 ? CriterioMigracion.FECHA_SUC_CONTRATO : CriterioMigracion.FECHA_SUC_CUENTA);
    if (fec && mda == 2 && tar && !suc && !cto && !bin)                   return Optional.of(CriterioMigracion.FECHA_TARJETA);
    if (bin && mda == 0 && !suc && !cto && !tar) {
        if (!fec) return Optional.of(CriterioMigracion.BIN);
        // con fecha: si es continuación de una búsqueda por BIN, se ignora la fecha
        boolean continuaBin = m.anterior().fecha() == null && m.masMovimientos();
        return Optional.of(continuaBin ? CriterioMigracion.BIN : CriterioMigracion.FECHA_BIN);
    }
    return Optional.empty();                                              // 77 CONSULTA INVALIDA
}
