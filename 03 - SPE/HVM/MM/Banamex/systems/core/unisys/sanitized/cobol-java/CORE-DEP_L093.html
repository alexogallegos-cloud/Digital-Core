CORE-DEP · L093 a Java 21 Spring Boot

L093 mantiene un buzón de 2016 posiciones compartido por todos los programas que la enlazan. Los programas de aplicación depositan un mensaje con GUARDAMSG indicando folio, longitud y estado, lo confirman con RESUELVEMSG cuando reciben respuesta o lo eliminan con BORRAMSG. Un proceso asíncrono que entra por ENVIA_CANCELACIONES revisa el buzón cada pocos segundos y reenvía por COMS cada mensaje cuyo tiempo de espera venció, marcándolo como cancelado, hasta que se confirma o se cumplen las horas de vigencia; en ese caso lo escribe en una bitácora de rechazos y libera la posición. Al descargarse la librería vuelca los pendientes a CORE-DEP/FILE/I93/MOVTOS/PEND y al arrancar los recupera.

    private static final int MAX_LONGITUD_ABSOLUTA = 2994;

    /** SETMSGSZ: solo crece la longitud; horas fuera de 1..24 se toman como 2; el CSI 32 vive la mitad. */
    public synchronized int setMsgSz(int longitudAplicacion, int horas) {
        if (longitudAplicacion > Math.min(MAX_LONGITUD_ABSOLUTA, maxLongitud)) {
            maxLongitud = Math.min(MAX_LONGITUD_ABSOLUTA, longitudAplicacion);   // RESIZE ... RETAIN
        }
        int h = (horas < 1 || horas > 24) ? 2 : horas;
        Duration v = Duration.ofHours(h);
        if ("32".equals(props.csiLocal())) {          // IF YO = "32" FOR 2
            v = v.dividedBy(2);
        }
        vigencia = v;
        return 0;
    }
@Component
@RequiredArgsConstructor
class PersistenciaBuzon {

    private final BuzonCancelaciones buzon;
    private final PendientesRepository pendientes;   // reemplaza CORE-DEP/FILE/I93/MOVTOS/PEND

    /** ARRANCAN: recarga lo que FIN_LIB dejó en disco y borra el archivo (CLOSE ... PURGE). */
    @PostConstruct
    @Transactional
    void recupera() {
        pendientes.encabezado().ifPresent(h -> {
            buzon.restauraParametros(h.maxLongitud(), h.vigencia(), h.tiempoEspera());
            Instant minimo = Instant.now().minus(h.vigencia()).plus(Duration.ofHours(1));
            for (var p : pendientes.findAll()) {
                // OLVIDA = MAX(original, ahora - vigencia + 1 h): lo recuperado vive al menos una hora más
                Instant base = p.insercion().isAfter(minimo) ? p.insercion() : minimo;
                buzon.restaura(new EntradaBuzon(p.folio(), p.cancelacionEnviada(), p.mensaje(),
                                                p.ultimoEnvio(), base));
            }
            pendientes.deleteAllInBatch();
        });
    }
}
    /** FIN_LIB (EPILOG): si hay posiciones ocupadas, las guarda con su contenido para el siguiente arranque. */
    @PreDestroy
    @Transactional
    void vuelca() {
        var vivas = buzon.instantanea();                      // copia consistente del mapa
        if (vivas.isEmpty()) return;
        pendientes.guardaEncabezado(buzon.maxLongitud(), buzon.vigencia(), buzon.tiempoEspera());
        pendientes.saveAll(vivas.stream().map(Pendiente::de).toList());
    }

    /**
     * Códigos de retorno de GUARDAMSG: 0 guardado, 1 sin lugar o lock ocupado, 2 folio duplicado,
     * 3 mayor a MAX_SZ, 4 mayor que el arreglo recibido, 5 folio menor a 1.
     */
    public int guardaMsg(long folio, int longitud, int estado, byte[] mensaje) {
        if (longitud > maxLongitud) return 3;
        if (longitud > mensaje.length) return 4;
        if (folio < 1) return 5;
        if (entradas.size() >= CAPACIDAD) return 1;
        Instant ahora = Instant.now();
        var nueva = new EntradaBuzon(folio, estado == 1, Arrays.copyOf(mensaje, longitud), ahora, ahora);
        // putIfAbsent hace atómicas la búsqueda de duplicado y el alta, que en el fuente están separadas.
        if (entradas.putIfAbsent(folio, nueva) != null) return 2;
        sts.insercion();                                         // STS[0,0] := * + 1
        return 0;
    }
    /** 0 resuelto, 1 lock ocupado o estado distinto, 2 folio no encontrado. */
    public int resuelveMsg(long folio, int estado) {
        if (folio <= 0) return 2;                          // KEY := 48"FFFFFFFFFFFF": no coincide con ningún folio
        var resultado = new int[] {2};
        entradas.computeIfPresent(folio, (k, e) -> {
            if (e.cancelacionEnviada() != (estado == 1)) { // MEM[J].[47:1] NEQ ESTADO
                resultado[0] = 1;
                return e;                                  // se conserva la entrada
            }
            sts.respuesta(Duration.between(e.insercion(), Instant.now()));   // STS[1,0], STS[2,0]
            resultado[0] = 0;
            return null;                                   // libera la posición
        });
        return resultado[0];
    }

    private final BitacoraRechazos bitacora;          // archivo CORE-DEP/FILE/L93/RECHOPEN|RECHAZOS
    private final ScheduledExecutorService reloj = Executors.newSingleThreadScheduledExecutor();
    private volatile boolean activo;                  // ACTIVO

    /** ENVIA_CANCELACIONES: lo arranca el programa asíncrono; aquí, el ciclo de vida del contenedor. */
    @Override public void start() {
        activo = true;
        programaSiguiente(buzon.tiempoEspera());
    }

    private void programaSiguiente(Duration espera) {
        if (!activo) return;
        reloj.schedule(() -> { programaSiguiente(revisa()); }, espera.toMillis(), TimeUnit.MILLISECONDS);
    }

    /** CIERRA_LOG: rota la bitácora de rechazos si se escribió algo. */
    @EventListener(RotaBitacoraEvent.class) void rota() { bitacora.cierraYRota(); }

    @Override public void stop()  { activo = false; reloj.shutdown(); bitacora.cierra(); }   // SNIF
    @Override public boolean isRunning() { return activo; }
}
    /** REVISA: devuelve cuánto esperar hasta la siguiente revisión. */
    private Duration revisa() {
        Instant ahora = Instant.now();
        Instant limiteReenvio = ahora.minus(buzon.tiempoEspera());   // LIMITE
        Instant limiteOlvido  = ahora.minus(buzon.vigencia());       // T_60
        buzon.estadisticas().vuelta();                               // VUELTAS := * + 1
        Instant proximo = null;                                      // MENOR

        if (buzon.usuariosEnlazados() > 1) {                         // CUANTOS_ME_USAN > 1
            for (var e : buzon.instantanea()) {
                if (e.ultimoEnvio().isBefore(limiteReenvio)) {
                    envia(e, limiteReenvio, limiteOlvido);           // segmento s10
                } else if (proximo == null || e.ultimoEnvio().isBefore(proximo)) {
                    proximo = e.ultimoEnvio();
                }
            }
        }
        Instant siguiente = (proximo == null) ? ahora.plus(buzon.tiempoEspera())
                                              : proximo.plus(buzon.tiempoEspera());
        Duration espera = Duration.between(ahora, siguiente);
        return espera.isNegative() ? Duration.ZERO : espera;
    }
    private void envia(EntradaBuzon e, Instant limiteReenvio, Instant limiteOlvido) {
        var marcada = buzon.marcaEnviada(e.folio(), limiteReenvio);   // MEM[J].[47:1] := 1; VENCE := LIMITE
        if (marcada.isEmpty()) return;                                // la resolvieron mientras tanto
        boolean enviado = comsSalida.send(MessageBuilder.withPayload(e.mensaje())
                .setHeader("coms.destino", "OUTPUT OF LINEA").build());
        buzon.estadisticas().envio();                                 // ENVIOS := * + 1
        if (!enviado) log.warn("Reenvío fallido folio {}", e.folio()); // el fuente ignora K := SEND(...)

        if (e.insercion().isBefore(limiteOlvido)) {                   // vigencia agotada
            buzon.estadisticas().desechado();                         // STS[3,0]
            bitacora.escribe(LocalTime.now(), e.folio(),
                    Arrays.copyOf(e.mensaje(), Math.min(168, e.mensaje().length)));
            buzon.libera(e.folio());
        }
    }
@Component
@RequiredArgsConstructor
public class TiempoEsperaAdaptativo {

    private static final Duration MINIMO = Duration.ofSeconds(3);     // TIME_OUT_DEF
    private static final Duration MAXIMO = Duration.ofSeconds(60);    // no existe en el fuente; propuesta
    private final BuzonCancelaciones buzon;
    private final ReenvioCancelaciones asincrono;
    private int vecesSinAsincrono;

    /** USO: cada 2 segundos rota estadísticas y ajusta TIEMPO según el tiempo medio de respuesta. */
    @Scheduled(fixedRate = 2_000)
    void ajusta() {
        var ventana = buzon.estadisticas().rota();                    // STS[K,2] := STS[K,1] := STS[K,0]
        if (ventana.respuestas() > 0) {
            Duration medio = ventana.tiempoTotal().dividedBy(ventana.respuestas());
            Duration t = buzon.tiempoEspera();
            Duration nuevo = medio.compareTo(t.dividedBy(2)) > 0
                    ? t.multipliedBy(3).dividedBy(2)                  // TIEMPO := 1.5 * TIEMPO
                    : mayor(MINIMO, t.multipliedBy(2).dividedBy(3));  // MAX(TIME_OUT_DEF, 2*TIEMPO/3)
            buzon.tiempoEspera(nuevo.compareTo(MAXIMO) > 0 ? MAXIMO : nuevo);
        }
        if (!asincrono.isRunning() && ++vecesSinAsincrono > 2) {
            vecesSinAsincrono = 0;
            log.warn("060103 01 CORE-DEP L093 PRECAUCION: FAVOR DE ACTIVAR PROGRAMA ASINCRONO");
        }
    }

    private static Duration mayor(Duration a, Duration b) { return a.compareTo(b) >= 0 ? a : b; }
}

Riesgos de migración
Condiciones de carrera en altas y confirmaciones

GUARDAMSG revisa duplicados antes de tomar el lock global (líneas 92-95) y RESUELVEMSG y BORRAMSG no vuelven a comparar el folio después de bloquear la posición (líneas 123-126 y 149). Puede haber folios duplicados o confirmaciones aplicadas a la entrada equivocada si coinciden en el tiempo.

Tiempo de espera sin tope

USO multiplica TIEMPO por 1.5 cada dos segundos mientras el tiempo medio de respuesta supere la mitad del plazo (líneas 490-492), sin límite superior.

Pérdida de pendientes ante caída

Los mensajes solo se persisten en el epílogo FIN_LIB (líneas 454-477); una terminación anormal de la librería pierde las cancelaciones pendientes.

Datos sensibles fuera de memoria

La bitácora de rechazos guarda hasta 168 caracteres de cada mensaje vencido (líneas 294-297) y las opciones de volcado activadas en ARRANCAN (líneas 410-413) incluyen el contenido del buzón en los dumps.

Consulta de estado sin respuesta

INFORMA construye el informe pero el SEND está comentado (línea 275); la consulta por COMS no recibe respuesta.

Capacidad de 2016 posiciones, longitud máxima 2994, tiempo de espera inicial de 3 segundos, vigencia por omisión de 2 horas, CSI 32 con vigencia a la mitad, destino COMS OUTPUT OF LINEA y títulos de archivos de bitácora y pendientes.
