@Component
@RequiredArgsConstructor
public class UmbralesRetencionTasklet implements Tasklet {

    private final ControlSistemaClient l001;       // "CONSISDIA IN CORE-GLL001"
    private final CalendarioBancario fechas;       // "DAME_ANTHAB2K IN FECHAS"
    private final P153Properties props;

    @Override
    public RepeatStatus execute(StepContribution contribution, ChunkContext ctx) {
        var params = ctx.getStepContext().getJobParameters();
        // La fecha de proceso viene de L001 para el sistema 500; un TASKVALUE > 0 la sustituía.
        LocalDate fechaProceso = Optional.ofNullable((LocalDate) params.get("fechaProceso"))
                .orElseGet(() -> l001.fechaProceso(1, 500)
                        .orElseThrow(() -> new ProcesoAbortadoException("CONSISDIA SIN FECHA")));

        // Clave de semana 65 y de festivos 1: calendario bancario con días inhábiles.
        LocalDate umbral = fechas.diaHabilAnterior(fechaProceso,
                props.claveSemana(), props.claveFestivos());       // 65, 1
        ctx.getStepContext().getStepExecution().getJobExecution()
           .getExecutionContext().put("fechaOut", umbral);         // WKS-FECHAOUT
        return RepeatStatus.FINISHED;
    }
}

@Configuration
public class DepuracionBifinJobConfig {

    /** 200-1000-PROCESO: un step por set, en el mismo orden; el HI se valida entre steps. */
    @Bean
    Job depuracionBifin(JobRepository repo, Step umbrales, Step reiniciaGlobales,
                        Step totProd, Step alarmas, Step posicion, Step protCob, Step domi,
                        Step batch, Step ctlEnvio, Step ctlGlobalDir, Step tdMigCap, Step ticketsBn) {
        return new JobBuilder("CORE-GLP153", repo)
                .start(umbrales)
                .next(reiniciaGlobales)     // 0500-INICIALIZA-GLOBALES
                .next(totProd)              // B01 TOTPROD   < fechaOut
                .next(alarmas)              // B03 ALARMAS   < fechaOut
                .next(posicion)             // B02 POSICION  < fechaOut
                .next(protCob)              // B07 PROTCOB   ids 1 y 2, ventana 15
                .next(domi)                 // B10 DOMI      id 3, ventana 15
                .next(batch)                // B05 BATCH     < fechaOut
                .next(ctlEnvio)             // B06 CTLENVIO  < fechaOut
                .next(ctlGlobalDir)           // B04 CTLGLOBALDIR ventana 3
                .next(tdMigCap)             // B08 TDMIGCAP  ventana 365
                .next(ticketsBn)            // B09 TICKETSBN
                .build();
        // 9999-VALIDA-HI: JobOperator.stop() detiene el job al terminar el chunk en curso.
    }
}
@Bean
Step totProd(JobRepository repo, PlatformTransactionManager tx, EntityManagerFactory emf,
             @Value("#{jobExecutionContext['fechaOut']}") LocalDate fechaOut) {
    // LOCK FIRST / LOCK NEXT B01SXTOTPROD AT B01-TOT-FECINF < WKS-FECHAOUT
    var lector = new JpaCursorItemReaderBuilder<TotProd>()
            .name("totProdDepurables").entityManagerFactory(emf)
            .queryString("select t from TotProd t where t.fecInf < :fechaOut order by t.id")
            .parameterValues(Map.of("fechaOut", fechaOut))
            .build();

    return new StepBuilder("totProd", repo)
            .<TotProd, TotProd>chunk(props.commitCada(), tx)     // END-TRANSACTION cada 100 borrados
            .reader(lector)
            .writer(chunk -> chunk.forEach(em()::remove))        // DELETE CORE-GLB01TOTPROD
            .faultTolerant()
            .skip(OptimisticLockingFailureException.class).skipLimit(0)   // un error de borrado falla el step
            .build();
}

@Component
@RequiredArgsConstructor
public class ReiniciaGlobalesTasklet implements Tasklet {

    private static final int CONTADORES = 30;           // B00-GLO-NCTLBIFIN(1..30)
    private final GlobalesBifinRepository globales;

    /** 0500-INICIALIZA-GLOBALES: LOCK del registro global, ceros en el arreglo y STORE. */
    @Override
    @Transactional
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        GlobalesBifin g = globales.findForUpdate()                       // LOCK BIFINDB
                .orElseThrow(() -> new ProcesoAbortadoException(" ERROR AL BLOQUEAR GLOBALES "));
        for (int i = 1; i <= CONTADORES; i++) {
            g.setContadorControl(i, 0L);                                 // MOVE ZEROS
        }
        return RepeatStatus.FINISHED;                                    // STORE BIFINDB al commit
    }
}
@Component
@Endpoint(id = "p153")
@RequiredArgsConstructor
public class P153AdminEndpoint {

    private final JobOperator operador;
    private final JobExplorer explorador;

    /** Equivale a HI 4: la ejecución en curso se detiene al cerrar el chunk activo. */
    @WriteOperation
    @PreAuthorize("hasRole('OPERADOR_CORE-GL')")
    public String hi(@Selector int valor) throws Exception {
        if (valor != 4) return "HI IGNORADO";                        // solo W88-HI-4 tiene efecto
        for (Long id : explorador.findRunningJobExecutions("CORE-GLP153").stream()
                                 .map(JobExecution::getId).toList()) {
            operador.stop(id);                                       // commit del chunk y STOPPED
        }
        return "TERMINACION NORMAL";                                 // 800100-HI-SALIDA
    }
}

Riesgos de migración
Errores de base no fatales

Los ON EXCEPTION de LOCK, DELETE, LOCK BIFINDB y STORE BIFINDB solo escriben en bitácora (por ejemplo, líneas 1072-1079, 1084-1088 y 1290-1302). La depuración continúa aunque falle el reinicio de globales, y un error persistente en LOCK NEXT puede dejar un ciclo sin salida.

Salida por HI con transacción abierta

9999-VALIDA-HI (líneas 1511-1515) cierra la base y termina sin END-TRANSACTION; hasta 99 borrados (999 en B08) quedan en una transacción no confirmada al momento del cierre.

Fecha de proceso sustituible por TASKVALUE

Si la tarea arranca con TASKVALUE mayor que cero, ese valor reemplaza la fecha de proceso (líneas 894-895) sin validación. Una fecha futura borraría registros vigentes.

Cálculos de fecha sin verificación

Si CONSISDIA o THECALENDAR fallan, el programa sigue con el valor previo de la fecha (líneas 891-892, 969-970, 1372-1373 y 1390-1391).

Comparación de fechas sin siglo

B07 y B10 se comparan contra WKS-FECHACALC-06, los 6 dígitos de la derecha de una fecha de 8 (líneas 109-112, 1174 y 1529).
