CORE-GL · P360 a Java 21 Spring Boot

P360 recibe por parámetro el número de sistema y el pack donde P330 dejó los archivos extraídos. Resuelve con la librería de control el nombre físico de la base de saldos del sistema, la abre en actualización y, para cada una de las seis estructuras, ordena el archivo plano con SORT, lo recorre creando y guardando un registro DMSII por renglón, y al final lo purga. Cualquier error al guardar termina la base con DMTERMINATE. Al terminar imprime un listado con el número de registros integrados por estructura y un importe de control por estructura.

/** WKS-PARAMETROS: sistema PIC 9(03) y pack X(12), recibidos al invocar el programa. */
public record ParametrosIntegracion(int sistema, String pack) {
    JobParameters toJobParameters(LocalDate fechaProceso) {
        return new JobParametersBuilder()
                .addLong("sistema", (long) sistema)
                .addString("pack", pack)
                .addLocalDate("fechaProceso", fechaProceso)
                .toJobParameters();
    }
}

@Configuration
public class IntegracionSaldosJobConfig {

    /** 00-0100-CTL-PROCESO: verificación, seis estructuras en orden fijo, listado y cierre. */
    @Bean
    Job integracionSaldos(JobRepository repo, Step verificacion,
                          Step b20, Step b21, Step b70, Step b71, Step b72, Step b80,
                          Step listadoControl) {
        return new JobBuilder("CORE-GLP360", repo)
                .start(verificacion)            // 01-0000-VERIFICACION
                .next(b20).next(b21).next(b70)  // 02 .. 04
                .next(b71).next(b72).next(b80)  // 05 .. 07
                .next(listadoControl)           // 09-0000-PROC-IMPRESION
                .build();                       // 10-0000-PROC-TERMINA: cierre de recursos por Spring
    }

    /** Disparo por llegada de archivos: reemplaza la invocación con parámetros desde el WFL. */
    @Bean
    IntegrationFlow llegadaExtraccion(JobLauncher launcher, Job integracionSaldos, P360Properties p) {
        return IntegrationFlow.from(Files.inboundAdapter(new File(p.directorioExtraccion()))
                        .patternFilter("*.done"), e -> e.poller(Pollers.fixedDelay(60_000)))
                .transform(File.class, f -> new JobLaunchRequest(integracionSaldos,
                        ParametrosIntegracion.desdeMarca(f).toJobParameters(p.fechaProceso())))
                .handle(new JobLaunchingGateway(launcher))
                .get();
    }
}

@Component
@RequiredArgsConstructor
public class FechaProcesoResolver {

    private final ControlSistemasClient control;   // CONSISDIA IN LIB-CONTROL (CORE-GLB01SISDIA)

    /**
     * Si la tarea trae VALUE distinto de cero, ese valor es la fecha de proceso;
     * si no, se consulta el control diario del sistema.
     */
    public LocalDate resuelve(int sistema, int csi, OptionalLong valueTarea) {
        if (valueTarea.isPresent() && valueTarea.getAsLong() != 0) {
            return LocalDate.parse(String.valueOf(valueTarea.getAsLong()),
                    DateTimeFormatter.BASIC_ISO_DATE);           // AAAAMMDD
        }
        var r = control.consultaSistemaDia(sistema, csi);
        if (r.result() != 0) {
            // El fuente no revisa W77-RESULT-LIBCON aquí; en Java se falla explícitamente.
            throw new ControlSistemaException("CONSISDIA RESULT " + r.result());
        }
        return r.fechaProceso();
    }
}
@Component
@RequiredArgsConstructor
public class BaseSaldosResolver {

    private final ControlSistemasClient control;     // B04SISTEM / CONSISMEN IN LIB-CONTROL
    private final Map<String, DataSource> bases;     // una por base CORE-GLBD11SDOSnnn

    /** 01-0130-ABRE-SALDOS: devuelve la base de saldos del sistema o falla con el motivo. */
    public DataSource resuelve(int sistema) {
        var b04 = control.sistema(sistema);
        if (b04.result() != 0 || b04.estatus() != 3) {
            // En el fuente esta rama no existe: no hay mensaje ni baja de la tarea.
            throw new BaseSaldosException("SISTEMA " + sistema + " NO DISPONIBLE EN B04");
        }
        var b03 = control.sistemaMensual(sistema);
        if (b03.result() == 1) throw new BaseSaldosException("NO EXITE EL NUMERO DE SISTEMA EN BASE MENSUAL");
        if (b03.result() != 0) throw new BaseSaldosException("ERROR EN BASE DE CONTROL");
        if (!Set.of(1, 2, 5, 6).contains(b03.tipoBd()))
            throw new BaseSaldosException("EL SISTEMA NO UTILIZA BASE DE SALDOS");
        String nombre = b03.nombreBdSaldos();
        if (nombre == null || nombre.isBlank())
            throw new BaseSaldosException("NO EXISTE EN NOMB. DE BD SALDOS");
        return Optional.ofNullable(bases.get(nombre))
                .orElseThrow(() -> new BaseSaldosException("BASE " + nombre + " NO CONFIGURADA"));
    }
}

/** REG-B72POSCONTA: llave de 10 campos y cuatro importes S9(15)V9(02) COMP. */
public record PosicionContableRecord(
        int csi, LocalDate fecha, int banco, int moneda, int fideicomiso,
        int producto, int instrumento,
        String cuenta,              // 9(12): como String para conservar ceros
        String claveContable,       // 9(06)
        int secuencia,              // 9(02)
        int naturalezaCuenta,       // 9(02)
        BigDecimal saldoAnterior,   // S9(15)V9(02) escala 2, con signo
        BigDecimal cargos,
        BigDecimal abonos,
        BigDecimal saldoActual) {

    /** Los campos COMP de Unisys se decodifican con un lector binario propio. */
    static PosicionContableRecord decode(UnisysBinaryReader r) {
        return new PosicionContableRecord(r.comp(4), r.fechaAmd(r.comp(8)), r.comp(4),
                r.comp(4), r.comp(4), r.comp(4), r.comp(4),
                r.compAsString(12), r.compAsString(6), r.comp(2), r.comp(2),
                r.compDecimal(15, 2), r.compDecimal(15, 2),
                r.compDecimal(15, 2), r.compDecimal(15, 2));
    }
}
@Component
@RequiredArgsConstructor
public class EstructuraStepFactory {

    private final JobRepository repo;
    private final PlatformTransactionManager tx;

    /** Un step por estructura: ordenar, leer, guardar y purgar el archivo plano. */
    public <T> Step paso(EstructuraSaldos<T> e) {
        return new StepBuilder("integra" + e.nombre(), repo)
                .<T, T>chunk(1_000, tx)
                .reader(new SortedRecordReader<>(e.archivo(), e.decoder(), e.ordenNatural()))
                .writer(e.writer())                       // CREATE + STORE
                .listener(new PurgaArchivoListener(e.archivo()))   // CLOSE ... WITH PURGE
                .listener(new TotalesControlListener(e))
                .build();
    }
}

// B20: orden AM, CON, PRD, INS, MON; equivale a SORT A20-SORT ... USING / GIVING R20SDOMENCON.
static final Comparator<SaldoMensualContrato> ORDEN_B20 =
        Comparator.comparingInt(SaldoMensualContrato::anioMes)
                  .thenComparing(SaldoMensualContrato::contrato)
                  .thenComparingInt(SaldoMensualContrato::producto)
                  .thenComparingInt(SaldoMensualContrato::instrumento)
                  .thenComparingInt(SaldoMensualContrato::moneda);
@Bean
JdbcBatchItemWriter<SaldoMensualContrato> writerB20(DataSource saldos) {
    return new JdbcBatchItemWriterBuilder<SaldoMensualContrato>()
            .dataSource(saldos)
            .sql("""
                INSERT INTO core-glb20sdomencon
                  (key_am, key_con, key_prd, key_ins, key_mon, suc_pro, sta_mov, apu_imp,
                   cnt_mov_01, sd_01, /* ... */ cnt_mov_10, sd_10)
                VALUES (:anioMes, :contrato, :producto, :instrumento, :moneda, :sucursal,
                        :estatusMov, :apuntadorImp, :cnt01, :sd01, /* ... */ :cnt10, :sd10)
                """)
            .itemSqlParameterSourceProvider(SaldoMensualContrato::parametros)  // 10 pares CNTMOV/SD
            .assertUpdates(true)
            .build();
}

// STORE ... ON EXCEPTION -> CALL SYSTEM DMTERMINATE.
// Spring Batch: la excepción revierte el chunk y deja el step en FAILED; no hay skip ni retry,
// para conservar la semántica de "todo o nada" por estructura que el operador espera.
@Component
@StepScope
public class TotalesControlListener implements ItemWriteListener<PosicionContableRecord> {

    private long integrados;                                   // WS-INTEGRADOS-B72
    private BigDecimal totalSaldoActual = BigDecimal.ZERO.setScale(2);   // WS-TOTALB72SDOACT

    @Override
    public void afterWrite(Chunk<? extends PosicionContableRecord> chunk) {
        for (var r : chunk) {
            integrados++;
            totalSaldoActual = totalSaldoActual.add(r.saldoActual());     // suma con signo
        }
    }

    @AfterStep
    ExitStatus publica(StepExecution se) {
        se.getExecutionContext().putLong("integradosB72", integrados);
        se.getExecutionContext().put("totalB72SdoAct", totalSaldoActual);
        return se.getExitStatus();
    }
}

public final class FechasY2k {

    /** A2K-BASE-YEAR VALUE 50: AA < 50 es 20AA, AA >= 50 es 19AA. */
    public static final int PIVOTE = 50;

    private static final DateTimeFormatter AAMMDD = new DateTimeFormatterBuilder()
            .appendValueReduced(ChronoField.YEAR, 2, 2, 1950)   // 50..99 -> 1950..1999, 00..49 -> 2000..2049
            .appendPattern("MMdd")
            .toFormatter();

    /** A2K-CONV-AMD-TO-CAMD: 0 se conserva como "sin fecha" y 999999 como fecha máxima. */
    public static Optional<LocalDate> deAammdd(int aammdd) {
        if (aammdd == 0)      return Optional.empty();
        if (aammdd == 999999) return Optional.of(LocalDate.MAX);       // 99999999 en el original
        return Optional.of(LocalDate.parse("%06d".formatted(aammdd), AAMMDD));
    }

    private FechasY2k() {}

Riesgos de migración
Totales de control sin signo

Los acumuladores WS-TOTALB20SDOSD2 a WS-TOTALB72SDOACT (líneas 486-490) son PIC 9(18)V9(02) sin signo, mientras los importes de origen son S9(13)V9(02) y S9(15)V9(02) con signo. Cada ADD guarda el valor absoluto, así que el listado de control puede no cuadrar con la base.

Error de B04 sin tratamiento

En 01-0130-ABRE-SALDOS, los IF de B04SISTEM (W77-RESULT-LIBCON = 0) y de estatus 3 no tienen rama ELSE. Si fallan, no hay mensaje ni baja de la tarea y la base de saldos no se abre.

Integración no atómica entre estructuras

No hay transacción ni punto de reinicio: un error en una estructura deja las anteriores integradas, la actual parcial y los archivos planos de las anteriores ya purgados.

Fecha de proceso sin validar

El resultado de CONSISDIA no se revisa (líneas 1560-1564) y un VALUE de tarea distinto de cero se toma como fecha sin validación.

Hardcodes de rutas

Ruta de la librería de mensajes en S000, ruta de respaldo de LIB-CONTROL en CMEMP y la plantilla de títulos S{sistema}/FILE/EXTR/... están compiladas. Se trasladan a application.yml.
