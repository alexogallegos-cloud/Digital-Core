/** Misma interfaz que exponía la librería CORE-DEPL035 a través de este copybook. */
public interface MonitoreoAplicaciones {

    /** S038L035_01_PROG_INI: WS-S038L035-01-ENTRADA -> ID-PROG-S 9(08). */
    ResultadoMapli<Long> inicioPrograma(InicioPrograma entrada);

    ResultadoMapli<Void>    finPrograma(long idPrograma);                         // _02_PROG_FIN
    ResultadoMapli<Integer> inicioActividad(InicioActividad entrada);             // _03_ACTI_INI
    ResultadoMapli<Void>    finActividad(long idPrograma, int numActividad);      // _04_ACTI_FIN
    ResultadoMapli<Void>    avance(long idPrograma, int numActividad, String txt);// _05_ACTU_REG
    ResultadoMapli<Long>    registrosDeDataset(String dataset, String base);      // _06_NUM_REGS

    MonitoreoAplicaciones NO_OP = new NoOpMonitoreo();
}

public record ResultadoMapli<T>(long codigo, T valor) {          // WS-S038L035-RESULT

}
/** Los PERFORM 2000038-INI/FIN-PROG del anfitrión pasan a ser un listener del Job. */
@Component
@RequiredArgsConstructor
public class MapliJobListener implements JobExecutionListener {

    private final MonitoreoAplicaciones mapli;

    @Override
    public void beforeJob(JobExecution job) {
        var r = mapli.inicioPrograma(InicioPrograma.de(job.getJobParameters()));
        if (r.ok()) job.getExecutionContext().putLong("mapli.idPrograma", r.valor());
        else log.warn("MAPLI inicio de programa RESULT={}", r.codigo());   // el anfitrión decide
    }

    @Override
    public void afterJob(JobExecution job) {
        if (job.getExecutionContext().containsKey("mapli.idPrograma")) {
            mapli.finPrograma(job.getExecutionContext().getLong("mapli.idPrograma"));
        }
    }
}
/** Una actividad MAPLI equivale a un Step de Spring Batch. */
@Component
@RequiredArgsConstructor
public class MapliStepListener implements StepExecutionListener {

    private final MonitoreoAplicaciones mapli;

    @Override
    public void beforeStep(StepExecution step) {
        long idProg = step.getJobExecution().getExecutionContext().getLong("mapli.idPrograma", 0L);
        if (idProg == 0) return;
        var r = mapli.inicioActividad(new InicioActividad(
                idProg,
                step.getJobExecution().getExecutionContext().getLong("mapli.regsAProcesar." + step.getStepName(), 0L),
                0, TipoActividad.de(step), 0,                         // NUM-ACTI-E: lo asigna MAPLI
                Texto.ajusta(step.getStepName(), 40), ""));           // NOM-ACTI-E X(40)
        if (r.ok()) step.getExecutionContext().putInt("mapli.numActividad", r.valor());
    }

    @Override
    public ExitStatus afterStep(StepExecution step) {
        if (step.getExecutionContext().containsKey("mapli.numActividad")) {
            mapli.finActividad(step.getJobExecution().getExecutionContext().getLong("mapli.idPrograma"),
                               step.getExecutionContext().getInt("mapli.numActividad"));
        }
        return step.getExitStatus();
    }
}
/** S038L035_05_ACTU_REG se invoca una vez por registro procesado. */
@Component
@RequiredArgsConstructor
public class MapliAvanceListener implements ItemProcessListener<Object, Object>, ChunkListener {

    private final MonitoreoAplicaciones mapli;
    private final MeterRegistry registry;

    @Override
    public void afterProcess(Object in, Object out) {
        registry.counter("mapli.registros", "step", pasoActual()).increment();   // costo local
    }

    /** Al cerrar cada chunk se publica el avance acumulado a MAPLI. */
    @Override
    public void afterChunk(ChunkContext ctx) {
        var step = ctx.getStepContext().getStepExecution();
        mapli.avance(idPrograma(step), numActividad(step),
                     "LEIDOS " + step.getReadCount() + " ESCRITOS " + step.getWriteCount());
    }
}
/** S038L035_06_NUM_REGS: DATASET-E X(17) y NOM-BASE-E X(57) -> NUM-REGS-S S9(11) BINARY. */
@Component
@RequiredArgsConstructor
class ConteoDataset {

    private final JdbcTemplate jdbc;
    private final Map<String, String> tablaPorDataset;   // mapli.datasets: CORE-DEPB28COMICOB -> core-dep_comisiones_cobradas

    /** En DMSII el total sale de la estructura; en relacional se usa la estadística, no un COUNT(*). */
    long registros(String dataset, String base) {
        String tabla = tablaPorDataset.get(dataset.strip());
        if (tabla == null) return 0L;
        Long n = jdbc.queryForObject(
                "select reltuples::bigint from pg_class where relname = ?", Long.class, tabla);
        return n == null ? 0L : n;
    }
}
Riesgos de migración
Resultados nunca evaluados

Los seis párrafos que llaman a CORE-DEPL035 dejan el código en WS-S038L035-RESULT sin revisarlo; la evaluación depende de cada uno de los programas que incluyen el copybook.

Versión de librería no garantizada

Si DAME_TIT falla (líneas 25-35) el programa sigue y liga CORE-DEPL035 con el título que tenga declarado, posiblemente una versión no vigente.

Fin de programa no registrado en abortos

El aviso de fin depende de que el anfitrión llegue a ejecutar 2000038-FIN-PROG-EN-MAPLI; los abortos por DMTERMINATE o STATUS -1 dejan el programa abierto en MAPLI.

Costo por registro

2000038-ACTU-REG-EN-MAPLI implica una llamada entre librerías por cada registro procesado por el anfitrión.
