    private final TarifasS080Client s080;       // 20000080-L700-TARIFAS / L710-CATALOGOS
    private final ModoCompilacion modo;          // sustituye $SET OMIT (BATCHP130, BATCHIPAB)

    public ParametrosFiscales carga(LocalDate fechaBase) {
        var b = ParametrosFiscales.builder()
                .tipoCambioFix(s080.tipoCambioFix(fechaBase))          // 20020050
                .diasAnio(DiasAnio.de(fechaBase.getYear(), fechaBase,
                        equipo.nombre(), calendarioProps))             // 20020030
                .salarioMinimo(s080.salarioMinimo(fechaBase))          // 20020100
                .exencionIsrMn(exencionPorUma(fechaBase))              // 20020110
                .factorIsr(s080.factorIsr(fechaBase))                  // 20020120 tarifa 254
                .porcentajeIsr(s080.porcentajeIsr(fechaBase));         // 20020130 tarifa 255
        if (modo.esP130() || modo.esIpab()) {
            b.factorIsrExtranjeros(s080.factorIsrExtranjeros(fechaBase)); // 20020140 tarifa 259
        }
        // Catálogo 279: tarifa de referencia DLS (154 LIBOR, 260 FFE, 261 SOFR).
        Optional<TarifaDls> dls = s080.tarifaReferenciaDls(fechaBase);  // A601000-DAME-QTASA-DLS
        b.cetes(s080.cetes(fechaBase)).cetesPromedio(s080.cetesPromedio(fechaBase));
        dls.ifPresentOrElse(
            t -> b.tasaDls(s080.tasa(t, fechaBase)).tasaDlsPromedio(s080.tasaPromedio(t, fechaBase)),
            () -> b.tasaDls(BigDecimal.ZERO).tasaDlsPromedio(BigDecimal.ZERO)); // error en cat. 279
        return b.valorUdi(s080.valorUdi(fechaBase)).build();            // 20020070
    }
}
/** 20020030-CALCULA-BISIESTO. WS-DIAS-ANUAL y WS-DIAS-ANUAL-EDIT (días x 100). */
public record DiasAnio(int dias, int diasPorCien) {

    public static DiasAnio de(int anio, LocalDate fechaBase, String equipo, CalendarioProps props) {
        boolean bisiesto = Year.isLeap(anio);           // misma regla 4 / 100 / 400 del fuente
        boolean excepcion2016 = props.equiposExcepcion2016().contains(equipo.trim())
                && fechaBase.isAfter(LocalDate.of(2016, 11, 27))
                && anio == 2016;
        return (!bisiesto || excepcion2016)
                ? new DiasAnio(365, 36500)
                : new DiasAnio(366, 36600);
    }

/** 20020110-DAME-UMA-ANUAL: la parte exenta de ISR en MN son 5 UMA anuales. */
private BigDecimal exencionPorUma(LocalDate fechaBase) {
    TarifaRespuesta r = s080.consulta(TarifaConsulta.builder()
            .clave(props.tarifaUmaAnual())        // 252
            .producto(9991).instrumento(2).moneda(0)
            .fechaVigencia(fechaBase).registros(1)
            .build());

    BigDecimal umaAnual = BigDecimal.ZERO;           // WS-CAP-EXEN-ISR-MN-UMA conserva su valor previo si falla
    if (r.resultado() == 0 || r.resultado() == 1) {
        // CAMPO-03 llega como entero con DEC-CAMPO-03 decimales implícitos
        umaAnual = new BigDecimal(r.campo(3).entero()).movePointLeft(r.campo(3).decimales());
    }
    // PIC 9(08)V99 COMP: multiplicar por 5 conserva la escala; sin ROUNDED
    return umaAnual.multiply(BigDecimal.valueOf(props.vecesUmaExencion()))   // 5
                   .setScale(2, RoundingMode.DOWN);
}

/** 20500000-CAPITALIZACION, preparación del contrato. */
private ContextoContrato prepara(Contrato c, LocalDate fechaBase) {
    int mes = fechaBase.getMonthValue();
    int diasMes = (mes >= 1 && mes <= 12) ? TABLA_DIAS_MES[mes - 1] : 30;   // WKS-TAB-DIAS-MES

    boolean estatusBeneficencia = c.statusBeneficencia() == 3 || c.statusBeneficencia() == 8;
    Art61 art61 = c.esCuentaGlobal()
            ? (estatusBeneficencia ? Art61.CUENTA_GLOBAL_ART61 : Art61.CUENTA_GLOBAL)   // WS-ES-ART61-CG 1 / 2
            : (estatusBeneficencia ? Art61.ART61 : Art61.NO);                          // WS-ES-ART61 1 / 0
    return new ContextoContrato(c, diasMes, art61);
}
public void capitaliza(ContextoContrato ctx) {
    Optional<Instrumento> inst = instrumentos.de(ctx.contrato());        // 20520100-DAME-INSTRUMENTO
    if (inst.isEmpty()) return;                                         // WS-CANC-ERROR = 2: no existe B05
    Contrato c = ctx.contrato();

    switch (modo.tipoProceso()) {
        case DIARIO -> {
            if (!Set.of(0, 1, 5).contains(c.status())) return;
            boolean activo = c.status() == 0 || c.status() == 5;
            boolean bloqueo14 = c.tipoBloqueo() == 14 && c.importeSolBloqueo().signum() > 0;
            if (inst.get().rendimientoDiario()) {
                rendimientos.paga(ctx);                                  // 20520000-PAGA-RENDIMIENTOS
                if (activo ? !ctx.art61().esArt61() : bloqueo14) comisiones.diarias(ctx);
            } else {
                boolean exento = ctx.art61().esArt61() || c.esBeneficencia() || c.esGlobalMps();
                if (!exento && (activo || bloqueo14)) comisiones.diarias(ctx);  // 20500010
            }
        }
        case LINEA_PENDIENTE -> {
            boolean exento = ctx.art61().esArt61() || c.esBeneficencia() || c.esGlobalMps()
                    || c.esSubcuentaTesofe() || c.usoCuentaSubcuenta();
            if (!exento) comisiones.pendientesLinea(ctx);              // 20500015-COMIS-PTE-LINEA
        }
        default -> corte.procesa(ctx, inst.get());                      // 20500005-CON-CORTE
    }
}
/** 20520500-CALCULA-RENDIMIENTOS, primera parte: base gravable. */
BigDecimal promedioGravado(ContextoContrato ctx, Instrumento inst, BigDecimal promCiclo,
                           BigDecimal tasaBruta, ParametrosFiscales p) {
    Contrato c = ctx.contrato();
    if (c.esCuentaGlobal() || ctx.art61() == Art61.ART61) {
        return promCiclo;                                   // sin exención: todo grava
    }
    TipoPersona tp = c.tipoPersona();
    if (!inst.retieneImpuesto() || ctx.sinImpuesto()
            || tp.in(MORAL_NO_LUCRATIVA_1, FISICA_EXTRANJERA, MORAL_EXTRANJERA, FINANCIERA_EXTRANJERA)) {
        return BigDecimal.ZERO;                             // no hay base gravable sobre saldo
    }
    if (tp.in(MORAL_GENERAL, FISICA_EMPRESARIAL, FISICA_EMPRESARIAL_EXTRANJERA_EM)) {
        return promCiclo;                                   // postmortem: sin exención para estos tipos
    }
    if (modo.libS084() && tp.esFisicaNacional()
            && promCiclo.signum() > 0 && tasaBruta.signum() > 0) {
        return exencionS084.valida(ctx);                    // 10000-VALIDA-IMP-EXENT
    }
    BigDecimal exento = (c.moneda() == 1) ? p.exencionIsrMn() : p.exencionIsrDls();
    return promCiclo.compareTo(exento) > 0 ? promCiclo.subtract(exento) : BigDecimal.ZERO;
}

int diasPorCien(ContextoContrato ctx) {                     // WS-CAP-DIASPORCIEN
    return (ctx.contrato().esCuentaGlobal() || ctx.art61() == Art61.ART61)
            ? ctx.diasMes() * 100 : 36000;
}
private static final MathContext MC = MathContext.DECIMAL128;

/** WS-CAP-RENDBRTO PIC 9(12)V99 con ROUNDED. */
BigDecimal rendimientoBruto(BigDecimal promCiclo, BigDecimal tasaBruta, int diasPorCien, int diasCiclo) {
    return promCiclo.multiply(tasaBruta, MC)
            .divide(BigDecimal.valueOf(diasPorCien), MC)
            .multiply(BigDecimal.valueOf(diasCiclo), MC)
            .setScale(2, RoundingMode.HALF_UP);
}

/** WS-CAP-IMPUESTO PIC 9(12)V99 con ROUNDED. */
BigDecimal isr(ContextoContrato ctx, Instrumento inst, BigDecimal promGravado,
               BigDecimal rendBruto, int diasCiclo, ParametrosFiscales p) {
    if (diasCiclo <= 0) return BigDecimal.ZERO;
    TipoPersona tp = ctx.contrato().tipoPersona();
    if (ctx.art61() != Art61.ART61) {
        if (!inst.retieneImpuesto() || ctx.sinImpuesto() || tp == MORAL_NO_LUCRATIVA_1) {
            return BigDecimal.ZERO;
        }
        if (tp.in(FISICA_EXTRANJERA, MORAL_EXTRANJERA, FINANCIERA_EXTRANJERA)) {
            // extranjeros: tasa de retención (tarifa 259) sobre el rendimiento
            return rendBruto.multiply(p.factorIsrExtranjeros()).setScale(2, RoundingMode.HALF_UP);
        }
    }
    // nacionales y Artículo 61: factor anual (tarifa 254) sobre el saldo gravado, prorrateado
    return promGravado.multiply(p.factorIsr(), MC)
            .divide(BigDecimal.valueOf(p.diasAnio().diasPorCien()), MC)
            .multiply(BigDecimal.valueOf(diasCiclo), MC)
            .setScale(2, RoundingMode.HALF_UP);
}

/** 20530510-CALCULA-MANEJO-CTA. Concepto 5 del instrumento; posición 03 del esquema S016. */
public Optional<Comision> calcula(ContextoContrato ctx, EsquemaS016 esq, Instrumento inst) {
    Cobro cobro = esq.cobro(3);
    BigDecimal tasa;
    BigDecimal importe;
    switch (cobro.clave()) {
        case 2 -> {                                        // tasa de S016, importe de la tabla
            tasa = cobro.valor();
            importe = tablaManejo(ctx);                    // 20530520-IMPCOM-CAP
            if (tasa.signum() == 0 && !ctx.contrato().esPersonalLegacyCore()) noCobro(ctx);
        }
        case 1 -> {                                        // importe fijo de S016
            tasa = CIEN;
            importe = cobro.valor();
            if (importe.signum() == 0 && !ctx.contrato().esPersonalLegacyCore()) noCobro(ctx);
        }
        default -> {                                       // tabla, topada a la diferencia contra saldo
            tasa = CIEN;
            importe = tablaManejo(ctx).min(ctx.difManejoVsSaldo());   // W77-DIF-MANCTA-SDO
        }
    }
    if (importe.signum() == 0 || tasa.signum() == 0) {
        if (!modo.esLinea() && ctx.comisionNoCobrada()) return calculadora.calcula(ctx, CERO_GRAVADOS);
        return Optional.empty();
    }
    return exencionPorCompras(ctx, inst);                  // 20530511
}

private BigDecimal tablaManejo(ContextoContrato ctx) {
    return ctx.contrato().esBancaInstitucional()
            ? tarifas.manejoInstitucional(ctx.indices())    // WS-COM-MANEJO-CI
            : tarifas.manejo(ctx.indices());                // WS-COM-MANEJO
}
/** 20530511-EXEMAN-X-COMPRAS. */
private Optional<Comision> exencionPorCompras(ContextoContrato ctx, Instrumento inst) {
    int comprasRequeridas = tarifas.comprasParaExentarManejo(ctx.indices());   // WS-249-COMP-MAN
    int claveExencion = inst.claveComision(45);
    if (comprasRequeridas > 0 && claveExencion > 0
            && comprasRequeridas <= ctx.contadorComisiones(24)) {             // WS06-CONT-COMIS(24)
        // se exenta: importes en cero y clave de exención 45 del instrumento
        return calculadora.calcula(ctx, Comision.exenta(claveExencion));
    }
    return calculadora.calcula(ctx, Comision.cobrable());                     // 20550100-CALCULA-COMISION
}
Riesgos de migración
Comportamiento distinto según directivas de compilación

Bloques completos dependen de $SET OMIT con CAPITALIZA, BATCHP130, BATCHIPAB y LIBS084L. En las líneas 6405-6411 la directiva cambia la forma del IF del cálculo de la base gravable. Cada programa host compila una variante distinta del mismo motor.

Cálculo de exención fuera de la condición de éxito

En 20020110-DAME-UMA-ANUAL el COMPUTE de la exención (línea 453) se ejecuta aunque S080 haya respondido con error, usando el último valor de la UMA en memoria.

Saldo de UMAS tomado de base de datos

En P130, para cheques de cuenta de producto y personas físicas, la línea 6428 sustituye el saldo promedio gravado por B03-UMAS-EXE, un campo de la base. El nombre sugiere un importe exento y se asigna a la variable del gravado.

Divisor anual dependiente del equipo

20020030-CALCULA-BISIESTO fuerza 365 días en 2016 para cinco nombres de equipo compilados en el código.
