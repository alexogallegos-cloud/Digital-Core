CORE-GL · L002R4 a Java 21 Spring Boot

L002R4 exporta tres puntos de entrada: CARGAMOV1 para movimientos, CARGASALDOS para saldos y CONTROLES para órdenes de control. CARGAMOV1 recibe un buffer de hasta 540 bytes, valida que el sistema del movimiento sea el de la instancia de la librería, elige el canal de día (0 a 9) según la fecha contable y ejecuta la función pedida: grabar, eliminar, iniciar, terminar, reblocar, precierre o cierre. Al grabar escribe en LOGS y DESS del canal, amplía los archivos cuando quedan menos de cinco registros libres y cada mil movimientos sincroniza el nivel con L001 vía CONSISDIA. Los rechazos se escriben en el archivo de errores y se reportan al Command Post de forma escalonada.

/** Contrato que exponía la librería: EXPORT CARGAMOV1, CARGASALDOS, CONTROLES. */
public interface Registras {

    /** CARGAMOV1(DATOS): devuelve RESULT y reescribe el buffer con la respuesta (ALOG2). */
    Resultado cargaMovimiento(byte[] datos);

    /** CARGASALDOS(DATOS) */
    Resultado cargaSaldos(byte[] datos);

    /** CONTROLES(DATOS): órdenes de operación (HI) recibidas por PROC_CONTROL. */
    int controles(byte[] datos);

    record Resultado(int codigo, byte[] respuesta) {}
}

@Service
@RequiredArgsConstructor
public class RegistrasLifecycle implements SmartLifecycle {

    private final CambioFechaLock cambiaFecha;   // CAMBIA_FECHA
    private final ArranqueRegistras arranque;    // LEVANTA: alta en B05PROCESOS, CONSISDIA, apertura de LOG/DES
    private final CierreRegistras cierre;        // BAJA
    private volatile boolean activo;

    @Override public void start() {
        cambiaFecha.liberaSiQuedoTomado();       // IF LOCKSTATUS(CAMBIA_FECHA) > 0 THEN UNLOCK
        arranque.levanta();
        activo = true;                           // FREEZE(CONTROL, PROC_CONTROL): la librería queda disponible
    }
    @Override public void stop()  { activo = false; cierre.baja(); }
    @Override public boolean isRunning() { return activo; }
}
@Component
@RequiredArgsConstructor
class ControlRegistras {

    private static final int HI_BAJA = 4;
    private final Registras registras;
    private final AtomicInteger usuariosActivos;     // MYSELF.LIBRARYUSERS

    /** Sustituye a WAITANDRESET((2), MYSELF.EXCEPTIONEVENT) + MYSELF.TASKVALUE. */
    @ManagedOperation(description = "Equivalente a <mix> HI n sobre la librería")
    public void hi(int numHi) {
        byte[] datControl = String.format("%04d%08d", numHi, 0).getBytes(StandardCharsets.US_ASCII);
        registras.controles(datControl);             // CONTROLES(DAT_CONTROL)
        if (numHi == HI_BAJA) {
            esperaUsuarios();
        }
    }

    /** WHILE MYSELF.LIBRARYUSERS > 0 DO WAITANDRESET((2), ...) */
    private void esperaUsuarios() {
        while (usuariosActivos.get() > 0) {
            LockSupport.parkNanos(Duration.ofSeconds(2).toNanos());
        }
    }
}

@Service
@RequiredArgsConstructor
public class CargaMovimientoService implements Registras {

    static final int TAMANO_BUFFER = 540;
    private final RegistrasProperties props;        // sist-lib, csi-lib, sistemas-sin-ajuste
    private final CambioFechaLock cambiaFecha;

    @Override
    public Resultado cargaMovimiento(byte[] datos) {
        // REPLACE POINTER(ALOG1) BY DATOS FOR SIZEDATC1, con SIZEDATC1 tope 540
        byte[] alog1 = Arrays.copyOf(datos, TAMANO_BUFFER);
        byte[] alog2 = Arrays.copyOf(datos, TAMANO_BUFFER);
        int largo = Math.min(datos.length, TAMANO_BUFFER);

        if (!props.sistemasSinAjuste().contains(props.sistLib())) {   // 500, 403, 404
            Fixed.putNibbles(alog1, 342, 3, 2);      // POINTER(ALOG1,4)+342: posición en dígitos de 4 bits
            Fixed.putNibbles(alog1, 345, 3, 2);
        }
        cambiaFecha.esperaLibre(Duration.ofMillis(300));   // WHILE LOCKSTATUS(CAMBIA_FECHA) > 0 DO WAIT((.3))

        var mov = MovimientoIdf.of(alog1);           // IDFSISTFAN, IDFFECCONT, IDFFUNCION, ...
        if (mov.sistFan() != props.sistLib()) {
            return rechazo(1, alog2, largo,
                "SISTEMA NO HABILITADO EN CORE-GL, SISTEMA : %04d".formatted(mov.sistema()));
        }
        mapeo(mov);                                   // MAPEO
        return procesa(mov, alog1, alog2, largo);
    }
}
@Component
@RequiredArgsConstructor
class CanalDiaResolver {

    private final RegistrasProperties props;   // equipos-beta: [ACYPBETA, VDMBETA]
    private final EstadoDias dias;             // FEC_DIA[0:9], FECHAPROC, DIA_SEMANA

    record Canal(int numDia, int result) {}

    Canal resuelve(long fecCont) {
        boolean beta = props.equiposBeta().contains(props.nombreEquipo());
        if (!beta && fecCont == dias.fechaProc()) {
            return new Canal(dias.diaSemana(), 0);
        }
        if (beta || fecCont > dias.fechaProc()) {
            for (int i = 0; i <= 9; i++) {           // WHILE I_I := * + 1 LEQ 9 AND NOT BOOL_I
                if (dias.fecDia(i) == fecCont) return new Canal(i, 0);
            }
        }
        // Fecha anterior a la de proceso o no encontrada en los diez canales:
        // RESULT = 4 pero el movimiento se dirige al canal del día de la semana.
        return new Canal(dias.diaSemana(), 4);
    }
}
    /** CASE FUNCIONVAL OF ... ELSE: FUNCION NO VALIDA. */
    private int ejecuta(Funcion f, MovimientoIdf mov, Canal canal) {
        switch (f) {
            case GRABA        -> cargaMemoria(mov, canal);          // 1
            case ELIMINA      -> elimina(mov, canal);               // 2
            case INICIA       -> inicia(mov, canal);                // 11
            case TERMINA      -> termina(mov, canal);               // 12
            case ELIM_X_PROC  -> eliminaPorProceso(mov, canal);     // 21
            case ELIM_X_AUT   -> eliminaPorAutorizacion(mov, canal);// 22
            case BLOQUE_GRANDE, BLOQUE_NORMAL -> reblockade(f, canal); // 31, 32
            case PRECIERRE    -> preFinal(canal);                   // 97
            case CIERRE       -> cierre(canal);                     // 98
        }
        return 0;
    }

    enum Funcion {
        GRABA(1), ELIMINA(2), INICIA(11), TERMINA(12), ELIM_X_PROC(21), ELIM_X_AUT(22),
        BLOQUE_GRANDE(31), BLOQUE_NORMAL(32), PRECIERRE(97), CIERRE(98);

        final int codigo;
        Funcion(int c) { codigo = c; }

        static Optional<Funcion> of(int c) {
            return Arrays.stream(values()).filter(f -> f.codigo == c).findFirst();
        }
    }

        }

    }

    /** REBLOCKADE: 31 = bloque grande sin sincronizar (carga masiva); 32 = bloque chico con SYNCHRONIZE OUT. */
    void reblockade(Funcion f, Canal canal) {
        cargaMemoria(movPendiente, canal, grabaMov);          // CARGAMEMORY antes de cerrar
        modoEscritura.set(canal.numDia(), switch (f) {

            default -> throw new IllegalStateException(f.name());
        });
    }

Riesgos de migración
Fecha contable no válida que de todas formas se graba

Cuando la fecha contable es anterior a la de proceso o no está en los diez canales, RESULT queda en 4 y NUMDIAC1 en DIA_SEMANA (líneas 5680-5690), pero el CASE de funciones se ejecuta con RESULT = 0 OR RESULT = 4 (línea 5700). El movimiento se registra en el canal del día y a la vez se reporta como error.

Espera indefinida sobre CAMBIA_FECHA

La línea 5628 espera en ciclos de 0.3 segundos mientras el candado esté tomado, sin límite. Un candado que no se libera detiene el registro de movimientos de todos los sistemas alimentadores.

Cierre del día sin propagación de error

FINAL reintenta CONSISDIA tres veces sin pausa y, si falla, solo escribe en bitácora (líneas 5579-5599). El llamador recibe RESULT = 0.

Avisos al Command Post comentados en el punto de error

Las llamadas a L002COMMPOST de sistema no habilitado y función no válida están comentadas (líneas 5635 y 5721); el aviso sale después por el bloque común de las líneas 5812-5832, que solo avisa la primera vez y cada cien errores del mismo tipo.
