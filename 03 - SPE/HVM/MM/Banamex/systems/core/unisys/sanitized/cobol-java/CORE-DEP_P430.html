CORE-DEP · P430 a Java 21 Spring Boot

P430 obtiene la fecha de proceso de CORE-DEPB02CONTROL, o del valor de la tarea si se le pasa, y termina de forma anormal si no es fin de mes. Escribe un header propio en la cinta y lee primero el archivo de Monterrey y después el de Valle de México, ambos de 60 posiciones con registros binarios. Valida que la fecha de cada header sea la de proceso, convierte cada detalle de saldo (tipo 2) o de movimiento (tipo 3) a formato DISPLAY, traduce cinco claves de transacción antiguas a las vigentes y comprueba que el trailer de cada archivo cuadre en registros e importe. Al final escribe un trailer consolidado con totales y conteos de clientes con y sin número de cliente.

/** REG-GERENCIAL de entrada (60 bytes, campos COMP). El tipo está en GER-TIPO-REG. */
public sealed interface RegistroGerencial
        permits RegistroGerencial.Header, RegistroGerencial.Saldo,
                RegistroGerencial.Movimiento, RegistroGerencial.Trailer {

    /** Tipo 1: GER-CONTROL. */
    record Header(String numSistema, String nomSistema,          // X(04), X(19)
                  LocalDate fechaProceso, LocalDate fechaOperacion) implements RegistroGerencial {}

    /** Tipo 2: GER-INFORMACION (el comentario del fuente lo llama movimiento, pero el layout es de saldo). */
    record Saldo(String cuenta, String cliente,                  // 9(12) COMP como String
                 LocalDate fechaUltimoMov, int moneda, int persona,
                 BigDecimal saldoActual, BigDecimal saldoPromedio,   // 9(12)V99 -> escala 2
                 BigDecimal tasa,                                    // 9(03)V9(03) -> escala 3
                 int sucursal) implements RegistroGerencial {
        boolean tieneCliente() { return !cliente.chars().allMatch(ch -> ch == '0'); }
    }

    /** Tipo 3: GER-MOVIMIENTOS. */
    record Movimiento(String contrato, LocalDate fecha, LocalDate fechaValor,
                      int sucTransaccion, int sucPromotora, int claveTransaccion,
                      BigDecimal importe, int moneda,
                      BigDecimal interesCancelacion,                  // 9(12)V99
                      int tipoDeposito) implements RegistroGerencial {}

    /** Tipo 4: GER-TRAILLER. */
    record Trailer(long numRegistros, BigDecimal importe) implements RegistroGerencial {}   // 9(10), 9(16)V99
}

@Component
@RequiredArgsConstructor
class CierreMesDecider implements JobExecutionDecider {

    private final ControlCaptacionRepository b02;   // CORE-DEPB02CONTROL
    private final CalendarioBancario locsup;

    /** 20002000-FECHA-PRO + 20002100-PROY-FEC + 20002110-DIAPRO. */
    @Override
    public FlowExecutionStatus decide(JobExecution je, StepExecution se) {
        LocalDate fechaLinea = b02.findFechaLinea()
                .orElseThrow(() -> new ProcesoAbortadoException("ERROR REG. DE CONTROL VACIO"));
        LocalDate forzada = je.getJobParameters().getLocalDate("fecha");   // WS-VAL-FEC

        LocalDate fechaProceso;
        LocalDate siguiente;
        if (forzada != null) {
            fechaProceso = forzada;
            siguiente = locsup.sumaDias(forzada, 1);          // función 13, formato 13
        } else {
            fechaProceso = locsup.retrocede(fechaLinea, 1);   // función 15, formato 12
            siguiente = fechaLinea;
        }
        je.getExecutionContext().put("fechaProceso", fechaProceso);
        boolean cierre = fechaProceso.getMonthValue() != siguiente.getMonthValue();
        return new FlowExecutionStatus(cierre ? "CIERRE" : "NO_ES_CIERRE");
    }
}
@Bean
Step fusiona(JobRepository repo, PlatformTransactionManager tx, P430Properties props, ResourceLoader rl,
             @Value("#{jobExecutionContext['fechaProceso']}") LocalDate fecha,
             FusionGerencialProcessor procesador, FlatFileItemWriter<SalidaGerencial> cinta,
             EncabezadoYTrailerListener extremos) {

    Resource[] entradas = props.entradas().stream()
            .map(e -> rl.getResource(props.resuelve(e.ruta(), fecha)))
            .toArray(Resource[]::new);

    return new StepBuilder("fusionGerencial", repo)
            .<RegistroGerencial, SalidaGerencial>chunk(100, tx)        // BLOCK CONTAINS 100 RECORDS
            .reader(new MultiResourceItemReaderBuilder<RegistroGerencial>()
                    .name("gerencialMtyVdm").resources(entradas)
                    .comparator((a, b) -> 0)                            // conserva el orden MTY, VDM
                    .delegate(lectorBinarioGerencial())                 // registros de 60 bytes con COMP
                    .build())
            .processor(procesador)
            .writer(cinta)
            .listener(extremos)       // 50001000-REPORTE escribe el header; 50005000-FINAL el trailer
            .build();
}

/** 50004000-CHECO-ARC dentro del procesador. */
public SalidaGerencial process(RegistroGerencial r) {
    return switch (r) {
        case RegistroGerencial.Header h     -> { validaHeader(h); yield null; }   // no se copia
        case RegistroGerencial.Saldo s      -> detalle(s);
        case RegistroGerencial.Movimiento m -> detalle(m);
        case RegistroGerencial.Trailer t    -> { validaTrailer(t); yield null; } // no se copia
    };
}

@Component
@StepScope
public class FusionGerencialProcessor implements ItemProcessor<RegistroGerencial, SalidaGerencial> {

    @Value("#{jobExecutionContext['fechaProceso']}") private LocalDate fechaProceso;
    private int headersVistos;                     // WS-4VEZ
    private TotalesPlaza plaza;                    // WS-NUM-GERE, WS-GER-SDO-PROM

    /** 50004100-HEADER: el primer header se asume de MTY y el segundo de VDM. */
    private void validaHeader(RegistroGerencial.Header h) {
        String etiqueta = (headersVistos++ == 0) ? "MTY" : "VDM";
        if (!h.fechaOperacion().equals(fechaProceso)) {
            throw new ProcesoAbortadoException("FEC EXT DIF FEC INT ARC.    " + etiqueta);   // DMTERMINATE
        }
        plaza = TotalesPlaza.nueva(etiqueta);      // MOVE 0 TO WS-GER-SDO-PROM WS-NUM-GERE
    }
    /** 50004300-DETALLE. Un solo acumulador suma saldos (tipo 2) e importes de movimiento (tipo 3). */
    private SalidaGerencial detalle(RegistroGerencial r) {
        plaza.cuentaRegistro();                                    // WS-NUM-GERE y WS-NUM-GERE-TOT
        return switch (r) {
            case RegistroGerencial.Saldo s -> {
                plaza.sumaImporte(s.saldoActual());                // WS-GER-SDO-PROM: suma el saldo actual
                plaza.sumaCliente(s.tieneCliente(), s.saldoActual());   // CTES-SI / CTES-NO
                yield SalidaGerencial.saldo(s);                    // D-GER-INFORMACION, DISPLAY
            }
            case RegistroGerencial.Movimiento m -> {
                plaza.sumaImporte(m.importe());
                yield SalidaGerencial.movimiento(m, ConversionClaves.convierte(m.claveTransaccion()));
            }
            default -> throw new IllegalStateException();
        };
    }

RN-CORE-DEP-762P430 concilia la vista gerencial fusionando los archivos X/Gerencial de los dos CSIs (`GERENCIAL-MTY` de Monterrey y `GERENCIAL-VDM` de México/VDM) en una sola cinta consolidada (`GERENCIAL-TAPE`, registro de 90 caracteres). Consolidar ambos CSIs en un solo archivo permite un reporte gerencial nacional único enviado a ETL.REG-GERENCIAL: control, saldo, movimiento y trailer sobre el mismo buffer · Títulos de los dos archivos gerenciales y de la cinta fusionada · Lectura de MTY y después de VDM, con despacho por tipo de registro · Header de cada plaza: fecha de operación igual a la de proceso · Detalle: acumulación por tipo y conversión a DISPLAY · Cuadre del trailer por plaza y trailer consolidado
Riesgos de migración
Aborto como control de flujo

Cuando no es cierre de mes el paso termina con DMTERMINATE (líneas 624-626). Un aborto esperado se confunde con una falla en la bitácora del WFL.

Comentario contradictorio sobre los tipos de registro
