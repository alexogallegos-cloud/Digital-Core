Exporta seis procedimientos. Los programas de aplicación guardan un mensaje por folio con GUARDAMSG, lo confirman con RESUELVEMSG o lo borran con BORRAMSG, y fijan tamaño y vigencia con SETMSGSZ. El programa asíncrono llama a ENVIA_CANCELACIONES, que revisa periódicamente el buzón de 2016 posiciones, reenvía por COMS los mensajes sin respuesta después del tiempo de espera y descarta a una bitácora los que rebasan la vigencia. Un procedimiento en segundo plano ajusta cada 40 segundos el tiempo de espera según el tiempo de respuesta observado, y el contenido del buzón se vuelca a disco al terminar la librería y se recarga al iniciar.

/**
 * Estado compartido de la librería (SHARING=SHAREDBYALL): una sola instancia por nodo.
 * Los arreglos paralelos MEM, SZ, VENCE, OLVIDA y CONTEXTO se agrupan en un record por posición.
 */
@Component
public class BuzonCancelaciones {

    static final int CAPACIDAD = 2016;              // N = 2015: posiciones 0..2015
    static final int PALABRAS_RESUMEN = 42;         // M = 41: 42 palabras de 48 bits = 2016 bits

    /** Una posición del buzón. */
    record Entrada(int folio,
                   boolean enviado,                 // MEM[J].[47:1]: 1 = ya se reenvió (cancelado)
                   byte[] mensaje,                  // CONTEXTO[J,*] con longitud SZ[J]
                   Instant ultimoEnvio,             // VENCE[J]
                   Instant insercion) {}            // OLVIDA[J]: se conserva para el vencimiento

    // MEM + MASKSEARCH + RESUMEN + LK_MEM se reemplazan por un mapa concurrente por folio:
    // la búsqueda es O(1) y la atomicidad por folio la da el propio mapa.
    private final ConcurrentHashMap<Integer, Entrada> porFolio = new ConcurrentHashMap<>(CAPACIDAD);
    private final Semaphore lugares = new Semaphore(CAPACIDAD);   // límite duro de 2016 mensajes

    private final AtomicBoolean asincronoActivo = new AtomicBoolean(false);   // ACTIVO
    private final BlockingQueue<Evento> eventos = new LinkedBlockingQueue<>();  // CIERRA_LOG, IN_EV, TSK_EV
    private final Estadisticas sts = new Estadisticas();                       // STS[0:3,0:2]
}

public enum ResultadoGuarda { GUARDADO /*0*/, SIN_LUGAR /*1*/, DUPLICADO /*2*/,
                              EXCEDE_MAXIMO /*3*/, EXCEDE_BUFFER /*4*/, FOLIO_INVALIDO /*5*/ }

/** GUARDAMSG(FOLIO, MSG_SZ, EDO, MSG). Misma firma y mismos códigos que la librería. */
public ResultadoGuarda guarda(int folio, int tamano, boolean estado, byte[] mensaje) {
    if (tamano > config.maxTamano()) return ResultadoGuarda.EXCEDE_MAXIMO;
    if (tamano > mensaje.length)     return ResultadoGuarda.EXCEDE_BUFFER;
    if (folio < 1)                   return ResultadoGuarda.FOLIO_INVALIDO;

    if (!lugares.tryAcquire()) return ResultadoGuarda.SIN_LUGAR;    // no hay posición libre
    Instant ahora = reloj.instant();
    var nueva = new Entrada(folio, estado, Arrays.copyOf(mensaje, tamano), ahora, ahora);

    // putIfAbsent hace atómicos la búsqueda de duplicado y el alta (en el ALGOL no lo son).
    if (porFolio.putIfAbsent(folio, nueva) != null) {
        lugares.release();
        return ResultadoGuarda.DUPLICADO;
    }
    sts.insercion();                                                  // STS[0,0] := *+1
    return ResultadoGuarda.GUARDADO;
}
public enum ResultadoResuelve { RESUELTO /*0*/, OCUPADO_O_ESTADO_DISTINTO /*1*/, NO_ENCONTRADO /*2*/ }

/** RESUELVEMSG(FOLIO, ESTADO): libera la posición solo si el bit de estado coincide. */
public ResultadoResuelve resuelve(int folio, boolean estadoEsperado) {
    if (folio <= 0) return ResultadoResuelve.NO_ENCONTRADO;           // KEY := 48"FFFFFFFFFFFF"
    var resultado = new AtomicReference<>(ResultadoResuelve.NO_ENCONTRADO);

    porFolio.computeIfPresent(folio, (f, e) -> {                     // lock de la posición
        if (e.enviado() != estadoEsperado) {
            resultado.set(ResultadoResuelve.OCUPADO_O_ESTADO_DISTINTO);
            return e;                                                 // se conserva
        }
        Duration respuesta = Duration.between(e.insercion(), reloj.instant());  // cruza medianoche sin ajuste manual
        sts.respuesta(respuesta);                                     // STS[1,0] y STS[2,0]
        resultado.set(ResultadoResuelve.RESUELTO);
        return null;                                                  // borra la entrada
    });
    if (resultado.get() == ResultadoResuelve.RESUELTO) lugares.release();
    return resultado.get();
}

public synchronized void configura(int tamanoAplicacion, int horas) {
    if (tamanoAplicacion > Math.min(props.tamanoMaximoTope(), config.maxTamano())) {
        config.maxTamano(Math.min(props.tamanoMaximoTope(), tamanoAplicacion));
    }
    int h = (horas < 1 || horas > 24) ? props.vigenciaDefaultHoras() : horas;
    Duration vigencia = Duration.ofHours(h);
    if (props.csiVigenciaMedia().contains(host.csi())) vigencia = vigencia.dividedBy(2);

}

/** ENVIA: se llama para cada posición ocupada en cada revisión. */
private Optional<Instant> procesa(Entrada e, Instant limiteReenvio, Instant limiteVigencia) {
    if (!e.ultimoEnvio().isBefore(limiteReenvio)) {
        return Optional.of(e.ultimoEnvio());           // aún no toca: candidato a próxima revisión (MENOR)
    }
    var reenviada = new Entrada(e.folio(), true, e.mensaje(), limiteReenvio, e.insercion());
    salida.envia(e.mensaje());                         // SEND(COMS_OUT, ..., CONTEXTO[J,*])
    envios.incrementAndGet();                          // ENVIOS

    if (e.insercion().isBefore(limiteVigencia)) {      // venció: OLVIDA < T_60
        sts.deshecho();                                // STS[3,0]
        bitacoraRechazos.escribe(reloj.instant(), e.folio(),
                Arrays.copyOf(e.mensaje(), Math.min(168, e.mensaje().length)));
        if (porFolio.remove(e.folio(), e)) lugares.release();
        return Optional.empty();
    }
    porFolio.replace(e.folio(), e, reenviada);
    return Optional.empty();
}
/** REVISA: solo se recorre si hay usuarios además del propio asíncrono. */
@Scheduled(fixedDelayString = "${l091.revision-ms:5000}")
void revisa() {
    if (usuarios.get() <= 1) return;                  // CUANTOS_ME_USAN > 1
    vueltas.incrementAndGet();                        // VUELTAS
    Instant ahora = reloj.instant();
    Instant limiteReenvio  = ahora.minus(config.tiempoEspera());   // LIMITE
    Instant limiteVigencia = ahora.minus(config.vigencia());       // T_60

    Instant menor = null;
    for (Entrada e : porFolio.values()) {             // sustituye RESUMEN + LOCK(LK_MEM[J],2)
        Optional<Instant> prox = procesa(e, limiteReenvio, limiteVigencia);
        if (prox.isPresent() && (menor == null || prox.get().isBefore(menor))) menor = prox.get();
    }
    // NEXT_TIME := MENOR.[19:20] + 5; si no hubo candidatos, MENOR = ahora + TIEMPO.
    proximaRevision = (menor == null ? ahora.plus(config.tiempoEspera()) : menor).plusSeconds(5);
}
/** ENVIA_CANCELACIONES: el programa asíncrono cede su hilo a la librería hasta recibir el código 99. */
public void ejecutaAsincrono(CanalComs canal) {
    asincronoActivo.set(true);
    try {
        while (true) {
            Duration espera = Duration.between(reloj.instant(), proximaRevision)
                    .abs().compareTo(config.tiempoEspera()) < 0
                    ? Duration.between(reloj.instant(), proximaRevision).abs() : config.tiempoEspera();
            switch (eventos.espera(espera)) {                     // WAIT(DELTA, CIERRA_LOG, IN_EV, TSK_EV)
                case TIEMPO      -> revisa();                      // 01
                case CIERRA_LOG  -> bitacoraRechazos.rota();       // 02: RECHOPEN a RECHAZOS + consecutivo
                default -> {
                    var m = canal.recibeSinEsperar();              // RECEIVE(COMS_IN[DONTWAIT])
                    if (m.codigo() == 99) return;                  // fin pedido por COMS
                    if (m.codigo() == 0) canal.responde(informe(m)); // INFO # o resumen del proceso
                }
            }
        }
    } finally {

/** USO: corre cada 40 segundos mientras la librería está congelada (FREEZE CONTROL). */
@Scheduled(fixedRate = 40_000)
void ajustaUso() {
    usuarios.set(libraryUsers.get());                              // MYSELF.LIBRARYUSERS
    sts.rotaVentana();                                             // STS[K,2] := STS[K,1] := STS[K,0]

    long respuestas = sts.respuestasEnVentana();
    if (respuestas > 0) {
        Duration promedio = sts.tiempoRespuestaEnVentana().dividedBy(respuestas);
        Duration actual = config.tiempoEspera();
        config.tiempoEspera(promedio.compareTo(actual.dividedBy(2)) > 0
                ? actual.multipliedBy(3).dividedBy(2)                      // TIEMPO := 1.5 * TIEMPO
                : max(TIEMPO_ESPERA_DEFAULT, actual.multipliedBy(2).dividedBy(3)));
    }
    if (!asincronoActivo.get() && ciclosSinAsincrono.incrementAndGet() > 2) {
        ciclosSinAsincrono.set(0);
        log.warn("060103 01 CORE-DEP L091 25MTP002 PRECAUCION: FAVOR DE ACTIVAR PROGRAMA ASINCRONO");
        alertas.publica("L091_SIN_ASINCRONO");
    }
}
/** ARRANCAN (al iniciar la librería) y FIN_LIB (epílogo): el buzón sobrevive a una baja ordenada. */
@PostConstruct
void recarga() throws IOException {
    if (!Files.exists(props.archivoPendientes())) return;            // PEND.RESIDENT
    var snap = pendientes.lee(props.archivoPendientes());           // MAX_SZ, HORAS_VENCE, TIEMPO + entradas
    config.restaura(snap.maxTamano(), snap.vigencia(), snap.tiempoEspera());
    // A cada mensaje recargado se le garantiza al menos una hora más de vida.
    Instant minimo = reloj.instant().minus(config.vigencia()).plus(Duration.ofHours(1));
    for (Entrada e : snap.entradas()) {
        var ajustada = new Entrada(e.folio(), e.enviado(), e.mensaje(), e.ultimoEnvio(),
                e.insercion().isAfter(minimo) ? e.insercion() : minimo);   // MAX(A_PEND[3], T_60)
        if (lugares.tryAcquire()) porFolio.put(e.folio(), ajustada);
    }
    Files.delete(props.archivoPendientes());                         // CLOSE(PEND, PURGE)
}

@PreDestroy
void vuelca() throws IOException {
    if (porFolio.isEmpty()) return;
    pendientes.escribe(props.archivoPendientes(),                    // CORE-DEP/FILE/I91/MOVTOS/PEND.
            new Snapshot(config.maxTamano(), config.vigencia(), config.tiempoEspera(),
                         List.copyOf(porFolio.values())));
}

Riesgos de migración
Duplicados por búsqueda fuera del lock

GUARDAMSG busca el folio (línea 91) antes de tomar el interlock global (línea 94); dos altas simultáneas del mismo folio pueden ocupar dos posiciones.

Pérdida del buzón ante una falla

El volcado a CORE-DEP/FILE/I91/MOVTOS/PEND. solo ocurre en el epílogo FIN_LIB; una terminación anormal de la librería pierde todas las cancelaciones pendientes.

Campo de día inconsistente en la recarga

En ARRANCAN (línea 434) el día se inserta en [34:15], distinto del [35:16] que usan GUARDAMSG, REVISA y ENVIA; el límite de vigencia de los mensajes recargados poco después de medianoche queda mal calculado.

Dependencia del programa asíncrono

Sin el asíncrono activo no hay reenvíos ni descartes; la única señal es un DISPLAY cada tres ciclos de 40 segundos. Con 2016 posiciones el buzón se satura y las altas devuelven 1.

Reenvío de mensajes ya vencidos

ENVIA manda el mensaje antes de evaluar si venció (líneas 287 y 289), por lo que cada mensaje vencido se reenvía una última vez justo antes de descartarse.

Hardcodes

Capacidad 2016, tope de 2994 caracteres, vigencia default de 2 horas, la regla del CSI 32, los títulos CORE-DEP/LIST/L091, CORE-DEP/FILE/L91/RECHOPEN y CORE-DEP/FILE/I91/MOVTOS/PEND., y las estaciones DEFAULTSTATION y OUTPUT OF LINEA están compilados.
