@Configuration
class RendimientosCpeJobConfig {

    @Bean
    Job rendimientosCpe(JobRepository repo, Step abreControl, Step esperaCpesec, Step generaCpeind,
                        Step cargaParametrosFiscales, Step archivoSat, Step actualizaCuentas,
                        Step actualizaUltimoGrupo, P310Properties props) {
        // IF B02-NUM-CSI = 10: el resto de los CSI abren, leen control y terminan.
        JobExecutionDecider esVdm = (je, se) -> new FlowExecutionStatus(
                je.getExecutionContext().getInt("miCsi") == props.csiProcesador() ? "VDM" : "OTRO");

        return new JobBuilder("CORE-DEPP310", repo)
                .start(abreControl)                      // 140-BASE-DE-DATOS + 152-VALIDA-B02CONTROL
                .next(esVdm)
                    .on("OTRO").end()                    // 305-CIERRA-BASE
                .from(esVdm).on("VDM").to(esperaCpesec)  // 154-VALIDA-RESIDE-CPESEC
                    .next(generaCpeind)                  // 154-GENERA-CPEIND
                    .next(cargaParametrosFiscales)       // 140 / 150 / 154-CARGA-DATOS-S080
                    .next(archivoSat)                    // 180-GENERA-ARCHIVO-SAT
                    .next(actualizaCuentas)              // 200-ACTUALIZA-CUENTAS
                    .next(actualizaUltimoGrupo)          // lo que 300-TERMINA resuelve con un segundo recorrido
                .end()
                .build();
    }
}

/** 152-VALIDA-B02CONTROL: P310 trabaja con el mes anterior a B02-FECHA-LINEA. */
public record PeriodoCpe(LocalDate finMesAnterior) {

    static PeriodoCpe desde(LocalDate fechaLinea) {
        // YearMonth resuelve enero a diciembre del año anterior y febrero bisiesto
        // con la regla gregoriana completa (2100 no es bisiesto).
        return new PeriodoCpe(YearMonth.from(fechaLinea).minusMonths(1).atEndOfMonth());
    }

    /** Etiquetas de archivo: CPESEC y CPEIND usan PIC 9(6) y el MOVE desde 9(8) deja AAMMDD. */
    String etiquetaArchivo() { return finMesAnterior.format(DateTimeFormatter.ofPattern("yyMMdd")); }

    /** SALDOSCPE usa X(08): AAAAMMDD completo. */
    String etiquetaSat()     { return finMesAnterior.format(DateTimeFormatter.BASIC_ISO_DATE); }

    /** SAT-FECHA del detalle: AAAAMM (WS-FECHA-PROC-AAAAMM). */
    String periodoSat()      { return finMesAnterior.format(DateTimeFormatter.ofPattern("yyyyMM")); }
}

@Component

public class FactoresIsr {

    private final TarifasS080Client s080;          // 20000080-L700-TARIFAS
    private final P310Properties props;

    public record Factores(BigDecimal estandar, BigDecimal extranjero) {}

    /** Tarifa 254 (estándar) y 259 (residentes en el extranjero), producto 9991, instrumento 1. */
    public Factores cargar(LocalDate fechaVigencia) {
        BigDecimal t254 = consulta(props.tarifaIsrEstandar(), fechaVigencia);       // 254
        BigDecimal t259 = consulta(props.tarifaIsrExtranjero(), fechaVigencia)      // 259
                .divide(BigDecimal.valueOf(100), 4, RoundingMode.DOWN);             // / 100 y PIC 99V9999 trunca
        return new Factores(t254, t259);
    }

    private BigDecimal consulta(int tarifa, LocalDate vigencia) {
        var r = s080.tarifa(310, tarifa, 9991, 1, 0, vigencia);
        if (r.resultado() != 0 && r.resultado() != 1) {
            // El fuente deja el factor en 0 y continúa: no habría retención.
            throw new ParametroFiscalNoDisponibleException(tarifa, r.resultado());
        }
        // ENT-CAMPO / 10 ** DEC-CAMPO: punto fijo a decimal, sin pérdida.
        return new BigDecimal(BigInteger.valueOf(r.entero()), r.decimales())
                .setScale(4, RoundingMode.DOWN);                                    // destino 99V9999
    }
}

@Component
@StepScope
@RequiredArgsConstructor
class EsperaCpesecTasklet implements Tasklet {

    private final P310Properties props;                 // espera: PT40S, tiempo-maximo: PT6H
    @Value("#{jobExecutionContext['periodo']}") private PeriodoCpe periodo;

    /** 154-VALIDA-RESIDE-CPESEC: WAIT (40) hasta que el archivo exista, sin límite en el fuente. */
    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) throws Exception {
        Path cpesec = props.rutaCpesec(periodo);
        Instant limite = Instant.now().plus(props.cpesecTiempoMaximo());
        while (!Files.exists(cpesec)) {
            if (Instant.now().isAfter(limite))
                throw new ArchivoNoRecibidoException(cpesec);   // el fuente esperaría indefinidamente
            Thread.sleep(props.cpesecEspera().toMillis());
        }
        return RepeatStatus.FINISHED;
    }
}

@Component
class CpesecValidator {
    void valida(List<RegistroCpesec> regs, String tituloEsperado) {
        if (regs.isEmpty()) throw new ProcesoAbortadoException("ARCHIVO VACIO " + tituloEsperado);
        var h = regs.get(0);
        if (h.tipo() != 1) throw new ProcesoAbortadoException("NO TIENE REG 01 HEADER " + h.tipo());
        if (!tituloEsperado.equals(h.nombreArchivo()))
            throw new ProcesoAbortadoException("EL NOMB-ARCH EN HEADER " + h.nombreArchivo()
                    + " DIFIERE AL EXTERNO " + tituloEsperado);
        var t = regs.get(regs.size() - 1);
        if (t.tipo() != 99) throw new ProcesoAbortadoException("NO TIENE REG 99 TRAILER " + t.tipo());
        long detalles = regs.stream().filter(r -> r.tipo() == 2).count();
        if (t.registrosDetalle() != detalles)
            throw new ProcesoAbortadoException("ERROR EN TRAILER EN NO. DE REGS " + t.registrosDetalle());
    }
}

 * SAT-DETALLE: 150 posiciones, grupo de 10 dígitos sin separador y el resto separado por comas.
 * Los importes llevan punto decimal implícito: no es un CSV estándar.
 */
@Component
class SaldosSatLineAggregator implements LineAggregator<CuentaCpe> {

    @Override
    public String aggregate(CuentaCpe c) {
        String cuerpo = String.join(",",
                "2" + "%010d".formatted(c.grupo()),                    // TIPOREG + AREA..GPO = B39-GRUPO
                "%04d".formatted(c.subGrupo()),                        // SAT-CTE
                "%04d".formatted(c.sucursalMda()),                     // primeros 4 de B39-NUM-MDA
                "%08d".formatted(c.cuentaMda()),                       // últimos 8
                Implicito.de(c.saldoPromedio(), 16, 2),                // 9(16)V99
                Implicito.de(c.tasaNeta152(), 2, 3),                   // 9(02)V999
                Implicito.de(c.rendimientoPagado152(), 12, 2),         // 9(12)V99
                "%02d".formatted(c.opcionGrupo()),                     // B37-OPCION
                "%04d".formatted(c.rangoGrupo()),                      // B37-RANGO
                c.periodo());                                          // AAAAMM
        return cuerpo + " ".repeat(6);                                 // FILLER X(06)
    }
}

final class Implicito {
    /** Formatea sin punto, truncando como MOVE a un PIC más corto (RoundingMode.DOWN). */
    static String de(BigDecimal v, int enteros, int decimales) {
        BigDecimal t = v.setScale(decimales, RoundingMode.DOWN);
        return ("%0" + (enteros + decimales) + "d").formatted(t.unscaledValue());
    }
}

@Component
@StepScope
@RequiredArgsConstructor
public class CuentaCpeProcessor implements ItemProcessor<CuentaCpe, CuentaCpe> {

    private final ContratoCaptacionRepository b03;
    private final HistoricoRepository b06;
    private final CpeMonterreyRepository cpeind;
    private final CalculoRendimientoCpe calculo;     // 230-ACTUALIZA-B39-VDM / -MTY
    private final P310Properties props;

    @Override
    public CuentaCpe process(CuentaCpe b39) {
        return switch (b39.numCsi()) {
            case 4 -> cpeind.findById(b39.contrato())
                    .map(m -> calculo.desdeMonterrey(b39, m))
                    .orElseGet(() -> b39.excluir());                          // STATUS 4
            case 10 -> b03.findForUpdate(b39.contrato())
                    .filter(c -> c.getNumProducto() == 1 && c.getNumInstrum() == 3)   // MTDP-2466
                    .filter(c -> !props.statusBeneficiarioArt61().contains(c.getStaBenef()))  // 3 y 8
                    .flatMap(c -> b06.findForUpdate(c.getNumContrato()).map(h -> calculo.desdeVdm(b39, c, h)))
                    .orElseGet(() -> b39.excluir());
            default -> null;                                                   // otros CSI: no se tocan
        };
    }
}

Riesgos de migración
Doble recorrido de todas las cuentas

300-TERMINA (línea 1048) vuelve a ejecutar 200-ACTUALIZA-CUENTAS desde el inicio de B39 para cerrar el último grupo. Todas las cuentas se procesan y graban dos veces, L422 se invoca dos veces por cuenta y, con un solo grupo, B37 no se actualiza nunca.

Factor de ISR en cero si falla el catálogo

140-CARGA-FACTOR-ISR y 150-CARGA-FACTOR-ISR-EXTR no tratan un resultado de L700 distinto de 0 o 1: el factor queda en 0 y se graba como tasa de ISR en B39. Además la tarifa 259 se divide entre 100 y la 254 no, lo que supone unidades distintas en el catálogo.

Errores de base de datos tratados como exclusión

En 210-ACTUALIZA-CUENTAS cualquier estado distinto de cero al bloquear B03 o B06 marca la cuenta con estatus 4 (líneas 841-864), igual que si no existiera. Un bloqueo o un error transitorio deja la cuenta sin rendimientos en el mes.

Esperas sin límite

La apertura de la base reintenta cada 60 segundos mientras el error sea 39 (líneas 551-553) y la espera de CPESEC revisa cada 40 segundos (línea 599); ninguna tiene tope.

Validación comentada en el índice de Monterrey

El DMTERMINATE para detalles de tipo inválido está comentado (línea 680); esos contratos no entran a CPEIND y quedan con estatus 4.

Archivo SAT previo a la actualización

180-GENERA-ARCHIVO-SAT corre antes de 200-ACTUALIZA-CUENTAS, no filtra estatus y usa los campos -152 de B39. El archivo no refleja exclusiones ni valores calculados en la corrida.

Hardcodes

CSI 10 como procesador y 04 como origen, tarifas 254 y 259, catálogo 9991, tipos de persona 11, 12 y 15, opciones 1, 2 y 5, estatus de beneficiario 3 y 8, producto 1 instrumento 3, concepto L422 (04), fallback de cobro 2 y tarifa 100, identificador S152 y subíndices 50 y 12 de B06.
