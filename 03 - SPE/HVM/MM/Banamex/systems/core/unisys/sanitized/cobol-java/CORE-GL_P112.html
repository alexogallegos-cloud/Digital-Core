CORE-GL · P112 a Java 21 Spring Boot

P112 recibe el número de sistema a procesar, obtiene la fecha de proceso de la librería de control de CORE-GL (o del TASKVALUE si se manda), y carga desde el catálogo paramétrico S080BD01CON las monedas, leyendas, parámetros por clave de transacción y la relación transacción-guía contable, que graba en un archivo indexado temporal. Lee el log de movimientos del día, descompone cada movimiento en hasta cinco pares clave-importe, los ordena por libro, producto, moneda, clave de transacción y esquema contable, y totaliza con cortes de control. Por cada total imprime la leyenda, la naturaleza en S028 y, si la clave afecta a CORE-GL, busca la guía contable y marca REL-TRAN-GUIA CONTABLE INEXISTENTE cuando no existe.

/** WKS-RS-LLAVE X(16): cinco campos numéricos de ancho fijo. El orden lexicográfico del SORT
 *  coincide con el orden numérico campo por campo, por eso basta un Comparator encadenado. */
public record LlaveTotal(int libro,          // 9(02)
                         int producto,       // 9(04)
                         int moneda,         // 9(02)
                         int cveTran,        // 9(04)
                         int esqCon)         // 9(04)
        implements Comparable<LlaveTotal> {

    private static final Comparator<LlaveTotal> ORDEN =
            Comparator.comparingInt(LlaveTotal::libro)
                      .thenComparingInt(LlaveTotal::producto)
                      .thenComparingInt(LlaveTotal::moneda)
                      .thenComparingInt(LlaveTotal::cveTran)
                      .thenComparingInt(LlaveTotal::esqCon);

    @Override public int compareTo(LlaveTotal o) { return ORDEN.compare(this, o); }
}

@Component
@StepScope
@RequiredArgsConstructor
public class MovimientoAPartidasProcessor implements ItemProcessor<MovimientoCORE-GL, List<PartidaPunteo>> {

    @Value("#{jobParameters['sistema']}") private int sistema;   // WKS-PARAM-SIS

    @Override
    public List<PartidaPunteo> process(MovimientoCORE-GL m) {
        int libro = m.indConta();
        if (sistema == 403 && libro == 2) libro = 0;              // regla propia del S403
        int producto = (sistema == 87) ? 87 : m.producto();       // S087 fuerza producto 87

        if (!(m.funcion() == 1 && m.status() == 1)) return null;  // solo altas procesadas

        int prod = (libro == 1 || libro == 2) ? m.fideicomiso() : producto;
        var partidas = new ArrayList<PartidaPunteo>(5);
        for (var e : m.elementos()) {                             // A00-R01-ELE-CVEIMP OCCURS 5
            if (e.cveTran() <= 0) continue;
            partidas.add(new PartidaPunteo(
                new LlaveTotal(libro, Pic.trunca(prod, 4), Pic.trunca(m.moneda(), 2),
                               Pic.trunca(e.cveTran(), 4), e.esqCon()),
                Pic.truncaImporte(e.importe(), 14, 2)));          // 9(16)V99 a 9(14)V99
        }
        return partidas.isEmpty() ? null : partidas;
    }
}

/** MOVE numérico COBOL: conserva los dígitos de la derecha y descarta los de la izquierda. */
final class Pic {
    static int trunca(long v, int digitos) { return (int) (v % (long) Math.pow(10, digitos)); }
    static BigDecimal truncaImporte(BigDecimal v, int enteros, int dec) {
        BigDecimal limite = BigDecimal.TEN.pow(enteros);
        return v.setScale(dec, RoundingMode.DOWN).remainder(limite);
    }
}

@Component
@RequiredArgsConstructor
public class ArranquePunteoTasklet implements Tasklet {

    private final ControlSistemaPort libControl;   // CONSISDIA IN LIB-CONTROL (CORE-GLB01SISDIA)
    private final HostInfo host;                   // CSIID IN SOPORTECOMS

    @Override
    public RepeatStatus execute(StepContribution sc, ChunkContext ctx) {
        var params = ctx.getStepContext().getJobParameters();
        int sistema = ((Number) params.get("sistema")).intValue();         // USING WKS-PARAM-SIS
        var r = libControl.consultaSistemaDia(sistema, host.csi(), 1);    // WKS-B01-FUNCION = 1
        switch (r.resultado()) {
            case 0 -> { }
            case 1 -> throw new ProcesoAbortadoException("SISTEMA NO DADO DE ALTA");     // STATUS = -1
            default -> throw new ProcesoAbortadoException("ERROR EN BASE DE CONTROL");
        }
        // TASKVALUE con fecha explícita tiene prioridad; si es cero se usa la de la base de control.
        Long fechaParam = (Long) params.get("fechaProceso");              // WKS-PARAM-FCH 9(08)
        LocalDate fecha = (fechaParam != null && fechaParam > 0)
                ? LocalDate.parse(String.valueOf(fechaParam), DateTimeFormatter.BASIC_ISO_DATE)
                : r.fechaProceso();                                       // WKS-B01-FECPRO
        ctx.getStepContext().getStepExecution().getJobExecution()
           .getExecutionContext().put("fechaProceso", fecha);
        return RepeatStatus.FINISHED;
    }
}

@Component
@RequiredArgsConstructor
public class CatalogosS080Loader {

    private static final int LIMITE_CLAVES = 9999;            // OCCURS 9999 de WKS-T-PARAMETROS-TRANS2
    private final S080CatalogoPort s080;                      // L710_CONSUL_DETALLE IN S080BD01CON
    private final RelacionTranGuiaRepository relaciones;

    /** 12200-CARGA-TRANSACCIONES: parámetros por clave de transacción del sistema. */
    public Map<Integer, ParametroTransaccion> parametros(int sistema, int catalogo) {
        var tabla = new HashMap<Integer, ParametroTransaccion>();
        for (var reg : s080.recorre(catalogo, sistema)) {     // pagina mientras RESULT = 1
            int cve = reg.dato(2);
            if (cve >= LIMITE_CLAVES) {
                log.warn("ERROR CAT # 6 ( TRAN X SIS ) S{} LIMITE 9999 CVETRAN {}", sistema, cve);
                continue;                                     // el fuente avisa una sola vez y descarta
            }
            tabla.put(cve, new ParametroTransaccion(reg.dato(3), reg.dato(4),
                                                    reg.dato(11), reg.dato(12), reg.dato(13)));
        }
        return tabla;
    }

    /** 12300-CARGA-RELTRANGUIA y 12425-GRABA-CAT: alta sin duplicados. */
    @Transactional
    public void relacionTranGuia(int catalogo) {
        for (var reg : s080.recorre(catalogo, 0)) {
            var llave = new RelacionTranGuia.Llave(reg.dato(1), reg.dato(2),
                    reg.dato(3) / 10_000, reg.dato(3) % 10_000, reg.dato(4), reg.dato(5));
            if (!relaciones.existsById(llave)) {
                relaciones.save(new RelacionTranGuia(llave, reg.dato(6)));
            }
        }
    }
}

@Component
@StepScope
@RequiredArgsConstructor
public class TotalizadorPorClave implements ItemProcessor<PartidaPunteo, TotalPorClave> {

    private final ReportePunteo reporte;            // V00-REP-CVES
    private LlaveTotal anterior;                    // WKS-LLAVE-ANT
    private BigDecimal importe = BigDecimal.ZERO;   // W77-IMPMOV
    private long movimientos;                       // W77-NUMMOV

    /** Emite el total acumulado cuando cambia cualquiera de las cinco dimensiones. */
    @Override
    public TotalPorClave process(PartidaPunteo p) {
        LlaveTotal k = p.llave();
        if (anterior == null) { reinicia(k, p); reporte.encabezado(k); return null; }
        if (k.equals(anterior)) {
            importe = importe.add(p.importe());     // suma exacta, escala 2
            movimientos++;
            return null;
        }
        var total = new TotalPorClave(anterior, importe, movimientos);    // 002200-IMP-DET
        if (k.libro() != anterior.libro() || k.producto() != anterior.producto()
                || k.moneda() != anterior.moneda()) {
            reporte.corte(anterior, k);             // 2600 / 2500 / 2400 y 2300-IMP-ENCABEZADO
        }
        reinicia(k, p);
        return total;
    }

    private void reinicia(LlaveTotal k, PartidaPunteo p) {
        anterior = k; importe = p.importe(); movimientos = 1;
    }
    // El último total se emite en el afterStep (PERFORM 002200-IMP-DET de la línea 2223).
}
@Component
@RequiredArgsConstructor
public class RenglonPunteoProcessor implements ItemProcessor<TotalPorClave, RenglonPunteo> {

    private final Map<Integer, ParametroTransaccion> parametros;   // WKS-T-PARAMETROS-TRANS2
    private final LeyendasTransaccion leyendas;                    // TRANS1..TRANS4 (4 x 3000)
    private final GuiaContableResolver guias;

    @Override
    public RenglonPunteo process(TotalPorClave t) {
        var p = parametros.getOrDefault(t.llave().cveTran(), ParametroTransaccion.VACIO);
        String nombre = leyendas.porIndice(p.numLeyenda());        // 0 o > 12000 -> espacios
        String naturaleza = switch (p.naturalezaS028()) {
            case 1 -> "CAR";
            case 2 -> "ABO";
            default -> "N/C";
        };
        boolean afectaCORE-GL = p.indCORE-GL() == 2;
        String error = afectaCORE-GL ? guias.diagnostico(t.llave()) : "";
        return new RenglonPunteo(t.llave().cveTran(), t.llave().esqCon(), t.importe(),
                t.movimientos(), nombre, naturaleza, afectaCORE-GL ? "SI" : "NO", error);
    }
}

    public String porIndice(int i) {
        return (i <= 0 || i > 12_000 || i > leyendas.size()) ? "" : leyendas.get(i - 1);
    }

    static final String SIN_GUIA = "REL-TRAN-GUIA CONTABLE INEXISTENTE";
    private final RelacionTranGuiaRepository relaciones;
    @Value("#{jobParameters['sistema']}") private int sistema;

    /** 2900-LEE-ESQCON. Las equivalencias por sistema van a application.yml en la versión final. */
    public String diagnostico(LlaveTotal k) {
        int mon = k.moneda(), libro = k.libro();
        if (sistema == 403 || sistema == 404) {
            if (mon == 1 && (libro == 0 || libro == 3 || libro == 7 || libro == 8)) libro = 0;
            else if (mon == 3 && (libro == 0 || libro == 3))                        libro = 0;
            else if ((mon == 3 || mon == 1) && (libro == 1 || libro == 2))          { /* sin cambio */ }
            else if (mon == 3 && (libro == 7 || libro == 8))                        libro = 2;
            else { mon = 0; libro = 0; }
        }
        if ((sistema == 264 || sistema == 703 || sistema == 18 || sistema == 17) && mon > 1) {
            mon = 0;
        }
        var llave = new RelacionTranGuia.Llave(sistema, k.cveTran(), k.esqCon(), mon, libro, 0);
        return relaciones.existsById(llave) ? "" : SIN_GUIA;
    }

Riesgos de migración
Truncamiento silencioso al armar la llave

En 001000-ARMA-ARCHIVO y 001100-ARMA-REGISTRO (líneas 2618-2647) se mueven CVETRAN 9(06), FIDEICO 9(06), MONEDA 9(04) e IMPORTE 9(16)V99 a campos de 4, 4, 2 y 14 enteros. Claves o importes que excedan esos tamaños se agrupan o reportan mal sin aviso.

Posible ciclo sin fin en la carga de la relación transacción-guía

12300-CARGA-RELTRANGUIA solo marca fin cuando la librería S080 regresa 0 (línea 2516) y no reporta otros códigos; el PERFORM de la línea 2318 se repetiría indefinidamente ante un error.

Log vacío no detiene el proceso

El AT END del primer READ (líneas 2146-2155) solo escribe un mensaje; el programa sigue y emite un reporte sin información.

Claves de transacción fuera de rango descartadas

Las claves con valor 9999 o mayor no se cargan en la tabla de parámetros (línea 2465) y solo se avisa la primera; esas claves salen sin leyenda, naturaleza ni búsqueda de guía.

Reglas por sistema compiladas

Los sistemas 87, 403, 404, 264, 703, 18 y 17 tienen tratamientos especiales en el código (líneas 2611-2615 y 3005-3026). Se trasladan a configuración.

  • Confirmar si la sustitución de producto por fideicomiso para los libros 1 y 2 aplica a todos los sistemas o solo a fiduciario.
