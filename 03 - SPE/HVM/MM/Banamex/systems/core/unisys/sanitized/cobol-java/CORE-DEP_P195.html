Lee I01-RECHAZOS (276 caracteres), cuyo tipo de registro identifica el origen del rechazo (01 prelínea, 02 CORE-GL, 03 S408, 04 sin confirmar, 05 recuperaciones, 10 y 11 aceptados con línea de crédito forzada). Según el tipo de proceso del TASKVALUE, imprime cada registro en su listado y copia los de prelínea, recuperación y aceptados a un archivo intermedio, ajustando la sucursal y el nodo de impresión cuando el nodo no pertenece al CSI. Ordena ese archivo por tipo, nodo, sucursal y autorización y genera listados con corte por sucursal y por nodo con totales de abonos, cargos y línea de crédito forzada. En el proceso 1 genera además el archivo de devoluciones para el sistema 711, y en el proceso 2 el listado de movimientos borrados.

/** 10000020-VALIDA-TASKVALUE: fecha AAAAMMDD, CSI y tipo de proceso. */
@Component
public class P195JobParametersValidator implements JobParametersValidator {

    @Override
    public void validate(JobParameters p) throws JobParametersInvalidException {
        int tipo = p.getLong("tipoProceso").intValue();
        if (tipo < 0 || tipo > 2) {
            throw new JobParametersInvalidException("ERROR EN VALUE TIPOP=" + tipo);
        }
        String fecha = p.getString("fechaBatch");        // vacía: se toma de CORE-DEPB02CONTROL
        Long csi = p.getLong("csi");
        if (fecha != null || csi != null) {
            LocalDate f = LocalDate.parse(fecha, DateTimeFormatter.BASIC_ISO_DATE);
            if (f.getYear() <= 1994 || (csi != 4 && csi != 10)) {
                throw new JobParametersInvalidException("ERR FECH/CSI EN VALUE FECHA " + fecha + " CSI " + csi);
            }
        }
    }
}
@Configuration
public class RuteoRechazosJobConfig {

    /** 50000000-PROCESO-PRINCIPAL. */

                .end()
                .build();
    }
}

/** 50003000-PROCESO-ARCH: decide los destinos de cada rechazo. */
@Component
@StepScope
public class RechazoClassifier implements Classifier<Rechazo, ItemWriter<? super Rechazo>> {

    @Value("#{jobParameters['tipoProceso']}") private int tipoProceso;
    private final Map<Destino, ItemWriter<Rechazo>> writers;

    @Override
    public ItemWriter<? super Rechazo> classify(Rechazo r) {
        List<Destino> destinos = switch (tipoProceso) {
            case 0, 2 -> switch (r.tipo()) {
                case CORE-GL       -> List.of(Destino.LISTADO_CORE-GL);          // R02
                case S408          -> List.of(Destino.LISTADO_S408);          // R03
                case SIN_CONFIRMAR -> List.of(Destino.LISTADO_SIN_CONFIRMAR); // R04
                case PRELINEA, RECUPERACION, ACEPTADO_LCF_PRELINEA, ACEPTADO_LCF_RECUPERA
                                   -> Destino.dePrelinea(r.tipo(), tipoProceso); // ver 50004000
                default            -> List.of();                              // 6 a 9: sin destino
            };
            default -> (r.tipo() == TipoRechazo.PRELINEA || r.tipo() == TipoRechazo.ACEPTADO_LCF_PRELINEA)
                    ? Destino.conS711(Destino.dePrelinea(r.tipo(), tipoProceso))
                    : List.of();                                              // proceso 1
        };
        return items -> destinos.forEach(d -> writers.get(d).write(items));
    }
}
/** 50004000-PRELINEA: listado de rechazos o de aceptados y, según el proceso, archivo por sucursal. */
static List<Destino> dePrelinea(TipoRechazo t, int proceso) {
    boolean rechazo = t == TipoRechazo.PRELINEA || t == TipoRechazo.RECUPERACION;      // 1 o 5
    List<Destino> d = new ArrayList<>();
    if (proceso == 0 || proceso == 2) {
        d.add(rechazo ? Destino.LISTADO_PRELINEA : Destino.LISTADO_ACEPTADOS_LCF);     // R01 o R09
    }
    boolean deLinea = t == TipoRechazo.PRELINEA || t == TipoRechazo.ACEPTADO_LCF_PRELINEA;     // 1 o 10
    boolean deRecup = t == TipoRechazo.RECUPERACION || t == TipoRechazo.ACEPTADO_LCF_RECUPERA; // 5 o 11
    if (proceso == 0 || (proceso == 1 && deLinea) || (proceso == 2 && deRecup)) {
        d.add(Destino.ARCHIVO_PARA_SUCURSAL);                                           // I02-PARASUC
    }
    return d;
}

@Component
@RequiredArgsConstructor
class NodoImpresionResolver implements ItemProcessor<RechazoPrelinea, RechazoPrelinea> {
    private final P195Properties props;
    private final SucursalesClient s080;                    // 42000080-DATOS-S080
    @Value("#{jobParameters['csi']}") private int csi;

    @Override
    public RechazoPrelinea process(RechazoPrelinea r) {
        String refAlfa = r.refAlfanumerica().replace('\u0000', ' ');      // INSPECT ... @00@ BY " "
        int sucursal = r.sucOpero() != 0 ? r.sucOpero() : Integer.parseInt(r.autorizacion().substring(0, 4));
        int nodo = s080.nodoImpresion(sucursal);
        var cfg = props.nodos().get(csi);
        if (!cfg.validos().contains(nodo)) {
            sucursal = cfg.sucursalOmision();
            nodo = cfg.nodoOmision();
        }
        return r.conSucursalNodo(sucursal, nodo, refAlfa);
    }
}
/** 50004200-ARCHIVO-S711 y 50004300-ARMA-ARCHS711 (solo proceso 1). */
@Component
class DevolucionesS711Processor implements ItemProcessor<RechazoPrelinea, DevolucionS711> {
    private final AtomicLong secuencia = new AtomicLong();        // W77-SEQ-S711

    @Override
    public DevolucionS711 process(RechazoPrelinea r) {
        // SISTORIG es X(004) y se compara con el literal 711: el fuente espera "711 ".
        if (!"711".equals(r.sistemaOrigen().strip()) || r.claveTran() != 2002) return null;
        Medio m = Medio.de(r.numMedio());                          // WKS-SUC-MDA y WKS-CTA-MDA
        return new DevolucionS711(secuencia.incrementAndGet(), r.refAlfanumerica(),
                m.sucursal(), m.cuenta(), r.refNumerica(), r.importe(),
                r.claveRechazo(), r.moneda(), r.nombre());
    }
}

record TotalesSucursal(BigDecimal abonos, int numAbonos, BigDecimal cargos, int numCargos,
                       BigDecimal lcForzada) {

    static TotalesSucursal vacio() { return new TotalesSucursal(ZERO2, 0, ZERO2, 0, ZERO2); }

    TotalesSucursal mas(RechazoPrelinea r) {
        int c = r.claveTran();
        boolean abono = (c > 999 && c < 2000) || (c > 2999 && c < 4000);
        boolean cargo = (c > 1999 && c < 3000) || (c > 3999 && c < 5000);
        BigDecimal lcf = esAceptado(r.tipo()) ? lcForzada.add(r.importeLcForzada()) : lcForzada;
        if (abono) return new TotalesSucursal(abonos.add(r.importe()), numAbonos + 1, cargos, numCargos, lcf);
        if (cargo) return new TotalesSucursal(abonos, numAbonos, cargos.add(r.importe()), numCargos + 1, lcf);
        return new TotalesSucursal(abonos, numAbonos, cargos, numCargos, lcf);   // fuera de rango: no suma
    }
}

Riesgos de migración
Reasignación de nodos compilada

Los nodos válidos por CSI y las sucursales por omisión 904 y 907 están en el código (líneas 1621-1632). Un nodo nuevo hace que sus rechazos se impriman en otra sucursal.

Comparación de texto contra número en el sistema 711

WKS-I01-R01-SISTORIG es PIC X(004) y se compara con 711 (línea 1644). Si el campo trae ceros a la izquierda, las devoluciones del sistema 711 no se generan.

Claves fuera de rango en totales

Las claves de transacción fuera de 1000 a 4999 se listan pero no suman a abonos ni a cargos (líneas 1899-1907), por lo que los totales por sucursal pueden no cuadrar con el detalle.

Trailers sin validar

Ni I01-RECHAZOS ni I05-BORRADOS validan conteos de trailer; un archivo truncado se procesa sin aviso.

Tipos de registro sin destino

Los tipos 6 a 9 documentados en el registro no tienen ruta en 50003000-PROCESO-ARCH y se descartan en silencio.
