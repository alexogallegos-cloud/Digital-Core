/** REORGANIZADOR y GARBAGE nunca se abren: el WFL solo pregunta IS RESIDENT. */
public interface CatalogoObjetos {
    boolean existe(String titulo, String familia);                 // FILE ... IS RESIDENT
    void elimina(String titulo, String familia);                   // REMOVE #S FROM familia
    void renombra(String origen, String destino, String familia);  // CHANGE #S1 TO #S2 FROM familia
}

@Component
@RequiredArgsConstructor
public class ArtefactosReorgTarjetas {

    private final ReorgTarjetasProperties props;
    private final CatalogoObjetos catalogo;

    /** S1: objeto que deja BUILDREORG. */
    public String reorganizador() { return "REORGANIZATION/" + props.base(); }

    /** S2: objeto con el nombre definitivo del programa P694. */
    public String garbage() {
        return "CORE-DEP/OBJECT/P694/" + props.base() + "/" + props.versionObjeto();
    }

    public boolean reorganizadorResidente() { return catalogo.existe(reorganizador(), props.familiaResidencia()); }
    public boolean garbageResidente()       { return catalogo.existe(garbage(), props.familiaResidencia()); }
}
/** Step 1. En el WFL es un REMOVE ciego; aquí se respalda antes de borrar. */
@Component
@RequiredArgsConstructor
public class LimpiaObjetosPreviosTasklet implements Tasklet {

    private final ArtefactosReorgTarjetas artefactos;
    private final CatalogoObjetos catalogo;
    private final RespaldoArtefactos respaldo;
    private final ReorgTarjetasProperties props;

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        for (String titulo : List.of(artefactos.reorganizador(), artefactos.garbage())) {
            if (catalogo.existe(titulo, props.familiaResidencia())) {
                respaldo.guarda(titulo, props.familiaResidencia());   // no existe en el fuente
                catalogo.elimina(titulo, props.familiaResidencia());  // REMOVE #S1, #S2 FROM CAPTACION
            }
        }
        return RepeatStatus.FINISHED;
    }
}

/** IF T1 IS COMPILEDOK: en el fuente la rama falsa termina el job en silencio. */
public class CompiladoOkDecider implements JobExecutionDecider {
    @Override
    public FlowExecutionStatus decide(JobExecution job, StepExecution step) {
        return Boolean.TRUE.equals(job.getExecutionContext().get("compiladoOk"))
                ? new FlowExecutionStatus("COMPILADO")
                : new FlowExecutionStatus("NO_COMPILADO");
    }
}

/** DO WAIT(15); IF REORGANIZADOR IS RESIDENT THEN CHANGE ...; UNTIL GARBAGE IS RESIDENT */
@Component
@RequiredArgsConstructor
public class PromueveObjetoTasklet implements Tasklet {

    private final ArtefactosReorgTarjetas artefactos;
    private final CatalogoObjetos catalogo;
    private final ReorgTarjetasProperties props;   // sondeo.intervalo = 15s, sondeo.maximo = configurable

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) throws InterruptedException {
        Instant limite = Instant.now().plus(props.sondeo().maximo());   // el WFL no tiene límite
        do {                                                            // post-test: al menos una espera
            Thread.sleep(props.sondeo().intervalo().toMillis());        // WAIT(15)
            if (artefactos.reorganizadorResidente()) {
                catalogo.renombra(artefactos.reorganizador(), artefactos.garbage(),
                                  props.familiaResidencia());           // CHANGE #S1 TO #S2 FROM CAPTACION
            }
            if (Instant.now().isAfter(limite)) {
                throw new ObjetoNoGeneradoException(artefactos.garbage());
            }
        } while (!artefactos.garbageResidente());
        return RepeatStatus.FINISHED;
    }
}

@Bean
Job reorgTarjetas(JobRepository repo, Step limpia, Step construye, Step promueve) {
    return new JobBuilder("CORE-DEP-REORG-GARBAGE-TARJETAS", repo)
            .start(limpia).next(construye)
            .next(new CompiladoOkDecider())
                .on("COMPILADO").to(promueve)
            .from(new CompiladoOkDecider())
                .on("NO_COMPILADO").fail()        // el WFL termina sin aviso; aquí queda FAILED
            .end().build();
}
Riesgos de migración
Borrado previo sin respaldo

La línea 20 elimina el reorganizador anterior y el objeto productivo P694 antes de generar el reemplazo. Si BUILDREORG falla, el pack queda sin objeto y el job termina sin aviso (líneas 38 y 46).

Fallo de compilación silencioso

Cuando T1 no es COMPILEDOK, el flujo salta a END JOB sin DISPLAY, ABORT ni código de retorno. No hay ON TASKFAULT ni manejo de excepciones en ninguna parte del job.

Ciclo de espera sin límite

El DO ... UNTIL GARBAGE IS RESIDENT de las líneas 39 a 45 no tiene contador ni timeout. La versión Java agrega un límite configurable.

AUTOSWAP sobre estructuras productivas

El objeto generado lleva AUTOSWAP=TRUE y COPY BACK=TRUE sobre tres estructuras de la base de tarjetas. La ventana de indisponibilidad no se controla en este job; ocurre al ejecutar el objeto P694.

Hardcodes de versión y pack

Versiones 25MTP001 y 25MTP003, pack CAPTACION y familia REORG están escritos en el cuerpo. Se trasladan a application.yml. No hay credenciales embebidas en el fuente.
