CORE-GL · P073 a Java 21 Spring Boot

Al arrancar resuelve por CTLVERS los títulos de sus librerías, carga desde LIB-CONTROL la tabla de sistemas habilitados para punteo cuyo día está en estatus 3 y residen en el CSI de la máquina, y enlaza la librería L011 de cada uno. Luego espera mensajes COMS en ciclos de 60 segundos. Cada mensaje dirigido al servicio 151-1-2 trae sistema, fecha, sucursal, caja, moneda y una llave de continuación; el programa llama a CONSULTA_CI de la L011 del sistema y responde con encabezado SA2, la misma llave y hasta 23 movimientos de 75 posiciones. El operador puede prender y apagar un archivo de monitor o terminar el programa con comandos HI.

@ConfigurationProperties("p073")
public record P073Properties(int encabezadoLgcr, int encabezadoSa2, Monitor monitor,
                             Set<Integer> sistemasPunteo, Duration esperaMensaje) {

    public record Monitor(String titulo) {}

    public P073Properties {
        sistemasPunteo = Set.copyOf(sistemasPunteo);
        if (esperaMensaje == null) esperaMensaje = Duration.ofSeconds(60);   // W77-ESPERACOMS-TIEMPO
    }
}
/** WKS-MSG-SAL-NIVEL2: respuesta de 1835 bytes; 23 renglones REDEFINES de 75 posiciones. */
public record RespuestaConsultaMovs(
        int sucursal, LocalDate fecha, int caja, int moneda, int sistema,
        Continuacion continuacion,              // AUTCON 9(08) + NUMREG 9(02)
        List<MovimientoGl> movimientos) {       // máximo 23 (W77-NUMMOVS)

    public static final int MAX_MOVIMIENTOS = 23;

    public RespuestaConsultaMovs {
        if (movimientos.size() > MAX_MOVIMIENTOS)
            throw new IllegalArgumentException("WKS-SNIV2-DATOS OCCURS 23");
        movimientos = List.copyOf(movimientos);
    }

    public record Continuacion(long autorizacion, int numRegistro) {}

    public record MovimientoGl(
            int claveTransaccion,                // 9(06)
            int naturaleza,                      // 9(01)
            BigDecimal importe,                  // 9(14)V9(02) sin signo -> escala 2
            long autorizacionAplicativa,         // 9(08)
            String cuenta,                       // 9(20): String, conserva ceros
            int sucursalPromotora,               // 9(04)
            String referencia) {}                // 9(20)

    /** Longitud lógica que hoy se envía: 34 fijos + 75 por movimiento + encabezados. */
    public int longitud(P073Properties p) {
        return 34 + 75 * movimientos.size() + p.encabezadoLgcr() + p.encabezadoSa2();
    }
}

@Component
@RequiredArgsConstructor
public class SistemasPunteoRegistry {

    private static final int ESTATUS_DIA_ABIERTO = 3;     // WKS-B04-ESTATUS = 3
    private final LibControlClient libControl;            // B04SISTEM, CONSISMEN, CONSISDIA
    private final P073Properties props;
    private final Map<Integer, SistemaActivo> activos = new ConcurrentHashMap<>();

    @PostConstruct
    void carga() {
        int miCsi = libControl.csiDelHost();               // CSIID OF SOPORTECOMS
        for (Sistema s : libControl.sistemas()) {          // B04SISTEM función 02, hasta RESULT > 0
            if (!props.sistemasPunteo().contains(s.numero())) continue;
            var men = libControl.consisMen(s.numero());    // función 01: modo y CSI de residencia
            var dia = libControl.consisDia(s.numero());    // función 01: fecha contable
            boolean resideAqui = (men.modo() == 1 && men.csiResidencia() == miCsi)
                              || (men.modo() == 0 && (men.csiResidencia() == 0 || men.csiResidencia() == miCsi));
            if (s.estatus() == ESTATUS_DIA_ABIERTO && resideAqui) {
                activos.put(s.numero(), new SistemaActivo(s.numero(), dia.fechaContable(),
                                                          men.modo(), men.csiResidencia()));
            }
        }
        log.info("Sistemas de punteo activos: {}", activos.keySet());   // W77-TOTAL-SISTEMAS
    }

    public Optional<SistemaActivo> busca(int sistema) { return Optional.ofNullable(activos.get(sistema)); }
}
@Configuration
@RequiredArgsConstructor
public class ConsultaMovsFlowConfig {

    private final ConsultaMovsHandler handler;

    /** ENABLE INPUT COMS-IN + ESPERACOMS + RECEIVE/SEND: un flujo request-reply. */
    @Bean
    IntegrationFlow consultaMovimientos(ConnectionFactory cf, P073Properties props) {
        return IntegrationFlow
            .from(Jms.inboundGateway(cf)
                     .destination("CORE-GL.CONSULTA.MOVS")          // servicio 151-1-2
                     .replyTimeout(props.esperaMensaje().toMillis()))
            .filter(MensajeSa2.class, MensajeSa2::esParaServicio151_1_2,
                    f -> f.discardFlow(d -> d.handle(handler::rechazoDestino)))   // RESULT = 2
            .handle(MensajeSa2.class, (m, h) -> handler.atiende(m))
            .get();
    }
}
// Los comandos HI 2/3/4/6 de la DECLARATIVES pasan a endpoints de Actuator:
// /actuator/monitor (on/off) y el apagado ordenado del contenedor.
/** Encabezado SA2 recibido. */
public record MensajeSa2(int origenSistema, int origenServicio, int origenSubservicio,
                         int destinoSistema, int destinoServicio, int destinoSubservicio,
                         PeticionConsultaMovs peticion) {

    /** Sólo se atiende lo dirigido a 151-1-2; el resto se contesta con resultado 2. */
    public boolean esParaServicio151_1_2() {
        return destinoSistema == 151 && destinoServicio == 1 && destinoSubservicio == 2;
    }
}

@Component
public class ComsStatusPolicy {
    /** Status de RECEIVE que se consideran mensaje válido (0, 89, 92, 93) y fin (99). */
    private static final Set<Integer> ACEPTADOS = Set.of(0, 89, 92, 93);

    public Accion evalua(int statusKey) {
        if (ACEPTADOS.contains(statusKey)) return Accion.PROCESA;
        if (statusKey == 99) return Accion.TERMINA;                 // MOVE 1 TO W77-FIN
        log.error("ERROR EN RECEIVE COMS status={}", statusKey);
        return Accion.DESCARTA;
    }
}

    private final RespuestaSa2Builder respuesta;

    public MensajeSa2 atiende(MensajeSa2 m) {
        var p = m.peticion();
        if (sistemas.busca(p.sistema()).isEmpty() || !l011PorSistema.containsKey(p.sistema())) {
            return respuesta.rechazo(m);                                 // RESULT = 2
        }
        var req = new ConsultaCi(1, p.fecha(), p.sucursal(), p.caja(), p.moneda(), p.continuacion());
        ResultadoL011 r = l011PorSistema.get(p.sistema()).consultaCi(req);   // W77-RESULT-L011
        if (r.codigo() != 0) {
            log.warn("CONSULTA SISTEMA {} FN 1 FECHA {} SUC {} CAJA {} MONEDA {} RESULT L011 {}",
                     p.sistema(), p.fecha(), p.sucursal(), p.caja(), p.moneda(), r.codigo());
            return respuesta.rechazo(m);
        }
        List<MovimientoGl> movs = r.movimientos().stream()
                .takeWhile(x -> x.autorizacionAplicativa() > 0)          // corta en el primer AUTAPL = 0
                .limit(RespuestaConsultaMovs.MAX_MOVIMIENTOS)
                .toList();
        return respuesta.ok(m, new RespuestaConsultaMovs(p.sucursal(), p.fecha(), p.caja(),
                p.moneda(), p.sistema(), r.continuacion(), movs));       // RESULT = 1

Riesgos de migración
Fin erróneo que no detiene el programa

900000-FIN-MAL (líneas 1752-1758) ejecuta 999999-FIN, que deshabilita el programa en COMS, pero no pone W77-FIN ni hace STOP RUN. Si falla la validación de COMS en 020400-COMS, el programa continúa y entra al ciclo de atención con el programa ya deshabilitado.

Doble ejecución de 999999-FIN con HI 4

El HI 4 ejecuta 999999-FIN dentro de 030100-ACEPTA-MENSAGE (línea 1465) y otra vez desde 010100-INICIA-CONTROL (línea 1040).

Código muerto del sistema 252

Hay enlace y llamada para la L011 del sistema 252 (líneas 1212 y 1677), pero 252 no está en W88-SISTEMAS-PUNTEO (líneas 40-55).

Valores de protocolo compilados

Servicio 151-1-2, prefijo 15173, encabezados de 48 y 37 posiciones, CSI 32 tratado como 10 (línea 1096), espera de 60 segundos y títulos por defecto de las librerías L011 están escritos en el código.

Tabla de sistemas fija durante la vida del programa

Los sistemas se cargan una sola vez al arrancar (020210-CARGA-SISTEMAS); cambios de estatus o de CSI posteriores no se reflejan hasta reiniciar.
