/** E01-REG-S111-PREALTA: un registro de 40 bytes; la vista depende de E01-S111-TIP-REG. */
public sealed interface SolicitudS111
        permits SolicitudS111.Header, SolicitudS111.Movimiento, SolicitudS111.PrealtaTef,
                SolicitudS111.Trailer, SolicitudS111.Desconocido {

    int tipo();

    /** Tipo 00. Fecha AAAAMMDD y sistemas origen y destino. */
    record Header(LocalDate fecha, String sisOrigen, String sisDestino) implements SolicitudS111 {
        public int tipo() { return 0; }
    }

    /** Tipos 01 alta, 02 bloqueo, 03 desbloqueo, 04 eliminación (vista E01-S111-DETALLE). */
    record Movimiento(int tipo, Pan tarjeta, String contrato, int sucPromotora) implements SolicitudS111 {}

    /** Tipo 05: alta de prealta TEF (vista E01-S111-DETALLE2). */
    record PrealtaTef(Pan tarjeta, String numTef) implements SolicitudS111 {
        public int tipo() { return 5; }
    }

    /** Tipo 99. */
    record Trailer(long numRegistros) implements SolicitudS111 {
        public int tipo() { return 99; }
    }

    record Desconocido(int tipo, String raw) implements SolicitudS111 {}

    static SolicitudS111 parse(String r) {
        int tipo = Integer.parseInt(r.substring(0, 2));
        return switch (tipo) {
            case 0      -> new Header(LocalDate.parse(r.substring(2, 10), BASIC_ISO_DATE),
                                      r.substring(10, 14), r.substring(14, 18));
            case 1, 2, 3, 4 -> new Movimiento(tipo, Pan.of(r.substring(2, 18)),
                                      r.substring(18, 30), Integer.parseInt(r.substring(30, 34)));
            case 5      -> new PrealtaTef(Pan.of(r.substring(2, 18)), r.substring(18, 34));
            case 99     -> new Trailer(Long.parseLong(r.substring(2, 10)));
            default     -> new Desconocido(tipo, r);
        };
    }
}

@Configuration
public class AltasBloqueosJobConfig {

    /** 00000000-PRINCIPAL: el TASKVALUE trae fecha (CAMD) y tipo de proceso. */
    @Bean
    Job core-depp176(JobRepository repo, Step inicio, JobExecutionDecider modo,
                 Step aplicaS111, Step avanzaFechaControl, Step aplicaTef, Step fin) {
        return new JobBuilder("CORE-DEPP176", repo)
                .start(inicio)                                         // 10000000-INICIO
                .next(modo)
                    .on("PREALTAS").to(aplicaS111).next(avanzaFechaControl).next(fin)
                .from(modo).on("TEF").to(aplicaTef).next(fin)
                .from(modo).on("NINGUNO").to(fin)                      // decomiso S111 activo u otro valor
                .end().build();
    }

    @Bean
    JobExecutionDecider modo() {
        return (job, step) -> {
            int tipo = job.getJobParameters().getLong("tipoProceso").intValue();   // WKS-PARAM-TIP-PROC
            boolean decomisoS111 = (Integer) job.getExecutionContext().get("decomisoS111") != 0;   // B02-USO-FUTURO-05
            if (tipo == 1 && !decomisoS111) return new FlowExecutionStatus("PREALTAS");
            if (tipo == 2) return new FlowExecutionStatus("TEF");
            return new FlowExecutionStatus("NINGUNO");
        };
    }
}

@Component
@StepScope
@RequiredArgsConstructor
public class SolicitudS111Processor implements ItemProcessor<SolicitudS111, ResultadoSolicitud> {

    private final CoberturaSucursal cobertura;     // 20103010-VALIDA-COBERTURA / S080
    private final PrealtaService prealtas;
    private final BloqueoService bloqueos;
    private final PrealtaTefService tef;
    private final ContadoresP176 cont;

    /** 20103000-LEO-DETALLE. */
    @Override
    public ResultadoSolicitud process(SolicitudS111 s) {
        return switch (s) {
            case SolicitudS111.Movimiento m -> {
                cont.detalles++;
                Optional<Integer> csi = cobertura.csiDe(m.sucPromotora());
                if (csi.isEmpty()) yield ResultadoSolicitud.rechazo(m, "SUCURSAL ERRONEA");
                yield switch (m.tipo()) {
                    case 1    -> prealtas.alta(m, csi.get());
                    case 2, 3 -> bloqueos.aplica(m);
                    case 4    -> prealtas.elimina(m);
                    default   -> throw new IllegalStateException();
                };
            }
            case SolicitudS111.PrealtaTef t -> {
                cont.totalTef++;
                yield t.numTef().chars().allMatch(ch -> ch == '0')
                        ? ResultadoSolicitud.omitida(t, "TEF EN CERO")      // WS-CONT-TEFCERO
                        : tef.valida(t);
            }
            case SolicitudS111.Trailer tr      -> ResultadoSolicitud.fin(tr);
            case SolicitudS111.Header h        -> ResultadoSolicitud.ignorado(h);
            case SolicitudS111.Desconocido d   -> ResultadoSolicitud.rechazo(d, "REGISTRO ERRONEO");
        };
    }
}
@Service
@RequiredArgsConstructor
public class PrealtaService {

    private final PrealtaRepository repo;        // CORE-DEPBD04TARJETAS.CORE-DEPB03PREALTAS
    private final Clock clock;

    /** 20103020-CGA-PREALTAS. Cada alta es una transacción independiente. */
    @Transactional(propagation = Propagation.REQUIRES_NEW)
    public ResultadoSolicitud alta(SolicitudS111.Movimiento m, int csi) {
        if (repo.existsById(m.tarjeta().value())) {
            // DMSII DUPLICATES: se libera el registro y se informa; no aborta.
            return ResultadoSolicitud.aviso(m, "TARJETA YA EXISTE");          // Pan se enmascara al imprimirse
        }
        try {
            repo.save(new Prealta(m.tarjeta().value(), m.contrato(), m.sucPromotora(), csi,
                                  Ctx.fechaLinea()));                        // B03P-FEC-PREALTA = B02-FECHA-LINEA
            return ResultadoSolicitud.aplicada(m);
        } catch (DataIntegrityViolationException carrera) {
            return ResultadoSolicitud.aviso(m, "TARJETA YA EXISTE");
        }
        // Cualquier otro error de base propaga y deja el step en FAILED (ERROR AL GRABAR B03P + DMTERMINATE).
    }
}
@Service
@RequiredArgsConstructor
public class BloqueoService {

    private static final int TIPO_BLOQUEO_S111 = 86;
    private final ContratoRepository contratos;      // CORE-DEPBD01CAPTACION.CORE-DEPB03CONTRATOS
    private final HistoricoRepository historicos;    // CORE-DEPB06HISTORICO
    private final BloqueoProperties props;           // producto 66, instrumento 11

    /** 20103030-CGA-BLOQUEO + 20103021-ACT-BLOQUEO. */
    @Transactional
    public ResultadoSolicitud aplica(SolicitudS111.Movimiento m) {
        Contrato c = contratos.findForUpdate(m.contrato())
                .orElse(null);
        if (c == null) return ResultadoSolicitud.aviso(m, "NO EXISTE REG EN B03 DE BD01");
        if (c.producto() != props.producto() || c.instrumento() != props.instrumento()) {
            return ResultadoSolicitud.aviso(m, "CONTRATO NO PERTENECE AL PRODUCTO -A-");
        }
        Historico h = historicos.findForUpdate(m.contrato())
                .orElseThrow(() -> new AvisoException("NO EXISTE REG EN B06 DE BD01"));
        if (m.tipo() == 2 && c.status() == 0) {                       // bloqueo
            c.setStatus(1);
            c.setTipoBloqueo(TIPO_BLOQUEO_S111);
            c.setFecCambioProducto(null);                             // MOVE ZEROS TO B03-FEC-CAMBPROD
            h.setFecBloqueo(Ctx.fechaLinea());                        // B06-FEC-BLOQUEO
        } else if (m.tipo() == 3 && c.status() == 1 && c.tipoBloqueo() == TIPO_BLOQUEO_S111) {
            c.setStatus(0);                                           // desbloqueo
            c.setTipoBloqueo(0);
            c.setFecCambioProducto(null);
            h.setFecBloqueo(null);
        } else {
            return ResultadoSolicitud.aviso(m, "CONTRATO YA ESTA BLOQUEADO/DESBLOQUEADO");
        }
        return ResultadoSolicitud.aplicada(m);   // B03 y B06 se graban juntos al confirmar la transacción
    }
}

            case 20   -> ResultadoSolicitud.rechazo(t, "NO EXISTE TEF");   // WS-TIPO-RECHA = 2
            default   -> ResultadoSolicitud.ignorado(t);                   // el COBOL no hace nada
        };
    }
}
Riesgos de migración
Número de tarjeta completo en bitácora y listados

Los mensajes de las líneas 968-972, 1004-1006, 1031-1034, 1055-1058, 1084-1087, 1123-1128, 1148-1151 y 1239-1242 escriben E01-S111-DET-TARJETA completo, y el listado de rechazos imprime WS-NUM-TAR PIC 9(16) (línea 1284). Es un hallazgo de PCI DSS; la versión Java enmascara el PAN por diseño.

Transacciones sin cerrar en duplicados

En 20103020-CGA-PREALTAS y 30100000-CGA-PREALTAS-TEF la rama de DUPLICATES ejecuta FREE pero no END-TRANSAC (líneas 1029-1035 y 1237-1243). El comportamiento depende de cómo 90400000-BEGIN-TRANSAC-NOAUD trata una transacción ya abierta, código que no viene en el fuente.

Conteo del trailer sin validar

La comparación de E01-S111-TRL-NUM-REG contra el número real de registros está comentada (línea 845). Un archivo incompleto con trailer se aplica sin aviso.

Subíndice fuera de rango en la búsqueda de BIN

El ciclo de 20103060-VALIDAR-CATBIN-S080 evalúa la tabla con índice 1001 (líneas 1185-1187). La definición activa de WKS-BIN-TBL no está en el fuente; la versión comentada (líneas 407-423) tiene 1000 ocurrencias.

Hardcodes

Producto 66 e instrumento 11 para bloqueos, tipo de bloqueo 86, medio de entrada 80 en B04P, fecha mínima 20070901 del parámetro y las rutas de archivo están compilados. Se trasladan a application.yml.

Pendiente de revisión SME
  • Confirmar el significado de B02-USO-FUTURO-05 como indicador de decomiso del S111 (WS-B02-DECOM-S111): el nombre del campo es genérico y el fuente no lo documenta.
  • Confirmar en el copy CORE-DEP/INC/WOR/25MTP004 la definición activa de WKS-BIN-TBL y su número de ocurrencias.
  • Revisar con SME el modo 2 (20103030-LEO-DETALLE-TEF y 80000000-BUSCA-CONTR-B03): no se incluyó en esta demostración y su condición anidada sobre I01-TEF-DET-CSI (líneas 981-995) compara contra WKS-B02-NUM-CSI y B02-NUM-CSI, dos campos distintos.
  • Confirmar que la eliminación tipo 04 debe borrar la prealta por número de contrato (90400003-B03PSXCTO-LOCK) y no por tarjeta; si un contrato tiene varias tarjetas en prealta solo se elimina la primera que encuentre el LOCK.
  • Validar qué hace 90400000-BEGIN-TRANSAC-NOAUD cuando ya hay una transacción abierta, para medir el impacto real del caso de duplicados.
