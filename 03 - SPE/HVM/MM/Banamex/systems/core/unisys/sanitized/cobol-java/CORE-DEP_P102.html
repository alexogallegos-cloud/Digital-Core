@Component
@RequiredArgsConstructor
public class FinDiaS408Tasklet implements Tasklet {

    private static final int FUNCION_FIN_DIA = 37;
    private static final int RS_NO_DISPONIBLE = 96;

    private final LineaCreditoClient s408;      // antes CALL "LINCRED" (S408L011LCRD)
    private final P102Properties props;          // reintentos: 4, espera: 5s

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        var req = SolicitudS408.control(FUNCION_FIN_DIA, 102, FechaLote.de(ctx), 30, 0);
        var retry = RetryTemplate.builder()
                .maxAttempts(props.reintentosS408() + 1)
                .fixedBackoff(props.esperaS408())                 // WAIT(05)
                .retryOn(S408NoDisponibleException.class)         // RS = 96
                .build();
        RespuestaS408 r = retry.execute(x -> {
            RespuestaS408 resp = s408.ejecuta(req);
            if (resp.status() == RS_NO_DISPONIBLE) throw new S408NoDisponibleException(resp);
            return resp;
        });
        if (r.status() != 0) {
            // ERR CALL(37) FIN DIA S408 + 70000050-MENSAJE
            throw new FinDiaS408Exception(r.status(), r.lineaError());
        }
        return RepeatStatus.FINISHED;
    }
}

@Component
@StepScope
@RequiredArgsConstructor
public class PagoLineaCreditoProcessor implements ItemProcessor<ContratoB03, ContratoB03> {

    private final ControlCaptacionRepository control;   // CORE-DEPB02CONTROL
    private final ModoPruebaProperties modo;
    private boolean hayPagosPendientes;                  // W77-HAY-PAGOLC

    @BeforeStep
    void leeBandera(StepExecution se) {
        hayPagosPendientes = control.findControl().pgoPendS408() == 1
                || modo.forzarPagosPendientes();              // WKS-CON-LCPEND
    }

    @Override
    public ContratoB03 process(ContratoB03 c) {
        // ... cargo pendiente de sábado, P010 y sello GCE (siempre)
        if (hayPagosPendientes && (c.indPendLc() > 0 || c.sdoPendiente().signum() > 0)) {
            pagaLineaCredito(c);                          // 50110010-SIGUE-ACT
        }
        return c.modificado() ? c : null;                 // WKS-SI-B03 = 1 -> STORE
    }
}
@Component
public class DisponibleCalculator {

    /**
     * 50111300-CALCULA-DISPONIBLE. WKS-SDO-DISPONIBLE es PIC 9(14)V99 sin signo.
     * B03-SDOPROT-BLOQ no se resta: ya viene descontado del saldo actual.
     */
    public BigDecimal disponible(ContratoB03 c) {
        BigDecimal d = c.sdoActual()
                .subtract(c.cgoPendSab())
                .subtract(c.autPendiente())
                .subtract(c.pcobpPendte())
                .subtract(c.sdoDaactTes())
                .subtract(c.sdoBloqGce())
                .setScale(2, RoundingMode.DOWN);          // COMPUTE sin ROUNDED
        if (c.sdoActual().signum() < 0) {
            log.warn("S/ACT NEG ENT/P102 CTO {} S/ACT={} S/PEN={} S/BLOQ={}",
                     c.numContrato(), c.sdoActual(), c.sdoPendiente(), c.sdoProtBloq());
        }
        return d.max(BigDecimal.ZERO);                    // IF WKS-SDO-DISPONIBLE < 0 MOVE 0
    }
}
/** 50112050-PAGA-SDOLC: abono del disponible a la línea de crédito vía S408. */
private RespuestaS408 llamaPago(ContratoB03 c, BigDecimal disponible, Contexto ctx) {
    var req = SolicitudS408.builder()
            .funcion(35).paso(102)
            .fechaProceso(ctx.fechaBase())
            .sucursalOperacion(c.sucPromotora())
            .caja(1)
            .importe(disponible)                 // no se descuenta B03-SDOPROT-BLOQ
            .tipoProceso(30).subproceso(0)
            .autorizacionCORE-DEP(ctx.miAutorizacion())
            .fechaHoraTransaccion(ctx.inicioMaquina())
            .build();
    return retryS408.execute(x -> {              // RS 96: reintento con espera de 5 s
        RespuestaS408 r = s408.ejecuta(req);
        if (r.status() == 96) throw new S408NoDisponibleException(r);
        return r;
    });
}

private void pagaLineaCredito(ContratoB03 c) {
    BigDecimal disponible = calculadora.disponible(c);
    if (disponible.signum() <= 0) return;        // 50112000: solo con disponible
    RespuestaS408 r = llamaPago(c, disponible, ctx);
    if (r.status() == 0) aplicaPago(c, disponible, r);
    else errorPago(c, disponible, r);            // 50112100: ajusta IND-LINCREDIT e IND-PENDLC
}
/** 50112110-APLICA-SDOLC. Importes S9(13)V99 -> BigDecimal escala 2. */
private void aplicaPago(ContratoB03 c, BigDecimal disponible, RespuestaS408 r) {
    BigDecimal pago = disponible.subtract(r.impExcedido());       // WKS-PAGO-S408
    BigDecimal aLinea, aSaldoAut;                                  // -LC y -ST
    if (pago.signum() > 0) {
        if (r.impAut().signum() > 0) {
            aSaldoAut = r.impAut();
            aLinea = pago.subtract(r.impAut());
        } else {
            aLinea = pago;
            aSaldoAut = BigDecimal.ZERO;
        }
    } else {
        aLinea = aSaldoAut = BigDecimal.ZERO;
        if (pago.signum() < 0) {
            log.error("ERR PAG/LC/S408 RS={} CTO={} PGO={} EXE={}", r.status(), c.numContrato(),
                      pago, r.impExcedido());
            pago = BigDecimal.ZERO;
        }
    }
    if (aLinea.signum() > 0 || aSaldoAut.signum() > 0) {
        movimientos.abonoS408(c, aLinea, aSaldoAut);   // WKS-TOT-ABO-S408-LC / -ST
        c.restaSaldoActual(pago);                     // SUBTRACT WKS-PAGO-S408 FROM B03-SDO-ACTUAL
    }
    if (r.resExist() != 99 && r.resExist() != c.indLinCredit()) {
        c.setIndLinCredit(r.resExist());             // S408 corrige el indicador del contrato
    }
    if (c.indPendLc() > 0 && (r.saldo().signum() == 0 || disponible.signum() == 0)) {
        c.setIndPendLc(0);                           // ya no hay pendiente que pagar
    }
}

Riesgos de migración
IF de pagos pendientes comentado

En 50000000-PROCESO-PRINCIPAL la condición IF W77-HAY-PAGOLC = 1 está comentada (línea 2069), así que el barrido completo de B03 corre en cada ejecución. El pago solo se condiciona por contrato en la línea 2508.

Disponible sin signo

WKS-SDO-DISPONIBLE es PIC 9(14)V99 sin signo (línea 669). Si el estándar se cumple, un disponible negativo se guarda como valor absoluto y el piso en cero de la línea 2586 no actúa, lo que podría generar un pago a la línea de crédito mayor al saldo real del contrato.

Forzador de pagos admitido en Producción

00001000-INICIO-PGM solo rechaza en Producción los forzadores de librería (líneas 1143-1151). WKS-CON-LCPEND = 1 en el TASKVALUE fuerza pagos pendientes en Producción.

Alto de operador y fin de día no garantizado

Tras cuatro respuestas RS 96 de S408 el programa hace STOP y espera a consola (líneas 2648 y 2020). Un status distinto de 0 y 96 en la función 37 se registra pero el programa termina normalmente sin fin de día a S408.

Hardcodes

Hora límite 1901 para cargos pendientes de sábado (línea 2479), caja 01, funciones 35 y 37, tipo de proceso 30, claves 1004 y 1098, y los nodos 50002E01, 50099E01 y 50003S02 están compilados. Se trasladan a application.yml.
