CORE-DEP · P197 a Java 21 Spring Boot

P197 recibe por contenido un parámetro de 30 posiciones con la opción (01 a 99 y familias alfanuméricas), la fecha AAAAMMDD y datos complementarios. Con el HOSTNAME de la propia tarea deduce si corre en desarrollo, SIT, UAT o producción y cuál es el host y CSI destino, y lee de CORE-DEPB02CONTROL si hay envío al S274. Según la familia de la opción arma un comando ADD para la librería XFER, lanza un WFL con SYSTEM/DUMPALL a cinta o envía el archivo por INTELARSND con reintentos. El resultado de XFER o DUMPALL se devuelve en el TASKVALUE de la tarea.

/** WKS-PARAM-ENT: "OO AAAAMMDD /NN ..." recibido por contenido desde el WFL. */
public record SolicitudTransferencia(String opcion, LocalDate fecha,
                                     Optional<Integer> consecutivo, String extra) {

    public enum Familia { HOST_MTY_VDM, HOST_VDM_MTY, DUMPALL_CINTA, INTELAR, AL_S274,
                          EXTRACCION_E, OTRA }

    private static final Set<String> OPC1 = rango("01", "27");                       // MTY a VDM
    private static final Set<String> OPC2 = Set.of("40","41","51","52","53","54","55","56","57","58");
    private static final Set<String> OPC3 = rango("81", "89");                       // DUMPALL
    private static final Set<String> OPC5 = Set.of("30", "31");                     // hacia S274

    public Familia familia() {
        if (OPC1.contains(opcion)) return Familia.HOST_MTY_VDM;
        if (OPC2.contains(opcion)) return Familia.HOST_VDM_MTY;
        if (OPC3.contains(opcion)) return Familia.DUMPALL_CINTA;
        if (OPC5.contains(opcion)) return Familia.AL_S274;
        return Familia.INTELAR;     // OPC4, A, B, C, D, E, M, P, R: 7003000-ENVIA-S022
    }

    /** Los nombres de archivo usan PIC 9(06): un MOVE desde 9(08) deja AAMMDD. */
    public String fechaAammdd() { return fecha.format(DateTimeFormatter.ofPattern("yyMMdd")); }
}

@ConfigurationProperties("p197")
public record P197Properties(Map<String, Dumpall> dumpall, Map<String, String> intelar) {

    public record Dumpall(String job, String archivo, String pack, String cinta) {
        /** Reproduce "RUN SYSTEM/DUMPALL ("COPY ... ON pack TO UL (KIND=TAPE,TITLE=...)")". */
        public String comando(String aammdd) {
            return "BEGIN JOB  %s; RUN SYSTEM/DUMPALL (\"COPY %s ON %s TO UL (KIND=TAPE,TITLE=%s)\")"
                    .formatted(job, archivo.replace("{aammdd}", aammdd), pack,
                               cinta.replace("{aammdd}", aammdd));
        }
    }
}
/** Topología que P197 deduce de ATTRIBUTE HOSTNAME OF MYSELF. */
@ConfigurationProperties("p197.topologia")
public record TopologiaHosts(Map<String, Host> hosts) {

    public enum Ambiente { DESA, SIT, UAT, PROD }
    public record Host(Ambiente ambiente, int csiOrigen, int csiDestino, String hostDestino) {}

    /** Un host no registrado se trata como UAT y sin transferencia (W77-SIN-TRANSF = 1). */
    public Optional<Host> resuelve(String hostname) {
        return Optional.ofNullable(hosts.get(hostname.replace(".", "").trim()));
    }
}

@Component
public class ValidadorSolicitud {

    private final int anioMin, anioMax;   // hoy compilados: > 2016 y < 2061

    public ValidadorSolicitud(@Value("${p197.anio-min:2017}") int anioMin,
                              @Value("${p197.anio-max:2060}") int anioMax) {
        this.anioMin = anioMin; this.anioMax = anioMax;
    }

    /** 1000200-VALIDA-PARAM: opción conocida y fecha AAAAMMDD válida. */
    public SolicitudTransferencia valida(String paramWfl) {
        var s = SolicitudParser.parse(paramWfl);          // lanza si la fecha no es numérica
        int anio = s.fecha().getYear();
        if (!OpcionesConocidas.contiene(s.opcion()) || anio < anioMin || anio > anioMax) {
            throw new ParametroInvalidoException("ERR PARAM RECIBIDO <" + paramWfl + ">");
        }
        if ("96".equals(s.opcion()) && s.consecutivo().isEmpty()) {
            throw new ParametroInvalidoException("OPCION 96 FECHA AAAAMMDD , CONSECUTIVO /nn");
        }
        return s;
    }
}

@Bean
IntegrationFlow transferencia(ValidadorSolicitud v, TopologiaHosts topo) {
    return IntegrationFlow.from("p197.entrada")
            .transform(String.class, v::valida)
            .enrichHeaders(h -> h.headerFunction("host", m -> topo.resuelve(hostname())))
            .route(SolicitudTransferencia.class, SolicitudTransferencia::familia, r -> r
                    .channelMapping(Familia.HOST_MTY_VDM, "xfer")
                    .channelMapping(Familia.HOST_VDM_MTY, "xfer")
                    .channelMapping(Familia.AL_S274, "xferS274")
                    .channelMapping(Familia.DUMPALL_CINTA, "dumpall")
                    .defaultOutputChannel("intelar"))
            .get();
}
@Bean
IntegrationFlow xferS274(XferGateway xfer, Control274 control) {
    // OPC5 solo transfiere si B02-NVOHOST-S274 habilita el host del ambiente, o si la opción es 31
    return IntegrationFlow.from("xferS274")
            .filter(SolicitudTransferencia.class,
                    s -> control.envioAlS274Habilitado() || "31".equals(s.opcion()),
                    f -> f.discardChannel("nullChannel"))       // NEXT SENTENCE: no hace nada
            .handle(xfer, "envia")
            .get();
}

@Service
@RequiredArgsConstructor
public class XferAdapter {
    private final FileTransferClient transferencias;   // sustituye ENVIA_XFER IN XFER
    private final Control274 control;

    @ServiceActivator(inputChannel = "xfer")
    public int envia(SolicitudTransferencia s, ComandoXfer cmd) {
        if ("17".equals(s.opcion()) && control.indicadorEnviosS711() > 0) {
            return 0;                                   // la opción 17 se omite si B02-IND-ENVIOS711 > 0
        }
        int rs = transferencias.copia(cmd);
        if (rs > 0) {
            log.error(" ERR CALL LIB XFER RS={} TRANSFERIR MANUAL: {}", rs, cmd.texto());
        } else {
            log.info(" ENVIO {}{} {}({})", cmd.usuarioDestino(), cmd.archivoDestino(),
                     cmd.hostDestino(), cmd.packDestino());
        }
        return rs;                                      // W77-SAL-RESULT -> TASKVALUE
    }
}
@Service
@RequiredArgsConstructor
public class DumpallAdapter {
    private final P197Properties props;
    private final OrquestadorClient orquestador;    // sustituye CALL SYSTEM WFL

    @ServiceActivator(inputChannel = "dumpall")
    public int lanza(SolicitudTransferencia s) {
        var plantilla = props.dumpall().get(s.opcion());
        if (plantilla == null) return 9;              // opción fuera de 81..89
        var ejecucion = orquestador.lanza(plantilla.comando(s.fechaAammdd()));
        // El COBOL devuelve 0 sin esperar ni revisar el WFL; aquí se espera el resultado.
        return ejecucion.esperaResultado(Duration.ofMinutes(30)).exitoso() ? 0 : 1;
    }
}

public class IntelarAdapter {

    private final IntelarClient intelar;             // INTELARSND IN ADMONXFERS
    private final ArchivoResolver archivos;          // ATTRIBUTE RESIDENT

    private final RetryTemplate reintento = RetryTemplate.builder()
            .maxAttempts(2)                            // UNTIL WKS-INTENTOS = 2
            .fixedBackoff(60_000)                      // WAIT (60)
            .retryOn(IntelarException.class)
            .build();

    /** 9000000-ENVIA-INTELAR. */
    public void envia(EnvioIntelar e) {
        try {
            reintento.execute(ctx -> {
                int rs = intelar.envia(e.archivo());
                if (rs != 0) throw new IntelarException(rs);
                log.info("ARCHIVO {} ENVIADO", e.nombre());
                return null;
            });
        } catch (IntelarException ex) {
            // Mensaje al operador; el TASKVALUE no cambia y el paso termina normal.
            log.error("ERROR AL ENVIAR ARCHIVO {} RS={} TRANSFERIR MANUAL OPC={}",
                      e.nombre(), ex.rs(), e.opcionManual());
        }
    }

    /** 9000001-ENVIA-O-RECHAZA: solo envía si el archivo existe. */
    public void enviaSiResidente(EnvioIntelar e) {
        if (archivos.existe(e.archivo())) envia(e);
        else log.error("ERROR AL ENVIAR ARCHIVO {} NO RESIDENTE", e.nombre());
    }
}
