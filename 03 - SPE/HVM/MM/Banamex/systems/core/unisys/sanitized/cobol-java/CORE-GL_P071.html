@Component
@RequiredArgsConstructor
class SistemasHabilitados {

    private static final int MAX_SISTEMAS = 50;          // WKS-SISTEMAS-RESIDENTES OCCURS 50
    private final LibControlClient libControl;            // B04SISTEM, CONSISMEN, CONSISDIA
    private final SistemasPunteoProperties props;
    @Value("${p071.csi-host}") private int csiHost;
    private List<SistemaHabilitado> tabla;

    @PostConstruct
    void carga() {
        tabla = libControl.sistemas().stream()            // B04SISTEM FUNCION 02 hasta fin
                .filter(s -> props.sistemasPunteo().contains(s.numero()))
                .flatMap(s -> libControl.mensual(s.numero())
                        .flatMap(m -> libControl.diario(s.numero()).map(d -> new Datos(s, m, d)))
                        .stream())
                .filter(x -> x.sistema().estatus() == 3 && resideAqui(x.mensual()))
                .map(x -> new SistemaHabilitado(x.sistema().numero(), x.diario().fecCon(),
                        x.mensual().modSis(), x.mensual().resCsi()))
                .limit(MAX_SISTEMAS)                       // IF W77-IND-SIS < 51
                .toList();
    }

    /** MODSIS 1: reside solo en su CSI. MODSIS 0: CSI 0 (todos) o el propio nodo. */
    private boolean resideAqui(SistemaMensual m) {
        return (m.modSis() == 1 && m.resCsi() == csiHost)
            || (m.modSis() == 0 && (m.resCsi() == 0 || m.resCsi() == csiHost));
    }
}

@Component
class FiltroMensajeSa2 {

    enum Resultado { PROCESA, RECHAZA, FIN, ERROR }

    Resultado evalua(int statusKey, EncabezadoSa2 sa2) {
        if (statusKey == 99) return Resultado.FIN;                    // MOVE 1 TO W77-FIN
        if (!Set.of(0, 89, 92, 93).contains(statusKey)) return Resultado.ERROR;

        // Fuente: SISTEMA = 151 AND CODSERV = 1 AND SUBCODSERV = 1 OR 2
        // COBOL lo evalúa como (151 AND 1 AND SUB = 1) OR (SUB = 2).
        boolean comoEnFuente = (sa2.sistema() == 151 && sa2.codServ() == 1 && sa2.subCodServ() == 1)
                            || sa2.subCodServ() == 2;
        boolean comoSeQuiso  = sa2.sistema() == 151 && sa2.codServ() == 1
                            && (sa2.subCodServ() == 1 || sa2.subCodServ() == 2);
        if (comoEnFuente != comoSeQuiso) {
            log.warn("Mensaje SA2 aceptado solo por la precedencia del fuente: {}", sa2);
        }
        return comoEnFuente ? Resultado.PROCESA : Resultado.RECHAZA;   // RECHAZA: result 2, long 24
    }
}

class ConsultaTotalesService {

    private final SistemasHabilitados sistemas;
    private final TotalesClientClassifier librerias;
    private final SistemasPunteoProperties props;

    RespuestaTotales consulta(SolicitudTotales s) {
        var resp = RespuestaTotales.nueva(s.nivel(), props.renglonesPorRespuesta());
        for (SistemaHabilitado sis : sistemas.posterioresA(s.sistemaContinuacion())) {   // > SISCON
            if (resp.llena()) break;                                      // W77-IND > 25
            ResultadoL011 r = librerias.classify(sis.numero()).totales(s.conFecha(s.fecha()));
            resp.agrega(sis, r);                                           // 030321-RECIBE-RESPLIB-FUN1/2
        }
        if (resp.vacia()) {
            return RespuestaTotales.sinDatos(s.nivel());                   // result 2, error(1) = 1
        }
        return resp;
    }
}

public interface TotalesCiClient {
    ResultadoL011 totales(ParametrosL011 p);    // TOTALESCI IN L011-Snnn
}

@Component
class TotalesClientClassifier implements Classifier<Integer, TotalesCiClient> {

    private final Map<Integer, TotalesCiClient> porSistema;

    TotalesClientClassifier(Map<String, TotalesCiClient> beans) {   // beans "l011-s017", "l011-s018", ...
        this.porSistema = beans.entrySet().stream().collect(Collectors.toUnmodifiableMap(
                e -> Integer.parseInt(e.getKey().substring("l011-s".length())), Map.Entry::getValue));
    }

    @Override
    public TotalesCiClient classify(Integer sistema) {
        // Sistema sin librería: el fuente devuelve 3 (MOVE 3 TO W77-RESULT-L011).
        return porSistema.getOrDefault(sistema, p -> ResultadoL011.error(3));
    }
}

Riesgos de migración
Filtro de destino con precedencia de AND sobre OR

La condición de las líneas 1614-1616 acepta cualquier mensaje con subservicio 2, sin importar el sistema o servicio de destino.

Código de error truncado a un dígito

W77-RESULT-L011 (PIC S9(11) BINARY) se mueve a campos PIC 9(01) en las líneas 1725, 1887 y 1909. Errores 10, 20 o mayores pueden llegar al canal como cero.

Contadores truncados

Los contadores de cargos y abonos pasan de PIC 9(08) en la librería a PIC 9(06) en la respuesta sin ON SIZE ERROR.

Tabla de sistemas con tope silencioso

La tabla de sistemas residentes tiene 50 posiciones; los sistemas que excedan ese número se descartan sin aviso.

Hardcodes

Lista de 18 sistemas de punteo, 25 renglones por respuesta, encabezados de 48 y 77 bytes, el sistema 711 como fin de continuación y la cadena de 18 librerías L011. Se trasladan a application.yml y a beans.
