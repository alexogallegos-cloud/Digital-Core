CORE-GL · L012 a Java 21 Spring Boot

L012 se congela con FREEZE(CONTROL, CONTROL_LIB) y expone tres puntos de entrada: FUNCIONL12 para comandos de operación, ENVIA_CD para el resultado que devuelve TANDEM y CONFIG_DISPERSION para ajustar el bloque de envío y la espera entre ciclos. En cada ciclo lee los movimientos del archivo CTDCORE-DEP, filtra los elegibles por tipo de proceso, función e indicador de envío a GlobalDirect, y por cada ocurrencia contable autorizada en el catálogo de claves arma el detalle MT942 con indicador DR, CR, RD o RC. Cada mensaje queda registrado en el dataset DMSII CORE-GLB04CTLGLOBALDIR con estatus 0, se actualiza con el resultado del depósito (99 o resultado más 10) y después con la respuesta de TANDEM (1 aceptado, 2 rechazado, 3 duplicado). Una pasada de reenvío recorre la bitácora y vuelve a mandar todo lo que esté en estatus de error.

/** Librería L012: misma superficie que exportaba (FUNCIONL12, ENVIA_CD, CONFIG_DISPERSION). */
@Service
@RequiredArgsConstructor
public class GlobalDirectBridge implements SmartLifecycle {

    private final DispersionProperties props;
    private final ArranqueBridge arranque;          // VERSIONES + VERIFICA_B05 + INICIA
    private final DespachadorMt942 despachador;     // BUSCA_MOVTOS / ENVIA_MOVIMIENTOS
    private final ReenvioBitacora reenvio;          // pasada de reenvío del evento 1
    private final TaskScheduler scheduler;

    private volatile int bloqueEnvios;              // BLOCK_ENVIOS
    private volatile Duration espera;               // TMPO_ESPERA
    private ScheduledFuture<?> ciclo;
    private volatile boolean running;

    @Override
    public void start() {
        arranque.verificaInstanciaUnica();          // VERIFICA_B05: otra L012 viva aborta el arranque
        var estado = arranque.inicia();             // INICIA: día contable, catálogos, archivos
        bloqueEnvios = props.bloquePorEquipo()
                .getOrDefault(estado.equipo(), props.bloqueDefault());   // VDMALFA./MONBETA. = 300, resto = 150
        espera = estado.espera();
        ciclo = scheduler.scheduleWithFixedDelay(this::cicloControl, espera); // WAITANDRESET(TMPO_ESPERA, ...)
        running = true;                              // FREEZE(CONTROL, CONTROL_LIB)
    }

    @Override
    public void stop() {                             // BAJA: persiste tuning y cierra archivos
        ciclo.cancel(false);
        arranque.baja(bloqueEnvios, espera);
        running = false;
    }

    @Override public boolean isRunning() { return running; }
}
    /**
     * CONFIG_DISPERSION(BLKENV, TESP). Solo cambia los valores positivos;
     * el original no asigna valor de retorno (INTEGER PROCEDURE sin resultado).
     */
    public void configDispersion(int bloque, int esperaSegundos) {
        if (bloque > 0) {
            this.bloqueEnvios = bloque;
        }
        if (esperaSegundos > 0) {
            this.espera = Duration.ofSeconds(esperaSegundos);
            ciclo.cancel(false);
            ciclo = scheduler.scheduleWithFixedDelay(this::cicloControl, espera);
        }
        log.info("Dispersión: {} envíos por ciclo, espera {}", bloqueEnvios, espera);
    }

@Component
@RequiredArgsConstructor
public class SalvoBuenCobroPolicy {

    private final SbcProperties sbc;   // l012.sbc.claves: [3009, 3036, 2005, 2006, 2007]
                                       // l012.sbc.claves-post-liberacion: [2005, 2006, 2007]

    /**
     * VALIDA_CVE(CVE, ECTA, CONTA). ECTA y CONTA se reciben en la macro pero no intervienen.
     * Fuera del ciclo SBC se excluyen las claves SBC; durante el ciclo solo pasan ellas;
     * ya liberado el SBC, las tres claves de cargo vuelven a ser elegibles.
     */
    public boolean admite(int clave, EstadoSbc estado) {
        boolean esSbc = sbc.claves().contains(clave);
        if (!estado.liberandoSbc() && !esSbc) {
            return true;
        }
        if (estado.liberandoSbc() && esSbc) {
            return true;
        }
        return estado.sbcLiberado() && sbc.clavesPostLiberacion().contains(clave);
    }
}

@Component

public class ElegibilidadMovimiento {

    private static final Set<Integer> CARGO_POR_CLAVE = Set.of(2005, 2006, 2007);
    private final CatalogoClaves catalogo;          // ACLAVES, cargado de S080 con L710_CONSUL_DETALLE
    private final SalvoBuenCobroPolicy sbc;

    /** Primer IF de ENVIA_MOVIMIENTOS: tipo de proceso, función 1 e indicador de envío GlobalDirect. */
    public boolean movimientoElegible(MovimientoCaptacion m) {
        return m.tipoProceso() < 20
                && m.funcion() == 1
                && (m.indicadorEnvioCd() == 2 || m.indicadorEnvioCd() == 3);   // ISOLATION BNE 2023 R10
    }

    /** WHILE INDICE < 6 AND CLAVE_BNX > 0: ocurrencias 1..5, se corta en la primera clave 0. */
    public List<OcurrenciaGlobalDirect> ocurrencias(MovimientoCaptacion m, EstadoSbc estado) {
        var salida = new ArrayList<OcurrenciaGlobalDirect>();
        for (var oc : m.ocurrencias()) {                     // ya limitado a 5
            if (oc.clave() <= 0) break;
            var cat = catalogo.get(oc.clave());
            if (cat.indicadorGlobal() != 1 && cat.indicadorGlobal() != 2) continue;   // descarte silencioso
            if (!sbc.admite(oc.clave(), estado)) continue;
            boolean cargo = cat.afectacionContable() == 1 || CARGO_POR_CLAVE.contains(oc.clave());
            salida.add(new OcurrenciaGlobalDirect(oc.indice(), oc.clave(), cargo,
                    oc.importe()));                          // IDFENT_OCUR + IDFDEC_OCUR -> BigDecimal escala 2
        }
        return salida;
    }
}
/** Tipo de movimiento SWIFT del tag :61: según alta o eliminación y cargo o abono. */
public enum IndicadorMovimiento {
    DR, CR, RD, RC;

    static IndicadorMovimiento de(int statusMovimiento, boolean cargo) {
        boolean alta = statusMovimiento == 1;           // IDFSTATUS = 1; cualquier otro valor es eliminación
        if (alta) return cargo ? DR : CR;
        return cargo ? RD : RC;
    }
}

@Service
@RequiredArgsConstructor
public class RespuestaTandemService {

    private final BitacoraEnvios bitacora;
    private final EstadoDespacho despacho;      // ULTIMO_MSG_ENV, RECHAZADO

    @Transactional
    public void enviaCd(int resultado, int numSecuencia, LocalDate fechaProceso) {
        if (numSecuencia > despacho.ultimoMensajeEnviado()) {
            // El ALGOL solo descarta si además está prendido el monitor; aquí se descarta siempre.
            metricas.respuestaFueraDeRango().increment();
            log.warn("Secuencia {} mayor a la última enviada {}", numSecuencia,
                     despacho.ultimoMensajeEnviado());
            return;
        }
        switch (resultado) {
            case 1 -> bitacora.actualizaEstatus(fechaProceso, numSecuencia, 1);      // aceptado
            case 2 -> {                                                                  // rechazado
                despacho.marcaRechazo();                                                 // RECHAZADO := TRUE
                bitacora.actualizaEstatus(fechaProceso, numSecuencia, 2);
            }
            case 3, 4 -> bitacora.actualizaEstatus(fechaProceso, numSecuencia, 3);   // duplicado
            default -> metricas.respuestaNoContemplada(resultado).increment();       // sin cambio de estatus
        }
    }
}

public class ReenvioBitacora {

    private static final Set<Integer> REENVIABLES = Set.of(12, 11, 2, 0, 99);   // "ERROR EN TRANSF."
    private final ControlEnvioRepository repo;
    private final DespachadorMt942 despachador;
    private final ReenvioProperties props;      // l012.reenvio.ciclos-sin-respuesta: 150, pausa: 20s, max-reintentos

    /** Se dispara al completar 150 ciclos (TOT_ENVIOS) o cuando ENVIA_CD reportó un rechazo. */
    @Transactional
    public void reenvia(LocalDate fechaProceso, int sistema, int bloque) throws InterruptedException {
        Thread.sleep(props.pausa().toMillis());                      // WAIT((20))
        int enBloque = 0;
        try (Stream<ControlEnvioGlobalDirect> pendientes =
                 repo.streamByFechaEnvioOrderByCtlEnvio(fechaProceso)) {   // SET B04SXCTLENVIO
            for (var reg : (Iterable<ControlEnvioGlobalDirect>) pendientes::iterator) {
                if (despachador.desligado()) break;
                if (!REENVIABLES.contains(reg.estatus())) continue;
                if (reg.reintentos() >= props.maxReintentos()) {     // no existe en el ALGOL
                    alertas.intervencionManual(reg);
                    continue;
                }
                despachador.reenvia(reg);                            // relee el registro físico CTLFILE
                reg.incrementaReintentos();                          // ACTZA_BIFIN(1, REINTENTOS, ...)
                if (++enBloque >= bloque) { enBloque = 0; Thread.sleep(1_000); }
            }
        }
    }

Riesgos de migración
Reintentos sin techo

REINTENTOS se incrementa en la línea 7056 y se graba en la bitácora, pero ningún punto del fuente lo compara contra un máximo. Un mensaje que nunca llega a estatus 1 se reenvía en cada pasada de forma indefinida.

Validación de secuencia condicionada al monitor

En ENVIA_CD (líneas 4399-4400) la respuesta con número de secuencia mayor al último enviado solo se descarta si el monitor está prendido. Con el monitor apagado actualiza la bitácora con un número de mensaje que L012 no ha mandado.

Reenvío de mensajes posiblemente entregados

El filtro de reenvío (líneas 7042-7045) incluye los estatus 0 y 99, que no son errores: 99 significa que la TCP Carrier aceptó el mensaje. La protección contra duplicados depende de TANDEM.

Ciclos de espera sin límite

WHILE CREA_BIFIN > 0 DO WAIT((.01)) en la línea 6371 y la espera del candado ACT_BIFIN en OBTEN_EVE no tienen tope; un error persistente de DMSII congela la tarea de control.

Tuning persistido que no se aplica

INICIA lee B00-GLO-TIEMESP y B00-GLO-SISTGLOBAL (líneas 4941-4942) pero fija TMPO_ESPERA en 1 (línea 4944) y el bloque principal sobrescribe BLOCK_ENVIOS por nombre de equipo (líneas 7132-7136).

Hardcodes de infraestructura

Nombres de equipo ACYPGAMA., MONALFA., VDMALFA. y MONBETA., nodos destino 21 y 22, bloques 300 y 150, umbral de 150 ciclos, pausa de 20 segundos y las cinco claves de Salvo Buen Cobro están compilados; se trasladan a application.yml.
