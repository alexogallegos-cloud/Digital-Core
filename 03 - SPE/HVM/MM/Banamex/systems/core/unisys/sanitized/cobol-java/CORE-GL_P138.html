CORE-GL · P138 a Java 21 Spring Boot

P138 carga el archivo de control (CORE-GL)CORE-GL/FILE/CONTROL/CORP, que trae en el header la fecha de proceso y en el detalle la última fecha reportada de cada sistema, y procesa uno o todos los sistemas (parámetro 999). Para cada fecha hábil desde la siguiente a la última reportada hasta la del header, verifica que existan los archivos E01 de los CSI 04 y 10 (solo el 10 para S701), los concatena, los ordena por moneda y cuenta, y emite un reporte con cortes por subcuenta, cuenta y moneda, con saldo anterior, cargos, abonos y saldo actual. Al terminar cada fecha reescribe en el control la fecha reportada. Si un archivo no existe, pregunta al operador si continúa o reintenta.

@Component
@RequiredArgsConstructor
class PlanPosicionGlobal {

    private final ControlCorpRepository control;         // (CORE-GL)CORE-GL/FILE/CONTROL/CORP
    private final CalendarioBancario calendario;

    /** 000000-INICIO: arma la lista de (sistema, fecha inicial) a procesar. */
    List<Pendiente> plan(int sistemaParametro, Optional<LocalDate> fechaTaskvalue) {
        ControlCorp c = control.lee();                   // header + detalle hasta el sistema 999
        Stream<ControlCorp.Detalle> sistemas = (sistemaParametro == 999)
                ? c.detalles().stream()
                : c.detalles().stream().filter(d -> d.sistema() == sistemaParametro);
        List<Pendiente> plan = sistemas
                .map(d -> new Pendiente(d.sistema(),
                        fechaTaskvalue.filter(f -> sistemaParametro != 999)   // 000150-FECHA-UNO
                                      .orElse(calendario.siguienteHabil(d.ultimaFecha())),
                        c.fechaProceso(), fechaTaskvalue.isPresent()))
                .toList();
        if (plan.isEmpty()) {
            throw new ProcesoAbortadoException("EL P138 NO PROCESA EL SISTEMA: " + sistemaParametro);
        }
        return plan;
    }

    record Pendiente(int sistema, LocalDate desde, LocalDate hasta, boolean soloUnaFecha) {}
}
/** 000050-UNO repetido UNTIL W88-FECHA-FIN, como iterador de fechas hábiles. */
static Stream<LocalDate> fechas(PlanPosicionGlobal.Pendiente p, CalendarioBancario cal) {
    if (p.soloUnaFecha()) {
        return p.desde().isAfter(p.hasta()) ? Stream.empty() : Stream.of(p.desde());
    }
    return Stream.iterate(p.desde(), f -> !f.isAfter(p.hasta()), cal::siguienteHabil);
}

@Component
@StepScope
@RequiredArgsConstructor
class FechaPosicionTasklet implements Tasklet {
    private final ArchivosCsi archivos;
    private final ReportePosicion reporte;
    private final ControlCorpRepository control;

    @Value("#{jobExecutionContext['sistema']}") private Integer sistema;
    @Value("#{jobExecutionContext['fecha']}")   private LocalDate fecha;

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        if (archivos.disponibles(sistema, fecha)) {        // 001000-INICIAL: W77-RESP = 3
            reporte.genera(sistema, fecha);                 // 002000-PROCESO
            control.registraFecha(sistema, fecha);          // 008100-ACT-FECHAP
        } else {
            reporte.sinInformacion(sistema, fecha);         // 001000-PROCESO: no se actualiza el control
        }
        return RepeatStatus.FINISHED;
    }
}
@Component
@RequiredArgsConstructor
class CalendarioBancario {
    private final DiaInhabilRepository inhabiles;   // reemplaza THECALENDAR IN LOCSUP

    /** Función 13, formato 12 (AAAAMMDD), incremento "00000001": siguiente día hábil. */
    LocalDate siguienteHabil(LocalDate f) {
        LocalDate d = f.plusDays(1);
        while (d.getDayOfWeek() == DayOfWeek.SATURDAY || d.getDayOfWeek() == DayOfWeek.SUNDAY
                || inhabiles.existsById(d)) {
            d = d.plusDays(1);
        }
        return d;
    }
}
@Repository
@RequiredArgsConstructor
class ControlCorpRepository {
    private final ControlCorpStore store;        // archivo CORP hoy; tabla de control en destino

    /** 008100-ACT-FECHAP: busca el detalle del sistema y reescribe su fecha. */
    @Transactional
    public void registraFecha(int sistema, LocalDate fecha) {
        ControlCorp.Detalle d = store.detalles().stream()
                .takeWhile(x -> x.sistema() != 999)
                .filter(x -> x.sistema() == sistema)
                .findFirst()
                .orElseThrow(() -> new ProcesoAbortadoException(
                        "NO SE ENCONTRO REGISTRO DE CONTROL DEL S" + sistema));
        store.reescribe(d.conFecha(fecha));      // REWRITE ... INVALID KEY -> STATUS -1
    }
}

@Component
@RequiredArgsConstructor
class ArchivosCsi {
    private final P138Properties props;

    List<String> titulos(int sistema, LocalDate fecha) {
        return props.csiPara(sistema).stream()
                .map(csi -> props.archivoSuma()
                        .replace("{sistema:000}", "%03d".formatted(sistema))
                        .replace("{csi:00}", csi)
                        .replace("{fecha:yyMMdd}", fecha.format(DateTimeFormatter.ofPattern("yyMMdd"))))
                .toList();
    }
}
enum Disponibilidad { DISPONIBLE, CONTINUAR_SIN_ARCHIVO }   // W77-RESP = 3 / 1

@Component
@RequiredArgsConstructor
class ResidenciaArchivo {
    private final ResourceLoader rl;
    private final P138Properties props;     // p138.sin-archivo: CONTINUAR | REINTENTAR | FALLAR

    /** 8000-CHECA-RESIDENCIA + 8500-VALIDA-RESIDENCIA sin ACCEPT de consola. */
    Disponibilidad verifica(String titulo) throws InterruptedException {
        Resource r = rl.getResource(titulo);
        int intentos = 0;
        while (!r.exists()) {
            log.warn("NO EXISTE ARCHIVO {} (C) CONTINUO o (R) REINTENTO", titulo);
            switch (props.sinArchivo()) {
                case CONTINUAR -> { return Disponibilidad.CONTINUAR_SIN_ARCHIVO; }
                case FALLAR    -> throw new ArchivoFaltanteException(titulo);
                case REINTENTAR -> {
                    if (++intentos > props.reintentos()) throw new ArchivoFaltanteException(titulo);
                    Thread.sleep(props.esperaReintento().toMillis());
                }
            }
        }
        return Disponibilidad.DISPONIBLE;
    }
}
@Bean
@StepScope
ItemReader<SaldoCuenta> posicionOrdenada(ArchivosCsi archivos, ResourceLoader rl,
        @Value("#{jobExecutionContext['sistema']}") Integer sistema,
        @Value("#{jobExecutionContext['fecha']}") LocalDate fecha) throws Exception {

    Resource[] csis = archivos.titulos(sistema, fecha).stream()
            .map(rl::getResource).toArray(Resource[]::new);        // 04 y luego 10 (701: solo 10)

    var concatenado = new MultiResourceItemReaderBuilder<SaldoCuenta>()
            .name("sumaCsis").resources(csis)
            .delegate(SaldoCuenta.lector())                        // A01-REG-SD-ARCH, 90 bytes
            .build();

    // SORT A01-SORT ASCENDING KEY A01-R00-MON, A01-R00-CUENTA
    List<SaldoCuenta> todos = new ArrayList<>();
    concatenado.open(new ExecutionContext());
    for (SaldoCuenta s; (s = concatenado.read()) != null; ) todos.add(s);
    concatenado.close();
    todos.sort(Comparator.comparingInt(SaldoCuenta::moneda).thenComparing(SaldoCuenta::cuenta12));
    return new ListItemReader<>(todos);
}

/** A01-REG-SD-ARCH. Cuenta de 12 = cuenta 9(04) + subcuenta 9(08). */
record SaldoCuenta(int moneda, String cuenta12,
                   BigDecimal saldoAnterior, BigDecimal cargos,      // S9(16)V99 -> escala 2
                   BigDecimal abonos, BigDecimal saldoActual) {
    String cuenta4() { return cuenta12.substring(0, 4); }
}

Riesgos de migración
Intervención manual en un batch

8500-VALIDA-RESIDENCIA detiene el proceso en un ACCEPT de consola hasta que el operador responde C o R (líneas 1137-1148). Cualquier otra respuesta repite la pregunta.

Truncamiento de totales al imprimir

Los acumuladores admiten 17 enteros y las máscaras de las líneas 540-570 solo 14. Totales iguales o mayores a 100 billones pierden los dígitos altos en el reporte.

Tabla de control sin tope

WKS-T-CTRL OCCURS 10 (línea 299) se llena sin validar el índice en 000010-LEE-CONTROL.

Fecha de consulta del catálogo anulada

En 005200-IMPRIME-SCTA la fecha de consulta se asigna y se sobrescribe con cero en la línea siguiente (líneas 1058-1059).

Hardcodes

Sistema 701 con un solo CSI, CSI 04 y 10, el literal 10 en el nombre del archivo, la ruta del control CORP, el catálogo 182/183, la cuenta 5000, el offset 500000000000 y los nombres de moneda 1, 3 y 5 están compilados.

Pendiente de revisión SME
  • Confirmar si el ACCEPT de 8500-VALIDA-RESIDENCIA se responde hoy desde la consola del operador o desde algún automatismo del WFL, para decidir la política por omisión en Java.
  • La primera lectura de 008100-ACT-FECHAP después de OPEN I-O devuelve el header del control; validar que su contenido nunca pueda confundirse con un detalle.
  • En 005400-IMPRIME-MONEDA el MOVE LOW-VALUES TO WKS-TOTALES-MON también limpia WKS-MON-ANT, WKS-CTA-ANT y WKS-CUENTA-ANT, que viven en el mismo grupo (líneas 584-592). Hoy no tiene efecto porque 005100 los reasigna enseguida, pero conviene que el SME lo confirme.
  • RN-CORE-GL-415 describe la función 13 de THECALENDAR como suma de un día hábil; no se tiene la documentación de LOCSUP para confirmarlo.
