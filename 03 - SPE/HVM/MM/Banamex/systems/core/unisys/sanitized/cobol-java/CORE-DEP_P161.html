P161 abre en consulta las bases de captación y de tarjetas, lee el control B02 y, salvo que el indicador de baja de S111 esté encendido, calcula la fecha de lote como el día hábil anterior a la fecha de línea, o usa la fecha recibida en el TASKVALUE si es válida. Escribe un header y recorre todos los contratos de B03: para los de producto 66 instrumento 11 clasifica el movimiento del día en alta, bloqueo, cambio de producto o cancelación, obtiene la tarjeta con la librería L040 y escribe un detalle por cada clasificación que cumple; para los 66 instrumento 17 no migrados escribe las altas del día con la tarjeta armada desde B03. Después recorre B04PTEF y escribe los TEF con medio de entrega 81 actualizados ese día. Cierra con un trailer que lleva el número de detalles.

@Component
@RequiredArgsConstructor
public class FechaLoteTasklet implements Tasklet {

    private final CalendarioBancario calendario;   // LOCSUP función 15, formato 13
    private final P161Properties props;

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        var ec = ctx.getStepContext().getStepExecution().getJobExecution().getExecutionContext();
        LocalDate fechaLinea = (LocalDate) ec.get("fechaLinea");            // B02-FECHA-LINEA
        LocalDate proyectada = calendario.habilAnterior(fechaLinea, 1);   // 10010500-PROYHABANT

        var param = Optional.ofNullable(ctx.getStepContext().getJobParameters().get("fecha"))
                .map(Object::toString).filter(s -> !s.isBlank())
                .map(s -> LocalDate.parse(s, DateTimeFormatter.BASIC_ISO_DATE));
        LocalDate lote = proyectada;
        if (param.isPresent() && param.get().isAfter(props.fechaMinimaParametro())) {
            if (!param.get().isBefore(proyectada)) {
                throw new ProcesoAbortadoException("FECHA ERRONEA, ES MAYOR A LA FECHA LINEA "
                        + " FECHA PARAM: " + param.get() + " FECHA LINEA: " + fechaLinea);   // DMTERMINATE
            }
            lote = param.get();
        }
        ec.put("fechaLote", lote);          // reemplaza a WKS-B02-FECHA-LOTE en todas las comparaciones
        return RepeatStatus.FINISHED;
    }
}

/** I01-REG-S111-APLICADO: 40 posiciones; el tipo de registro decide la vista. */
public sealed interface RegistroS111 {

    /** Tipo 00. */
    record Header(LocalDate fecha, String sistemaOrigen, String sistemaDestino, int csiOrigen)
            implements RegistroS111 {}

    /** Tipos 01 alta, 02 bloqueo, 03 cambio de producto, 04 cancelación, 05 alta 66/17. */
    record Movimiento(TipoMovimiento tipo, String tarjeta,      // 9(016)
                      String contrato,                          // 9(012)
                      String sucursalPromotora) implements RegistroS111 {}   // 9(004)

    /** Tipo 06: tarjeta y número de TEF de B04PTEF. */
    record Tef(String tarjeta, String numeroTef) implements RegistroS111 {}

    /** Tipo 99: número de registros de detalle. */
    record Trailer(long detalles) implements RegistroS111 {}

    enum TipoMovimiento { ALTA(1), BLOQUEO(2), CAMBIO_PRODUCTO(3), CANCELACION(4), ALTA_66_17(5);
        final int codigo; TipoMovimiento(int c) { codigo = c; } }

    static String encode(RegistroS111 r) {
        return switch (r) {
            case Header h     -> "00" + Fixed.aaaammdd(h.fecha()) + h.sistemaOrigen()
                                      + h.sistemaDestino() + "%02d".formatted(h.csiOrigen()) + " ".repeat(20);
            case Movimiento m -> "%02d".formatted(m.tipo().codigo) + m.tarjeta() + m.contrato()
                                      + m.sucursalPromotora() + " ".repeat(6);
            case Tef t        -> "06" + t.tarjeta() + t.numeroTef() + " ".repeat(6);
            case Trailer t    -> "99" + "%08d".formatted(t.detalles()) + " ".repeat(30);
        };
    }
}

@Component
@StepScope
@RequiredArgsConstructor
public class ContratoS111Processor implements ItemProcessor<ContratoB03, List<RegistroS111>> {

    private final HistoricoB06Repository b06;
    private final ClasificadorMovimientos clasificador;   // 20102010-GENERA-DETALLE
    private final AltaCuentaMaestra1717 alta6617;          // 20102013-VALIDA-66-17

    @Override
    public List<RegistroS111> process(ContratoB03 c) {
        if (c.producto() == 66 && c.instrumento() == 11) {
            return b06.findByContrato(c.numContrato())
                    .map(h -> clasificador.clasifica(c, h))
                    .orElseGet(() -> {
                        log.warn("EL CONTRATO: <{}> NO EXISTE EN B06", c.numContrato());
                        return List.of();
                    });
        }
        if (c.producto() == 66 && c.instrumento() == 17 && !c.migradoS111()) {
            return alta6617.evalua(c);
        }
        return List.of();
    }
}

@Component
@StepScope
@RequiredArgsConstructor
public class ClasificadorMovimientos {

    @Value("#{jobExecutionContext['fechaLote']}") private LocalDate lote;
    private final TarjetaResolver tarjetas;   // 20102011-BUSCA-TARJETA (L040)

    /** Cada IF del fuente es independiente: un contrato puede generar varios registros el mismo día. */
    public List<RegistroS111> clasifica(ContratoB03 c, HistoricoB06 h) {
        var tipos = new ArrayList<TipoMovimiento>(4);
        if (lote.equals(c.fechaApertura()))                                       tipos.add(ALTA);
        if (lote.equals(h.fechaBloqueo()) && c.status() == 1 && c.tipoBloqueo() == 85) tipos.add(BLOQUEO);
        if (lote.equals(c.fechaCambioProducto()) && c.status() == 0)              tipos.add(CAMBIO_PRODUCTO);
        if (Set.of(2, 3, 4).contains(c.status()) && lote.equals(h.fechaCancelacion())
                && c.restricciones() == 0)                                        tipos.add(CANCELACION);

        return tipos.stream()
                .flatMap(t -> tarjetas.de(c.numContrato())
                        .map(tar -> new RegistroS111.Movimiento(t, tar, c.numContrato(), c.sucursalPromotora()))
                        .stream())
                .map(RegistroS111.class::cast)
                .toList();
    }
}
@Component
@RequiredArgsConstructor
public class TarjetaResolver {
    private final TarjetasClient l040;   // CORE-DEPL040 DA-TARJETA, expuesto como servicio

    /** 20102011: si L040 no devuelve tarjeta, el movimiento se omite con aviso. */
    Optional<String> de(String contrato) {
        var r = l040.daTarjeta(contrato);
        if (!r.ok()) {
            log.warn("NO SE ENCONTRO LA TARJETA DEL CONTRATO: {}", contrato);
            return Optional.empty();
        }
        return Optional.of(r.tarjeta());          // 9(016)
    }
}

@Component
@StepScope
public class AltaCuentaMaestra1717 {
    @Value("#{jobExecutionContext['fechaLote']}") private LocalDate lote;

    /** 20102013 / 20102014: alta del día; la tarjeta se arma con BIN (6) y serie (10) de B03. */
    List<RegistroS111> evalua(ContratoB03 c) {
        if (!lote.equals(c.fechaApertura())) return List.of();
        String tarjeta = "%06d".formatted(c.medioAccesoBin()) + "%010d".formatted(c.medioAccesoSerie());
        return List.of(new RegistroS111.Movimiento(ALTA_66_17, tarjeta, c.numContrato(),
                c.sucursalPromotora()));
    }
}

Riesgos de migración
Errores de base tratados como fin de archivo

En 20102000-BUSCA-TARJETAS (línea 380) y 30005000-BARRE-B04TEF (línea 475), cualquier WS-DB-ERROR termina el barrido sin mensaje. Un error DMSII produce un archivo truncado con trailer coherente.

Movimientos perdidos sin rastro

Los contratos 66/11 sin B06 o sin tarjeta en L040 solo generan un mensaje al operador; no hay archivo de rechazos ni conteo en el trailer.

Baja de S111 silenciosa

Con B02-USO-FUTURO-05 distinto de cero el programa no genera archivo ni emite mensaje de fin; un valor espurio en ese campo de uso futuro detendría el envío a S111 sin aviso.

Fecha sin siglo en el título

WKS-ETIQ-S111-FECHA es PIC 9(006) y recibe la fecha de 8 dígitos; el nombre del archivo lleva AAMMDD.

Hardcodes

Productos 66/11 y 66/17, tipo de bloqueo 85, estatus 2, 3 y 4, medio de entrega 81, la fecha mínima 20070830 y la ruta (CORE-DEP)CORE-DEP/FILE/S111/ están compilados. Se trasladan a application.yml.
