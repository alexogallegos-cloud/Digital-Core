P189 identifica por HOSTNAME si corre en Valle de México (nodo 10, integra S084) o en Monterrey (nodo 04, integra S087). Une por SORT el archivo CORE-DEP local con el S084 o S087, descarta los detalles con cliente 0 o 999999999999 y deja un archivo intermedio que P197 transfiere a la otra plaza. Recorre ese archivo y el que llega de la plaza foránea para crear o actualizar cada cliente en B09PMOTOR con su último movimiento y estatus, y marca la beneficencia cuando el último movimiento es anterior a la fecha de corte del Artículo 61. Después reinicia a cero el estatus de los clientes no tocados en el día, escribe en Monterrey el archivo MOTOR y actualiza en B06 la fecha de último movimiento y el estatus por contrato.

            .<RegistroIntercambio, RegistroIntercambio>chunk(5000, tx)

            .build();

final class FiltroClienteErroneo {
    static RegistroIntercambio process(RegistroIntercambio r) {
        return (r instanceof RegistroIntercambio.Detalle d && d.clienteErroneo()) ? null : r;
    }
}

/** WS-CORE-DEPB09PMOTOR en memoria: se arma mientras llegan detalles del mismo cliente. */
public final class ClienteEnCurso {
    String cliente;
    LocalDate fechaCreacion;
    LocalDate fecUltMov = LocalDate.MIN;
    UltimoMovimiento ultimo;
    int estatus;                       // WS-B09P-STA-CTE
    LocalDate ultActLocal, ultActForaneo;

    /** 50000000-CARGA-B09P. */
    void acumula(RegistroIntercambio.Detalle d, LocalDate fechaProceso, boolean cargaLocal) {
        cliente = d.cliente();
        fechaCreacion = fechaProceso;
        if (!d.fecUltMov().isBefore(fecUltMov)) {          // NOT <: en empate gana el último leído
            fecUltMov = d.fecUltMov();
            ultimo = UltimoMovimiento.de(d);              // sistema, contrato, producto, instr., moneda...
        }
        if (estatus == 0) estatus = d.estatusContrato();   // el primer bloqueo del día se conserva
        if (cargaLocal) ultActLocal = fechaProceso; else ultActForaneo = fechaProceso;
    }
}
// Tras acumular, el detalle se marca aplicado (STALOC/STAFOR = 1) y se reescribe en el archivo
// (REWRITE REG-A02-CORE-DEP): en Java, un ItemWriter que actualiza la tabla de staging.
@Component
@RequiredArgsConstructor
public class ConsolidaClienteProcessor {

    private final ClienteMotorRepository motor;     // CORE-DEPB09PMOTOR (CORE-DEPBD04TARJETAS)
    private final Articulo61Policy art61;

    /** 50000000-GRABA-B09P + 50000000-ACT-B09P en una transacción por cliente. */
    @Transactional
    public void graba(ClienteEnCurso c, boolean cargaLocal) {
        ClienteMotor m = motor.findByNumCliente(c.cliente)
                .orElseGet(() -> ClienteMotor.nuevo(c));            // B09PMOTOR-CREATE
        if (c.fecUltMov.isAfter(m.getFecUltMov())) {                  // nunca retrocede
            m.setUltimoMovimiento(c.fecUltMov, c.ultimo);
        }
        if (m.getStaCte() == 0) m.setStaCte(c.estatus);
        art61.marca(m);                                              // 50000000-MARCA-STABENF
        if (cargaLocal) m.setUltActLoc(c.ultActLocal); else m.setUltActFor(c.ultActForaneo);
        // Líneas 1696-1699: el IF que pone el estatus en cero queda anulado por la
        // asignación incondicional siguiente. Se reproduce el efecto real:
        m.setStaCte(c.estatus);
    }
}
/** 50000000-MARCA-STABENF: Artículo 61 de la Ley de Instituciones de Crédito. */
@Component
public class Articulo61Policy {

    private LocalDate fechaCorte;   // WKS-FEC-ART61: viene del header CORE-DEP vía el cliente dummy 999999999999

    public void marca(ClienteMotor m) {
        boolean sinCorte = fechaCorte == null;                       // equivale a WKS-FEC-ART61 = 0
        if (m.getStaBenf() == 0) {
            if (m.getStaCte() == 0 && !sinCorte && m.getFecUltMov().isBefore(fechaCorte)) {
                m.setStaBenf(1);          // sin movimiento desde antes del corte y desbloqueado
            }
        } else if ((!sinCorte && m.getFecUltMov().isAfter(fechaCorte)) || m.getStaCte() == 1) {
            m.setStaBenf(0);              // volvió a moverse o quedó bloqueado
        }
    }
}
@Component
@RequiredArgsConstructor
public class ReinicioEstatusTasklet implements Tasklet {

    private final ClienteMotorRepository motor;
    private final Articulo61Policy art61;
    @Value("#{jobExecutionContext['fechaProceso']}") private LocalDate fechaProceso;

    /**
     * 50000000-INI-STA-CTE. Los archivos solo transportan bloqueos; un cliente que no llegó
     * hoy por ninguna plaza se considera desbloqueado. Se confirma cada 5000 registros
     * (WKS-MAX-TRANS), igual que el END-TRANSACTION del fuente.
     */
    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        try (Stream<ClienteMotor> todos = motor.streamAll()) {
            Lotes.de(todos, 5000).forEach(lote -> desbloquea(lote));
        }
        return RepeatStatus.FINISHED;
    }

    @Transactional
    void desbloquea(List<ClienteMotor> lote) {
        for (var m : lote) {
            boolean tocadoHoy = fechaProceso.equals(m.getUltActLoc())
                    || fechaProceso.equals(m.getUltActFor());
            if (!tocadoHoy && m.getStaCte() > 0) {
                m.setStaCte(0);
                art61.marca(m);
            }
        }
    }
}
@Component
@RequiredArgsConstructor
public class HistoricoB06Writer {

    private final ClienteMotorRepository motor;
    private final HistoricoContratoRepository b06;     // CORE-DEPB06HISTORICO (CORE-DEPBD01CAPTACION)

    /** 60000000-BARRE-B03 + 60000000-VAL-ACT-B06. WKS-MAX-TRANS-BD01 = 1: commit por contrato. */
    @Transactional
    public void sincroniza(Contrato b03) {
        motor.findByNumCliente(b03.getNumCliente()).ifPresent(m ->
            b06.findForUpdateByContrato(b03.getNumContrato()).ifPresent(h -> {
                boolean cambia = false;
                if (m.getFecUltMov().isAfter(h.getFechXcte())) {
                    h.setFechXcte(m.getFecUltMov());
                    cambia = true;
                }
                if (m.getStaCte() != h.getStaXcte()) {
                    h.setStaXcte(m.getStaCte());
                    cambia = true;
                }
                if (!cambia) b06.liberaBloqueo(h);              // FREE CORE-DEPB06HISTORICO
            }));
    }
}
