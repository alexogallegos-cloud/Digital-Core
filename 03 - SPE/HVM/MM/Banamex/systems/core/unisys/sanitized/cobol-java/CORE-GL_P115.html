CORE-GL · P115 a Java 21 Spring Boot

Lee INFOICA, la información de cuentas por aplicar que llega de cada nodo, y valida cada movimiento campo por campo: moneda, CSI y aplicación de origen y destino, estación, cuenta, autorización, fecha contable, importe, código de servicio y estatus. Los registros con error, las operaciones locales sin respuesta o anuladas y los enviados o denegados se apartan en LOG-ERROR; las operaciones locales válidas van a LOG-TRANSMISION y las foráneas a LOG-SORT. Con SORT por llave resume las foráneas en el archivo indexado COMPENSACION (operado y aplicado, cargos y abonos) y copia cada movimiento al LOG-CSI; con otro SORT arma la interfaz TRANSMISION de S028 con registro de control, movimientos por sistema, sucursal y terminal, y trailer. La rama de pendientes y casi todos los reportes impresos están comentados en el fuente.

@Component
@RequiredArgsConstructor
public class ValidadorMovimientoIca {

    private final P115Properties props;   // csi-local (PTER-CSI-WS), csis-validos

    /** 21212-CHECA-CSI: si el CSI local viene como destino, se intercambian origen y destino. */
    MovimientoIca normaliza(MovimientoIca m) {
        if (m.csiOrigen() != props.csiLocal() && m.csiDestino() == props.csiLocal()) {
            return m.conOrigenDestino(m.csiDestino(), m.csiOrigen(), m.aplDestino(), m.aplOrigen());
        }
        return m;
    }

    /** 21300-REVISA-LOG, primer bloque: cada falla deja una letra en R2D1-ERR(n). */
    EnumSet<ErrorIca> valida(MovimientoIca m) {
        var e = EnumSet.noneOf(ErrorIca.class);
        if (m.moneda() <= 0)                                   e.add(ErrorIca.K_MONEDA);
        if (!props.csisValidos().contains(m.csiOrigen())
                || m.csiOrigen() != props.csiLocal())          e.add(ErrorIca.A_CSI_ORIGEN);
        if (!props.csisValidos().contains(m.csiDestino()))     e.add(ErrorIca.C_CSI_DESTINO);
        if (m.aplOrigen() <= 0 || m.aplOrigen() > 999)         e.add(ErrorIca.B_APL_ORIGEN);
        if (m.aplDestino() <= 0 || m.aplDestino() > 999)       e.add(ErrorIca.D_APL_DESTINO);
        if (m.estacion().isBlank())                            e.add(ErrorIca.E_ESTACION);
        if (m.cuentaDestino().isBlank())                       e.add(ErrorIca.G_CUENTA);
        if (m.claveAutorizacion().isBlank())                   e.add(ErrorIca.F_AUTORIZACION);
        if (m.fechaContable() == null)                         e.add(ErrorIca.L_FECHA_CONTABLE);
        if (m.importe() == null || m.importe().signum() == 0)  e.add(ErrorIca.I_IMPORTE);
        if (m.servicio() == CodigoServicio.INVALIDO)           e.add(ErrorIca.H_SERVICIO);
        if (m.estatus() == EstatusTransaccion.INVALIDO)        e.add(ErrorIca.J_ESTATUS);
        return e;                                              // no vacío -> HAY-ELIMINA
    }
}

@Component
public class ClasificadorMovimiento implements ItemProcessor<MovimientoIca, Destino> {

    @Override
    public Destino process(MovimientoIca m) {
        var errores = validador.valida(m);
        String motivo = errores.isEmpty() ? null : ErrorIca.letras(errores);
        boolean local = m.csiOrigen() == m.csiDestino();
        if (local && m.estatus() != EstatusTransaccion.OPERADO) {   // local solo vale si está operado
            motivo = switch (m.estatus()) {
                case PENDIENTE -> "LOCAL S/RESP";
                case ENVIADO   -> "LOCAL ANULAD";
                default        -> "LOCAL ER/STA";
            };
        }
        if (motivo != null) {
            boolean enviadoODenegado = m.estatus() == EstatusTransaccion.ENVIADO
                    || m.estatus() == EstatusTransaccion.DENEGADO;
            return new Destino.Error(m, motivo, !enviadoODenegado);   // 21330-SUMA-ELIMINADOS
        }
        acumuladores.suma(m, local ? Ambito5.LOCAL : Ambito5.de(m.estatus()));   // 21320 (O, A, P)
        return local ? new Destino.Transmision(m) : new Destino.Foraneo(m);
    }
}

List<RegistroTransmision> armaLogTransmision(MovimientoIca m, LocalDate fechaProceso) {
    var base = new RegistroTransmision(m.servicio(), m.importe());
    var salida = new ArrayList<RegistroTransmision>(3);
    salida.add(base.porSistema(m.fechaContable(), m.aplOrigen()));          // tipo 1, ámbito 1
    salida.add(base.porNodo(m.nodoOrigen(), m.estacion()));                 // tipo 2
    if (m.fechaContable().equals(fechaProceso)) {                           // WS-FECHA-INV-PROC
        salida.add(base.porTerminal(m.aplOrigen(), m.sucursalOrigen(),      // tipo 3: alimenta S028
                                    m.moneda(), m.estacion()));
    }
    return salida;
}

@Component
@StepScope
@RequiredArgsConstructor
public class CompensacionNacionalWriter implements ItemWriter<MovimientoIca> {

    private final LogCsiWriter logCsi;            // 23225-GRABA-LOG-CSI
    private final CompensacionRepository repo;

    /** 23220-SUMARIZA: todo va al LOG-CSI; solo aplicados y operados suman a la compensación. */
    @Override
    @Transactional
    public void write(Chunk<? extends MovimientoIca> chunk) {
        for (var m : chunk) {
            logCsi.escribe(m);
            Ambito ambito = switch (m.estatus()) {
                case APLICADO -> Ambito.APLICADO;
                case OPERADO  -> Ambito.OPERADO;
                default       -> null;            // pendientes y demás no compensan
            };
            if (ambito != null) {
                acumulaPorLlave(CompensacionId.de(m), ambito, m.servicio(), m.importe());
            }
        }
    }

    /** 23240-SUMA-CARGO-ABONO: las eliminaciones restan conteo e importe. */
    static Cifras aplica(Cifras c, CodigoServicio s, BigDecimal imp) {
        return switch (s) {
            case CARGO      -> c.conCargos(c.numCargos() + 1, c.impCargos().add(imp));
            case ABONO      -> c.conAbonos(c.numAbonos() + 1, c.impAbonos().add(imp));
            case ELIM_CARGO -> c.conCargos(c.numCargos() - 1, c.impCargos().subtract(imp));
            case ELIM_ABONO -> c.conAbonos(c.numAbonos() - 1, c.impAbonos().subtract(imp));
            case INVALIDO   -> c;                  // ya filtrado en la validación
        };
    }
}
    /** 23250-ACTUALIZA-COMP-NAL: WRITE y, si la llave ya existe, REWRITE (23280). */
    private void acumulaPorLlave(CompensacionId id, Ambito ambito,
                                 CodigoServicio servicio, BigDecimal importe) {
        var comp = repo.findById(id).orElseGet(() -> Compensacion.nueva(id));   // DUMMY-COMP = 0
        comp.cifras().merge(ambito, aplica(Cifras.cero(), servicio, importe), Cifras::suma);
        comp.validaCapacidad();   // S9(15)V99: falla explícita en lugar de truncar
        repo.save(comp);
    }

Riesgos de migración
Truncamiento en acumuladores de compensación

IMP-CARGOS-COMP e IMP-ABONOS-COMP (líneas 308-309) son S9(15)V99 y reciben importes de 9(16)V99 (línea 1025) sin ON SIZE ERROR. Los acumuladores de conteo ACUM y TOT son S9(06) (líneas 2114-2125), menores que los S9(07) de COMPENSACION.

Rama de pendientes desactivada

El archivo PENDIENTES, el SORT4, 25000-IMPRIME-PENDIENTES y el filtro de antigüedad (líneas 4005, 4172-4177, 4627-4630) están comentados con *ASM. Las partidas pendientes se clasifican pero no se persisten ni se reportan.

Reportes impresos comentados

Los CLOSE de REPORTE0 a REPORTE7 y la impresión de títulos y totales (líneas 3995-3997, 4600-4607) están comentados: los cortes y acumuladores se calculan pero la carátula y los reportes de transmisión no se imprimen.

Validación de aplicación más estricta que el campo

21300-REVISA-LOG rechaza aplicaciones mayores a 999 (líneas 4107 y 4112) aunque el campo es 9(04) y COMPENSACION admite hasta 9999.

Hardcodes de plazas

Los CSI válidos 1, 2, 4, 6 y 10 están en cuatro niveles 88; en 10000-INICIA-PROCESO el CSI 32 se convierte en 4 y el 12 en 10 (líneas 3393-3400). Se trasladan a application.yml.

Pivote Y2K fijo

A2K-BASE-YEAR = 50 (línea 3308) decide el siglo de las fechas de 6 dígitos como FECHA-COMP; en Java el pivote queda explícito en la conversión.
