/**
 * A01-CORE-GLMOV: ACCESS RANDOM con ACTUAL KEY = número relativo de registro (W77-ULT-REG-LEIDO).
 * A02-CORE-GLDES: ACCESS RANDOM por W77-LOGKEY, sin buffers (RESERVE 0 AREAS).
 */
public final class LogMovimientosFile implements AutoCloseable {

    public static final int LONG_REGISTRO = 450;          // CORE-GL/FORMATO/ARCHIVO/MOVTOS
    private final FileChannel canal;

    public LogMovimientosFile(Path ruta) throws IOException {
        this.canal = FileChannel.open(ruta, StandardOpenOption.READ);   // OPEN INPUT
    }

    /** READ ... INVALID KEY: vacío si el registro aún no existe. */
    public Optional<ByteBuffer> lee(long numRegistro) throws IOException {
        var buf = ByteBuffer.allocate(LONG_REGISTRO);
        long pos = (numRegistro - 1) * LONG_REGISTRO;     // ACTUAL KEY es base 1
        if (canal.read(buf, pos) < LONG_REGISTRO) return Optional.empty();
        return Optional.of(buf.flip());
    }

    @Override public void close() throws IOException { canal.close(); }
}

/** A00-R01-ELE-CVEIMP OCCURS 5 en el log y WKS-TAB-CVEIMP OCCURS 5 en la interfaz con L020. */
public record ClaveImporte(int cveTran, int indLey, int esqCon, BigDecimal importe) {

    // Capacidad de la interfaz WKS-MOV-DATOS, más chica que la del log.
    static final int MAX_CVETRAN_L020 = 9_999;                                   // 9(04) frente a 9(06)
    static final BigDecimal MAX_IMPORTE_L020 = new BigDecimal("99999999999999.99"); // 9(14)V99 frente a 9(16)V99

    /** MOVE A00-R01-xxx (n) TO WKS-MOV-xxx (n): COBOL trunca dígitos altos sin aviso; aquí se rechaza. */
    public ClaveImporte paraL020() {
        if (cveTran > MAX_CVETRAN_L020 || importe.compareTo(MAX_IMPORTE_L020) > 0) {
            throw new TruncamientoInterfazException(this);
        }
        return this;
    }
}

@Component
@RequiredArgsConstructor
@Slf4j
public class ControlDiaTasklet implements Tasklet {

    private final LibControlClient control;   // CONSISDIA / CONSISMEN IN LIB-CONTROL

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        int sistema = ctx.getStepContext().getJobParameters().get("sistema") instanceof Long l ? l.intValue() : 0;
        var dia = control.consisdia(sistema);
        if (dia.status() > 0) log.error("ERROR EN LIB-CONTROL CORE-GLB01SISDIA {}", dia.status());   // el fuente no aborta

        if (dia.fecCon().isBefore(dia.fecPro151())) {
            // El mensaje dice DESCONTINUO PROCESO, pero el fuente no detiene la tarea.
            log.error("ERROR EN FECHAS CONTABLE Y PROCESO DEL CORE-GL {} {}", dia.fecCon(), dia.fecPro151());
        }

        var men = control.consismen(sistema);
        if (men.status() > 0) throw new ProcesoAbortadoException("ERROR EN LIB-CONTROL CORE-GLB03SISMEN " + men.status());
        boolean baseNoDisponible = switch (men.stabdsal()) {
            case 4, 5, 6 -> throw new ProcesoAbortadoException("ERROR NO SE HA EJECUTADO P197 : " + men.stabdsal());
            case 3       -> throw new ProcesoAbortadoException("ERROR PROCESO YA EJECUTADO : " + men.stabdsal());
            case 99      -> true;                // W77-BD-SDOS = 1
            default      -> false;
        };
        control.marcaStabdsal(sistema, 2);       // CONSISMEN función 9: base en proceso

        var job = ctx.getStepContext().getStepExecution().getJobExecution().getExecutionContext();
        job.put("dia", dia);
        job.put("baseNoDisponible", baseNoDisponible);
        return RepeatStatus.FINISHED;
    }
}
@Component
@RequiredArgsConstructor
@Slf4j
public class ReposicionTasklet implements Tasklet {

    private final L020Client l020;          // CORE-GLLIB020 IN LIB-L020
    private final P025Properties props;      // p025.lib-l020-prefijo: CORE-GLL020S

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        var job = ctx.getStepContext().getStepExecution().getJobExecution().getExecutionContext();
        if ((Boolean) job.get("baseNoDisponible")) return RepeatStatus.FINISHED;   // ni título ni consulta

        var dia = (ControlDia) job.get("dia");
        // Llamada "vacía": sólo sistema y fecha contable, para que L020 informe dónde se quedó.
        var r = l020.aplica(MovimientoSaldo.consulta(dia.sistema(), dia.fecPro151()));
        long desde = 0;                                             // MOVE ZEROS TO W77-ULT-REG-LEIDO
        if (r.codigo() == 999) {
            job.put("baseNoDisponible", true);                      // W77-BD-SDOS = 1
        } else if (r.codigo() > 0) {
            log.error("ERROR EN CALL L020 {}", r.codigo());          // el fuente no aborta
        } else if (r.ultimaAutCORE-GL() > 0) {
            desde = r.ultimaAutCORE-GL() + 1;                          // ADD 1 y la lectura suma otro: registro aut + 2
        }
        // Con autorización 0 la posición queda en 0 y la siguiente lectura vuelve al header (ver pendientes).
        job.put("ultimoRegistro", desde);
        return RepeatStatus.FINISHED;
    }
}

@Component
@StepScope
@RequiredArgsConstructor
public class LogSeguidorReader implements ItemStreamReader<RegistroLog> {

    private static final int FIN_DEFINITIVO = 98, FIN_PARCIAL = 99;
    private final LogMovimientosFile log;
    private final P025Properties props;          // p025.espera-log: 180s (WAIT (180))
    private long ultimo;                         // W77-ULT-REG-LEIDO

    @Override public void open(ExecutionContext ec) { ultimo = ec.getLong("ultimoRegistro", 1L); }
    @Override public void update(ExecutionContext ec) { ec.putLong("ultimoRegistro", ultimo); }

    @Override
    public RegistroLog read() throws Exception {
        while (true) {
            var reg = log.lee(ultimo + 1).map(RegistroLog::parse);
            if (reg.isEmpty()) return null;                       // INVALID KEY: W77-FIN = 1
            var r = reg.get();
            if (r.funcion() == FIN_DEFINITIVO) {
                var sig = log.lee(ultimo + 2).map(RegistroLog::parse);
                if (sig.isEmpty() || sig.get().funcion() == FIN_PARCIAL) return null;   // 98 seguido de 99: fin
                ultimo += 2;
                return sig.get();                                 // 98 sin 99: sigue con el siguiente
            }
            if (r.funcion() == FIN_PARCIAL) {
                Thread.sleep(props.esperaLog().toMillis());       // el log todavía se está escribiendo
                continue;                                         // relee la misma posición
            }
            ultimo++;
            return r;
        }
    }
}
@Component
@RequiredArgsConstructor
public class EdoCtaProcessor implements ItemProcessor<RegistroLog, RegistroLog> {

    private final P025Properties props;
    // p025.edocta-override:
    //   producto: 84
    //   instrumento: 15
    //   claves-sin-edocta: [13,14,15,16,17,23,25,26,28,33,34,35,36,37,38]
    //   claves-con-edocta: [18,19,20,21,22,27,29,31,32]

    @Override
    public RegistroLog process(RegistroLog r) {
        var o = props.edoctaOverride();
        int indEdoCta = r.indEdoCta();
        if (r.producto() == o.producto() && r.instrumento() == o.instrumento()) {
            int cve = r.claves().get(0).cveTran();          // sólo la primera de las cinco
            if (o.clavesSinEdocta().contains(cve) && indEdoCta == 1) indEdoCta = 0;
            else if (o.clavesConEdocta().contains(cve) && indEdoCta == 0) indEdoCta = 1;
        }
        boolean aplica = r.funcion() == 1 && r.status() == 1 && indEdoCta == 1;
        return aplica ? r.conIndEdoCta(indEdoCta) : null;    // null = no se envía a L020
    }
}

public class L020Writer implements ItemWriter<RegistroLog> {

    private final L020Client l020;
    private final DatosAdicionalesReader des;    // 800000-LEE-CORE-GLDES
    @Value("#{jobExecutionContext['baseNoDisponible']}") private boolean baseNoDisponible;

    @Override
    @Transactional                                  // una unidad por chunk sobre la base de saldos
    public void write(Chunk<? extends RegistroLog> chunk) {
        if (baseNoDisponible) return;               // W77-BD-SDOS = 1: el log se recorre sin aplicar nada
        for (var r : chunk) {
            var mov = r.aMovimientoSaldo();
            if (r.apuDes() > 0) mov = mov.conDatosAdicionales(des.lee(r.apuDes() + 1));
            var res = l020.aplica(mov);
            if (res.codigo() > 0 && res.codigo() != 999) {
                throw new ProcesoAbortadoException("ERROR EN CALL LIB-L020 %d AUTORIZACION : %d"
                        .formatted(res.codigo(), res.ultimaAutCORE-GL()));   // SET MYSELF(STATUS) TO -1
            }
            // 999 en un movimiento se ignora (NEXT SENTENCE)
        }
    }
}

    /** 800000-LEE-DES: W77-LOGKEY = A00-R01-APUDES + 1. */
    public DatosAdicionales lee(long logKey) {

                .map(DatosAdicionales::parse)
                .orElseThrow(() -> {

                    // contenido previo del buffer A01-R01-DATOS (datos del movimiento anterior).
                    log.error("ERROR ARCHIVO DESCRIPCIONES LOGKEY : {}", logKey);
                    return new DatosAdicionalesNoEncontradosException(logKey);
                });
    }

Riesgos de migración
Truncamiento de clave e importe hacia L020

El log define CVETRAN 9(06) e IMPORTE 9(16)V99 (líneas 112 y 116) y la interfaz WKS-MOV-DATOS los recibe en 9(04) y 9(14)V99 (líneas 1250 y 1255). Los MOVE de 350100-CALL-L020 truncan dígitos altos sin aviso.

Mensajes de paro que no detienen

El error de CONSISDIA (línea 2301), la fecha contable menor que la de proceso (líneas 2309-2318, cuyo texto dice DESCONTINUO PROCESO) y el error de CTLVERS al buscar LIB-L020 (líneas 2412-2417) sólo se despliegan y el proceso sigue.

Datos adicionales de otro movimiento

Si la lectura de A02-CORE-GLDES falla (líneas 2647-2655), el error no se despliega y se envían a L020 los datos adicionales que quedaron en el buffer del registro anterior.

Recorrido sin aplicación

Con STABDSAL 99 al inicio o resultado 999 en la llamada inicial a L020, W77-BD-SDOS = 1 bloquea todas las llamadas a L020 y la lectura de descripciones; el programa recorre el log completo y termina sin aplicar saldos ni avisar.

Espera sin tope y hardcodes

El ciclo de WAIT (180) no tiene límite. Producto 84, instrumento 15, las dos listas de claves de 310100-PROCESA, el prefijo CORE-GLL020S y el título de LIB-DISP están compilados; se trasladan a application.yml.
