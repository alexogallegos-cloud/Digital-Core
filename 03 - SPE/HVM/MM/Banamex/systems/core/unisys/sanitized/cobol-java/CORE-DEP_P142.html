Abre CORE-DEPBD07ATRIBUCTA en modo de consulta, toma la fecha de proceso del registro de control B02 (o del TASKVALUE de la tarea si viene informado) e identifica el CSI del equipo en que corre. Escribe un header tipo 1, un detalle tipo 2 por cada contrato de B01 cuyo primer dato numérico valga 0 o 1, con cinco grupos de atributos numéricos y cinco alfanuméricos, y un trailer tipo 9 con el número de detalles. Los contratos con un primer dato distinto quedan fuera del archivo y se anotan en el archivo PROBLEMAS, que es el único canal de errores no fatales.

/** Registro fijo de 1200 bytes que P142 entrega a Teradata. Tipo en la posición 1. */
public sealed interface CtorepRecord
        permits CtorepRecord.Header, CtorepRecord.Detalle, CtorepRecord.Trailer {

    int LONGITUD = 1200;

    String toFixed();

    /** TREG = 1. FEC-GEN-ARCH recibe en la práctica la fecha de proceso (ver 40006000-OPEN-CTOREP). */
    record Header(String sisOrigen, LocalDate fecGenArch, int csiOrigen) implements CtorepRecord {
        public String toFixed() {
            return Fixed.of(LONGITUD)
                    .num(1, 1).alfa(sisOrigen, 4).fechaAmd(fecGenArch).num(csiOrigen, 2)
                    .spaces(1185).build();
        }
    }

    /** Grupo numérico: seis sellos de fecha/hora 9(08), DATO 9(12) y FILLER X(20) = 80 bytes. */
    record GrupoNumerico(long fecProAct, long fecMaqAct, long horMaqAct,
                         long fecProAct2, long fecMaqAct2, long horMaqAct2,
                         long dato) {}

    /** Grupo alfanumérico: seis sellos 9(08), DATO X(40) y FILLER X(20) = 108 bytes. */
    record GrupoAlfa(long fecProAct, long fecMaqAct, long horMaqAct,
                     long fecProAct2, long fecMaqAct2, long horMaqAct2,
                     String dato) {}

    /** TREG = 2. 77 bytes de cabecera + 5 x 80 + 5 x 108 + FILLER X(183) = 1200. */
    record Detalle(int csiContrato, int csiCobertura,
                   String numContrato,                 // 9(12): String para conservar ceros
                   long fecProAlt, long fecMaqAlt, long horMaqAlt,
                   long fecProAct, long fecMaqAct, long horMaqAct,
                   int indPantalla, long fecActPan,
                   List<GrupoNumerico> numericos,       // OCCURS 5 TIMES
                   List<GrupoAlfa> alfas)               // OCCURS 5 TIMES
            implements CtorepRecord {
        public Detalle {
            if (numericos.size() != 5 || alfas.size() != 5)
                throw new IllegalArgumentException("CTOREP exige exactamente 5 grupos de cada tipo");
        }
        public String toFixed() { return CtorepLayout.detalle(this); }
    }

    /** TREG = 9. Solo cuenta detalles: el header y el trailer no se incluyen. */
    record Trailer(long totalDetalles) implements CtorepRecord {
        public String toFixed() {
            return Fixed.of(LONGITUD).num(9, 1).num(totalDetalles, 8).spaces(1191).build();
        }
    }
}

@Component
@StepScope
@RequiredArgsConstructor
public class FechaProcesoTasklet implements Tasklet {

    private final ControlAtributosRepository control;  // CORE-DEPB02CONTROL (OPEN INQUIRY)
    private final CalendarioBancario calendario;        // reemplaza THECALENDAR IN LOCSUP

    @Value("#{jobParameters['fechaProceso']}") private String fechaParametro;  // WS-VAL-FEC

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        var b02 = control.findFirst()
                .orElseThrow(() -> new ProcesoAbortadoException("ERROR REG. DE CONTROL VACIO"));

        LocalDate fechaPro;
        LocalDate fechaProx;
        if (fechaParametro != null) {                         // IF WS-VAL-FEC > 0
            fechaPro  = LocalDate.parse(fechaParametro, DateTimeFormatter.BASIC_ISO_DATE);
            fechaProx = calendario.siguienteHabil(fechaPro);   // LOCSUP función 13, formato 13, +1
        } else {
            fechaPro  = b02.fechaLote();                       // B02-FECHA-LOTE
            fechaProx = b02.fechaLinea();                      // B02-FECHA-LINEA
        }

        // WS-PRO-MM NOT = WS-FEC-PROX-MM
        boolean esMensual = fechaPro.getMonthValue() != fechaProx.getMonthValue();
        // 20002120-SEMANAL: LOCSUP función 6 devuelve 0=domingo .. 6=sábado
        boolean esSemanal = diaSemanaUnisys(fechaPro) > diaSemanaUnisys(fechaProx);

        var ec = ctx.getStepContext().getStepExecution().getJobExecution().getExecutionContext();
        ec.put("fechaProceso", fechaPro);
        ec.put("esSemanal", esSemanal);   // calculados como en el fuente; P142 no los consume
        ec.put("esMensual", esMensual);
        return RepeatStatus.FINISHED;
    }

    private static int diaSemanaUnisys(LocalDate d) {
        return d.getDayOfWeek().getValue() % 7;           // DayOfWeek: 1=lunes .. 7=domingo
    }
}

@Component
@StepScope
@RequiredArgsConstructor
public class ContratoCtorepProcessor implements ItemProcessor<ContratoAtributos, CtorepRecord> {

    private final HostCsiResolver host;
    private final ProblemasLog problemas;

    /** 50003300-WRITE-ARCHIVO. */
    @Override
    public CtorepRecord process(ContratoAtributos b01) {
        long dato1 = b01.numericos().get(0).dato();          // B01-NUM-DATO(1)
        if (dato1 != 0 && dato1 != 1) {
            // El fuente pasa el valor a WKS-NUM-DATO PIC 9(02): solo se conservan dos dígitos.
            problemas.anota("ERROR NUM DATO(1): %02d, EN CTO: %s"
                    .formatted(dato1 % 100, b01.numContrato()));
            return null;                                     // no se escribe ni se cuenta
        }
        return new CtorepRecord.Detalle(
                b01.csiContrato(), host.csiLocal(),          // CSI-COBERTURA = CSI del equipo
                b01.numContrato(),
                b01.fecProAlt(), b01.fecMaqAlt(), b01.horMaqAlt(),
                b01.fecProAct(), b01.fecMaqAct(), b01.horMaqAct(),
                b01.indPantalla(), b01.fecActPan(),
                GrupoMapper.numericos(b01),                  // 50003300-LEO-NUM
                GrupoMapper.alfas(b01));                     // 50003300-LEO-ALF
    }
}
final class GrupoMapper {

    /** 50003300-LEO-NUM / 50003300-NUM-GRP: copia 1:1 de las cinco ocurrencias. */
    static List<CtorepRecord.GrupoNumerico> numericos(ContratoAtributos b01) {
        return IntStream.range(0, 5)
                .mapToObj(i -> {
                    var g = b01.numericos().get(i);
                    return new CtorepRecord.GrupoNumerico(
                            g.fecProAct(), g.fecMaqAct(), g.horMaqAct(),
                            g.fecProAct2(), g.fecMaqAct2(), g.horMaqAct2(),
                            g.dato());                    // WKS-DET-NUM-FILLER se serializa como espacios
                })
                .toList();
    }

    // alfas(...) es idéntico sobre B01-ALFA-* (50003300-LEO-ALF / ALF-GRP).
}
@Bean
@StepScope
FlatFileItemWriter<CtorepRecord> ctorepWriter(P142Properties props, HostCsiResolver host,
        @Value("#{jobExecutionContext['fechaProceso']}") LocalDate fecha,
        CtorepContador contador) {
    String ruta = props.resuelve(props.ctorep(), host.csiLocal(), fecha);
    return new FlatFileItemWriterBuilder<CtorepRecord>()
            .name("ctorep")
            .resource(new FileSystemResource(ruta))
            .lineAggregator(CtorepRecord::toFixed)
            // 40006000-OPEN-CTOREP: header con la fecha de proceso, no la del reloj
            .headerCallback(w -> w.write(
                    new CtorepRecord.Header(props.sistemaOrigen(), fecha, host.csiLocal()).toFixed()))
            // 50003400-TRAILER-CTOREP: total = detalles escritos (WKS-CONTADOR)
            .footerCallback(w -> w.write(new CtorepRecord.Trailer(contador.total()).toFixed()))
            .shouldDeleteIfExists(true)   // OPEN OUTPUT
            .transactional(true)
            .build();
}

        anota("INICIA ARCHIVO PROBLEMAS");
    }

    /** 50009000-PROB-TIEMPO: renglón en blanco y renglón con fecha, hora y mensaje (80 bytes). */
    public void anota(String mensaje) {
        String renglon = SELLO.format(LocalDateTime.now(reloj)) + " " + mensaje;
        try {
            out.write(" ".repeat(80)); out.newLine();
            out.write(StringUtils.rightPad(StringUtils.left(renglon, 80), 80)); out.newLine();
        } catch (IOException e) { throw new UncheckedIOException(e); }
    }

    @Override
    public ExitStatus afterStep(StepExecution se) {
        anota("TERMINA ARCHIVO PROBLEMAS");
        try { out.close(); } catch (IOException e) { throw new UncheckedIOException(e); }  // CLOSE WITH SAVE
        return se.getExitStatus();
    }
}

Riesgos de migración
Escritura en PROBLEMAS antes de abrirlo

Los errores de apertura de CORE-DEPBD07ATRIBUCTA (líneas 14005-14009) y de B02 vacío (líneas 14089-14093) hacen PERFORM 50009000-PROB-TIEMPO, que escribe F02-REG, cuando F02-PROBLEMAS se abre hasta 50001000-ARCH-PRO (línea 14317). En esos casos el mensaje al archivo no puede quedar y la tarea podría terminar por un error de E/S antes del DMTERMINATE.

Indicadores de semana y mes calculados y no usados

WKS-ES-SEMANAL y WKS-ES-MENSUAL (líneas 13646-13647) se calculan en 20002100-PROY-FEC y 20002120-SEMANAL, pero ningún otro párrafo los consulta. WKS-ES-MENSUAL además solo se pone en 1. Las reglas RN-CORE-DEP-126 y RN-CORE-DEP-127 describen una lógica que no cambia la salida de P142.

Replicación entre hosts sin efecto

40003000-IDENTIFICA-HOST llena los pares de equipo origen y destino, pero WKS-PARAM-XFER-XX y WS-WFL-COPY no se envían a XFER ni a WFL, y el envío de PROBLEMAS por INTELARSND está comentado (líneas 14157-14158). P142 solo deja CTOREP en el directorio INTS; el transporte a Teradata ocurre fuera del programa.

Registros excluidos sin cifra de control

Los contratos con B01-NUM-DATO(1) distinto de 0 y 1 se anotan en PROBLEMAS pero el trailer solo cuenta lo escrito; Teradata no recibe un total de leídos contra exportados para conciliar.

Hardcodes de rutas y equipos

Prefijos CORE-DEP/FILE/INTS/50024S01 y 50004S01, CSI 10 y 04 y los ocho nombres de equipo están compilados. Se trasladan a application.yml.
