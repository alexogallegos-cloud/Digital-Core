/** Feed GBNP: registros de 98 bytes con campos separados por "~". */
public sealed interface GbnpRecord permits GbnpRecord.Header, GbnpRecord.Detalle, GbnpRecord.Trailer {

    String linea();

    record Header(LocalDate fechaBase, int secuencia, String csi,
                  GbnpProperties p) implements GbnpRecord {
        public String linea() {          // HDR~VDBAL~AAAAMMDD~SSS~CSI ~MX ~999
            return String.join("~", "HDR", p.handle(), fechaBase.format(BASIC_ISO_DATE),
                    "%03d".formatted(secuencia), "%-4s".formatted(csi), p.sistema(), p.claveBanco());
        }
    }

    record Detalle(String claveBanco, String contrato20, LocalDate fechaAplic,
                   BigDecimal saldo, String monedaIso, String diaCorte, char status) implements GbnpRecord {
        public String linea() {
            return String.join("~", "DET", claveBanco, contrato20, fechaAplic.format(BASIC_ISO_DATE),
                    Fixed.editado(saldo, 21, 2),        // ZZZ...9.99 con ceros a la izquierda (INSPECT)
                    monedaIso, Fixed.editado(saldo, 21, 2), diaCorte, String.valueOf(status));
        }
    }

    record Trailer(long registros) implements GbnpRecord {
        public String linea() { return "TRL~" + "%010d".formatted(registros); }
    }
}

                .from(siguienteSecuencia)

@Service
@RequiredArgsConstructor
public class ControlGbnpService {

    private final ControlCaptacionRepository b02;
    private final GbnpProperties props;
    private final WflGateway wfl;                       // CALL SYSTEM WFL USING WS-WFLP197

    /** 20006000-VALIDA-SERV / 20006100-REENVIA-ARC */
    @Transactional(readOnly = true)
    public DecisionGbnp decide(ParametroP108 p, LocalDate fechaBase) {
        ControlCaptacion c = b02.findControl().orElseThrow();
        String csi = (c.numCsi() == 10) ? "VDM" : "QRO";        // rotulado del header
        if (fechaBase.equals(c.gbnpSaldosFec())) {
            // Los archivos de la fecha ya existen: se reenvían según la secuencia guardada.
            int sec = c.gbnpSaldosSec();
            if (sec == 3) {                                   // cualquier parámetro
                reenvia(fechaBase, 1, 3);
                return DecisionGbnp.terminado(csi);            // WKS-FIN = 2
            }
            if (p.valor() == 1 && sec == 2) {
                reenvia(fechaBase, 1, 2);
                return DecisionGbnp.terminado(csi);
            }
            if (p.valor() == 1 && sec == 1) {
                reenvia(fechaBase, 1, 1);
                return DecisionGbnp.generaSecuencia(csi, 2);   // WKS-SEC2 = 1: falta la 002
            }
        }
        return DecisionGbnp.genera(csi);
    }

    private void reenvia(LocalDate fecha, int desde, int hasta) {
        for (int s = desde; s <= hasta; s++) {
            wfl.envia(props.wflEnvio(), fecha, s);
            if (s < hasta) sleepSegundos(5);                   // WAIT(5)
        }
    }
}
/** 50000200-ACT-B02: BEGIN-TRANSACTION, LOCK, STORE, END-TRANSACTION. */
@Transactional
public void registraEnvio(LocalDate fechaBase, int secuencia) {
    ControlCaptacion c = b02.findForUpdate();          // B02CONTROL-LOCK
    if (!fechaBase.equals(c.gbnpSaldosFec())) {
        c.setGbnpSaldosFec(fechaBase);
    }
    c.setGbnpSaldosSec(secuencia);                     // 1, 2 o 3
}   // si el STORE falla, el COBOL solo escribe mensaje y continúa; aquí la excepción hace rollback
/** 50004000-VA-SEC2: la secuencia 002 se genera solo cuando B00 marca el paso 105 en las tres posiciones. */
@Component
@RequiredArgsConstructor
public class EsperaP105Tasklet implements Tasklet {

    private final ControlPasoRepository b00;
    private final P108Properties props;          // intervalo 30 s y tope de espera

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) throws InterruptedException {
        Instant limite = Instant.now().plus(props.esperaMaximaP105());
        while (!b00.pasosEn(105, 105, 105)) {     // B00-NUM-PASO-01/02/03 = 105
            if (Instant.now().isAfter(limite)) {
                throw new EsperaAgotadaException("P105 no terminó en " + props.esperaMaximaP105());
            }
            Thread.sleep(props.intervaloP105().toMillis());   // WAIT(30)
        }
        return RepeatStatus.FINISHED;
    }
}

@Component
public class SaldoGbnpCalculator {

    private static final int ESCALA = 2;                        // WKS-SALDO-GBNP PIC 9(14)V99

    public BigDecimal calcula(Contrato c, int secuencia, ParametroP108 p,
                              TipoCorte corte, int diasInhabiles /* WS-DIAS-INHAXCTO */) {
        BigDecimal actual = c.sdoActual();
        if ((secuencia == 1 || secuencia == 2) && c.sdoPendiente().signum() > 0) {
            actual = actual.add(c.sdoPendiente());              // ADD B03-SDO-PENDIENTE TO B03-SDO-ACTUAL
        }
        BigDecimal base = actual.add(c.sdoProtBloq());
        if (corte.esTipo2() && p.valor() == 1) {
            // 20520205-SUMA-SDOINHAB suma los saldos finales de los días 1..N-1
            BigDecimal inhabiles = BigDecimal.ZERO;
            for (int i = 1; i < diasInhabiles; i++) {
                inhabiles = inhabiles.add(c.sdoFinInhab(i));
            }
            if (diasInhabiles == 0) throw new DiasInhabilesCeroException(c.numContrato());
            // COMPUTE sin ROUNDED: trunca a 2 decimales
            return base.add(inhabiles).divide(BigDecimal.valueOf(diasInhabiles), ESCALA, RoundingMode.DOWN);
        }
        return base.setScale(ESCALA, RoundingMode.DOWN);
    }
}

Saldos negativos enviados como positivos

WKS-SALDO-GBNP (línea 292) es 9(14)V99 sin signo y WKS-DET-GBNP-SDO es un editado ZZZ...9.99 sin signo. Un contrato sobregirado aparece en GBNP con saldo positivo.

Moneda no mapeada viaja en LOW-VALUES

50002100-MUEV-DET solo traduce 01 y 05 (líneas 1417-1421); con cualquier otra moneda el campo queda en LOW-VALUES y el detalle se escribe igual.

Espera sin límite del P105

50004000-VA-SEC2 consulta B00CTRLPASO con WAIT(30) hasta que los tres pasos valgan 105, sin tope de tiempo (líneas 1674-1703).

Errores de B02 no detienen el envío

50000200-ACT-B02 registra mensajes ante fallas de BEGIN, STORE o END-TRANSACTION pero no aborta; el archivo se envía aunque B02 no quede marcado.

Modificación del buffer de B03

50002300-CALCULA-SALDO suma B03-SDO-PENDIENTE sobre B03-SDO-ACTUAL en el área de la base abierta en UPDATE (línea 1604). No hay STORE de B03, pero cualquier lógica posterior sobre el mismo registro ve el saldo alterado.

Hardcodes

Clave de banco 999, handle VDBAL, sistema MX, buzón 50025S01, CSI 10 como VDM y el resto como QRO, WFL ENVIAGBNPSDOS y esperas WAIT(5) y WAIT(30) están compilados.
