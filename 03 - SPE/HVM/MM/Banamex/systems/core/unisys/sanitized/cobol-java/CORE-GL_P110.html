/** LOG151-COMP: ACCESS MODE IS RANDOM, ACTUAL KEY IS LOG-KEY (número relativo de registro). */
public final class ComplementoLog151 implements AutoCloseable {

    private static final int LONGITUD = 540;          // REG-LOG-COMP PIC X(540)
    private final FileChannel canal;

    public ComplementoLog151(Path archivo) throws IOException {
        this.canal = FileChannel.open(archivo, StandardOpenOption.READ);
    }

    /** READ ... INVALID KEY -> Optional.empty(). La llave es 1-based como en MCP. */
    public Optional<byte[]> leer(long llave) throws IOException {
        long pos = (llave - 1) * LONGITUD;
        if (llave < 1 || pos + LONGITUD > canal.size()) return Optional.empty();
        ByteBuffer buf = ByteBuffer.allocate(LONGITUD);
        canal.read(buf, pos);
        return Optional.of(buf.array());
    }

    @Override public void close() throws IOException { canal.close(); }
}

/** WKS-PARAMETROS: PROCEDURE DIVISION USING WKS-PARAMETROS. */
public record CorridaBit(TipoCorrida tipo, int sistema) {

    /** W77-TIPO: 1 = línea, 2 = batch (niveles 88 W88-TIPO-LINEA / W88-TIPO-BATCH). */
    public enum TipoCorrida { LINEA, BATCH;
        static TipoCorrida de(int v) {
            return switch (v) { case 1 -> LINEA; case 2 -> BATCH;
                default -> throw new IllegalArgumentException("Tipo de proceso no soportado: " + v); };
        }
    }

    /** W77-TIPO-PROCESO del movimiento: 0..19 línea, 20..99 batch. */
    public static TipoCorrida delMovimiento(int tipoProc) {
        return tipoProc <= 19 ? TipoCorrida.LINEA : TipoCorrida.BATCH;
    }
}

/** WKS-TR500-CIG: trailer tipo "9" de la BIT. */
public record TrailerBit(long registros,          // 9(13)
                         BigDecimal importe,       // 9(16)V9(2) -> escala 2
                         long claves) {            // 9(08)
    public static TrailerBit vacio() { return new TrailerBit(0, BigDecimal.ZERO.setScale(2), 0); }
}

/** Combina cada movimiento de LOG151 con su complemento (APUDES + 1). */
@Component
@StepScope
@RequiredArgsConstructor
public class Log151ConComplementoReader implements ItemStreamReader<MovimientoLog151> {

    private final FlatFileItemReader<RegistroLog151> log151;
    private final ComplementoLog151 complemento;

    @Override
    public MovimientoLog151 read() throws Exception {
        RegistroLog151 r = log151.read();
        if (r == null || r.funcion() == 99) return null;          // fin de archivo o función 99
        Optional<byte[]> comp = Optional.empty();
        if (r.apudes() > 0) {
            comp = complemento.leer(r.apudes() + 1);              // COMPUTE LOG-KEY = APUDES + 1
            if (comp.isEmpty()) {
                log.warn("Complemento inexistente para APUDES {}", r.apudes());   // W77-ERROR-BIT
            }
        }
        return new MovimientoLog151(r, comp.map(Complemento::parse).orElse(null));
    }
}

@Component
@StepScope
@RequiredArgsConstructor
public class BitItemProcessor implements ItemProcessor<MovimientoLog151, RegistroBit> {

    private final CorridaBit corrida;
    private final IntercompanyService intercompany;
    private final BitMapper mapper;

    @Override
    public RegistroBit process(MovimientoLog151 m) {
        Integer claveCia = 0;                                            // MOVE ZEROS TO WKS-CLAVE-CIA
        if (m.funcion() == 1 && (m.status() == 1 || m.status() == 2)) {
            claveCia = intercompany.claveDe(m.cliente(), m.contrato());  // 231000-CARGA-INTERCOMPANY
        }
        boolean coherente = CorridaBit.delMovimiento(m.tipoProc()) == corrida.tipo();
        boolean funcionValida = m.funcion() == 1 || m.funcion() == 2;
        boolean origenValido  = m.origen() == 1 || m.origen() == 3;     // cambio R08-001

        if (!(coherente && funcionValida && origenValido)) {
            return null;                                                 // no va a la BIT
        }
        return mapper.aBit(m, claveCia);                                 // 232000-GRABA-MOVSCIG
    }
}
@Service
@RequiredArgsConstructor
public class IntercompanyService {

    private final ClienteIntercompanyRepository repo;   // catálogo S080BD01CON (L710_CONSUL_DETALLE)
    private Map<Long, Integer> porCliente;              // WKS-TAB-CLIENTES

    @PostConstruct
    @Transactional(readOnly = true)
    void carga() {
        // En COBOL la tabla es OCCURS 2500 y la carga no tiene tope.
        // Si hay clientes repetidos gana el último, igual que el recorrido completo del fuente.
        porCliente = repo.findAllVigentes().stream()
                .collect(Collectors.toMap(ClienteIntercompany::cliente,
                                          ClienteIntercompany::inbu,
                                          (primero, ultimo) -> ultimo));
    }

    /** 21016-BUSCA-CLIENTE: la búsqueda es por cliente aunque la guarda pregunta por contrato. */
    public int claveDe(long cliente, long contrato) {
        if (contrato <= 0) return 0;
        return porCliente.getOrDefault(cliente, 0);
    }
}
@Component
public class BitMapper {

    private TrailerBit trailer = TrailerBit.vacio();

    public RegistroBit aBit(MovimientoLog151 m, int claveCia) {
        var b = RegistroBit.builder()
                .tipoReg('5')
                .funcion(m.funcion()).sistema(m.sistema())
                .producto(m.producto()).instrumento(m.instrumento())
                .sucProm(m.sucProm())
                .contrato(m.contratoTexto())                 // número de contrato como String
                .claveIntercompany(claveCia);
        BigDecimal suma = BigDecimal.ZERO.setScale(2);
        for (int i = 0; i < 5; i++) {                        // CVETRAN1..5, INDLEY, ESQCON, IMPORTE
            var g = m.grupo(i);
            b.grupo(i, g.cveTran(), g.indLey(), g.esqCon(), g.importe());
            suma = suma.add(g.importe());                    // ADD A00-R01-IMPORTEn TO WKS-TR500-TOT-IMP
        }
        if (m.apudes() > 0 && m.complemento() != null) {
            b.datosComplemento(m.complemento());             // nómina, contrato origen, referencias
        }
        trailer = new TrailerBit(trailer.registros() + 1, trailer.importe().add(suma), trailer.claves());
        return b.build();
    }
}
/** 30190-CUENTA-CVES con PERFORM VARYING W77-IND 1..5 UNTIL W77-BIT-SALIDA = 1. */
static int clavesParaTrailer(RegistroBit b) {
    if (b.funcion() != 1) {
        return 1;                                  // función distinta de 1: cuenta una y sale
    }
    int claves = 0;
    for (int i = 0; i < 5; i++) {
        var g = b.grupo(i);
        boolean vacio = g.cveTran() == 0 && g.importe().signum() == 0;
        if (vacio) {
            if (i == 0) { claves++; continue; }    // el primer grupo cuenta aunque venga vacío
            break;                                 // primer grupo vacío posterior: termina
        }
        claves++;
    }
    return claves;
}

Riesgos de migración
Primer registro de LOG151 sin evaluar

200000-PROCESO lee el primer registro INTO A01-R01-DATOS (línea 2819) y no INTO A00-R01-REGMOV, de modo que 230000-ESCOGE-MOVS evalúa por primera vez el área de movimiento sin datos del archivo. Si el primer registro de LOG151 es un movimiento y no un header, no llega a la BIT.

Error de complemento descartado

W77-ERROR-BIT se activa por INVALID KEY en LOG151-COMP (líneas 2829 y 2906) y se limpia en la línea 3276 sin consultarse. RN-CORE-GL-784 lo describe como marca de integridad rota, pero en el fuente no tiene efecto.

Tabla de clientes desbordable

WKS-CLIENTES-BU es OCCURS 2500 (línea 2079); el ciclo de búsqueda admite hasta 5000 y la carga 12610-CARGA-CLIENTES no tiene límite. Con más de 2500 clientes hay escritura fuera de la tabla.

Búsqueda intercompany sin corte

21016-BUSCA-CLIENTE mueve 5001 a W77-SALIDA (PIC 9) en lugar de a W77-BUSCA; el recorrido no se detiene y en clientes duplicados queda la última coincidencia. El costo es lineal por movimiento.

Transmisión sin verificación

700000-ENVIA-BIT ejecuta CALL SYSTEM WFL sin revisar resultado.
