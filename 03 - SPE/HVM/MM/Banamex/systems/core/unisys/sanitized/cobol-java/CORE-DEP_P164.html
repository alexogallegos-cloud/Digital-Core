El TASKVALUE trae una fecha opcional AAAAMMDD y el tipo de proceso (1 a 5; 0 corre ISI, réplica, B52 y B25 en ese orden). La fecha base sale de B02-FECHA-LOTE o del TASKVALUE, y si el mes de la fecha siguiente es distinto al de la base la corrida se considera mensual. En extracción, B52 se exporta completo en la corrida mensual y sólo con fecha de actualización igual a la base en la diaria, con acumulados que se ponen en cero cuando no son del día o del mes; B25 exporta los pagos cuyo último pago es de la fecha base, limpiando nulos de la referencia. En réplica, arma un archivo con la cola B53, lo transfiere al host contraparte definido por una tabla fija de nombres de equipo, espera el archivo de la otra cobertura y aplica sus importes a B52.

/** WKS-TASKVALUE PIC 9(10): AAAAMMDD (opcional, 0 si no viene) + tipo de proceso 9(02). */
public record P164Parametros(Optional<LocalDate> fecha, TipoProceso proceso) {

    public enum TipoProceso { TODOS, CTRLDEPRET, ISILOANS, REPLICA, PGOSPENDPE, RDORMANT }  // 0..5

    public static P164Parametros de(long taskValue) {
        long fech = taskValue / 100;                  // WKS-TSK-FECH
        int proc = (int) (taskValue % 100);           // WKS-TSK-PROCESO
        if (proc > 5) throw new TaskValueInvalidoException(taskValue);   // el fuente no lo valida: cae en TODOS
        var tipo = TipoProceso.values()[proc];
        if (fech == 0) return new P164Parametros(Optional.empty(), tipo);
        if (tipo == TipoProceso.REPLICA) {
            throw new TaskValueInvalidoException(taskValue);   // réplica con fecha: no permitido
        }
        int aa = (int) (fech / 10000), mm = (int) (fech / 100 % 100), dd = (int) (fech % 100);
        if (aa <= 2009 || mm < 1 || mm > 12 || dd < 1 || dd > 31) {
            throw new TaskValueInvalidoException(taskValue);   // "ERROR TASKVALUE" + DMTERMINATE
        }
        return new P164Parametros(Optional.of(LocalDate.of(aa, mm, dd)), tipo);  // rechaza 31 de febrero
    }
}
/** WKS-CTRLDEPRET-DETALLE, tipo de registro 2. Importes 9(16)V99 sin signo, escala 2. */

        BigDecimal limRetDia, BigDecimal limRetMen,
        LocalDate fecLimDia, LocalDate fecLimMen,
        int indBloqueo, LocalDate fecBloqueo,
        BigDecimal sdoConDep,
        long inteRetDia, long inteRetMen,   // 9(08): conteo de intentos, no importe
        LocalDate fecAcInte, int indReplica) {

    /** Formato fijo para Teradata: cada importe en 18 posiciones con 2 decimales implícitos. */
    public String aLinea() {
        return new FixedWidth(330)
                .num(1, 2).num(12, numCliente).num(2, csi).fecha(fecUpdate)

                .imp(18, acumDepDia).imp(18, acumDepMen).imp(18, acumRetDia).imp(18, acumRetMen)
                // ... resto de campos en el orden del layout, FILLER X(73)
                .toString();
    }

@Component

public class FechaBaseTasklet implements Tasklet {

    private final ControlCaptacionRepository b02;   // CORE-DEPB02CONTROL
    private final CalendarioBancario calendario;    // LOCSUP función 13, formato 12

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        var ec = ctx.getStepContext().getStepExecution().getJobExecution().getExecutionContext();
        var p = (P164Parametros) ec.get("parametros");
        var control = b02.find().orElseThrow();

        LocalDate base = control.fechaLote();
        LocalDate prox = control.fechaLinea();
        if (p.fecha().isPresent()) {                  // reproceso con fecha del operador
            base = p.fecha().get();
            prox = calendario.avanza(base, 1);        // 20040300-PROYECTA-FECHA
        }
        boolean mensual = prox.getMonthValue() != base.getMonthValue();   // WKS-ES-MENSUAL

        ec.put("fechaBase", base);
        ec.put("esMensual", mensual);
        return RepeatStatus.FINISHED;
    }

@Component
@StepScope
public class B52Processor implements ItemProcessor<B52CtrlDepRet, CtrlDepRetDetalle> {

    @Value("#{jobExecutionContext['fechaBase']}") private LocalDate fechaBase;
    @Value("#{jobExecutionContext['esMensual']}") private boolean esMensual;
    @Value("#{jobExecutionContext['csiLocal']}")  private int csiLocal;

    @Override
    public CtrlDepRetDetalle process(B52CtrlDepRet b52) {
        if (!esMensual && !b52.fecUpdate().equals(fechaBase)) {
            return null;                              // diario: sólo lo actualizado en la fecha base
        }
        return AcumuladosVigentes.aplica(b52, fechaBase, csiLocal);   // 50003300 + 50003500
    }
}
// Reader: RepositoryItemReader sobre CORE-DEPB52CTRLDEPRET (FIND NEXT desde BEGINNING).
// Un status de base distinto de 0 y 1 aborta el step (DMTERMINATE).
public final class AcumuladosVigentes {

    /** 50003500-LIMPIA-ACUMULADOS: el acumulado guardado sólo vale para su día o su mes. */
    static CtrlDepRetDetalle aplica(B52CtrlDepRet b, LocalDate base, int csi) {
        boolean delDia = b.fecAcum().equals(base);
        boolean delMes = YearMonth.from(b.fecAcum()).equals(YearMonth.from(base));
        LocalDate fecAcum = delMes ? b.fecAcum() : null;   // el bloque mensual decide la fecha final

        boolean inteDia = b.fecAcInte().equals(base);
        boolean inteMes = YearMonth.from(b.fecAcInte()).equals(YearMonth.from(base));

        return new CtrlDepRetDetalle(b.numCliente(), csi, b.fecUpdate(), b.tipPer(), b.ubiGeo(),
                delDia ? b.acumDepDia() : BigDecimal.ZERO.setScale(2),
                delMes ? b.acumDepMen() : BigDecimal.ZERO.setScale(2),
                delDia ? b.acumRetDia() : BigDecimal.ZERO.setScale(2),
                delMes ? b.acumRetMen() : BigDecimal.ZERO.setScale(2),
                b.fecCrea(), fecAcum,
                b.limDepDia(), b.limDepMen(), b.limRetDia(), b.limRetMen(),
                b.fecLimDia(), b.fecLimMen(), b.indBloqueo(), b.fecBloqueo(),
                delDia ? b.sdoConDep() : BigDecimal.ZERO.setScale(2),
                inteDia ? b.inteRetDia() : 0, inteMes ? b.inteRetMen() : 0,
                b.fecAcInte(), b.indReplica());
    }

@Component
@StepScope
public class B25Processor implements ItemProcessor<B25PagoPendiente, PagoPendienteDetalle> {

    private static final int LONG_REFER = 40;
    @Value("#{jobExecutionContext['fechaBase']}") private LocalDate fechaBase;

    @Override
    public PagoPendienteDetalle process(B25PagoPendiente b25) {
        if (!b25.fechaUltPago().equals(fechaBase)) return null;
        return PagoPendienteDetalle.de(b25.conReferAlfa(sinNulos(b25.referAlfa())));
    }

    /** 00000000-VALIDA-NUL: cada byte @00@ se sustituye por un espacio (no se elimina). */
    static String sinNulos(String refer) {
        var sb = new StringBuilder(LONG_REFER);
        for (int i = 0; i < LONG_REFER; i++) {
            char ch = i < refer.length() ? refer.charAt(i) : ' ';
            sb.append(ch == '\u0000' ? ' ' : ch);
        }
        return sb.toString();
    }
}

Riesgos de migración
Espera de réplica sin tiempo máximo

50004000-ARRIBO espera en ciclos de 300 segundos hasta que el archivo foráneo es residente o B00 marca fin de paso (130 o 181). No hay tope de intentos y el mensaje ERROR ARCHIVO NO ARRIBO (líneas 3097-3109) nunca se puede emitir.

Topología de hosts compilada

Los ocho nombres de equipo y sus contrapartes (líneas 1999-2059), los CSI 10 y 04 y el pack CAPTACION están en el código; un host fuera de la tabla genera un comando XFER sin origen ni destino.

Validación de versión fuera de la exención

En 40002000-TITLE-LIBRERIAS el punto de la línea 1919 hace que el IF de S000-CTR-CVEERROR (línea 1921) se evalúe también en hosts de prueba.

STORE de B52 sin reversa

Un error al guardar B52 durante la réplica (líneas 3609-3623) se registra en F02-PROBLEMAS y la transacción se cierra con END TRANSACTION; los acumulados de ese cliente divergen entre coberturas.

Delta diario dependiente de B52-FEC-UPDATE

La corrida diaria sólo exporta registros con fecha de actualización igual a la fecha base (línea 2409); los cambios de acumulados que no muevan esa fecha no llegan a Teradata hasta la corrida mensual.
