CORE-DEP · P155 a Java 21 Spring Boot

P155 exige que CORE-DEPB00CTRLPASO marque el paso 130 en los tres nodos y toma de CORE-DEPB02CONTROL la fecha de lote, la de línea y el CSI. Si el CSI es 04 y la fecha de línea cae en otro mes, recorre CORE-DEPB03CONTRATOS y escribe en CPESEC un detalle por cada contrato vigente de producto 1, instrumento 3 y moneda 1 que no sea de artículo 61, con datos de B06 y la comisión que devuelve la librería S016 L422. Después borra de B25 los pagos con estatus 1 o 2, con fecha de último cobro vencida o con saldo cero y fecha centinela, y deja cada borrado en el archivo REGBORRADOS. Al final procesa el archivo de clientes del S016 de los dos nodos y el archivo de bloqueos MODSTATUS, que pone bloqueo 66 o restricción 17 a los contratos. Un TASKVALUE con fecha permite además corregir manualmente la última fecha de archivo aplicado.

public record FechaProcesoCpe(LocalDate fechaLote, LocalDate fechaLinea) {

    /** WS-FIN-MES = 1 cuando la fecha de línea ya está en otro mes que la de lote. */
    public boolean esFinDeMes() {
        return fechaLote.getMonthValue() != fechaLinea.getMonthValue();
    }

    /**
     * WS-CPE-FECHA: último día del mes anterior a la fecha de línea. El fuente lo calcula con
     * una tabla de días por mes y bisiesto por divisibilidad entre 4; YearMonth aplica la regla
     * gregoriana completa, que coincide con la del fuente hasta el año 2099.
     */
    public LocalDate fechaCpe() {
        return YearMonth.from(fechaLinea).minusMonths(1).atEndOfMonth();
    }

@Component
@StepScope
@RequiredArgsConstructor
public class CpesecProcessor implements ItemProcessor<ContratoCaptacion, CpesecRecord.Detalle> {

    // WS-ES-CTAPROD VALUE 00010003001: producto 1, instrumento 3, moneda 1 (en application.yml)
    private final CpeProperties props;
    private final HistoricoRepository b06;          // CORE-DEPB06HISTORICO
    private final ComisionS016Client s016;          // L422 COMXCTA

    @Override
    public CpesecRecord.Detalle process(ContratoCaptacion b03) {
        if (b03.marcaClon() == 3) return null;                              // clones fuera
        boolean esCpe = b03.producto() == props.producto() && b03.instrumento() == props.instrumento()
                && b03.moneda() == props.moneda();
        boolean vigente = b03.status() == 0 || b03.status() == 1;           // W88-CTO-VIG
        boolean noArt61 = Set.of(0, 1, 6, 7).contains(b03.staBenef());      // W88-NO-ART61
        if (!(esCpe && vigente && noArt61)) return null;

        var com = s016.comisionPorCuenta(b03.contrato());
        var hist = b06.findByContrato(b03.contrato());                      // NOTFOUND -> ceros
        return new CpesecRecord.Detalle(b03.contrato(),
                hist.map(h -> h.contadorComisiones(50)).orElse(0),
                hist.map(Historico::diasCiclo).orElse(0),
                hist.map(h -> h.promedioCiclo(12)).orElse(BigDecimal.ZERO.setScale(2)),
                b03.tipoPersona(), b03.nombre(), b03.status(),
                b03.tasaAnterior(), b03.interesesCapitalizados(), b03.impuestoRetenido(),
                hist.map(Historico::tasaBruta).orElse(BigDecimal.ZERO.setScale(4)),
                com.claveCobro(), com.importe());
    }
}

@Component
@RequiredArgsConstructor
public class ComisionS016Client {

    private final L422Gateway l422;              // S016 L422, operación COMXCTA
    private final CpeProperties props;

    public record Comision(int claveCobro, BigDecimal importe) {}

    public Comision comisionPorCuenta(String contrato) {
        var r = l422.comisionPorCuenta(500, 155, 1, 3, contrato);    // sistema, paso, producto, instrumento
        if (r.resultado() != 0) {
            log.warn("L422 COMXCTA resultado {} para {}: se usa comisión por omisión", r.resultado(), contrato);
            return new Comision(props.claveCobroOmision(),          // 2
                                props.comisionOmision());           // 100.00, PIC 9(06)V99
        }
        return new Comision(r.claveCobro(4), r.comision(4).setScale(2, RoundingMode.DOWN));   // ocurrencia 04
    }
}

@Bean
Step depuraB25(JobRepository repo, PlatformTransactionManager tx,
               JpaCursorItemReader<PagoPendiente> b25,
               @Value("#{jobExecutionContext['fechaLote']}") LocalDate fechaLote,
               JpaItemWriter<PagoPendiente> borraB25,                     // DELETE

    // fecha centinela 99999999 de B25-FECHA-LTECOBR
    Predicate<PagoPendiente> seBorra = p ->
            p.status() == 1 || p.status() == 2
         || !p.fechaUltCobro().isAfter(fechaLote)                         // NOT > fecha de proceso
         || (p.saldo().signum() == 0 && p.fechaUltCobroCentinela());

    return new StepBuilder("depuraB25", repo)
            .<PagoPendiente, PagoPendiente>chunk(500, tx)
            .reader(b25)
            .processor(p -> seBorra.test(p) ? p : null)

@Component
@RequiredArgsConstructor
public class BloqueoExpedienteProcessor implements ItemProcessor<RegistroModStatus, ContratoCaptacion> {

    static final int BLOQUEO_EXPEDIENTE = 66, RESTRICCION_EXPEDIENTE = 17, BLOQUEO_EXCLUIDO = 15;

    private final ContratoRepository b03;
    private final HistoricoRepository b06;
    @Value("#{jobExecutionContext['fechaLote']}") private LocalDate fechaLote;

    @Override
    @Transactional
    public ContratoCaptacion process(RegistroModStatus r) {
        var cto = b03.findForUpdate(r.contrato()).orElse(null);
        if (cto == null || cto.tipoBloqueo() == BLOQUEO_EXCLUIDO) return null;
        var hist = b06.findForUpdate(r.contrato())
                .orElseThrow(() -> new ProcesoAbortadoException("NO HAY CTO EN B06 " + r.contrato()));

        switch (r.tipoMovimiento()) {
            case 1 -> { cto.bloquea(BLOQUEO_EXPEDIENTE); hist.setFechaBloqueo(fechaLote); }
            case 2 -> {
                cto.setRestricciones(RESTRICCION_EXPEDIENTE);
                if (!(cto.status() == 1 && cto.tipoBloqueo() == BLOQUEO_EXPEDIENTE)) {
                    cto.bloquea(BLOQUEO_EXPEDIENTE);                 // status 1, tipo 66
                    hist.setFechaBloqueo(fechaLote);
                }
            }
            default -> { return null; }                            // otro tipo: el fuente no enciende banderas
        }
        return cto;
    }
}
Riesgos de migración
Comisión por omisión de 100.00

Si L422 COMXCTA responde con error, 50450000 (líneas 1008-1010) escribe en CPESEC la clave de cobro 2 y una comisión de 100.00 sin ninguna marca. Los consumidores del archivo no pueden distinguirla de una comisión calculada.

Borrado de pagos con saldo

La condición B25-FECHA-LTECOBR NOT > WS-FECHA-PRO-AMD (línea 1055) borra pagos pendientes vencidos sin revisar el saldo. Confirmar que ese saldo ya se cobró o castigó en otro paso.

Auditoría posterior al borrado

Cada DELETE de B25 se confirma antes de escribir REGBORRADOS. Una falla de escritura deja registros borrados sin bitácora.

Reinicio frágil

20000150-VAL-PASO pone en cero el paso del nodo 01 al inicio; si el job falla después, un reinicio no pasa la validación del P130 y requiere corrección manual de B00CTRLPASO.

Hardcodes
