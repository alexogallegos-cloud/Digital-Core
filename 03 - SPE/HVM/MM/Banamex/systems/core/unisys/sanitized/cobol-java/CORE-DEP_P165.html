/** REG-PAGOS (estructura B24): 66 bytes, vista según PAGO-TREG. */
public sealed interface PagoRecord permits PagoRecord.Header, PagoRecord.Detalle, PagoRecord.Trailer {

    /** PAGO-TREG = 01. */
    record Header(LocalDate fechaProceso,      // 9(08) AAAAMMDD
                  String sistemaOrigen,        // "CORE-DEP"
                  int csi, int paso)           // paso = 165
            implements PagoRecord {}

    /** PAGO-TREG = 02. */
    record Detalle(int csiDestino,
                   String sucursalCuenta,      // 9(16): medio de acceso como String
                   String referencia,          // 9(12)
                   BigDecimal importeOriginal, // 9(10)V99 -> escala 2
                   BigDecimal importePagado)   // 9(10)V99
            implements PagoRecord {}

    /** PAGO-TREG = 03. */
    record Trailer(long registros,             // 9(08)
                   BigDecimal totalOriginal,   // 9(12)V99
                   BigDecimal totalPagado)     // 9(12)V99
            implements PagoRecord {}

    BigDecimal TOPE_DETALLE = new BigDecimal("9999999999.99");
}

@ConfigurationProperties("p165.pagos")
public record PagosProperties(String titulo, int csiDestino, String fechaEnNombre) {

    /** WS-ETIQ-PAG-SIS es PIC 9(03): de PAGO1-SISTEMA X(04) solo sobreviven los 3 dígitos de la derecha. */
    public String resuelve(String sistema, int csiOrigen, LocalDate fecha) {
        String sis3 = sistema.substring(sistema.length() - 3);
        return titulo.replace("{sistema}", sis3)
                     .replace("{csiOrigen}", "%02d".formatted(csiOrigen))
                     .replace("{csiDestino}", "%02d".formatted(csiDestino))
                     .replace("{fecha}", fecha.format(DateTimeFormatter.ofPattern(fechaEnNombre)));
    }
}

@Configuration
public class ResultadoDispersionJobConfig {

    /** PROCEDURE DIVISION USING WS-VAL-165: la fase llega como parámetro del job. */
    @Bean
    Job resultadoDispersion(JobRepository repo, Step controlLote,
                            Flow fase1Pagos, Flow fase2Resultados) {
        JobExecutionDecider porFase = (job, step) -> new FlowExecutionStatus(
                "1".equals(job.getJobParameters().getString("val165")) ? "FASE1" : "FASE2");

        return new JobBuilder("CORE-DEPP165", repo)
                .start(controlLote)                       // 20000000-OPEN: B02, fechas, etiquetas
                .next(porFase)
                    .on("FASE1").to(fase1Pagos)           // 50010000-ARC-PAGOS
                .from(porFase)
                    .on("*").to(fase2Resultados)          // 51000000 + 52000000 + 53000000
                .end()
                .build();
    }
}
@Component
@RequiredArgsConstructor
public class ControlLoteTasklet implements Tasklet {

    private final ControlCaptacionRepository b02;   // OPEN INQUIRY: solo lectura
    private final P165Properties props;

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        ControlCaptacion control = b02.findControl()
                .orElseThrow(() -> new ProcesoAbortadoException("ERROR REG. DE CONTROL VACIO"));

        BigDecimal tipoCambio = control.tcTcambio().setScale(4);   // 9(04)V9(04)
        int csi = control.numCsi();
        // CSI 10 es Valle de México; cualquier otro valor se rotula como Monterrey
        String plaza = (csi == props.csiValle()) ? "VDM" : "MTY";
        String plazaInversa = plaza.equals("VDM") ? "MTY" : "VDM";

        var jc = ctx.getStepContext().getStepExecution().getJobExecution().getExecutionContext();
        jc.put("tipoCambio", tipoCambio);
        jc.put("fechaLote", control.fechaLote());
        jc.put("csi", csi);
        jc.put("plaza", plaza);
        jc.put("plazaInversa", plazaInversa);
        jc.put("horaMin", props.horaMin());              // MOVE 2459 TO WS-HORAMIN
        jc.put("horaOrdenadoTodo", props.horaOrdTodo()); // MOVE 9999 TO WS-ETIQ-ORDTOD-HORA
        return RepeatStatus.FINISHED;
    }
}

/** 50011150-MUEVO-B24 */
public PagoPuente aPuente(PagoPendienteB24 b) {
    // WS-PAGO-IMPPAG es PIC 9(10)V99 sin signo: COBOL guarda el valor absoluto y trunca a 10 enteros.
    BigDecimal pagado = b.importe().subtract(b.saldo());
    if (pagado.signum() < 0) {
        log.warn("B24 {} con saldo mayor al importe; COBOL lo graba como {}", b.referencia(), pagado.abs());
        pagado = pagado.abs();
    }
    return PagoPuente.detalle(b.sistemaArch(), 24, b.medAcceso(), b.referencia(),
                              b.importe(), Fixed.trunca(pagado, 10, 2));
}

/** Un writer por sistema destino; la clave de clasificación es (sistema, estructura). */
@Component
@StepScope
public class PagosPorSistemaWriter implements ItemStreamWriter<PagoPuente> {

    private final PagosProperties props;
    private String sistemaActual;                 // WS-SISTEMA
    private ArchivoPagos archivo;                 // PAGOS (B24) o PAGOS2 (B25)

    @Override
    public void write(Chunk<? extends PagoPuente> chunk) {
        for (PagoPuente p : chunk) {
            if (!p.sistema().equals(sistemaActual)) {        // 50013100: corte de control
                if (archivo != null) archivo.cierraConTrailer();
                sistemaActual = p.sistema();
                archivo = ArchivoPagos.abre(p.estGen(),      // 24 -> header CORE-DEP/165/TREG 01
                        props.resuelve(p.sistema(), csi, fechaProceso));
            }
            if (p.detalle() == 1) archivo.escribeDetalle(p);
        }
    }

    @Override
    public void close() {                                    // AT END -> 50013400-GRABO-TRAILER
        if (archivo != null) archivo.cierraConTrailer();
    }
}
/** 50013400-GRABO-TRAILER */
void cierraConTrailer() {
    switch (estructura) {
        case B24 -> escribe(new PagoRecord.Trailer(registros,
                                                   totalOriginal,     // WS-SUMPAG-IMPORI 9(12)V99
                                                   totalPagado));     // WS-SUMPAG-IMPPAG 9(12)V99
        case B25 -> escribe(PagoB25.trailer(registrosB25));           // TIPREG 99: solo conteo
    }
    registros = 0; registrosB25 = 0;
    totalOriginal = BigDecimal.ZERO.setScale(2);
    totalPagado = BigDecimal.ZERO.setScale(2);
    cierra();                                                         // CLOSE ... WITH SAVE
}

/** 5200005700-ASIGNA-PRIORIDAD: Classifier de prioridad para el reporte. */
@Component
public class PrioridadClassifier implements Classifier<MovimientoOrdenado, Integer> {

    private static final int PRODUCTO_CTAMAE = 66, INSTRUM_TARJETA = 15, MN = 1;

    @Override
    public Integer classify(MovimientoOrdenado m) {
        boolean tarjeta = m.producto() == PRODUCTO_CTAMAE && m.instrumento() == INSTRUM_TARJETA;
        boolean mn = m.moneda() == MN;
        int p = m.prioridad();
        if (m.origenEas() == 2) {                              // 5200005701-PRIORIDAD-EAS
            if (p == 2 || p == 7 || p == 8) return tarjeta ? (mn ? 20 : 21) : 16;
            if (p == 4 || p == 9)           return tarjeta ? (mn ? 18 : 19) : 17;
        } else {                                               // 5200005702-PRIORIDAD-S111
            if (p == 2 || p == 7 || p == 8) return tarjeta ? (mn ? 9 : 10) : 2;
            if (p == 4 || p == 9)           return tarjeta ? (mn ? 7 : 8)  : 4;
            // prioridades 1 y 5 y producto 66 instrumento 11/12: líneas 4005-4029
        }
        // En COBOL WS-ORD-PRIORIDAD conserva el valor del registro anterior;
        // aquí se hace explícito y se falla para que el SME decida.
        throw new PrioridadNoAsignadaException(m);
    }
}
/** 53211220-IMPRIME-DETALLE: desglose de rechazos en MN y dólares. */
@Component
public class RechazoDolaresCalculator {

    private static final Set<Integer> EN_MN  = Set.of(7, 9, 18, 20);
    private static final Set<Integer> EN_DLS = Set.of(8, 10, 19, 21);

    public Desglose desglosa(MovimientoOrdenado m, BigDecimal tipoCambio /* 9(04)V9(04) */) {
        // WS-TB-IMPORTE suma los tres importes; la conversión solo usa IMPORTE e IMP-AUT
        BigDecimal total = m.importe().add(m.importe2()).add(m.impAut());
        BigDecimal base  = m.importe().add(m.impAut());

        if (EN_MN.contains(m.prioridad())) {
            if (tipoCambio.signum() == 0) throw new TipoCambioCeroException();
            BigDecimal dls = base.divide(tipoCambio, 2, RoundingMode.HALF_UP);   // COMPUTE ROUNDED
            return Desglose.dolares(total, dls, tipoCambio);
        }
        if (EN_DLS.contains(m.prioridad())) {
            BigDecimal mn = base.multiply(tipoCambio).setScale(2, RoundingMode.HALF_UP);
            return Desglose.dolares(mn, total, tipoCambio);
        }
        return Desglose.soloMn(total);
    }
}
Riesgos de migración
Prioridad heredada del registro anterior

En 5200005701-PRIORIDAD-EAS no hay rama para las prioridades 1 y 5, que sí entran a 5200005700 desde 52000057-VAL-PRIORIDAD (línea 3942). WS-ORD-PRIORIDAD no se reinicia por registro, así que esos movimientos EAS toman la prioridad del movimiento anterior y pueden caer en un renglón equivocado del reporte de cifras.

Truncamiento de importes en el archivo de pagos

PAGO1-IMPORI y PAGO1-IMPPAG son 9(12)V99 y se mueven a campos 9(10)V99 del detalle (líneas 2816-2817), mientras el trailer acumula en 9(12)V99. Un importe grande se trunca en el detalle pero no en el trailer.

Pérdida de signo en el importe pagado

WS-PAGO-IMPPAG es PIC 9(10)V99 sin signo (línea 515). Si el saldo de B24 supera al importe, el resultado negativo se graba como positivo.

Mismo título para PAGOS y PAGOS2 y corte solo por sistema

Ambos FD usan WS-ETIQ-PAGOS y el corte de 50013100 compara solo el sistema. Un sistema con pagos B24 y B25 produciría escritura sobre un archivo no abierto o un archivo sobrescrito.

Desglose en dólares sin protección

COMPUTE WS-VALDL ROUNDED divide entre WS-TIPCAM sin ON SIZE ERROR (línea 4417); un tipo de cambio en cero en B02 deja el resultado indefinido.

Hardcodes

CSI destino 10 en la ruta de pagos, CSI 10 como Valle de México, WS-HORAMIN 2459, hora 9999 del ordenado consolidado y el prefijo CORE-DEP/FILE/S están compilados.
