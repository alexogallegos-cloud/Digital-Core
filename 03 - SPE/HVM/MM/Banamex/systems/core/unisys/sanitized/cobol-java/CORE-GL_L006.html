/** SUMA_ARCHIVO: recorre LOG1 desde AUT+1 hasta la función 99. */
@Component
@RequiredArgsConstructor
class SumaArchivoLog {

    private final LectorLog1 log1;            // registros de 75 bytes, acceso por posición
    private final AcumuladorCruce acumulador;
    private final EstadoCruce estado;
    private final L006Properties props;       // l006.banco-legacycore: 2

    void ejecuta() {
        long aut = estado.autorizacion;
        RegistroLog r = log1.lee(++aut);
        while (r.funcion() != 99) {
            boolean altaActiva = r.funcion() == 1 && r.status() == 1;
            boolean origenValido = r.origen() == 1 || r.origen() == 2;
            boolean noAmbosLegacyCore = r.bancoDestino() != props.bancoLegacyCore()
                                  || r.bancoOrigen()  != props.bancoLegacyCore();
            if (altaActiva && origenValido && noAmbosLegacyCore) {
                acumulador.acumula(r, Signo.SUMA);         // SUMA_RESTA := 0
            }
            r = log1.lee(++aut);
        }
        estado.autorizacion = aut - 1;                     // AUT := * - 1
    }
}
@Component
@RequiredArgsConstructor
class AcumuladorCruce {

    private static final int LIQ_OTR_A_BNX = 1078;   // l006.claves.liquidacion-otr
    private static final int LIQ_BNX_A_OTR = 2129;   // l006.claves.liquidacion-bnx
    private final EstadoCruce estado;
    private final CatalogoClaves catalogo;           // ACLAVES: naturaleza por clave (S080)
    private final MovimientosCruceRepository movs;
    private final L006Properties props;

    /** ACUMULA_REG. Solo operaciones cruzadas: banco origen distinto de banco destino. */
    void acumula(RegistroLog r, Signo signo) {
        if (r.bancoOrigen() == r.bancoDestino()) return;
        ParCruce par = new ParCruce(r.bancoOrigen(), r.bancoDestino(), r.moneda());
        for (Ocurrencia o : r.ocurrencias()) {                 // hasta 5; corta en la primera clave 0
            if (o.clave() <= 0) break;
            BigDecimal importe = o.importeCentavos().movePointLeft(2).multiply(signo.factor());
            switch (o.clave()) {
                case LIQ_OTR_A_BNX -> movs.registra(r, o.clave(),
                        estado.liq1078.merge(par, importe, BigDecimal::add), estado.numLiq1078, par);
                case LIQ_BNX_A_OTR -> movs.registra(r, o.clave(),
                        estado.liq2129.merge(par, importe, BigDecimal::add), estado.numLiq2129, par);
                default -> {
                    var mapa = (r.bancoOrigen() == props.bancoLegacyCore()) ? estado.legacycore : estado.contraparte;
                    TotalesCruce t = mapa.computeIfAbsent(par, k -> new TotalesCruce());
                    switch (catalogo.naturaleza(o.clave())) {
                        case 1 -> { t.cargos = t.cargos.add(importe); t.numCargos++; }
                        case 2 -> { t.abonos = t.abonos.add(importe); t.numAbonos++; }
                        default -> log.warn("Clave {} sin naturaleza 1/2: se descarta", o.clave());
                    }
                }
            }
        }
    }
}

    RespuestaSaldos atiende(SolicitudSaldos s) {
        semaforos.tomaConsultaSaldos();               // RESET(CON_SDO)
        try {
            semaforos.esperaAnalisis();               // WAIT(ANALIZA)
            if (!base.abierta()) return RespuestaSaldos.error(5);       // base no disponible
            if (s.bancoOrigen() >= 13 || s.bancoDestino() >= 13 || s.moneda() >= 6)
                return RespuestaSaldos.error(7);                         // bancos o moneda inválidos
            return switch (s.funcion()) {
                case 0 -> op.inicializa(s);           // 9 si el par ya estaba inicializado
                case 1 -> op.consulta(s);
                case 2 -> op.yaInicializado(s) ? op.actualiza(s) : RespuestaSaldos.error(10);
                case 6 -> op.borraSaldosYMovimientos(s);
                case 7 -> op.grabaTodosYCierra(s);    // WAIT(SUMA) + triple ciclo + CLOSE
                default -> RespuestaSaldos.error(1);
            };
        } finally {
            semaforos.liberaConsultaSaldos();         // CAUSE(CON_SDO)
        }
    }

Riesgos de migración
Importes en punto flotante

Los acumuladores son DOUBLE y los importes del log se dividen entre 100 al acumular (líneas 738-783) y se multiplican por 100 al grabar (líneas 1402-1409). La aritmética binaria no garantiza centavos exactos; la versión Java usa BigDecimal.

Índices sin validar en la acumulación

ACUMULA_REG usa banco origen, banco destino y moneda del log como índices de arreglos [0:12,0:12,0:05] sin validar el rango; un valor fuera de rango provocaría una falla de índice en la librería compartida. SALDOS sí valida (líneas 1615-1617).

Descarte silencioso de claves

Claves cuya naturaleza en el catálogo no es 1 ni 2 no se acumulan en ningún total ni se reportan (líneas 756-784).

Comando de borrado sin confirmación

El valor 6 en TASKVALUE borra saldos y movimientos del día para el sistema 0500 compilado (líneas 1979-1989). La versión Java lo pone detrás de un endpoint con rol.

Hardcodes

Banco LegacyCore = 2, claves de liquidación 1078 y 2129, sistema 0500, límites de 12 bancos y 5 monedas, ciclo de 30 segundos y título por defecto de la librería CONTROL están compilados. Se trasladan a configuración. No hay credenciales en el fuente.

Pendiente de revisión SME
  • GRABAMOVTO graba en B15-MOV-IMPMOV el acumulado de liquidación del par (IMPMOV := LIQ_1078[...] y LIQ_2129[...], líneas 741 y 750), no el importe del movimiento individual. Confirmar con el SME si los consumidores de MOVIMIENTOS esperan el acumulado.
  • ACTSDO solo asigna su valor de retorno en la rama de error (línea 1415); en la rama de éxito no hay asignación. Confirmar qué devuelve la plataforma en ese caso (se asumió 0).
  • ANALIZA_FUNCION declara un SIGNO local INTEGER (línea 812) que oculta al SIGNO global REAL; ACUMULA_REG usa el global. No se detectó efecto, pero conviene confirmarlo al revisar ANALIZA_FUNCION completo, que no se transpiló aquí.

  • La regla RN-CORE-GL-1151 del brain describe L006 como librería genérica de acceso a datos; el fuente muestra una función específica de conciliación cruzada interbancaria. El inventario no trae reglas asociadas, por lo que ningún segmento cita reglas.
