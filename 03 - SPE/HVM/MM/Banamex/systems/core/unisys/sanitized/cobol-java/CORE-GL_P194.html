CORE-GL · P194 a Java 21 Spring Boot

P194 enlaza sus librerías por CTLVERS, obtiene el CSI y el nombre del equipo, y consulta el control diario CORE-GLB01SISDIA para tomar la fecha de proceso (o la toma del VALUE de la tarea). Con LOCSUP proyecta la fecha hacia atrás y la convierte a formato juliano; en producción calcula cuatro fechas de 1 a 4 días atrás y en los equipos de prueba una sola fecha desplazada cuatro días más. Por cada renglón del archivo CORE-GL/FILE/I10/DEPURAR/DIRECTORIOS separa usuario, directorio, pack, CSI y días, arma el título del directorio y, si el CSI coincide con el del equipo o el equipo es el CSI 32, llama a REMUEVE_ARCHIVOS con la fecha de retención.

/** Un renglón de CORE-GL/FILE/I10/DEPURAR/DIRECTORIOS: usuario.directorio.pack.csi.dias */
public record DirectorioADepurar(String usuario,     // WKS-USER X(04)
                                 String directorio,  // WKS-DIR  X(30)
                                 String pack,        // WKS-PACK X(17)
                                 int csi,            // WKS-CSI  X(02) REDEFINES 9(02)
                                 String dias) {      // WKS-DIAS X(02): P194 no lo usa (ver segmento s6)
}

@Component
public class DirectorioLineMapper implements LineMapper<DirectorioADepurar> {

    @Override
    public DirectorioADepurar mapLine(String linea, int numero) {
        String[] p = linea.split("\\.", -1);                    // UNSTRING ... DELIMITED BY "."
        if (p.length < 5) {
            throw new FlatFileParseException("Renglón sin cinco campos", linea, numero);
        }
        return new DirectorioADepurar(corta(p[0], 4), corta(p[1], 30), corta(p[2], 17),
                                      Integer.parseInt(p[3].strip()), corta(p[4], 2));
    }

    /** UNSTRING a un campo alfanumérico trunca por la derecha al tamaño del receptor. */
    private static String corta(String s, int n) { return s.length() > n ? s.substring(0, n) : s; }
}

@Configuration
public class DepuracionJobConfig {

    @Bean
    Job depuraDirectorios(JobRepository repo, Step fechasRetencion, Step depura) {
        return new JobBuilder("CORE-GLP194", repo)
                .start(fechasRetencion)        // 0100-LIBRERIAS + 0150-CALCULA-FECHA-ATRAS
                .next(depura)                  // 0200-LLAMA-BORRADO por renglón
                .build();
    }

    @Bean
    Step depura(JobRepository repo, PlatformTransactionManager tx,
                FlatFileItemReader<DirectorioADepurar> directorios,
                DepuracionProcessor processor, ItemWriter<SolicitudBorrado> borrado) {
        return new StepBuilder("depura", repo)
                .<DirectorioADepurar, SolicitudBorrado>chunk(1, tx)   // un directorio por transacción
                .reader(directorios)
                .processor(processor)
                .writer(borrado)
                .build();
    }
}

@Component
@RequiredArgsConstructor
public class FechasRetencionTasklet implements Tasklet {

    private final CalendarioBancario calendario;      // THECALENDAR IN LOCSUP
    private final P194Properties props;               // equipos-prueba: [ACYPBETA, VDMBETA, ...]
    private final HostInfo host;                      // CSIID IN SOPORTECOMS

    @Override
    public RepeatStatus execute(StepContribution sc, ChunkContext ctx) {
        LocalDate base = fechaBase(ctx);                              // VALUE de la tarea o WKS-B01-FECPRO
        boolean pruebas = props.equiposPrueba().contains(host.nombreEquipo());
        var fechas = new ArrayList<LocalDate>();
        LocalDate f = calendario.retrocede(base, 1);                  // primer 0150-CALCULA-FECHA-ATRAS
        if (pruebas) {
            fechas.add(f.minusDays(4));                               // COMPUTE WKS-FECHA = WKS-FECHA - 4
        } else {
            for (int i = 0; i < 4; i++) {                             // WKS-FECHA-1 .. WKS-FECHA-4
                fechas.add(f);
                f = calendario.retrocede(f, 1);
            }
        }
        var ec = ctx.getStepContext().getStepExecution().getJobExecution().getExecutionContext();
        ec.put("pruebas", pruebas);
        ec.put("fechasRetencion", fechas);
        return RepeatStatus.FINISHED;
    }
}
@Service
public class CalendarioBancario {

    private final LocsupPort locsup;   // adaptador mientras conviva el mainframe; luego, calendario propio

    public CalendarioBancario(LocsupPort locsup) { this.locsup = locsup; }

    /** THECALENDAR función 15, formato 13, desplazamiento "00000001". */
    public LocalDate retrocede(LocalDate desde, int dias) {
        return locsup.theCalendar(15, desde, dias, 13);
    }

    /** VALY_CAMBIAFORM entrega AAAA0DDD (formato 6); el programa lo reacomoda a AAAADDD para BORRA. */
    public static long aJulianoBorra(LocalDate f) {
        return f.getYear() * 1_000L + f.getDayOfYear();      // WKS-FECHA-JULIANA 9(07): AAAADDD
    }
}

/** (USUARIO)DIRECTORIO ON PACK.  Con DELIMITED BY SIZE los campos conservan sus espacios de relleno. */
static String tituloMcp(DirectorioADepurar d) {
    return "(" + rellena(d.usuario(), 4) + ")" + rellena(d.directorio(), 30)
         + " ON " + rellena(d.pack(), 17) + ".";
}

private static String rellena(String s, int n) {
    return String.format("%-" + n + "s", s);          // PIC X(n): relleno con espacios a la derecha
}
@Component
@StepScope
public class DepuracionProcessor implements ItemProcessor<DirectorioADepurar, SolicitudBorrado> {

    @Value("#{jobExecutionContext['pruebas']}")          private boolean pruebas;
    @Value("#{jobExecutionContext['fechasRetencion']}") private List<LocalDate> fechas;
    @Value("#{@hostInfo.csi()}")                         private int csiEquipo;

    @Override
    public SolicitudBorrado process(DirectorioADepurar d) {
        // Producción: WKS-DIAS-NUM se fuerza a 1 y la cadena IF 1..4 siempre elige WKS-FECHA-1.
        // Pruebas: la cadena completa está dentro del IF de producción y no se evalúa;
        //          queda la fecha única calculada al inicio. Los días del archivo no se usan nunca.
        LocalDate remocion = fechas.get(0);    // producción: WKS-FECHA-1; pruebas: fecha única
        if (csiEquipo != d.csi() && csiEquipo != 32) {
            return null;                                   // otro CSI: no se depura
        }
        return new SolicitudBorrado(tituloMcp(d), CalendarioBancario.aJulianoBorra(remocion));
    }
}
@Component
@RequiredArgsConstructor
public class BorradoWriter implements ItemWriter<SolicitudBorrado> {

    private final BorraPort borra;                 // REMUEVE_ARCHIVOS IN BORRA (CORE-GLL194)

    @Override
    public void write(Chunk<? extends SolicitudBorrado> chunk) {
        for (var s : chunk) {
            log.info("Depurando {} anteriores a {}", s.titulo(), s.fechaJuliana());   // DISPLAY WKS-DIRECTORIO
            int resultado = borra.remueveArchivos(s.titulo(), s.fechaJuliana());
            if (resultado != 0) {
                // El fuente ignora W77-RESULT; aquí se registra y se marca el ítem como fallido.
                throw new BorradoFallidoException(s.titulo(), resultado);
            }
        }
    }
}

Riesgos de migración
Días de retención del archivo ignorados

En 0200-LLAMA-BORRADO (líneas 830-844) la selección por días está dentro del IF de producción, después de forzar los días a 1. En pruebas se usa una fecha fija de cinco días atrás. El campo de días del archivo de control no tiene efecto.

Fecha forzada sin efecto

Cuando la tarea recibe VALUE, la fecha se mueve a WKS-FECHA pero no a WKS-FECHA1 (líneas 753-757), que es la entrada de THECALENDAR. La retención se calcula con el contenido previo de WKS-FECHA1.

Borrado sin verificación

No se revisa el resultado de REMUEVE_ARCHIVOS (línea 850) ni de VALY_CAMBIAFORM (línea 808). El CSI 32 tiene alcance sobre todos los directorios del archivo.

Fecha juliana inválida en pruebas

La resta de 4 sobre AAAA0DDD (línea 767) no considera el cambio de año.

Hardcodes

Lista de siete equipos de prueba (líneas 761-764), CSI global 32, sistema 500 en la consulta a CONSISDIA, título por omisión de CORE-GLL001 en CMEMP y título del archivo de control en el FD.
