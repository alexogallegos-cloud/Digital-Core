CORE-GL · P117 a Java 21 Spring Boot

Lee el LOG general del día (CORE-GL/FILE/LOGGRAL/AAMMDD, registros de 160 bytes) y agrupa los movimientos por clave de autorización en una tabla de hasta 100 renglones. Dentro de cada grupo cancela los pares operado y aplicado con el mismo importe, cuenta y tipo de movimiento, y aparta como diferencias no contables los pares de un movimiento con su eliminación. Lo que sobrevive con importe distinto de cero se escribe a un temporal y a S028MANIOBRA con el código de servicio invertido; el temporal se ordena y alimenta los reportes de diferencias con cortes por CSI, moneda, fecha y sistema, y el archivo HISTDIF con conteos e importes por sistema.

public enum EstatusTrn {
    ENVIADO(1), APLICADO(2), OPERADO(3), DENEGADO(4), PENDIENTE(5);
    final int codigo;
    EstatusTrn(int c) { codigo = c; }
}

/** Interpretación contable fija del código de servicio. */
public enum CodigoServicio {
    CARGO(10), ELIM_CARGO(11), ABONO(20), ELIM_ABONO(21);
    final int codigo;
    CodigoServicio(int c) { codigo = c; }

    /** Es la eliminación (reverso) de 'otro'. Se usa para netear en 2543. */
    boolean revierte(CodigoServicio otro) {
        return switch (this) {
            case ELIM_CARGO -> otro == CARGO;
            case ELIM_ABONO -> otro == ABONO;
            case CARGO      -> otro == ELIM_CARGO;
            case ABONO      -> otro == ELIM_ABONO;
        };
    }
}

@Component
@RequiredArgsConstructor
class AnalisisLogTasklet implements Tasklet {

    private final P117Properties props;
    private final ResourceLoader rl;
    private final AgrupadorPorAutorizacion agrupador;

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) throws Exception {
        LocalDate fecha = (LocalDate) ctx.getStepContext().getJobExecutionContext().get("fechaProceso");
        // NOMBRE-LOG-GRAL = "CORE-GL/FILE/LOGGRAL/" + AAMMDD + "."
        Resource log = rl.getResource(props.rutaLog().replace("{fecha}",
                fecha.format(DateTimeFormatter.ofPattern("yyMMdd"))));

        if (!log.exists()) {                                   // ATTRIBUTE RESIDENT
            throw new ProcesoAbortadoException("010501 LOG GRAL. AUSENTE");
        }
        try (var lector = LectorLog.abre(log)) {
            if (!lector.hayRegistros()) {
                throw new ProcesoAbortadoException("010106 NO HAY INFORMACION EN LOG GRAL.");
            }
            agrupador.procesa(lector);                         // PERFORM 2510 UNTIL IND-FIN-LOG = 1
        }
        return RepeatStatus.FINISHED;
    }
}

@Component
@RequiredArgsConstructor
class AgrupadorPorAutorizacion {

    static final int TOPE_GRUPO = 100;            // WS-TVM-OCC OCCURS 100 / 88 IND-TAB-VAL 0 THRU 99
    private final ConciliadorGrupo conciliador;
    private final ReporteDuplicados duplicados;
    private final SalidaDiferencias salida;      // TEMPFILE + S028MANIOBRA

    void procesa(LectorLog lector) {
        MovimientoLog actual = lector.siguiente();
        while (actual != null) {
            String clave = actual.claveAutorizacion();          // WS-NUM-AUT
            List<MovimientoLog> grupo = new ArrayList<>();
            while (actual != null && actual.claveAutorizacion().equals(clave)) {
                if (grupo.size() < TOPE_GRUPO && actual.csiOrigen() != actual.csiDestino()) {
                    grupo.add(actual);
                } else {
                    duplicados.imprime(actual);                 // intra-CSI o excedente del tope
                }
                actual = lector.siguiente();
            }
            List<MovimientoLog> diferencias = (grupo.size() > 1)
                    ? conciliador.concilia(grupo)                // 2540 y 2542
                    : grupo;
            diferencias.stream()
                    .filter(m -> m.importe().signum() != 0)     // 2550: importe distinto de cero
                    .forEach(salida::escribe);
        }
    }
}
@Component
class ConciliadorGrupo {

    /** 2540 / 2541 / 25411: cancela pares que cuadran; el primero que empata gana. */
    List<MovimientoLog> concilia(List<MovimientoLog> grupo) {
        MovimientoLog[] t = grupo.toArray(MovimientoLog[]::new);   // null = MOVE SPACES (cancelado)
        for (int i = 0; i < t.length; i++) {
            for (int j = i + 1; j < t.length; j++) {
                if (t[i] != null && t[j] != null && cuadranOperadoAplicado(t[i], t[j])) {
                    t[i] = null;
                    t[j] = null;
                }
            }
        }
        eliminaNoContables(t);                                        // 2542 / 2543
        return Arrays.stream(t).filter(Objects::nonNull).toList();
    }

    private static boolean cuadranOperadoAplicado(MovimientoLog a, MovimientoLog b) {
        boolean opApl = (a.estatus() == OPERADO && b.estatus() == APLICADO)
                     || (a.estatus() == APLICADO && b.estatus() == OPERADO);
        return opApl
                && a.importe().compareTo(b.importe()) == 0          // WS-TVM-IMPORTE 9(18)
                && a.cuentaDestino().equals(b.cuentaDestino())      // WS-TVM-ID-CTA
                && a.codServicio() == b.codServicio();              // cargo con cargo, abono con abono...
    }
}
    /** 2542 / 2543: mismo estatus, mismo importe y cuenta, y uno revierte al otro. */
    private void eliminaNoContables(MovimientoLog[] t) {
        for (int i = 0; i < t.length; i++) {
            for (int j = i + 1; j < t.length; j++) {
                if (t[i] == null || t[j] == null) continue;
                MovimientoLog a = t[i], b = t[j];
                boolean mismoEstatus = (a.estatus() == OPERADO && b.estatus() == OPERADO)
                                    || (a.estatus() == APLICADO && b.estatus() == APLICADO);
                if (mismoEstatus
                        && a.importe().compareTo(b.importe()) == 0
                        && a.cuentaDestino().equals(b.cuentaDestino())
                        && a.codServicio().revierte(b.codServicio())) {
                    reporteNoContable.imprimePar(a, b);   // 2580-IMPRIME-DIFNOCONT (R/151-117/004)
                    t[i] = null;
                    t[j] = null;
                }
            }
        }
    }

@Component
@RequiredArgsConstructor
class SalidaDiferencias {

    private final FlatFileItemWriter<MovimientoLog> tempfile;
    private final FlatFileItemWriter<ManiobraS028> s028Maniobra;
    private final CatalogoCcr ccr;                 // TABLA-CCRS: sucursal OCCA por CSI

    void escribe(MovimientoLog m) {
        tempfile.write(Chunk.of(m));                                   // WRITE REG-TEMP-LOG
        // El aplicado se orienta como el operado: se intercambian CSI origen y destino.
        MovimientoLog o = (m.estatus() == OPERADO) ? m : m.conCsiIntercambiados();
        if (o.csiOrigen() < 1 || o.csiOrigen() > ccr.limite()) return;   // LIM-CCRS = 20

        CodigoServicio inverso = switch (o.codServicio()) {
            case CARGO      -> CodigoServicio.ABONO;        // 10 -> 20
            case ABONO      -> CodigoServicio.CARGO;        // 20 -> 10
            case ELIM_CARGO -> CodigoServicio.ELIM_ABONO;   // 11 -> 21
            case ELIM_ABONO -> CodigoServicio.ELIM_CARGO;   // ELSE -> 11
        };
        var tipo1 = ManiobraS028.de(1, o, o.sucursalOrigen(), inverso);
        var tipo2 = ManiobraS028.de(2, o, ccr.sucursalOcca(o.csiOrigen()), inverso);
        s028Maniobra.write(Chunk.of(tipo1, tipo2));   // el segundo se escribe siempre
    }
}

record LlaveDiferencia(int csiOrigen, int moneda, LocalDate fechaContable, int aplOrigen,
                       int csiDestino, int aplDestino, String cuentaDestino,
                       int codServicio, BigDecimal importe, int estatus)
        implements Comparable<LlaveDiferencia> {

    static final Comparator<LlaveDiferencia> ORDEN = Comparator
            .comparingInt(LlaveDiferencia::csiOrigen)
            .thenComparingInt(LlaveDiferencia::moneda)
            .thenComparing(LlaveDiferencia::fechaContable)
            .thenComparingInt(LlaveDiferencia::aplOrigen)
            .thenComparingInt(LlaveDiferencia::csiDestino)
            .thenComparingInt(LlaveDiferencia::aplDestino)
            .thenComparing(LlaveDiferencia::cuentaDestino)
            .thenComparingInt(LlaveDiferencia::codServicio)
            .thenComparing(LlaveDiferencia::importe)
            .thenComparingInt(LlaveDiferencia::estatus);

    @Override public int compareTo(LlaveDiferencia o) { return ORDEN.compare(this, o); }

    /** Devuelve vacío si algún CSI es cero: el registro no se libera al sort. */
    static Optional<LlaveDiferencia> de(MovimientoLog m) {
        MovimientoLog o = (m.estatus() == EstatusTrn.OPERADO) ? m : m.conCsiIntercambiados();
        if (o.csiOrigen() == 0 || o.csiDestino() == 0) return Optional.empty();
        return Optional.of(new LlaveDiferencia(o.csiOrigen(), o.moneda(), o.fechaContable(),
                o.aplOrigen(), o.csiDestino(), o.aplDestino(), o.cuentaDestino(),
                o.codServicio().codigo, o.importe(), o.estatus().codigo));
    }
}

Riesgos de migración
Conciliación sin moneda

En 2541 y 2543 (líneas 2563-2570 y 2581-2588) los pares se cancelan por estatus, importe, cuenta y código de servicio, sin comparar WS-TVM-CVE-MON. Dos movimientos en monedas distintas con la misma cifra se dan por cuadrados y la diferencia no se reporta.

Validación de CSI anulada

La línea 2428 sobrescribe el CSI del host con la constante 02 antes de validarlo, por lo que el control de CSI inválido nunca se activa.

Tope de 100 movimientos por autorización

WS-TABLA-VERIF-MSG tiene 100 posiciones (líneas 751-786). Los excedentes de una clave de autorización van al reporte de duplicados y no se concilian ni se envían como diferencia al S028.

Aborto con salidas ya abiertas

Los errores de LOG ausente, LOG vacío o escritura de HISTDIF terminan la tarea con SET MYSELF(STATUS) TO -1 cuando HISTDIF y los reportes ya están abiertos. El destino debe definir si esos archivos parciales se descartan o se conservan.

Histórico que se sobrescribe

HISTDIF se crea con OPEN OUTPUT sobre un título fijo y se cierra WITH LOCK, de modo que cada ejecución reemplaza la anterior.

Catálogos compilados

Los CSI válidos (2, 4, 10 y 32), la tabla de 20 CCR con sus sucursales OCCA, los cinco sistemas emisores y el nodo 10 de los reportes están en el fuente. Se trasladan a application.yml.
