/**
 * EDO-ICA PIC 9(08). El SORT asciende por (llave, estado), así que el código numérico
 * define en qué orden llegan los estados de una misma transacción: primero los erróneos
 * (01 a 03), después la confirmación (05) y al final el procesamiento (06, 07).
 */
public enum EstadoIca {
    SOSPECHOSO(1), REVOCADO(2), POR_REVOCAR(3), ENTREGADO(4), CONFIRMADO(5),
    PROCESADO_ORIGEN(6), PROCESADO_DESTINO(7), RECIBIDO(8), ENVIADO(9), FLUJO(10), NINGUNO(11);

    private final int codigo;
    EstadoIca(int codigo) { this.codigo = codigo; }

    public boolean erroneo() { return this == SOSPECHOSO || this == REVOCADO || this == POR_REVOCAR; }

    public static EstadoIca de(int codigo) {
        return Arrays.stream(values()).filter(e -> e.codigo == codigo).findFirst()
                .orElseThrow(() -> new IllegalArgumentException("EDO-ICA desconocido: " + codigo));
    }

    /** Comparador equivalente a "ON ASCENDING KEY SD-LLAVE, SD-EDO-ICA". */
    public static final Comparator<EventoIca> ORDEN_SORT =
            Comparator.comparing(EventoIca::llave).thenComparingInt(e -> e.estado().codigo);
}

public record EventoIca(String llave,                 // SD-LLAVE X(27) dentro de MSG-SA2
                        EstadoIca estado,             // WS-ESTADO devuelto por REGISTRO-ICA
                        int aplicacion, int codServicio, int subServicio,   // MSG-COD-TRAN-1
                        Direccion direccion,          // MSG-DIRECCION
                        int codRespuesta,             // MSG-COD-RESPUESTA
                        boolean aplicada,             // SA2-ESTADO = 1
                        boolean interactivo,          // SA2-MODO = 1
                        String raw) {

    public enum Direccion { ENVIO, REGRESO, REGRESO_REVOCACION, OTRA }

    private static final Set<Integer> SERVICIOS_VALIDOS = Set.of(10, 11, 20, 21);   // COD-SERV-VALIDO
    private static final Set<Integer> RESPUESTAS_OK = Set.of(2, 4, 7);              // RESP, ENTREGADO, ACEPTADO

    public boolean servicioValido()   { return SERVICIOS_VALIDOS.contains(codServicio); }
    public boolean respuestaPositiva() { return RESPUESTAS_OK.contains(codRespuesta); }
}

@Configuration
@RequiredArgsConstructor
public class ExtraccionIcaFlowConfig {

    private final P114Properties props;

    /**
     * Secuencia principal: fecha (1010), apertura (1000/1100), extracción (2000), cierre (9000).
     * El canal de LIBICA se modela como un MessageSource que entrega un EventoIca por poll.
     */
    @Bean
    IntegrationFlow extraccionIca(IcaDiaMessageSource diaIca,           // ABREDIAICA / REGISTRO-ICA
                                  InfoIcaFileWriter infoIca) {
        return IntegrationFlow.from(diaIca, p -> p.poller(Pollers.fixedDelay(0).maxMessagesPerPoll(-1)))
                .filter(EventoIca.class, EventoIca::servicioValido)        // 2200-ARMA-SORT
                .aggregate(a -> a
                        .correlationStrategy(m -> ((EventoIca) m.getPayload()).llave())
                        .releaseStrategy(g -> diaIca.agotado())             // fin del día ICA
                        .outputProcessor(g -> TransaccionIca.de(g.getMessages())))  // orden por estado
                .handle(TransaccionIca.class, (t, h) -> ResolutorEstatus.resuelve(t)) // 2600 / 2610
                .handle(infoIca)                                            // 2640: header, detalle, trailer
                .get();
    }
}

public LocalDate fechaProceso(JobParameters params, int csi) {
    // TASKVALUE distinto de cero sustituye la fecha de la librería de control (reproceso manual).
    Long override = params.getLong("fechaOverride");
    if (override != null && override != 0) {
        return LocalDate.parse(String.valueOf(override), DateTimeFormatter.BASIC_ISO_DATE);
    }
    return control.consultaSistemaDia(props.sistemaControl(), csi)
            .map(SistemaDia::fechaProceso)
            .orElseThrow(() -> new ProcesoAbortadoException("ERROR AL CONSULTAR CORE-GLLIBCONTROL"));
}

public String nodo(int csi) {
    String n = props.nodos().get("%02d".formatted(csi));
    if (n == null) throw new ProcesoAbortadoException("CSI sin nodo ICA configurado: " + csi);
    return n;
}
@Component
@RequiredArgsConstructor
public class IcaDiaMessageSource implements MessageSource<EventoIca> {

    private final LibIcaClient libIca;                         // REGISTRO-ICA OF LIBICA
    private final MeterRegistry metricas;
    private volatile boolean agotado;

    /** 2100-METE-REG: estado mayor que cero es un evento; cero es fin; negativo es error. */
    @Override
    public Message<EventoIca> receive() {
        if (agotado) return null;
        var r = libIca.siguiente();
        if (r.estado() > 0) {
            EventoIca e = r.evento();
            if (e.servicioValido()) {                          // 2200-ARMA-SORT
                metricas.counter("p114.codigo.servicio", "cod", String.valueOf(e.codServicio()))
                        .increment();                          // ELEM-CODIG-SERV(MSG-TRC1-COD)
            }
            return MessageBuilder.withPayload(e).build();
        }
        agotado = true;
        libIca.cierraDia();                                    // CIERRA-DIA-ICA
        if (r.estado() < 0) {
            throw new IcaLecturaException(r.estado());          // el COBOL solo lo registra y sigue
        }
        return null;
    }

    public boolean agotado() { return agotado; }
}
/**
 * 2000-PROCESO: el COBOL calcula memoria y disco del SORT a partir de los atributos físicos
 * de CLASIFICA. En Java el ordenamiento ocurre en memoria dentro del agregador; si el volumen
 * lo exige, se usa un sort externo con límites explícitos y nombrados correctamente.
 */
@ConfigurationProperties("p114.sort")
public record SortProperties(int palabrasPorRegistroExtra,   // + 3
                             int factorMemoria,              // * 2000
                             int baseMemoria,                // + 1500
                             BigDecimal factorDisco) {       // * 3.5

    static final int BYTES_POR_PALABRA = 6;                   // palabra MCP de 48 bits

    long palabrasRegistro(int maxRecSize, int frameSize) {
        return frameSize == 8
                ? (maxRecSize + BYTES_POR_PALABRA - 1) / BYTES_POR_PALABRA   // redondeo hacia arriba
                : maxRecSize;
    }
    long memoria(long palabrasRegistro) {
        return (palabrasRegistro + palabrasPorRegistroExtra) * factorMemoria + baseMemoria;
    }
    long disco(long palabrasRegistro, long registros) {
        return factorDisco.multiply(BigDecimal.valueOf(palabrasRegistro * registros))
                .setScale(0, RoundingMode.HALF_UP).longValueExact();          // COMPUTE ... ROUNDED
    }
}

/** 2600-ARMA-REGISTROS + 2610-VERIFICA-RESPUESTA para todos los eventos de una misma llave. */
public final class ResolutorEstatus {

    public static List<InfoIcaDetalle> resuelve(TransaccionIca t) {
        var salida = new ArrayList<InfoIcaDetalle>();
        boolean erroneo = false, confirmado = false;              // WS-EDO-ERRONEO, WS-MSG-CONFIRMADO
        for (EventoIca e : t.eventosOrdenados()) {                // orden EstadoIca.ORDEN_SORT
            if (e.estado().erroneo()) erroneo = true;             // sospechoso, revocado o por revocar
            if (erroneo || !e.interactivo()) continue;            // no interactivas: nunca se extraen
            if (e.estado() == EstadoIca.CONFIRMADO) { confirmado = true; continue; }

            boolean procesado = (e.estado() == EstadoIca.PROCESADO_ORIGEN && confirmado)
                             || e.estado() == EstadoIca.PROCESADO_DESTINO;
            if (procesado && e.direccion() == Direccion.REGRESO
                    && e.respuestaPositiva() && e.aplicada()) {
                EstatusTrn st = e.estado() == EstadoIca.PROCESADO_ORIGEN
                        ? EstatusTrn.OPERADO      // 3
                        : EstatusTrn.APLICADO;    // 2
                salida.add(InfoIcaDetalle.de(e, st));             // 2630-ARMA-REG-INFOICA
            }
        }
        return salida;
    }
}
public record InfoIcaDetalle(int csiOrigen, int nodoOrigen, int csiDestino,
                             String tipoDispositivo, int sucursalOrigen, String cajaOrigen,
                             LocalDate fechaGeneracion, LocalTime horaGeneracion,
                             int aplOrigen, int aplDestino,      // TRC2 es origen, TRC1 es destino
                             int codServicio, int subServicio,
                             String cuentaDestino, BigDecimal importe, int moneda,
                             LocalDate fechaContable, long numTransaccion,
                             String cuentaAcceso, String referencia, String autorizacion,
                             EstatusTrn estatus) {

    /** A2K-CONV-AMD-TO-CAMD: AAMMDD a CCAAMMDD; A2K-BASE-YEAR = 50. 000000 y 999999 son centinelas. */
    static final DateTimeFormatter AAMMDD = new DateTimeFormatterBuilder()
            .appendValueReduced(ChronoField.YEAR, 2, 2, 1950)    // AA < 50 = 20xx, AA >= 50 = 19xx
            .appendPattern("MMdd").toFormatter();

    static InfoIcaDetalle de(EventoIca e, EstatusTrn st) {
        var sa2 = e.sa2();
        return new InfoIcaDetalle(e.hdr().csiOrigen(), e.hdr().computadoraOrigen(), e.hdr().csiDestino(),
                e.hdr().tipoSo(), sa2.sucursalOrigen(), sa2.cajaOrigen(),
                LocalDate.parse(sa2.fechaTrn(), AAMMDD), sa2.horaTrn(),
                e.trc2().aplicacion(), e.trc1().aplicacion(), e.codServicio(), e.subServicio(),
                sa2.cuenta(), sa2.importe(), sa2.moneda(),
                LocalDate.parse(sa2.fechaSolicitud(), AAMMDD), sa2.numTrn(),
                sa2.cuenta(), sa2.referencia(), e.autorizacion(), st);
    }
}
@Component
@RequiredArgsConstructor
public class InfoIcaFileWriter implements MessageHandler {

    private final InfoIcaFormato formato;      // registros de 160 bytes
    private final Path destino;                // CORE-GL/XFER/FILE/INFOICA/{ori}{des}/{aammdd}.
    private InfoIcaDetalle anterior;           // WS-REG-INFOICA-ANT
    private long escritos;                     // WS-NUM-REG

    /** 2640-ESCRIBE-INFOICA: solo se descarta un registro idéntico al inmediato anterior. */
    @Override
    public void handleMessage(Message<?> m) {
        for (InfoIcaDetalle d : (List<InfoIcaDetalle>) m.getPayload()) {
            if (!d.equals(anterior)) { formato.escribeDetalle(d); escritos++; }
            anterior = d;
        }
    }

    /** 2800-CREA-TRAILER + 2900-REESCRIBE-REG-UNO: el conteo se conoce al final. */
    public void cierra(HeaderInfoIca header) throws IOException {
        Path tmp = Files.createTempFile(destino.getParent(), "infoica", ".tmp");
        try (var out = Files.newBufferedWriter(tmp, StandardCharsets.ISO_8859_1)) {
            out.write(formato.header(header.conRegistros(escritos)));  // en lugar de REWRITE del registro 1
            formato.copiaDetalles(out);
            out.write(formato.trailer(escritos));                      // "TLR" + 9(07)
        }
        Files.move(tmp, destino, StandardCopyOption.ATOMIC_MOVE);      // el CORE-GL nunca ve un archivo a medias
    }
}

Riesgos de migración
Tamaños de SORT intercambiados

En 2000-PROCESO (líneas 1712-1713) MEMORY SIZE recibe el cálculo de disco y DISK SIZE el de memoria. El defecto pasa inadvertido con volúmenes chicos y se manifiesta en días de alto volumen.

Error de lectura tratado como fin de día

En 2100-METE-REG un estado negativo de la librería ICA solo genera un mensaje informativo y detiene la extracción; el INFOICA sale incompleto, con trailer coherente con lo leído, y el programa termina normal. P116 no puede detectarlo porque el trailer cuadra.

Deduplicación solo consecutiva

2640-ESCRIBE-INFOICA compara cada detalle contra el anterior; duplicados no contiguos pasan al archivo y a la compensación.

Nodos y sistema compilados

Los CSI 04, 12, 10 y 32 se traducen a MON y VDM y la fecha se consulta con el sistema 264 (líneas 1572 y 1593-1603). Con un CSI distinto el nodo del nombre del archivo no se asigna.

Código comentado del supervisor

El archivo SUPERVISOR y la actualización de fechas del guardián (9100-ACTUALIZA-GUARDIAN) están comentados con marcas *ASM; el control de fechas procesadas que hacían ya no existe en este programa.
