CORE-GL · P690 a Java 21 Spring Boot

Pide a la librería de control CORE-GLL001 la fecha contable, el pack del archivo de movimientos y una tabla de diez días con el número de movimientos de cada archivo; si la llamada falla, termina. Abre en lectura y escritura el archivo CORE-GL/FILE/MOVS del sistema recibido y la fecha contable, salta el encabezado y recorre tantos registros como indica el nivel del día. Cada movimiento con función 99 pasa a función 11 y status 1, con el sistema que lo procesó, la fecha contable y una autorización igual a su posición en el archivo menos uno, y se reescribe en sitio.

/** A00-R01-REGMOV (450 bytes). Solo se modelan los campos que P690 lee o escribe. */
public final class MovimientoCORE-GL {

    public static final int FUNCION_PENDIENTE = 99;
    public static final int FUNCION_CERRADO   = 11;

    private int funcion;            // A00-R01-FUNCION 9(02) COMP
    private int sistema;            // A00-R01-SISTEMA 9(04) COMP
    private long autCORE-GL;           // A00-R01-AUT-CORE-GL 9(08) COMP
    private LocalDate fecContable;  // A00-R01-FECCONT 9(08) COMP
    private long apuDes;            // A00-R01-APUDES (no lo toca P690)
    private int status;             // A00-R01-STATUS 9(02) COMP: 1 = completado
    private final byte[] raw;       // el resto del registro se reescribe intacto

    /** Transición que aplica 200-REGRABA. */
    public void cierra(int sistemaProceso, long autorizacion, LocalDate fechaContable) {
        if (funcion != FUNCION_PENDIENTE) throw new IllegalStateException("Solo se cierran movimientos 99");
        if (autorizacion > 99_999_999L) throw new ArithmeticException("AUT-CORE-GL excede 9(08)");
        this.funcion = FUNCION_CERRADO;
        this.sistema = sistemaProceso;
        this.autCORE-GL = autorizacion;
        this.fecContable = fechaContable;
        this.status = 1;
    }
}
/** WKS-CORE-GLB01SISDIA: respuesta de CONSISDIA IN CORE-GL-L001. */
public record ControlDiaCORE-GL(int sistema, LocalDate fecContable,
                             String nombreArchivoMov,   // WKS-B01-NOMARCMOV: P690 no lo usa
                             String packMov,            // WKS-B01-NOMPACMOV
                             List<NivelArchivo> dias) { // OCCURS 10

    public record NivelArchivo(LocalDate fecArchivoMov,  // WKS-B01-FECARCMOV
                               long nivelArchivoMov,     // WKS-B01-NIVARCMOV: registros a cerrar
                               long nivelBdMov, int statusLog) {}

    /** 100-BUSCA-IND-FECARCMOV: posición del día contable dentro de los 10 días. */
    public NivelArchivo nivelDelDia() {
        return dias.stream()
                .filter(d -> d.fecArchivoMov().equals(fecContable))
                .findFirst()
                .orElseThrow(() -> new ControlDiaInconsistenteException(
                        "FECCON %s no está en FECARCMOV".formatted(fecContable)));
    }
}

/** 020400-LIBRERIAS: resuelve CORE-GLL001 por catálogo de versiones y pide los datos del día. */
@Component
@RequiredArgsConstructor
public class ControlDiaTasklet implements Tasklet {

    private final ControlVersiones ctlvers;     // DAME_TIT IN CTRLVERS
    private final ControlCORE-GLClient control;    // CONSISDIA IN CORE-GL-L001
    private final AvisoOperador avisos;         // LJ IN LIB-DISP

    @Override
    public RepeatStatus execute(StepContribution c, ChunkContext ctx) {
        int sistema = ((Number) ctx.getStepContext().getJobParameters().get("sistema")).intValue();
        String destino = ctlvers.titulo("CORE-GLL001CTL")
                .orElse("(CORE-GL)CORE-GL/OBJECT/L001/CONTROL ON CMEMP.");   // respaldo fijo del original
        ResultadoControl r = control.consisdia(destino, sistema, 1);  // WKS-B01-FUNCION = 01
        if (r.codigo() > 0) {
            avisos.operador("020901", "NO PUEDO OBTENER FECHA DE PROCESO  ERROR EN CORE-GL-L001 S%03d, ERR #%02d"
                    .formatted(sistema, r.codigo()));
            throw new ProcesoAbortadoException(r.codigo());           // SET MYSELF(STATUS) TO -1
        }
        ctx.getStepContext().getStepExecution().getJobExecution()
           .getExecutionContext().put("controlDia", r.datos());
        return RepeatStatus.FINISHED;
    }
}
@Configuration
@EnableConfigurationProperties(P690Properties.class)
public class CierreMovimientosJobConfig {

    /** 000000-INIC-PROGRAMA. Parámetro del WFL: sistema (9(03)). */
    @Bean
    Job cierreMovimientos(JobRepository repo, Step controlDia, Step cierraPendientes,
                          NivelDelDiaDecider nivel) {
        return new JobBuilder("CORE-GLP690", repo)
                .start(controlDia)                         // 020400-LIBRERIAS
                .next(nivel).on("CON_NIVEL").to(cierraPendientes)
                .from(nivel).on("SIN_NIVEL").fail()        // FECCON fuera de la tabla de 10 días
                .end().build();
    }

    @Bean
    Step cierraPendientes(JobRepository repo, PlatformTransactionManager tx,
                          MovimientosCORE-GLReader reader, CierreMovimientoProcessor processor,
                          MovimientosCORE-GLRewriteWriter writer) {
        return new StepBuilder("cierraPendientes", repo)
                .<MovimientoCORE-GL, MovimientoCORE-GL>chunk(100, tx)    // BLOCK CONTAINS 100 RECORDS
                .reader(reader)          // salta el header y se detiene en NIVARCMOV del día o al EOF
                .processor(processor)
                .writer(writer)          // REWRITE en sitio; CLOSE ... WITH SAVE al cerrar el step
                .build();
    }
}
/** 500-LEE-A01-MOVSNNN: el contador cuenta todas las lecturas, incluido el header. */
@Component
@StepScope
class MovimientosCORE-GLReader implements ItemStreamReader<MovimientoCORE-GL> {
    private long contador;                                  // WKS-CONTADOR 9(09)
    @Value("#{jobExecutionContext['nivelDelDia']}") private long nivel;
    private final ArchivoMovimientos archivo;               // acceso por posición (ACTUAL KEY)

    public MovimientoCORE-GL read() {
        if (contador == 0) { archivo.leeEncabezado(); contador = 1; }   // lectura de cebado
        if (contador > nivel) return null;                              // UNTIL CONTADOR > NIVARCMOV
        contador++;
        return archivo.siguiente().map(m -> m.conPosicion(contador)).orElse(null);  // AT END
    }
}

/** 200-REGRABA. */
@Component
@StepScope
class CierreMovimientoProcessor implements ItemProcessor<MovimientoCORE-GL, MovimientoCORE-GL> {
    @Value("#{jobParameters['sistema']}")             private int sistema;
    @Value("#{jobExecutionContext['fecContable']}") private LocalDate fecContable;

    public MovimientoCORE-GL process(MovimientoCORE-GL m) {
        if (m.funcion() != MovimientoCORE-GL.FUNCION_PENDIENTE) return null;   // no se reescribe
        m.cierra(sistema, m.posicion() - 1, fecContable);                   // AUT-CORE-GL = CONTADOR - 1
        return m;
    }
}

/*
 * 210-REGRABA-REG no se traduce: su único PERFORM está comentado (líneas 734-735),
 * junto con un arranque fijo del contador en 38880. Era una variante de recuperación que:
 *   - cerraba solo FUNCION 99 con WKS-CONTADOR < 2000000,
 *   - no ponía SISTEMA, AUT-CORE-GL, FECCONT ni STATUS,
 *   - mostraba "ERROR EN REG. " + AUT-CORE-GL si el REWRITE fallaba.
 * Si operación necesitara reanudar desde una posición, el equivalente es reiniciar el step:
 * Spring Batch guarda la posición del reader en el ExecutionContext (ItemStream.update).
 */
@Override
public void update(ExecutionContext ctx) {
    ctx.putLong("p690.contador", contador);     // punto de reinicio en lugar de MOVE 38880
}

@Override
public void open(ExecutionContext ctx) {
    contador = ctx.getLong("p690.contador", 0L);
    archivo.posiciona(contador);                // equivalente al SEEK comentado en 727-728
}
Riesgos de migración
Error de reescritura ignorado

REWRITE A00-R01-REGMOV INVALID KEY NEXT SENTENCE (línea 752) descarta el error: un movimiento puede quedar en función 99 sin aviso y el job termina normal.

Búsqueda del día sin tope

El PERFORM VARYING de las líneas 721-723 no limita W77-IND a las 10 entradas de WKS-B01-CTL-DIA. Si la fecha contable no está en la tabla, el índice rebasa el arreglo.

Nombre de archivo compilado

El título usa el prefijo fijo CORE-GL/FILE/MOVS (líneas 679-687) y no el nombre WKS-B01-NOMARCMOV que entrega la librería de control. Solo el pack es dinámico.

Autorización ligada a la posición física

AUT-CORE-GL = WKS-CONTADOR - 1 (línea 749) depende del orden físico del archivo. Reordenar, filtrar o insertar registros antes de P690 cambia los números de autorización.

Fecha truncada en el título

WKS-SIS-FEC es PIC 9(06) y recibe WKS-B01-FECCON de 8 dígitos (línea 720); el nombre del archivo pierde el siglo. No se encontraron credenciales en el fuente.

Pendiente de revisión SME
  • El inventario clasifica P690 como A5 Orquestador de Flujos, pero el fuente es un actualizador en sitio de un archivo secuencial sin orquestación de otros pasos. Se tradujo como job de Spring Batch con decisor para respetar el arquetipo; valorar reclasificarlo como A2.
  • Confirmar que SET MYSELF(STATUS) TO -1 (línea 822) termina la tarea de inmediato; si la ejecución continuara, el programa abriría el archivo con el pack vacío.
  • El pseudocódigo cita las reglas RN-CORE-GL-625 a RN-CORE-GL-632, pero el inventario no asigna reglas del brain a este componente; por eso ningún segmento las referencia.
  • Confirmar si el nivel WKS-B01-NIVARCMOV cuenta solo movimientos o incluye el encabezado: con la lógica actual se procesan los registros 2 a NIVARCMOV + 1.
