CORE-DEP · P200 a Java 21 Spring Boot

P200 arma en memoria la tabla de rangos de saldo y tasas de cada instrumento consultando el tarifario S080 (tarifa 50), y calcula la tasa neta restando la retención de ISR a la tasa bruta. Después recorre todo el maestro de contratos de CORE-DEPBD01CAPTACION en modo consulta, selecciona los que están vigentes o tienen saldo, ubica a cada uno en su rango según el saldo o, en cierre de mes, según el promedio del ciclo, y escribe un archivo puente. El puente ordenado se consolida por producto, persona, rango y tasas pagadas en el archivo RANGOS con header y trailer de control, que se ordena otra vez y se imprime como reporte de validación.

@Component
@RequiredArgsConstructor
class AnalizadorTasasS080 {

    /**
     * 20041000-ANALIZA-TASAS. S080 entrega enteros con su número de decimales;
     * el COMPUTE sin ROUNDED trunca a la escala de WKS-IMPPTE (6) y el MOVE a la del destino.
     */
    RangoCrudo analiza(RenglonTarifaS080 r) {
        BigDecimal tasaBruta = BigDecimal.valueOf(r.tasa())
                .movePointLeft(r.decimalesTasa())
                .setScale(6, RoundingMode.DOWN);                 // WKS-TASABRUTA 9(04)V9(06)
        BigDecimal limInf = BigDecimal.valueOf(r.limiteInferior())
                .movePointLeft(r.decimalesInferior())
                .setScale(2, RoundingMode.DOWN);                 // WKS-LIMINF 9(12)V99
        BigDecimal limSup = BigDecimal.valueOf(r.limiteSuperior())
                .movePointLeft(r.decimalesSuperior())
                .setScale(2, RoundingMode.DOWN);                 // WKS-LIMSUP
        return new RangoCrudo(r.persona(), r.plazo(), tasaBruta, limInf, limSup);
    }
}

@Component
@RequiredArgsConstructor
class CalculoTasaNeta {
    private final P200Properties props;

    /** 20042000-CALCULA-TASAS: la retención de ISR se resta en puntos de tasa, con piso en cero. */
    BigDecimal tasaNeta(BigDecimal tasaBruta) {
        BigDecimal isr = props.tasaIsr().setScale(6, RoundingMode.DOWN);
        return tasaBruta.compareTo(isr) > 0
                ? tasaBruta.subtract(isr)            // resta exacta a 6 decimales
                : BigDecimal.ZERO.setScale(6);
    }
    // La variante que aplicaba 0.85 * ISR a personas morales está comentada en el fuente
    // (líneas 1013 y 1021-1028) y no se traduce.
}

@Component
class SeleccionPuente implements ItemProcessor<Contrato, RegistroPuente> {

    static boolean califica(Contrato c) {
        if (c.getMigradoS111() == 1) return false;                         // migrados a S111 nunca
        boolean statusVigente = c.getStatus() < 2 || c.getStatus() == 5;
        boolean conSaldo = c.getSaldoAnterior().add(c.getSaldoProtegidoAnterior()).signum() > 0;
        return statusVigente || conSaldo;
    }

    @Override
    public RegistroPuente process(Contrato c) {
        return califica(c) ? generaPuente.desde(c, historicos.de(c.getNumContrato())) : null;
    }
}
@Component
@RequiredArgsConstructor
class GeneraPuente {
    private final TablaTasas tabla;
    private final ProcesoContext ctx;        // esMensual: cambio de mes entre fecha línea y fecha batch

    RegistroPuente desde(Contrato c, Historico h) {
        TipoPersona p = c.esPersonaFisicaGrupo() ? TipoPersona.FISICA : TipoPersona.MORAL;  // WS03-88-TP-GFISICA
        BigDecimal total = c.getSaldoAnterior().add(c.getSaldoProtegidoAnterior());

        // Cierre de mes con ciclo anterior: califica por el promedio del ciclo (B06-PROM-CICLO(12)).
        boolean usaPromedio = c.getStatus() != 3 && ctx.esMensual() && h.getFecCicloAnt() > 0;
        BigDecimal saldoRango = usaPromedio ? h.getPromCiclo(12) : total;
        BigDecimal saldoProm  = saldoRango;
        BigDecimal saldoAct   = total;

        List<RangoTasa> rangos = tabla.de(c.instrumento(), p);
        int r = ubicaRango(saldoRango, rangos);
        RangoTasa rt = rangos.get(r - 1);

        boolean status3 = c.getStatus() == 3;
        return new RegistroPuente(c.instrumento(), p, r, rt.limiteInferior(), rt.limiteSuperior(),
                saldoAct, saldoProm,
                status3 ? BigDecimal.ZERO : c.getTasaAnterior(),     // TASANP: tasa neta pagada (B03)
                status3 ? BigDecimal.ZERO : h.getTasaBruta(),        // TASABP: tasa bruta pagada (B06)
                rt.tasaNeta().setScale(3, RoundingMode.DOWN));       // TASANT 9(03)V9(03): trunca a 3
    }

    /** Primer rango cuyo límite superior es mayor o igual al saldo; saldo cero o un rango, el 1. */
    static int ubicaRango(BigDecimal saldo, List<RangoTasa> rangos) {
        int sup = rangos.size();
        if (saldo.signum() == 0 || sup == 1) return 1;
        if (saldo.compareTo(rangos.get(sup - 1).limiteSuperior()) >= 0) return sup;
        for (int r = 1; r <= sup; r++) {
            if (saldo.compareTo(rangos.get(r - 1).limiteSuperior()) <= 0) return r;
        }
        return sup;   // inalcanzable por la condición anterior
    }
}

class GrupoRangoReader implements ItemReader<RangoTesoreria.Detalle> {
    private final PeekableItemReader<RegistroPuente> in;

    @Override
    public RangoTesoreria.Detalle read() throws Exception {
        RegistroPuente primero = in.read();
        if (primero == null) return null;
        var g = Grupo.inicia(primero);                                // 60010120-INICIA-REGS
        for (RegistroPuente sig; (sig = in.peek()) != null && g.mismaLlave(sig); ) {
            g.suma(in.read());                                        // 60012000-SUMA-IGUAL
        }
        return g.comoDetalle();                                       // 60015000-ESCRIBE-ARCHIVO
    }
}

Riesgos de migración
El archivo puente no se purga

La cabecera (línea 28) y RN-CORE-DEP-759 dicen que RGOSPTE se borra con CLOSE WITH PURGE, pero esa línea está comentada (1279) y la activa es CLOSE WITH SAVE (1280).

La tasa neta calculada no llega a Tesorería

WS-XTASANETA solo se copia al campo TASANT del puente (líneas 1238 y 1242), que ningún párrafo lee después; el archivo RANGOS lleva las tasas pagadas de B03 y B06 con tres decimales.

Tope de rango truncado

La literal 9999999999.99 se mueve a WS-XLMTESUP, PIC 9(09)V99 (líneas 407, 857 y 930); el valor almacenado es 999999999.99.

Reemplazo de header y trailer con WRITE

60051010 y 60051020 usan WRITE sobre I01-RANGOS abierto en I-O tras un READ; el resultado depende de la semántica de posicionamiento de MCP y no de un REWRITE explícito.

Errores que no abortan

Si S080 no tiene tasas para un instrumento, el programa asigna un rango único en cero y continúa (el DMTERMINATE de la línea 934 está comentado); los errores de WRITE solo cambian el STATUS a -1.

Hardcodes

Tarifa 50, 10 registros por llamada a S080, 30 instrumentos y 20 rangos, buzón 50004S15, sistema 0500, sucursales 3084 y 3667 por nodo y factores de memoria del SORT. Se trasladan a configuración.

Pendiente de revisión SME
  • RN-CORE-DEP-759 describe la purga del puente como comportamiento vigente; el código hace lo contrario. Confirmar con el SME cuál es el esperado y corregir la regla o el programa.
  • RN-CORE-DEP-758 afirma que la precisión de seis decimales reduce el error antes de reportar a Tesorería; en el fuente esa tasa no llega al archivo de salida. Validar la redacción de la regla.
  • Obtener el copybook que define WS-ISR-0 para conocer la tasa de ISR vigente y su mantenimiento.
  • Confirmar con un especialista MCP el efecto de WRITE (y no REWRITE) sobre un archivo secuencial abierto en I-O en 60051010 y 60051020.
  • Los dos SORT y el reporte de validación R01-RGOSDO no se incluyeron en esta demostración.
