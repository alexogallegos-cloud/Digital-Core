CORE-DEP · P106 a Java 21 Spring Boot

Recibe por parámetro la fecha AAMMDD, la opción de archivo (1 a 6), el tipo de búsqueda, un secuencial y la clave a buscar. Las opciones 1 a 4 recorren rechazos, log o incidentes y arman el reporte de transacciones encontradas. Las opciones 5 y 6 leen los archivos CANCELADOS 01 a 03 de la fecha: validan el header contra la fecha y el CSI, y por cada detalle la opción 5 invoca S016L428 para dejar la cuenta en estatus 31, mientras la opción 6 sólo marca el resultado 99. Cada detalle procesado se agrega a un archivo de reinicio que permite reanudar en el contrato siguiente; al final ese archivo se usa para imprimir el reporte de aceptados y rechazados y se purga.

/** PROCEDURE DIVISION USING WKS-IN-PARAM: 27 posiciones recibidas del WFL. */
public record P106Parametros(
        LocalDate fecha,          // WKS-IN-FECH 9(06) AAMMDD
        Opcion opcion,            // WKS-IN-FILE 9(01)
        int tipoBusqueda,         // WKS-IN-OPC: 1 tarjeta, 2 sucursal y cuenta, 3 autorización o contrato
        int secuencial,           // WKS-IN-SEC (OnDemand)
        String claveBusqueda) {   // WKS-IN-FIND 9(16): String para conservar ceros

    public enum Opcion { RECHAZOS, LOG, INCIDENTES, INCIDENTE_4, CANCELADOS_AVISA, CANCELADOS_CONTINGENCIA }

    /** REDEFINES WKS-RD-AUTORI: 8 de relleno + autorización 9(08). */
    public String autorizacion() { return claveBusqueda.substring(8, 16); }
    /** REDEFINES WKS-RD-CTA-CTO: sucursal 9(04) + cuenta 9(12). */
    public String sucursal()     { return claveBusqueda.substring(0, 4); }
    public String cuenta()       { return claveBusqueda.substring(4, 16); }

    public static P106Parametros parse(String p, int pivoteSiglo) {
        var aammdd = p.substring(0, 6);
        int aa = Integer.parseInt(aammdd.substring(0, 2));
        int anio = (aa >= pivoteSiglo ? 1900 : 2000) + aa;          // pivote explícito: p106.pivote-siglo
        var fecha = LocalDate.of(anio, Integer.parseInt(aammdd.substring(2, 4)),
                                 Integer.parseInt(aammdd.substring(4, 6)));
        int op = p.charAt(6) - '0';
        if (op < 1 || op > 6) throw new OpcionInvalidaException(op);   // "OPCION n INVALIDA"
        return new P106Parametros(fecha, Opcion.values()[op - 1], p.charAt(7) - '0',
                Integer.parseInt(p.substring(8, 11)), p.substring(11, 27));
    }
}
/** WKS-I10-REG-CANCEL: 140 bytes, tres vistas REDEFINES discriminadas por el primer dígito. */
public sealed interface RegistroCancelado
        permits RegistroCancelado.Header, RegistroCancelado.Detalle, RegistroCancelado.Trailer {

    record Header(String sisOrigen, String sisDestino, LocalDateTime creacion,
                  int siglo, String fechaContableAammdd,   // WKS-I10-SIG + WKS-110-FECH
                  int csiOrigen, int csiDestino) implements RegistroCancelado {}

    record Detalle(int producto, int instrumento, int moneda,
                   String contrato, String cliente) implements RegistroCancelado {}   // 9(12) como String

    record Trailer(long totalGlobal) implements RegistroCancelado {}

    static RegistroCancelado parse(String r) {
        return switch (r.charAt(0)) {
            case '1' -> Fixed.headerCancelado(r);
            case '3' -> new Detalle(Fixed.entero(r, 1, 4), Fixed.entero(r, 5, 4), Fixed.entero(r, 9, 2),
                                    r.substring(11, 23), r.substring(23, 35));
            default  -> new Trailer(Fixed.entero(r, 1, 8));   // cualquier otro tipo se lee como trailer
        };
    }
}

    @Bean
    Job p106(JobRepository repo, OpcionDecider decider, Step consultaRechazos, Step consultaLog,
             Step consultaIncidentes, Step reinicio, Step procesaCancelados, Step reporteCancelados,
             IncidenteExitListener incidente) {

                .listener(incidente)                                  // SET MYSELF (TASKVALUE) TO W77-ENC-INC
                .start(decider)
                    .on("RECHAZOS").to(consultaRechazos)              // hasta TIT-SEQ-RECH = 14
                .from(decider).on("LOG").to(consultaLog)
                .from(decider).on("INCIDENTES*").to(consultaIncidentes)  // opciones 3 y 4
                .from(decider).on("CANCELADOS*").to(reinicio)          // opciones 5 y 6
                    .next(procesaCancelados)                          // archivos 01 a 03
                    .next(reporteCancelados)
                .end()
                .build();
    }
}

        var ec = ctx.getStepContext().getStepExecution().getJobExecution().getExecutionContext();
        var ultimo = reinicio.ultimo();            // READ ... UNTIL AT END: se queda con el último
        if (ultimo.isPresent()) {

        }
        return RepeatStatus.FINISHED;

@Bean
@StepScope
MultiResourceItemReader<RegistroCancelado> canceladosReader(
        @Value("#{jobExecutionContext['secuenciaArchivo']}") Integer desde,
        P106Rutas rutas, ReinicioWriter reinicio, CanceladoLineReader delegado) {

    var existentes = new ArrayList<Resource>();
    for (int sec = desde; sec <= 3; sec++) {                 // UNTIL WKS-TIT-CPY-I10 > 3
        Resource r = rutas.cancelados(sec);                  // CORE-DEP/FILE/CANCELADOS/{csi}/{fecha}/{sec}.
        if (r.exists()) {
            existentes.add(r);
        } else {
            reinicio.anotaMensaje(" ARCHIVO " + r.getFilename() + " NO RESIDENTE", sec + 1);
        }
    }
    return new MultiResourceItemReaderBuilder<RegistroCancelado>()
            .name("cancelados")
            .resources(existentes.toArray(Resource[]::new))
            .comparator((a, b) -> 0)                         // conservar el orden 01, 02, 03
            .delegate(delegado)
            .build();
}
@Component
@StepScope
public class CanceladoProcessor implements ItemProcessor<RegistroCancelado, ResultadoCancelacion> {

    @Value("#{jobExecutionContext['contratoReinicio']}") private String contratoReinicio;
    @Value("#{jobParameters['opcion']}")                  private String opcion;
    private final P106Parametros params;
    private final int miCsi;
    private final S016CancelacionClient s016;
    private final ReinicioWriter reinicio;
    @Value("#{jobExecutionContext['secuenciaArchivo']}") private int secuencia;
    private boolean huboDetalle;                                // se reinicia al abrir cada archivo

    @Override
    public ResultadoCancelacion process(RegistroCancelado reg) {
        if (contratoReinicio != null) {                         // WKS-SI-HAY-REINICIO = 1
            if (reg instanceof RegistroCancelado.Detalle d && d.contrato().equals(contratoReinicio)) {
                contratoReinicio = null;                        // el propio contrato ya se avisó
            }
            return null;                                        // EXIT PERFORM: se salta
        }
        return switch (reg) {
            case RegistroCancelado.Header h -> {
                if (!h.fechaContableAammdd().equals(params.fecha().format(AAMMDD)) || h.csiOrigen() != miCsi) {
                    throw new HeaderInvalidoException("ERROR EN HEADER");   // W77-EOF = 1: deja el archivo
                }
                yield null;
            }
            case RegistroCancelado.Detalle d -> {
                huboDetalle = true;                                           // W77-SI-DETALLE = 1
                yield "CANCELADOS_AVISA".equals(opcion)
                        ? ResultadoCancelacion.de(d, s016.avisa(d))          // 001-AVISA-CANC-S016
                        : ResultadoCancelacion.de(d, 99);                    // opción 6: contingencia
            }
            case RegistroCancelado.Trailer t -> {                             // 001-VALIDA-DET
                if (!huboDetalle) reinicio.anotaMensaje(" ARCHIVO ... VACIO ", secuencia);
                yield null;
            }
        };
    }

    private static final DateTimeFormatter AAMMDD = DateTimeFormatter.ofPattern("yyMMdd");
}

    }
}

public class ReporteCanceladosProcessor implements ItemProcessor<RegistroReinicio, LineaReporteCancelados> {

    private long rechazados, aceptados;                 // W77-REPCAN-RCHZ / W77-REPCAN-ACEPT

    @Override
    public LineaReporteCancelados process(RegistroReinicio r) {
        boolean esMensaje = !r.mensaje().isBlank();     // archivo no residente o vacío
        if (!esMensaje) {
            if (r.resultadoS016() > 0) rechazados++; else aceptados++;
        }
        return esMensaje ? LineaReporteCancelados.mensaje(r.mensaje())
                         : LineaReporteCancelados.detalle(r, aceptados + rechazados);
    }

    @AfterStep
    ExitStatus publica(StepExecution se) {
        se.getJobExecution().getExecutionContext().putLong("rechazos", rechazados);   // 001-ENCIENDE-INC
        return se.getExitStatus();
    }
}
@Component
@RequiredArgsConstructor
public class ReinicioWriter implements ItemWriter<ResultadoCancelacion> {

    private final ReinicioRepository reinicio;   // OPEN EXTEND / OPEN OUTPUT + WRITE + CLOSE WITH SAVE
    @Value("#{jobParameters['fecha']}") private String fechaAammdd;
    @Value("#{jobExecutionContext['secuenciaArchivo']}") private int secuencia;

    @Override
    @Transactional
    public void write(Chunk<? extends ResultadoCancelacion> chunk) {
        for (var r : chunk) {
            reinicio.agrega(new RegistroReinicio(fechaAammdd, r.contrato(), secuencia,
                    r.producto(), r.instrumento(), r.moneda(), r.cliente(), r.resultado(), ""));
        }
    }

    /** Renglón de aviso (archivo no residente o vacío): el fuente reutiliza el último detalle leído. */
    public void anotaMensaje(String mensaje, int secuenciaSiguiente) {
        reinicio.agrega(new RegistroReinicio(fechaAammdd, "", secuenciaSiguiente, 0, 0, 0, "", 0, mensaje));
    }
}

Riesgos de migración
Reanudación que puede saltar un archivo completo

001-LEE-CANCELADOS salta registros hasta encontrar el contrato del último reinicio (líneas 2355-2369). Si el último renglón de reinicio es un aviso de archivo faltante o vacío, guarda el contrato del detalle anterior con la secuencia siguiente (líneas 2301-2307 y 2647-2651), y ese contrato no estará en el archivo siguiente: todo el archivo se saltaría sin avisar al S016.

Entrada a S016L428 sin limpiar

El MOVE ZEROS de WS-S016-0701-ENTRADA está comentado (línea 2447); los campos no asignados explícitamente arrastran valores de la llamada anterior.

Header inválido sin abortar

Un header con fecha o CSI distintos (líneas 2375-2393) sólo imprime ERROR EN HEADER y cierra ese archivo; la corrida continúa con el siguiente y no enciende el incidente si no hubo rechazos.

Hardcodes de hosts y plazas

Los cuatro hosts que definen CSI 10 (líneas 1627-1633), las sucursales 3667 y 3084 con sus nombres (líneas 2575-2585) y el título de LIBLJ están compilados. Se trasladan a application.yml.

Comparación de fecha sin siglo

El header se valida con WKS-110-FECH de seis dígitos contra WKS-IN-FECH; el campo de siglo WKS-I10-SIG no se usa.
