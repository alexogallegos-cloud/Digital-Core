@Service
@RequiredArgsConstructor
public class TotalesPorClaveTransaccionService {      // TOTXCVETRA, sin estado compartido

    private final MovDiaRepository movDia;             // BASEMOVDIA = CORE-GLBD10MOVDIA151
    private final ControlDiaService control;           // LIB-CONTROL / CONSISDIA

    @Transactional(readOnly = true)
    public ConsultaTotalesResponse consulta(ConsultaTotalesRequest rq) {
        int f = rq.funcion();
        if (f == 0 || f == 40 || f == 41) {
            return control.levanta(rq);                 // 00-0100-LEVANTA
        }
        if (!control.baseDisponible()) {               // WKS-INICIO / WKS-ABRE-SEM
            return vacia(control.inicializado() ? ResultadoL040.BASE_DATOS_NO_DISPONIBLE
                                                : ResultadoL040.FUNCION_INVALIDA);
        }
        return switch (FuncionL040.de(f).orElse(null)) {
            case SIS -> porSistema(rq);                                   // 01-0100-SIS
            case SIS_BORI_BDES_PROD -> porProducto(rq);                   // 02-0100-PRD
            // ... 3 a 14 con el mismo patrón ...
            case null, default -> vacia(ResultadoL040.FUNCION_INVALIDA);  // 15-20 y > 34
        };
    }
}
@Component
@RequiredArgsConstructor
public class ControlDiaService {

    private final LibControlClient libControl;     // CONSISDIA IN LIB-CONTROL, función 1
    private final MovDiaDataSourceRouter bases;    // la base MOVDIA cambia de título por sistema
    private final CsiProvider csi;                 // CSIID IN SOPORTECOMS

    /** 00-0100-LEVANTA: función 0 abre, 40 cierra, 41 reabre. */
    public ConsultaTotalesResponse levanta(ConsultaTotalesRequest rq) {
        return switch (rq.funcion()) {
            case 0 -> {
                if (bases.abierta(rq.sistema())) yield vacia(ResultadoL040.SIN_ERROR);
                var dia = libControl.consisdia(rq.sistema(), csi.numero());
                yield switch (dia.resultado()) {
                    case 0 -> dia.estadoBaseSemana() == 0
                            ? vacia(bases.abre(rq.sistema(), dia.nombreBaseSemana()))   // 0 o 7
                            : vacia(ResultadoL040.BASE_DATOS_NO_DISPONIBLE);            // 7
                    case 1 -> vacia(ResultadoL040.SISTEMA_NO_DADO_DE_ALTA);             // 8
                    default -> vacia(ResultadoL040.BASE_CONTROL_NO_DISPONIBLE);         // 6
                };
            }
            case 40 -> vacia(bases.cierra(rq.sistema()));
            case 41 -> vacia(bases.abre(rq.sistema(), null));
            default -> vacia(ResultadoL040.FUNCION_INVALIDA);
        };
    }
}
@Component
@RequiredArgsConstructor
public class MovDiaDataSourceRouter {

    private final MovDiaProperties props;      // reintento-espera: 20s, max-intentos: 5
    private final Map<Integer, DataSource> abiertas = new ConcurrentHashMap<>();

    /** 00-0130-ABRE-MOVDIA: OPEN UPDATE con reintento si la base está en uso exclusivo. */
    @Retryable(retryFor = BaseEnUsoExclusivoException.class,
               maxAttemptsExpression = "${movdia.max-intentos:5}",
               backoff = @Backoff(delayExpression = "${movdia.reintento-espera:20000}"))
    public ResultadoL040 abre(int sistema, String nombreBase) {
        if (nombreBase == null || nombreBase.isBlank()) return ResultadoL040.BASE_DATOS_NO_DISPONIBLE;
        abiertas.computeIfAbsent(sistema, s -> conecta(nombreBase));   // DMSTATUS 39 -> excepción reintentable
        return ResultadoL040.SIN_ERROR;
    }

    @Recover
    ResultadoL040 agotado(BaseEnUsoExclusivoException e, int sistema, String nombreBase) {
        log.error("ERROR EN BASE BD10 : ARCHIVO DE CONTROL EN USO EN FORMA EXCLUSIVA {}", e.detalle());
        return ResultadoL040.BASE_DATOS_NO_DISPONIBLE;   // el fuente reintenta sin fin
    }

    public ResultadoL040 cierra(int sistema) {
        var ds = abiertas.remove(sistema);
        return ds == null || Conexiones.cierra(ds) ? ResultadoL040.SIN_ERROR : ResultadoL040.ERROR_DMSII;
    }
}

final class PaginaTotales {
    private final int max; private final boolean agrupaPorCve;
    private final List<TotalRenglon> renglones = new ArrayList<>();
    private ConsultaTotalesRequest.Continuacion continuacion;
    private boolean agotada = true;

    void acumula(MovDia r) {
        continuacion = r.llave();                          // WKS-L40-*CON := registro actual
        int i = renglones.size() - 1;
        if (agrupaPorCve && i >= 0 && renglones.get(i).cveTra() == r.cveTra()) {
            var t = renglones.get(i);
            renglones.set(i, t.suma(r.numMov(), r.impMov()));   // COMPUTE ... = B50 + acumulado
        } else if (renglones.size() < max) {
            renglones.add(TotalRenglon.de(r));                  // 01-0115-MUEVE-DATOS
        } else {
            agotada = false;                                    // W77-IND = 101: no se copia
        }
    }
    boolean llena() { return renglones.size() == max && !agrupaPorCve; }

Riesgos de migración
Estado compartido en una librería SHARED BY ALL

WKS-INICIO, WKS-ABRE-SEM, W77-CVETRA-ANT y la posición actual del índice B50SXTOTXCVE viven en WORKING-STORAGE de una única instancia compartida (línea 5). Dos tareas que paginen al mismo tiempo comparten el cursor. La versión Java es sin estado y la continuación viaja en la petición.

Reintento infinito al abrir MOVDIA

En 00-0130-ABRE-MOVDIA (líneas 1372-1382) el error DMSII 39 deja W77-BASE en cero y el contador de intentos se reinicia tras el aviso, así que la apertura se repite cada 20 segundos sin límite.

Funciones fuera de rango

Las funciones 15 a 20 ejecutan 00005-FUNCION-ERROR y caen por flujo a 00010-LIMPIA-ARREGLO con subíndice cero (líneas 1270-1283). Las funciones 35 a 39 y mayores a 41 no tienen destino en el GO TO DEPENDING y regresan resultado 0.

Escritura posible fuera del arreglo

En la función 21 el segundo ciclo de 01-0100-SIS (líneas 1447-1452) usa W77-CVETRA-ANT sin asignarlo; si coincide con la clave actual, 01-0110-CONSIS escribe el renglón 101 de un arreglo de 100.

Condición siempre verdadera

En 01-0155-FF-CONT (líneas 1578-1584) un IF DMSTATUS(NOTFOUND) anidado dentro de otro idéntico hace inalcanzable su rama ELSE.

Desbordamiento silencioso

Las sumas de 01-0110-CONSIS (líneas 1479-1482) no tienen ON SIZE ERROR sobre 9(08) y S9(13)V99.

Apertura en modo actualización

La librería abre CORE-GLBD10MOVDIA151 con OPEN UPDATE (línea 1369) aunque solo hace FIND; compite por bloqueos con los procesos que sí actualizan.
