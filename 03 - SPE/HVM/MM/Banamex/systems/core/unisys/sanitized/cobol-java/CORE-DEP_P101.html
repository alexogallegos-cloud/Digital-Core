/** W77-PARAM-WFL X(31): opción 9 + fecha 9(08) + CSI 9(02) + job 9(05) + paso X(15). */
public record ParametroWfl(Opcion opcion, LocalDate fechaBatch, int csi, int job, String paso) {

    public enum Opcion {
        INICIO_WFL(1), INICIO_PASO(2), FIN_PASO(3), VALIDA_REINICIO(4), FIN_WFL(5),
        INCIDENTE(6), PROCESO_ESPECIAL(7), FIN_PASO_SIN_ROLLBACK(8);
        final int codigo;
        Opcion(int c) { codigo = c; }
    }

    static ParametroWfl parse(String p) {                     // ("2200501150499999CORE-DEPP130CORTCOM")
        if (p == null || p.length() != 31) throw new ParametroErroneoException(p);
        return new ParametroWfl(Opcion.values()[Character.digit(p.charAt(0), 10) - 1],
                LocalDate.parse(p.substring(1, 9), DateTimeFormatter.BASIC_ISO_DATE),
                Integer.parseInt(p.substring(9, 11)),
                Integer.parseInt(p.substring(11, 16)),
                p.substring(16, 31));
    }
}

/** WKS-PARAM-SAL: "R B TEXTO-27"; se devuelve en TASKSTRING. */
public record RespuestaWfl(int opcReinicio, int opcRollback, String datos) {
    static RespuestaWfl ok()                 { return new RespuestaWfl(0, 0, "OK"); }
    static RespuestaWfl error(int c, String t) { return new RespuestaWfl(c, c, t); }
    String toTaskString() { return "%d %d %-27.27s".formatted(opcReinicio, opcRollback, datos); }
}
@Component
@RequiredArgsConstructor
class ValidadorParametro {

    private final ControlWflProperties props;   // csi-validos: [4, 10] · anio-minimo: 2004 · anio-maximo: 2099

    /** 1000200-VALIDA-PARAM: todas las condiciones a la vez; un solo mensaje si alguna falla. */
    Optional<ParametroWfl> valida(String crudo) {
        try {
            ParametroWfl p = ParametroWfl.parse(crudo);
            boolean ok = props.anioValido(p.fechaBatch().getYear())
                    && props.csiValidos().contains(p.csi())
                    && p.job() >= 1 && p.job() <= 99_999
                    && !p.paso().isBlank();
            if (!ok) throw new ParametroErroneoException(crudo);
            return Optional.of(p);
        } catch (DateTimeParseException | ParametroErroneoException | ArrayIndexOutOfBoundsException e) {
            lj.error("PARAMETRO ERRONEO ");
            return Optional.empty();
        } finally {
            lj.info("PARAMETRO RECIBIDO <" + crudo + ">");
        }
    }
}

@Service
@RequiredArgsConstructor
public class ControlWflService {

    private final CtrlWflRepository repo;
    private final PuntoSincroniaService sincronia;   // 5003210-ACCESA-BASE
    private final MarcaPasoService marcas;           // archivo ESTE PASO YA CORRIO HOY
    private final ProcesoEspecialService especial;   // ESPECIALENLOTED

    @Transactional
    public Respuesta despacha(ParametroWfl p) {
        switch (p.opcion()) {                          // 5002000-BUSCA-PROCESO
            case INICIO_WFL:       return Respuesta.de(iniciaWfl(p));
            case INCIDENTE:        throw new IncidentePasoException(p.paso());   // STATUS -1: DS del paso
            case PROCESO_ESPECIAL: return especial.lee(p);
            default:               break;
        }
        Optional<CtrlWfl> ctrl = repo.findById(new CtrlWflId(p.fechaBatch(), p.csi()));
        if (ctrl.isEmpty()) {                          // ATTRIBUTE RESIDENT = FALSE
            return Respuesta.de(p.opcion() == Opcion.VALIDA_REINICIO
                    ? new RespuestaWfl(1, 1, "SINREINICIO")
                    : RespuestaWfl.error(7, "NOHAYARCHIVO"));
        }
        return Respuesta.de(switch (p.opcion()) {      // 5003000-QUE-PROCESO
            case INICIO_PASO     -> iniciaPaso(ctrl.get(), p);
            case FIN_PASO        -> finPaso(ctrl.get(), p);
            case VALIDA_REINICIO -> validaReinicio(ctrl.get());
            case FIN_WFL         -> finWfl(ctrl.get());
            default              -> marcas.genera(ctrl.get(), p);  // opción 8: 5004000-VALGEN-ARCH-PASO
        });
    }
}
    /** 5003200-INICIO-PASO: el paso solo arranca si el WFL está en curso y no hay otro paso abierto. */
    RespuestaWfl iniciaPaso(CtrlWfl c, ParametroWfl p) {
        if (c.wflStatus != 1) return RespuestaWfl.error(9, "NOINICIOWFL");
        if (c.pasoStatus != 0 && c.pasoStatus != 2) return RespuestaWfl.error(8, "HAYPASOSINFIN");
        c.pasoStatus = 1;
        c.pasoRequiereRollback = true;
        c.pasoJob = proceso.jobNumber();             // ATTRIBUTE JOBNUMBER OF MYJOB
        c.pasoMix = proceso.mixNumber();             // ATTRIBUTE MIXNUMBER OF MYSELF
        c.pasoInicio = LocalDateTime.now(clock);
        c.pasoNombre = p.paso();
        repo.save(c);
        sincronia.marca();                           // punto de rollback en la base
        return RespuestaWfl.ok();
    }

/** 5005000-VAL-REINICIO: decide si el lote se reanuda y si el paso interrumpido requiere rollback. */
@Component
@RequiredArgsConstructor
public class ControlWflDecider implements JobExecutionDecider {

    private final CtrlWflRepository repo;

    @Override
    public FlowExecutionStatus decide(JobExecution job, StepExecution step) {
        var id = new CtrlWflId(job.getJobParameters().getLocalDate("fecha"),
                               job.getJobParameters().getLong("csi").intValue());
        return repo.findById(id)
                .filter(c -> c.wflStatus == 1)
                .map(c -> c.pasoStatus == 1
                        ? new FlowExecutionStatus("CON_ROLLBACK")      // 2 2 job mix fecha hora
                        : new FlowExecutionStatus("CON_REINICIO"))     // 2 1 CONREINICIO
                .orElse(new FlowExecutionStatus("SIN_REINICIO"));      // 1 1 SINREINICIO
    }

    static String datosRollback(CtrlWfl c) {                           // 5005400-CON-ROLLBACK
        return "%05d %05d %s %s".formatted(c.pasoJob, c.pasoMix,
                c.pasoInicio.toLocalDate().format(BASIC_ISO_DATE),
                c.pasoInicio.toLocalTime().format(DateTimeFormatter.ofPattern("HHmmss")));
    }
}

            return Respuesta.de(RespuestaWfl.error(5, "ERRARCHH"));   // ver riesgo: el COBOL no llena la respuesta
        }
        List<String> regs = Files.readAllLines(archivo, CP1047);
        if (regs.isEmpty() || !esHeaderValido(regs.get(0), p)) return Respuesta.de(RespuestaWfl.error(5, "ERRARCHH"));
        if (regs.size() < 2 || !regs.get(1).startsWith("03"))  return Respuesta.de(RespuestaWfl.error(5, "ERRARCHD"));
        var d = regs.get(1);
        return Respuesta.especial(d.substring(3, 7), d.substring(8, 33), d.substring(34, 46));  // PASOANT, PASOESP, VALUE
    }

    private boolean esHeaderValido(String h, ParametroWfl p) {
        return h.startsWith("01") && Integer.parseInt(h.substring(2, 4)) == p.csi()
                && h.substring(4, 12).equals(p.fechaBatch().format(BASIC_ISO_DATE))
                && h.substring(12, 16).equals("P101");
    }
}

Riesgos de migración
Punto de rollback sin verificación

En 5003210-ACCESA-BASE los errores de OPEN, BEGIN-TRANSACTION y END-TRANSACTION no se evalúan; WS-STATUS-BASE se asigna y nunca se consulta. El bloqueo y STORE de CORE-DEPB02CONTROL están comentados.

Verificación de versión desactivada

PERFORM 1000100-VERSIONES está comentado en la línea 771.

Respuesta indefinida en la opción 7

Cuando no existe ESPECIALENLOTED, el programa devuelve WKS-PARAM-SAL sin asignar los códigos de reinicio y rollback.

Inyección de pasos por archivo

Un archivo ESPECIALENLOTED con header válido hace que el WFL ejecute el paso que indique. No hay firma, lista blanca ni registro de quién lo colocó.

Validación de fecha incompleta

Día y mes se validan por separado (01-31 y 01-12); fechas inexistentes pasan la validación.

Hardcodes

Prefijo CORE-DEP/FILE/CTRLWFL/, nombre CTRLCORE-DEPWFLLOTE, ESPECIALENLOTED, valor inicial CORE-GLP17701 del nombre de marca, CSI 04 y 10, años 2004 a 2099 y MAXCARDS 9. Pasan a application.yml.
