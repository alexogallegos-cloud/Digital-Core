$ SET VERSION 06.001.06000
%$ SET TADS
BEGIN

INPUTHEADER     COMSINHDR;
OUTPUTHEADER    COMSOUTHDR;

LIBRARY TIME_OUT(LIBACCESS = BYTITLE);

PROCEDURE ENVIA_CANCELACIONES(COMS_IN,COMS_OUT,IN_EV,TSK_EV);
    INPUTHEADER  COMS_IN;
    OUTPUTHEADER COMS_OUT;
    EVENT        IN_EV, TSK_EV;
    LIBRARY TIME_OUT;

PROCEDURE INICIA;
    BEGIN
    EBCDIC ARRAY E_MSG, TITULO[0:71], AUX[0:20];
    EBCDIC VALUE ARRAY MY_ID("18MTP002");

    LIBRARY CTLVER(LIBACCESS=BYFUNCTION,FUNCTIONNAME="CTLVERS"),
            SHELL (LIBACCESS=BYFUNCTION,FUNCTIONNAME="SOPORTECOMS.");

    INTEGER PROCEDURE DAME_TIT(ID,TITULO);
        EBCDIC  ARRAY ID,TITULO[0];
        LIBRARY CTLVER;

    PROCEDURE CSIID(INFO);
        EBCDIC ARRAY INFO[0];
        LIBRARY SHELL;

    REPLACE AUX BY MYSELF.USERCODE;
    REPLACE E_MSG BY AUX UNTIL = ".","L093ASIN",48"00";
    IF DAME_TIT(E_MSG,TITULO) < 0 THEN
        DISPLAY("ERROR AL LLAMAR A CONTROL DE VERSIONES")
    ELSE
        REPLACE TIME_OUT.TITLE BY TITULO FOR 72;

    CSIID(E_MSG);
    MYSELF.OPTIONS:=* & 1 [VALUE(BDBASE):1]
                      & 1 [VALUE(FAULT) :1]
                      & 1 [VALUE(DSED)  :1]
                      & 1 [VALUE(ARRAYS):1];
    REPLACE TITULO BY "S500/LIST/P093/",E_MSG FOR 2,
        "00/R000/",TIME(15).[15:48],".";
    REPLACE MYSELF.BDNAME BY TITULO;
    ENABLE(COMSINHDR,"ONLINE");
    DELINKLIBRARY(CTLVER);
    DELINKLIBRARY(SHELL);
    END INICIA;

INICIA;

ENVIA_CANCELACIONES
    (COMSINHDR,COMSOUTHDR,MYSELF.DCIINPUTEVENT,MYSELF.DCITASKEVENT);

END.
