    SUBROUTINE FECHAS ( INTEGER IFECHAHOY );                                      00232000
       BEGIN                                                                      00233000
          CALCULA ( IFECHAHOY );                                                  00234000
          FECHAY := IFECHAHOY;                                                    00235000
          IDIA   := IFECHAHOY MOD 100;                                            00236000
          IMES   := IFECHAHOY DIV 100 MOD 100;                                    00237000
          IF IFESTIVO = 1 THEN FESTIVO    := FALSE                                00238000
          ELSE                 FESTIVO    := TRUE ;                               00239000
          CASE IDIASEM OF                                                         00240000
            BEGIN                                                                 00241000
               (0):DOMINGO  := TRUE;                                              00242000
               (1):LUNES    := TRUE;                                              00243000
               (2):MARTES   := TRUE;                                              00244000
               (3):MIERCOLES:= TRUE;                                              00245000
               (4):JUEVES   := TRUE;                                              00246000
               (5):VIERNES  := TRUE;                                              00247000
               (6):SABADO   := TRUE;                                              00248000
            END;                                                                  00249000
 %     CALCULO DE DIA ULTIMO HABIL DE MES                                         00250000
  IF IFECHAHOY DIV 100 MOD 10 NEQ IFECHAPROX DIV 100 MOD 10 THEN                  00251000
     DIA30 := TRUE                                                                00252000
  ELSE                                                                            00253000
     DIA30 := FALSE;                                                              00254000
 %     CALCULO DE PRIMER DIA HABIL DE MES                                         00255000
  IF IFECHAHOY DIV 100 MOD 10 NEQ IFECHAANT  DIV 100 MOD 10 THEN                  00256000
     DIA1MES := TRUE                                                              00257000
  ELSE                                                                            00258000
     DIA1MES := FALSE;                                                            00259000
 %   CALCULO DE PROCESO SEMANAL (FIN DE SEMANA)                                   00260000
  IF IDIAPROX < IDIASEM  THEN                                                     00261000
     DIAFINSEM := TRUE                                                            00262000
  ELSE                                                                            00263000
     DIAFINSEM := FALSE;                                                          00264000
 %     CALCULO DE PROCESO QUINCENAL                                               00265000
  IF IFECHAHOY MOD 100 = 15 OR                                                    00266000
     DIA30                  OR                                                    00267000
     ( 15 > IFECHAHOY MOD 100 AND 15 < IFECHAPROX MOD 100 )                       00268000
  THEN                                                                            00269000
     DIA15 := TRUE                                                                00270000
  ELSE                                                                            00271000
     DIA15 := FALSE;                                                              00272000
 %      CALCULO DE PROCESO MULTIPLO DE DIA 5 ( 5,10,15,20,25,30)                  00273000
  IF IFECHAHOY MOD 100 MOD 5 = 0 OR                                               00274000
     DIA30                       OR                                               00275000
     DIA15                       OR                                               00276000
     ( 5  > IFECHAHOY MOD 100 AND  5 < IFECHAPROX MOD 100 ) OR                    00277000
     ( 10 > IFECHAHOY MOD 100 AND 10 < IFECHAPROX MOD 100 ) OR                    00278000
     ( 25 > IFECHAHOY MOD 100 AND 25 < IFECHAPROX MOD 100 ) OR                    00279000
     ( 30 > IFECHAHOY MOD 100 AND 30 < IFECHAPROX MOD 100 )                       00280000
  THEN                                                                            00281000
     DIA5  := TRUE                                                                00282000
  ELSE                                                                            00283000
     DIA5  := FALSE;                                                              00284000
 %     CALCULO DE PROCESO AL DIA 24                                               00285000
  IF IFECHAHOY MOD 100 = 24  OR                                                   00286000
     ( 24 > IFECHAHOY MOD 100 AND 24 < IFECHAPROX MOD 100 )                       00287000
  THEN                                                                            00288000
     DIA24 := TRUE                                                                00289000
  ELSE                                                                            00290000
     DIA24 := FALSE;                                                              00291000
 %     CALCULO DE PROCESO AL DIA 25                                               00292000
  IF IFECHAHOY MOD 100 = 25  OR                                                   00293000
     ( 25 > IFECHAHOY MOD 100 AND 25 < IFECHAPROX MOD 100 )                       00294000
  THEN                                                                            00295000
     DIA25 := TRUE                                                                00296000
  ELSE                                                                            00297000
     DIA25 := FALSE;                                                              00298000
 %     CALCULO DE PROCESO AL DIA 23                                               00299000
  IF ( IFECHAHOY MOD 100 = 23  OR                                                 00300000
     ( 23 > IFECHAHOY MOD 100 AND 23 < IFECHAPROX MOD 100 ))                      00301000
     AND NOT ( DIA24 )                                                            00302000
  THEN                                                                            00303000
     DIA23 := TRUE                                                                00304000
  ELSE                                                                            00305000
     DIA23 := FALSE;                                                              00306000
 %     CALCULO DE PENULTIMO DIA HABIL DE MES                                      00307000
                                                                                  00308000
  IF IFECHAPROX = IFECHAFINMES THEN                                               00309000
     DIA29 := TRUE                                                                00310000
  ELSE                                                                            00311000
     DIA29 := FALSE;                                                              00312000
 %       CALCULO DE PRIMER DIA HABIL DE LA SEMANA                                 00313000
  IF IDIASEM  < IDIAANT THEN                                                      00314000
     DIA1SEM := TRUE                                                              00315000
  ELSE                                                                            00316000
     DIA1SEM := FALSE;                                                            00317000
 %       CALCULO DE PROCESO DE DIAS INHABILES ENTRE SEMANA                        00318000
  IF IDIASEM < IDIAPROX THEN  % ENCIENDE BOOL.DE DIAS PUENTE ENTRE SEM            00319000
   BEGIN                                                                          00320000
    CONT := IDIASEM;                                                              00321000
    DO BEGIN                                                                      00322000
     CASE CONT OF                                                                 00323000
      BEGIN                                                                       00324000
       (0):DOMINGO   := TRUE;                                                     00325000
       (1):LUNES     := TRUE;                                                     00326000
       (2):MARTES    := TRUE;                                                     00327000
       (3):MIERCOLES := TRUE;                                                     00328000
       (4):JUEVES    := TRUE;                                                     00329000
       (5):VIERNES   := TRUE;                                                     00330000
       (6):SABADO    := TRUE;                                                     00331000
      END;                                                                        00332000
     CONT := CONT + 1;                                                            00333000
    END UNTIL CONT = IDIAPROX ;                                                   00334000
   END;                                                                           00335000
 % CALCULO DE PROCESO DE DIAS INHABILES AL INICIO DE LA SEMANA                    00336000
  IF IDIAANT GEQ IDIASEM THEN % ENC. BOOLEAN PUENTE INICIO DE SEMANA              00337000
   BEGIN                                                                          00338000
    CONT := 1;                                                                    00339000
    DO BEGIN                                                                      00340000
     CASE CONT OF                                                                 00341000
      BEGIN                                                                       00342000
       (1):LUNES     := TRUE;                                                     00343000
       (2):MARTES    := TRUE;                                                     00344000
       (3):MIERCOLES := TRUE;                                                     00345000
       (4):JUEVES    := TRUE;                                                     00346000
       (5):VIERNES   := TRUE;                                                     00347000
      END;                                                                        00348000
     CONT := CONT + 1;                                                            00349000
    END UNTIL CONT GEQ IDIASEM ;                                                  00350000
   END;                                                                           00351000
 % CALCULO DE PROCESO DE DIAS INHABILES AL FIN DE LA SEMANA                       00352000
   IF IDIASEM < 5 AND IDIAPROX LEQ IDIASEM THEN % ENC.BOO.PUENT FIN SEM           00353000
   BEGIN                                                                          00354000
    CONT := IDIASEM;                                                              00355000
    DO BEGIN                                                                      00356000
     CASE CONT OF                                                                 00357000
      BEGIN                                                                       00358000
       (1):LUNES     := TRUE;                                                     00359000
       (2):MARTES    := TRUE;                                                     00360000
       (3):MIERCOLES := TRUE;                                                     00361000
       (4):JUEVES    := TRUE;                                                     00362000
       (5):VIERNES   := TRUE;                                                     00363000
      END;                                                                        00364000
     CONT := CONT + 1;                                                            00365000
    END UNTIL CONT = 6;                                                           00366000
   END;                                                                           00367000
END;                                                                              00368000
